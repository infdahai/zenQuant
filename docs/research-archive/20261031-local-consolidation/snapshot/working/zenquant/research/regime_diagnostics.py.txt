"""Causal, fixed-window market labels for research reporting only.

Labels do not participate in strategy decisions, risk limits, or admission.
Compounding disjoint days with a shared label is a diagnostic grouping, never
a tradable equity curve; no regime-specific drawdown is manufactured from it.
"""

from __future__ import annotations

import math
from collections.abc import Mapping, Sequence
from itertools import pairwise
from typing import Any

VERSION = "research-regime-diagnostics.v1"
BAR_NS = 14_400_000_000_000
DAY_NS = 6 * BAR_NS
MS_NS = 1_000_000
SMA_BARS = 720
MOMENTUM_BARS = 180
REGIMES = ("bull", "bear", "neutral", "UNKNOWN")


class _Issue(ValueError):
    def __init__(self, reason: str, path: str, status: str = "INVALID") -> None:
        super().__init__(reason)
        self.reason, self.path, self.status = reason, path, status


def _result(status: str, **fields: Any) -> dict[str, Any]:
    return {
        "schema_version": VERSION,
        "status": status,
        "scope": "regime_diagnostics_only",
        "missing": [],
        "blockers": [],
        **fields,
        "live_eligible": False,
    }


def _failed(issue: _Issue, **fields: Any) -> dict[str, Any]:
    key = "missing" if issue.status == "NOT_EVALUABLE" else "blockers"
    return _result(issue.status, **{key: [{"path": issue.path, "reason": issue.reason}]}, **fields)


def _mapping(value: Any, path: str) -> Mapping[str, Any]:
    if not isinstance(value, Mapping):
        raise _Issue("EXPECTED_OBJECT", path)
    return value


def _sequence(value: Any, path: str) -> Sequence[Any]:
    if value is None:
        raise _Issue("MISSING_EVIDENCE", path, "NOT_EVALUABLE")
    if not isinstance(value, Sequence) or isinstance(value, (str, bytes)):
        raise _Issue("EXPECTED_SEQUENCE", path)
    return value


def _time(value: Any, path: str) -> int:
    if isinstance(value, bool) or not isinstance(value, int):
        raise _Issue("EXPECTED_INTEGER_NANOSECONDS", path)
    return value


def _number(value: Any, path: str) -> float:
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        raise _Issue("EXPECTED_FINITE_NUMBER", path)
    try:
        number = float(value)
    except OverflowError as exc:
        raise _Issue("NONFINITE_NUMBER", path) from exc
    if not math.isfinite(number):
        raise _Issue("NONFINITE_NUMBER", path)
    return number


def _bars(value: Sequence[Mapping[str, Any]]) -> tuple[dict[int, dict[str, Any]], int]:
    records: dict[int, dict[str, Any]] = {}
    previous, offset = None, None
    for index, raw in enumerate(_sequence(value, "bars")):
        path = f"bars[{index}]"
        row = _mapping(raw, path)
        closed = _time(row.get("close_time_ns"), f"{path}.close_time_ns")
        if previous is not None and closed <= previous:
            raise _Issue("DUPLICATE_OR_UNORDERED_BAR", path)
        residue = closed % BAR_NS
        if residue not in (0, BAR_NS - MS_NS):
            raise _Issue("NOT_CONFIRMED_4H_BOUNDARY", path)
        current_offset = 0 if residue == 0 else MS_NS
        if offset is not None and current_offset != offset:
            raise _Issue("MIXED_BAR_CLOSE_CONVENTIONS", path)
        offset = current_offset
        if "open_time_ns" in row:
            opened = _time(row["open_time_ns"], f"{path}.open_time_ns")
            if opened % BAR_NS or closed - opened != BAR_NS - offset:
                raise _Issue("NOT_CONFIRMED_4H_INTERVAL", path)
        price = _number(row.get("close"), f"{path}.close")
        if price <= 0.0:
            raise _Issue("NONPOSITIVE_PRICE", path)
        assumed = "available_at_ns" not in row
        available = closed + MS_NS if assumed else _time(row["available_at_ns"], f"{path}.available_at_ns")
        if available < closed:
            raise _Issue("AVAILABLE_BEFORE_CONFIRMED_CLOSE", path)
        confirmed = row.get("confirmed", True)
        if not isinstance(confirmed, bool):
            raise _Issue("EXPECTED_BOOLEAN_CONFIRMED", path)
        records[closed] = {"close": price, "available": available, "assumed": assumed, "confirmed": confirmed}
        previous = closed
    return records, offset or 0


def assign_daily_regimes(bars: Sequence[Mapping[str, Any]], day_start_ns: Sequence[int]) -> dict[str, Any]:
    """Label day T using exactly 720 consecutive confirmed 4h closes before T.

    Source close timestamps may use exact 4h boundaries or Binance's boundary
    minus 1ms convention, consistently within the input. An absent availability
    timestamp explicitly assumes close + 1ms. Late/missing/unconfirmed bars in
    the required window yield UNKNOWN instead of shifting to an older window.
    """
    try:
        days = [_time(day, "day_start_ns") for day in _sequence(day_start_ns, "day_start_ns")]
        if any(day % DAY_NS for day in days) or any(right <= left for left, right in pairwise(days)):
            raise _Issue("DAYS_MUST_BE_ORDERED_UNIQUE_UTC_MIDNIGHTS", "day_start_ns")
        records, offset = _bars(bars)
        labels, missing = [], []
        for day in days:
            cutoff = day - offset if offset else day - BAR_NS
            timestamps = [cutoff - index * BAR_NS for index in range(SMA_BARS - 1, -1, -1)]
            window = [records.get(timestamp) for timestamp in timestamps]
            reason, regime = "FIXED_CAUSAL_RULE", "UNKNOWN"
            assumed = any(row is not None and row["assumed"] for row in window)
            if any(row is None for row in window):
                reason = "INSUFFICIENT_OR_MISSING_4H_BARS"
            elif any(not row["confirmed"] for row in window):
                reason = "UNCONFIRMED_4H_BAR"
            elif any(row["available"] > day for row in window):
                reason = "REQUIRED_BAR_NOT_AVAILABLE_AT_DAY_START"
            else:
                prices = [row["close"] for row in window]
                mean = math.fsum(price / SMA_BARS for price in prices)
                current, prior = prices[-1], prices[-1 - MOMENTUM_BARS]
                # Positive prices make direct comparisons equivalent to the
                # ratio sign, without overflow on extreme but finite prices.
                regime = (
                    "bull"
                    if current > mean and current > prior
                    else "bear"
                    if current < mean and current < prior
                    else "neutral"
                )
            label = {
                "day_start_ns": day,
                "regime": regime,
                "source_cutoff_ns": cutoff if regime != "UNKNOWN" else None,
                "availability_assumed": assumed,
                "reason": reason,
            }
            labels.append(label)
            if regime == "UNKNOWN":
                missing.append({"day_start_ns": day, "reason": reason})
        if not days:
            missing.append({"path": "day_start_ns", "reason": "NO_REQUESTED_DAYS"})
        return _result(
            "NOT_EVALUABLE" if missing else "PASS",
            labels=labels,
            missing=missing,
            method={
                "sma_bars": SMA_BARS,
                "momentum_bars": MOMENTUM_BARS,
                "timeframe": "4h",
                "frozen": True,
                "close_rule": "close_time_ns < T",
                "availability_rule": "available_at_ns <= T",
                "missing_availability_assumption": "close_time_ns + 1000000",
                "parameter_search": False,
            },
        )
    except _Issue as issue:
        return _failed(issue, labels=[])


def _labels(value: Any) -> dict[int, Mapping[str, Any]]:
    if isinstance(value, Mapping):
        if value.get("status") in ("INVALID", "FAIL", "STALE"):
            raise _Issue("UNUSABLE_REGIME_LABEL_SOURCE", "labels")
        value = value.get("labels")
    result = {}
    for index, raw in enumerate(_sequence(value, "labels")):
        path = f"labels[{index}]"
        row = _mapping(raw, path)
        day = _time(row.get("day_start_ns"), f"{path}.day_start_ns")
        if day % DAY_NS or day in result:
            raise _Issue("DUPLICATE_OR_NON_MIDNIGHT_LABEL", path)
        if not isinstance(row.get("regime"), str) or row["regime"] not in REGIMES:
            raise _Issue("UNKNOWN_REGIME_NAME", path)
        cutoff = row.get("source_cutoff_ns")
        if cutoff is not None and _time(cutoff, f"{path}.source_cutoff_ns") >= day:
            raise _Issue("NONCAUSAL_LABEL_CUTOFF", path)
        if row["regime"] != "UNKNOWN" and cutoff is None:
            raise _Issue("MISSING_LABEL_SOURCE_CUTOFF", path, "NOT_EVALUABLE")
        if not isinstance(row.get("availability_assumed"), bool):
            raise _Issue("MISSING_AVAILABILITY_DISCLOSURE", path, "NOT_EVALUABLE")
        result[day] = row
    return result


def _compound(values: Sequence[float]) -> float:
    try:
        result = math.expm1(math.fsum(math.log1p(value) for value in values))
    except (OverflowError, ValueError) as exc:
        raise _Issue("NONFINITE_COMPOUND_RETURN", "daily_returns") from exc
    if not math.isfinite(result):
        raise _Issue("NONFINITE_COMPOUND_RETURN", "daily_returns")
    return result


def attribute_regimes(report: Mapping[str, Any], labels: Any) -> dict[str, Any]:
    """Attribute return stamped t to the label at t - one UTC day.

    ``arithmetic_contribution`` is the sum of the regime's daily return
    fractions, not contribution to the compounded total. Worst segment means
    the complete compounded return of each contiguous same-regime run, not a
    peak-to-trough drawdown or worst subperiod. UNKNOWN days remain visible.
    """
    try:
        report = _mapping(report, "report")
        if not report.get("identity") or not isinstance(report["identity"], Mapping):
            raise _Issue("MISSING_REPORT_IDENTITY", "report.identity", "NOT_EVALUABLE")
        ruin = _mapping(report.get("ruin", {}), "report.ruin")
        if not isinstance(ruin.get("occurred", False), bool):
            raise _Issue("EXPECTED_BOOLEAN", "report.ruin.occurred")
        if ruin.get("occurred") is True:
            raise _Issue("RUIN", "report.ruin", "FAIL")
        assigned = _labels(labels)
        rows, missing, previous = [], [], None
        sampling = _mapping(report.get("daily_sampling", {}), "report.daily_sampling")
        for timestamp in _sequence(sampling.get("missing_midnights", []), "report.daily_sampling.missing_midnights"):
            missing.append(
                {
                    "time_ns": _time(timestamp, "report.daily_sampling.missing_midnights"),
                    "reason": "MISSING_DAILY_OBSERVATIONS",
                }
            )
        for index, raw in enumerate(_sequence(report.get("daily_returns"), "daily_returns")):
            path = f"daily_returns[{index}]"
            row = _mapping(raw, path)
            timestamp = _time(row.get("time_ns"), f"{path}.time_ns")
            if timestamp % DAY_NS or (previous is not None and timestamp <= previous):
                raise _Issue("DUPLICATE_UNORDERED_OR_NON_MIDNIGHT_RETURN", path)
            if previous is not None and timestamp != previous + DAY_NS:
                missing.append(
                    {
                        "path": "daily_returns",
                        "start_ns": previous + DAY_NS,
                        "end_ns": timestamp,
                        "reason": "MISSING_DAILY_OBSERVATIONS",
                    }
                )
            value = _number(row.get("value"), f"{path}.value")
            if value <= -1.0:
                raise _Issue("RUIN", path, "FAIL")
            day = timestamp - DAY_NS
            label = assigned.get(day)
            regime = label["regime"] if label is not None else "UNKNOWN"
            if regime == "UNKNOWN":
                missing.append(
                    {
                        "day_start_ns": day,
                        "reason": label.get("reason", "UNKNOWN_LABEL") if label else "MISSING_DAY_START_LABEL",
                    }
                )
            rows.append((timestamp, value, regime))
            previous = timestamp
        if not rows:
            missing.append({"path": "daily_returns", "reason": "NO_DAILY_RETURNS"})
        complete = bool(rows) and not any(item.get("reason") == "MISSING_DAILY_OBSERVATIONS" for item in missing)
        grouped = {}
        for regime in REGIMES:
            values = [value for _, value, name in rows if name == regime]
            segments, active, prior = [], [], None
            for timestamp, value, name in rows:
                if active and (name != regime or timestamp != prior + DAY_NS):
                    segments.append(_compound(active))
                    active = []
                if name == regime:
                    active.append(value)
                prior = timestamp
            if active:
                segments.append(_compound(active))
            try:
                arithmetic = math.fsum(values) if values else None
            except OverflowError as exc:
                raise _Issue("NONFINITE_ARITHMETIC_CONTRIBUTION", "daily_returns") from exc
            if arithmetic is not None and not math.isfinite(arithmetic):
                raise _Issue("NONFINITE_ARITHMETIC_CONTRIBUTION", "daily_returns")
            grouped[regime] = {
                "status": "PASS" if values and complete else "NOT_EVALUABLE",
                "reason": "OBSERVED_DAILY_DIAGNOSTIC"
                if values and complete
                else "MISSING_DAILY_OBSERVATIONS"
                if values
                else "NO_REGIME_OBSERVATIONS",
                "days": len(values),
                "compound_return": _compound(values) if values else None,
                "arithmetic_contribution": arithmetic,
                "worst_contiguous_segment_return": min(segments) if segments else None,
                "contiguous_segments": len(segments),
                "unit": "return_fraction",
            }
        absent = [regime for regime in REGIMES[:3] if grouped[regime]["days"] == 0]
        return _result(
            "NOT_EVALUABLE" if missing else "PASS",
            by_regime=grouped,
            missing=missing,
            daily_sampling_complete=complete,
            coverage={
                "total_days": len(rows),
                "unknown_days": grouped["UNKNOWN"]["days"],
                "missing_regimes": absent,
                "all_regimes_observed": not absent and grouped["UNKNOWN"]["days"] == 0,
            },
            compound_return_is_tradable_nav=False,
            compound_return_basis="PRODUCT_OF_OBSERVED_REGIME_DAYS_DIAGNOSTIC_ONLY",
            arithmetic_contribution_basis="SUM_OF_DAILY_RETURN_FRACTIONS_NOT_COMPOUNDED_PNL",
            worst_segment_basis="COMPLETE_CONTIGUOUS_SAME_REGIME_RUN_NOT_DRAWDOWN",
            gate_status="NOT_EVALUABLE",
            gate_reason="DIAGNOSTIC_ATTRIBUTION_DOES_NOT_PROVE_R3_COVERAGE_OR_ROBUSTNESS",
        )
    except _Issue as issue:
        return _failed(issue, by_regime={})
