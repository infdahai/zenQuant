"""Bounded, model-only inventory Q-learning diagnostic; no live-path claims."""

from __future__ import annotations

import hashlib
from typing import Any

import numpy as np
import pandas as pd

from zenquant.research.frontier_lab.common import DEV_END, SEEDS, TRAIN_END, load_bars

ACTIONS = (0, -1, 1)  # Deterministic ties prefer flat, never a random dev action.
EPOCHS = 8
TRAIN_FEE = 0.0005


def inventory_step(inventory: int, action: int, gap: float, intrabar: float, fee: float) -> tuple[float, int, float]:
    """Old inventory holds until next open, then modeled full fill sets inventory.

    This is a unit-NAV exposure model, not exchange contract/margin accounting.
    gap = next_open/current_close-1; intrabar = next_close/next_open-1.
    """
    if inventory not in ACTIONS or action not in ACTIONS:
        raise ValueError("inventory/action must be -1, 0 or 1")
    if not np.isfinite([gap, intrabar, fee]).all() or fee < 0:
        raise ValueError("invalid transition input")
    turnover = float(abs(action - inventory))
    factor = (1 + inventory * gap) * (1 + action * intrabar - fee * turnover)
    return float(factor), action, turnover


def market_states(frame: pd.DataFrame) -> tuple[np.ndarray, np.ndarray]:
    """Fixed causal bins: 6-bar momentum +/-1%, 42-bar log-vol >1.2%."""
    close = frame["close"].astype(float)
    momentum = np.log(close / close.shift(6))
    volatility = np.log(close).diff().rolling(42).std()
    trend = np.where(momentum > 0.01, 2, np.where(momentum < -0.01, 0, 1))
    state = trend * 2 + (volatility > 0.012).to_numpy(dtype=int)
    valid = np.isfinite(momentum) & np.isfinite(volatility)
    return state, valid.to_numpy(copy=True)


def _choice(values: np.ndarray) -> int:
    return ACTIONS[int(np.argmax(values))]


def train_q(
    states: np.ndarray,
    gap: np.ndarray,
    intrabar: np.ndarray,
    reward_end_ns: np.ndarray,
    *,
    seed: int,
    epochs: int = EPOCHS,
) -> np.ndarray:
    """Chronological tabular Q-learning, fixed budget, no dev-state bootstrap."""
    if (
        len(states) < 2
        or len({len(states), len(gap), len(intrabar), len(reward_end_ns)}) != 1
        or np.any(reward_end_ns >= TRAIN_END.value)
        or not np.isfinite([gap, intrabar]).all()
    ):
        raise ValueError("training reward must be finite and end before 2024")
    q = np.zeros((6, 3, 3))
    rng = np.random.default_rng(seed)
    for epoch in range(epochs):
        inventory = 0
        epsilon = 0.2 - 0.15 * epoch / max(epochs - 1, 1)
        for i, state in enumerate(states):
            inv_index = ACTIONS.index(inventory)
            action_index = int(rng.integers(3)) if rng.random() < epsilon else int(np.argmax(q[state, inv_index]))
            action = ACTIONS[action_index]
            factor, next_inventory, _ = inventory_step(inventory, action, gap[i], intrabar[i], TRAIN_FEE)
            # Last training reward is terminal: never bootstrap into dev.
            future = 0.0 if i == len(states) - 1 else float(np.max(q[states[i + 1], action_index]))
            target = factor - 1 + 0.97 * future
            q[state, inv_index, action_index] += 0.08 * (target - q[state, inv_index, action_index])
            inventory = next_inventory
    return q


def myopic_table(states: np.ndarray, gap: np.ndarray, intrabar: np.ndarray) -> np.ndarray:
    """Train-only one-interval mean reward; same states/actions/fee as Q learner."""
    table = np.zeros((6, 3, 3))
    for state in range(6):
        mask = states == state
        if not np.any(mask):
            continue
        for j, inventory in enumerate(ACTIONS):
            for k, action in enumerate(ACTIONS):
                factors = (1 + inventory * gap[mask]) * (
                    1 + action * intrabar[mask] - TRAIN_FEE * abs(action - inventory)
                )
                table[state, j, k] = float(np.mean(factors - 1))
    return table


def evaluate(
    states: np.ndarray,
    gap: np.ndarray,
    intrabar: np.ndarray,
    fee: float,
    policy: str,
    table: np.ndarray | None = None,
) -> dict[str, Any]:
    inventory = 0
    equity = [1.0]
    rewards, exposures = [], []
    turnover = 0.0
    changes = 0
    bankrupt = False
    for state, g, r in zip(states, gap, intrabar, strict=True):
        if policy in {"q", "myopic"}:
            if table is None:
                raise ValueError("table policy requires frozen table")
            action = _choice(table[state, ACTIONS.index(inventory)])
        elif policy == "trend":
            action = int(state // 2) - 1
        elif policy == "long":
            action = 1
        elif policy == "flat":
            action = 0
        else:
            raise ValueError("unknown policy")
        factor, inventory, traded = inventory_step(inventory, action, float(g), float(r), fee)
        turnover += traded
        changes += int(traded > 0)
        exposures.append(abs(inventory))
        if factor <= 0:
            bankrupt = True
            factor = 0.0
        rewards.append(factor - 1)
        equity.append(equity[-1] * factor)
        if bankrupt:
            break
    # Window ends with marked inventory, no fictitious same-close liquidation.
    nav = np.asarray(equity)
    returns = np.asarray(rewards)
    std = float(returns.std())
    losses = returns[returns < 0]
    downside = float(np.sqrt(np.mean(np.minimum(returns, 0) ** 2)))
    return {
        "bars": len(returns),
        "total_return": float(nav[-1] - 1),
        "max_drawdown": float(np.max(1 - nav / np.maximum.accumulate(nav))),
        "sharpe": float(returns.mean() / std * np.sqrt(365 * 6)) if std else 0.0,
        "sortino": float(returns.mean() / downside * np.sqrt(365 * 6)) if downside else None,
        "profit_factor": float(returns[returns > 0].sum() / -losses.sum()) if len(losses) else None,
        "turnover_units": turnover,
        "action_changes": changes,
        "mean_abs_exposure": float(np.mean(exposures)),
        "ending_inventory": inventory,
        "bankrupt": bankrupt,
    }


def synthetic_failures() -> dict[str, float]:
    """Counterexamples are mechanisms, never historical trading alpha."""
    state = np.tile([0, 4], 20)
    # An oracle alternates the right sign on tiny moves; double-sided flips erase it.
    returns = np.where(state == 0, -0.0005, 0.0005)
    free = evaluate(state, np.zeros(len(state)), returns, 0, "trend")
    paid = evaluate(state, np.zeros(len(state)), returns, 0.001, "trend")
    # Same endpoints: close=100; next open=110; next close=105.
    # A fictitious close fill earns +5%; the actual next-open long loses 4.55%.
    correct, _, _ = inventory_step(0, 1, 0.1, 105 / 110 - 1, 0)
    return {
        "free_churn_return": free["total_return"],
        "paid_churn_return": paid["total_return"],
        "incorrect_same_close_return": 105 / 100 - 1,
        "next_open_return": correct - 1,
    }


def run(context: dict[str, Any]) -> dict[str, Any]:
    frame = load_bars(context)
    states, valid = market_states(frame)
    gap = (frame["open"].shift(-1) / frame["close"] - 1).to_numpy()
    intrabar = (frame["close"].shift(-1) / frame["open"].shift(-1) - 1).to_numpy()
    reward_end = frame["close_time_ns"].shift(-1).to_numpy()
    valid &= np.isfinite(gap) & np.isfinite(intrabar) & np.isfinite(reward_end)
    train = valid & (reward_end < TRAIN_END.value)
    dev = valid & (frame["close_time_ns"].to_numpy() >= TRAIN_END.value) & (reward_end < DEV_END.value)
    train_states, dev_states = states[train], states[dev]
    train_gap, train_intrabar = gap[train], intrabar[train]
    tables = [train_q(train_states, train_gap, train_intrabar, reward_end[train], seed=s) for s in SEEDS]
    myopic = myopic_table(train_states, train_gap, train_intrabar)
    cost_results = {}
    for bps in (0, 5, 15):
        fee = bps / 10000
        rl = [evaluate(dev_states, gap[dev], intrabar[dev], fee, "q", table) for table in tables]
        baselines = {
            name: evaluate(dev_states, gap[dev], intrabar[dev], fee, name, myopic if name == "myopic" else None)
            for name in ("flat", "trend", "long", "myopic")
        }
        cost_results[str(bps)] = {
            "q_by_seed": {str(seed): metrics for seed, metrics in zip(SEEDS, rl, strict=True)},
            "baselines": baselines,
            "q_positive_seeds": sum(metrics["total_return"] > 0 for metrics in rl),
            "q_beats_myopic_seeds": sum(
                metrics["total_return"] > baselines["myopic"]["total_return"] for metrics in rl
            ),
            "q_median_return": float(np.median([metrics["total_return"] for metrics in rl])),
        }
    expensive = cost_results["15"]
    return {
        "direction_id": "d05",
        "hypothesis": "Inventory-aware sequential Q learning may outperform a same-state myopic policy after next-open fills and turnover costs.",
        "evidence_type": "mixed",
        "method": {
            "model": "18-state tabular Q learner; actions 0/-1/+1, modeled full fills, max abs exposure 1",
            "train": "pre2024 only; reward ends at following confirmed close; terminal Q bootstrap zero",
            "dev": "2024-25 previously observed diagnostic; frozen policies, no dev tuning",
            "transition": "close i decision; old inventory receives gap to open i+1; new inventory receives open-to-close i+1 return; proportional turnover fee at open",
            "state_bins": "6-bar log momentum +/-0.01; 42-bar log-return vol threshold 0.012; discrete modeled exposure state, not filled contract quantities",
            "epochs": EPOCHS,
            "updates_per_seed": EPOCHS * int(train.sum()),
            "seeds": SEEDS,
            "alpha": 0.08,
            "gamma": 0.97,
            "epsilon": "linear 0.20 to 0.05 across epochs",
            "training_fee_bps": 5,
            "threads": 1,
            "window_end": "marked inventory, no fabricated same-close terminal exit",
        },
        "metrics": {
            "train_rows": int(train.sum()),
            "dev_rows": int(dev.sum()),
            "last_train_reward_ns": int(reward_end[train].max()),
            "cost_bps": cost_results,
            "policy_sha256": {
                str(seed): hashlib.sha256(table.tobytes()).hexdigest()
                for seed, table in zip(SEEDS, tables, strict=True)
            },
            "frozen_action_tables": {
                str(seed): np.asarray(ACTIONS)[np.argmax(table, axis=2)].tolist()
                for seed, table in zip(SEEDS, tables, strict=True)
            },
            "synthetic_failures": synthetic_failures(),
        },
        "verdict": f"Diagnostic only: at 15bp {expensive['q_positive_seeds']}/5 Q policies positive, {expensive['q_beats_myopic_seeds']}/5 beat the myopic return; no robustness or tradable RL edge established.",
        "limitations": [
            "Unit-NAV inventory environment assumes full next-open fills and rebalanced exposure; no contract, cash, partial-fill, borrow or funding ledger.",
            "Turnover is charged only when the discrete exposure action changes; maintaining fixed short leverage requires additional rebalancing that this proxy does not account for. Return levels are not executable PnL.",
            "Abs exposure <=1 is not a liquidation guarantee; maintenance margin and intrabar liquidation absent, negative NAV transitions merely stop simulation.",
            "No drawdown throttle, spread/order-book/market-impact model; 0/5/15bp are turnover-cost sensitivity, not execution acceptance.",
            "Five seeds test learner stochasticity, not path robustness; repeated chronological training can overfit and hidden market dynamics violate a stationary Markov assumption.",
            "2024-25 has previously been observed; neither virgin OOS nor statistical significance after 12-direction selection; historical available-at unverified.",
            "No production changes, old RL campaign resumption, replay/live parity, promotion or real-capital eligibility.",
        ],
        "falsification": "Reject practical sequential value if frozen Q cannot consistently exceed same-state myopic and trend baselines at 15bp with controlled drawdown on separately frozen path perturbations.",
        "next_experiment": "Freeze a selected rule before new data; test train-only walk-forward and five deterministic path perturbations in an audited actual-fill ledger with funding/margin/kill conditions before considering deployment.",
        "input_paths": [],
        "rankable": False,
        "live_eligible": False,
        "promotion_eligible": False,
    }
