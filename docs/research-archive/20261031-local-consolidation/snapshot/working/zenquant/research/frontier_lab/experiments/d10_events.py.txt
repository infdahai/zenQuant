@dataclass(frozen=True)
class Contract:
    name: str
    resolver: str
    window: str
    event: str
    payouts: tuple[float, ...]


def parse_contract(raw: dict[str, Any], states: tuple[str, ...]) -> Contract:
    """Parse an explicit reviewed table, not free text or LLM semantic extraction."""
    if not states or len(set(states)) != len(states) or set(raw["payoffs"]) != set(states):
        raise ValueError("every possible settlement state must have an explicit payout")
    payouts = tuple(float(raw["payoffs"][state]) for state in states)
    if not all(np.isfinite(p) and 0 <= p <= 1 for p in payouts):
        raise ValueError("payouts must be finite fractions of collateral")
    if not all(raw.get(key) for key in ("name", "resolver", "window", "event")):
        raise ValueError("missing resolution identity")
    return Contract(raw["name"], raw["resolver"], raw["window"], raw["event"], payouts)


def compatible(contracts: list[Contract]) -> bool:
    return bool(contracts) and len({(c.resolver, c.window, c.event) for c in contracts}) == 1


def payoff_matrix(contracts: list[Contract]) -> np.ndarray:
    if not contracts or len({len(c.payouts) for c in contracts}) != 1:
        raise ValueError("contracts need a common explicit settlement state space")
    return np.asarray([c.payouts for c in contracts], dtype=float).T


def partition_check(contracts: list[Contract]) -> dict[str, Any]:
    matrix = payoff_matrix(contracts)
    binary = bool(((matrix == 0) | (matrix == 1)).all())
    total = matrix.sum(axis=1)
    return {
        "compatible_resolution": compatible(contracts),
        "binary_settlement": binary,
        "mutually_exclusive": bool(binary and (total <= 1).all()),
        "exhaustive": bool(binary and (total >= 1).all()),
        "valid_partition": bool(compatible(contracts) and binary and (total == 1).all()),
        "unit_bundle_payoffs": total.tolist(),
        "unit_bundle_floor": float(total.min()),
    }


def implication_check(antecedent: Contract, consequent: Contract) -> dict[str, Any]:
    """A implies B: long B YES plus A NO has payoff B + (1-A) >= 1."""
    a, b = np.asarray(antecedent.payouts), np.asarray(consequent.payouts)
    if a.shape != b.shape:
        raise ValueError("implication requires aligned states")
    bundle = b + 1 - a
    return {
        "compatible_resolution": compatible([antecedent, consequent]),
        "payoff_dominance": bool((a <= b).all()),
        "valid_implication": bool(compatible([antecedent, consequent]) and (a <= b).all()),
        "unit_bundle_payoffs": bundle.tolist(),
        "unit_bundle_floor": float(bundle.min()),
    }


def sweep(
    levels: list[tuple[float, float]], quantity: float, *, fee_rate: float, slippage: float = 0, side: str = "buy"
) -> dict[str, Any]:
    """Finite-depth FAK simulation; fees in collateral, rounded per synthetic fill.

    Quantity is delivered token inventory. A quote fill is not an atomic multi-leg
    guarantee. Slippage is an adverse price increment, distinct from book walking.
    """
    if side not in {"buy", "sell"} or not np.isfinite([quantity, fee_rate, slippage]).all():
        raise ValueError("invalid execution parameters")
    if quantity < 0 or fee_rate < 0 or slippage < 0:
        raise ValueError("negative execution parameters")
    if any(not np.isfinite([p, q]).all() or not 0 < p < 1 or q < 0 for p, q in levels):
        raise ValueError("invalid book level")
    levels = sorted(levels, reverse=side == "sell")
    remaining, notional, fee = quantity, 0.0, 0.0
    fills = []
    for quoted_price, depth in levels:
        delivered = min(depth, remaining)
        if delivered <= 0:
            continue
        price = quoted_price + slippage * (1 if side == "buy" else -1)
        if not 0 < price < 1:
            raise ValueError("slipped price outside binary market interval")
        charged = round(delivered * fee_rate * price * (1 - price), 5)
        fills.append({"quantity": delivered, "price": price, "fee": charged})
        notional += delivered * price
        fee += charged
        remaining -= delivered
        if remaining <= 1e-12:
            break
    return {
        "quantity_requested": quantity,
        "quantity_filled": quantity - remaining,
        "unfilled": remaining,
        "notional": notional,
        "fees": fee,
        "cash_cost": notional + fee if side == "buy" else -(notional - fee),
        "fills": fills,
    }


def execute_bundle(
    matrix: np.ndarray,
    books: list[list[tuple[float, float]]],
    quantity: float,
    *,
    fee_rate: float,
    slippage: float = 0.002,
    fixed_cost: float = 0.1,
) -> dict[str, Any]:
    matrix = np.asarray(matrix, dtype=float)
    if matrix.ndim != 2 or not matrix.size or matrix.shape[1] != len(books):
        raise ValueError("book and payoff dimensions disagree")
    if not np.isfinite(matrix).all() or (matrix < 0).any() or (matrix > 1).any() or fixed_cost < 0:
        raise ValueError("invalid payoff or fixed cost")
    legs = [sweep(book, quantity, fee_rate=fee_rate, slippage=slippage) for book in books]
    filled = np.asarray([leg["quantity_filled"] for leg in legs])
    payout = matrix @ filled
    cost = sum(leg["cash_cost"] for leg in legs) + fixed_cost
    return {
        "target_quantity_per_leg": quantity,
        "filled_quantities": filled.tolist(),
        "all_legs_filled": bool(np.allclose(filled, quantity, rtol=0, atol=1e-12)),
        "payoffs_by_state": payout.tolist(),
        "guaranteed_payoff_floor": float(payout.min()),
        "maximum_payoff": float(payout.max()),
        "cash_cost": cost,
        "fees": sum(leg["fees"] for leg in legs),
        "guaranteed_net_profit": float(payout.min() - cost),
        "legs": legs,
    }


def contracts_for_partition(states: tuple[str, ...], *, void: bool = False) -> list[Contract]:
    contracts = []
    for name in ("A", "B", "C"):
        payouts = {state: float(state == name) for state in states}
        if void:
            payouts["unknown_A"] = 0.5 if name == "A" else 0.0
        contracts.append(
            parse_contract(
                {"name": name, "resolver": "official_final_result", "window": "final_event", "event": "winner", "payoffs": payouts},
                states,
            )
        )
    return contracts


def implication_contracts(mismatch: str | None = None) -> tuple[tuple[str, ...], Contract, Contract]:
    states = ("below100", "100to110", "above110")
    if mismatch:
        states += ("A_true_B_false",)
    a_pay = dict(zip(states, [0, 0, 1] + ([1] if mismatch else []), strict=True))
    b_pay = dict(zip(states, [0, 1, 1] + ([0] if mismatch else []), strict=True))
    base = {"resolver": "official_close", "window": "same_close", "event": "price_threshold"}
    a = parse_contract({**base, "name": "price_above110", "payoffs": a_pay}, states)
    if mismatch == "resolver":
        base["resolver"] = "different_exchange_close"
    if mismatch == "timewindow":
        base["window"] = "next_day_close"
    b = parse_contract({**base, "name": "price_above100", "payoffs": b_pay}, states)
    return states, a, b


def synthetic_scenarios(seed: int) -> dict[str, Any]:
    rng = np.random.default_rng(seed)
    jitter = rng.uniform(-0.003, 0.003, 3)
    matrix = payoff_matrix(contracts_for_partition(("A", "B", "C")))
    asks = np.asarray([0.25, 0.30, 0.35]) + jitter
    books = [[(float(p), 40.0), (float(p + 0.02), 200.0)] for p in asks]
    rates = {"fee_free_scenario": 0.0, "finance_politics_scenario": 0.04, "crypto_scenario": 0.07}
    true = {name: execute_bundle(matrix, books, 50, fee_rate=rate) for name, rate in rates.items()}
    _, a, b = implication_contracts()
    implication_matrix = np.column_stack([b.payouts, 1 - np.asarray(a.payouts)])
    implication = execute_bundle(implication_matrix, [[(0.52, 100)], [(0.37, 100)]], 50, fee_rate=0.04)

    # Midpoints are hypothetical marks, not necessarily UI-displayed prices.
    midpoint_asks = np.asarray([0.35, 0.36, 0.39]) + jitter
    midpoint_bids = np.asarray([0.23, 0.25, 0.27]) + jitter
    midpoint = execute_bundle(matrix, [[(float(p), 100)] for p in midpoint_asks], 50, fee_rate=0.04)
    midpoint["midpoint_unit_cost"] = float(((midpoint_asks + midpoint_bids) / 2).sum())
    midpoint["apparent_midpoint_profit"] = 50 * (1 - midpoint["midpoint_unit_cost"])

    false_positives: dict[str, Any] = {}
    for reason, states in [("nonexhaustive", ("A", "B", "C", "other")), ("void_settlement", ("A", "B", "C", "unknown_A"))]:
        contracts = contracts_for_partition(states, void=reason == "void_settlement")
        false_positives[reason] = {
            "states": list(states),
            "semantics": partition_check(contracts),
            "execution": execute_bundle(payoff_matrix(contracts), books, 50, fee_rate=0.04),
        }
    for reason in ("resolver", "timewindow"):
        states, a_bad, b_bad = implication_contracts(reason)
        actual = np.column_stack([b_bad.payouts, 1 - np.asarray(a_bad.payouts)])
        false_positives[reason] = {
            "states": list(states),
            "semantics": implication_check(a_bad, b_bad),
            "execution": execute_bundle(actual, [[(0.52, 100)], [(0.37, 100)]], 50, fee_rate=0.04),
        }

    capacity_books = [[(float(p), 5.0), (float(p + 0.15), 100.0)] for p in asks]
    capacity = [execute_bundle(matrix, capacity_books, q, fee_rate=0.04) for q in (1, 2, 5, 10, 25, 50, 100)]
    profitable = [row for row in capacity if row["all_legs_filled"] and row["guaranteed_net_profit"] > 0]

    # Each independently submitted order sees a later book; no cross-leg atomicity.
    later_books = [books[0], [(float(asks[1] + 0.04), 20.0)], []]
    partial = execute_bundle(matrix, later_books, 50, fee_rate=0.04)
    missing = 50 - np.asarray(partial["filled_quantities"])
    completion = [
        sweep([(float(min(p + 0.20, 0.95)), 100)], float(q), fee_rate=0.04, slippage=0.002)
        for p, q in zip(asks, missing, strict=True)
    ]
    completion_cost = partial["cash_cost"] + sum(leg["cash_cost"] for leg in completion)
    unwind = [
        sweep([(float(p - 0.10), 15)], q, fee_rate=0.04, slippage=0.002, side="sell")
        for p, q in zip(asks, partial["filled_quantities"], strict=True)
    ]
    residual = np.asarray(partial["filled_quantities"]) - np.asarray([leg["quantity_filled"] for leg in unwind])
    net_spent = partial["cash_cost"] + sum(leg["cash_cost"] for leg in unwind)
    return {
        "seed": seed,
        "true_partition": {"semantics": partition_check(contracts_for_partition(("A", "B", "C"))), "fee_sensitivity": true},
        "true_implication": {"semantics": implication_check(a, b), "execution": implication},
        "midpoint_only_false_positive": midpoint,
        "semantic_false_positives": false_positives,
        "small_capacity": {
            "grid": capacity,
            "max_profitable_quantity_on_frozen_grid": max((row["target_quantity_per_leg"] for row in profitable), default=0),
            "best_grid_guaranteed_profit": max((row["guaranteed_net_profit"] for row in profitable), default=0),
        },
        "asynchronous_fills": {
            "initial_simultaneous_quote": true["finance_politics_scenario"]["guaranteed_net_profit"],
            "later_confirmed_fills": partial,
            "complete_at_adverse_asks": {"additional_legs": completion, "cash_cost": completion_cost, "guaranteed_net_profit": 50 - completion_cost},
            "unwind_at_available_bids": {
                "sell_legs": unwind,
                "residual_quantities": residual.tolist(),
                "residual_payoffs_by_state": (matrix @ residual).tolist(),
                "cash_still_spent": net_spent,
                "guaranteed_net_profit": float((matrix @ residual).min() - net_spent),
                "risk_flat": bool((residual == 0).all()),
            },
        },
    }


def audit_inventory(repo_root: Path) -> dict[str, Any]:
    """Filename inventory only; do not claim universal absence from arbitrary files."""
    roots = ("data", "results/market_exploration/20261005/inputs")

        str(p.relative_to(repo_root)) for root in roots for p in (repo_root / root).rglob("*") if p.is_file()
    )
    keywords = ("polymarket", "kalshi", "prediction_market", "event_market", "conditional_token")
    candidates = [p for p in paths if any(keyword in p.lower() for keyword in keywords)]
    return {
        "roots": list(roots),
        "files_enumerated": len(paths),
        "sorted_path_inventory_sha256": hashlib.sha256("\n".join(paths).encode()).hexdigest(),
        "candidate_name_keywords": list(keywords),
        "candidate_paths": candidates,
        "verified_prediction_market_feeds": 0,
        "real_feed_used": False,
        "empirical_alpha_status": "data_blocked_no_verified_feed",

        "required_missing_contract": ["versioned resolution text", "aligned multi-token L2 books", "market-specific fee metadata", "received-at timestamps", "confirmed fills", "final payouts/redemption"],
    }
