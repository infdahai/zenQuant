"""Frozen BTC -> GLD ETF volatility transfer; historical diagnostic only."""

from __future__ import annotations

from pathlib import Path
from typing import Any

import numpy as np
import pandas as pd
import pyarrow.parquet as pq

from zenquant.research.frontier_lab.common import DEV_END, TRAIN_END, load_bars, prediction_metrics, ridge_predict

DAY_NS = 86400 * 10**9
FEATURES = ["abs_return", "mean_abs_5", "mean_abs_20", "vol_20", "vol_60", "abs_momentum_20"]
SOURCES = [
    "https://www.spdrgoldshares.com/usa/gld/",
    "https://www.cmegroup.com/market-regulation/files/gold-futures-and-options-fact-card.pdf",
    "https://datamine.new.cmegroup.com/catalog/dataset/78549b1567e23e7350a179adfdb4e283",
    "https://www.cmegroup.com/datamine/datamine-list-api.html",
]


def btc_daily(bars: pd.DataFrame) -> pd.DataFrame:
    """Retain complete UTC days only; all six constituent closes precede availability."""
    frame = bars.copy()
    frame["date_ns"] = (frame["open_time_ns"] // DAY_NS) * DAY_NS
    rows = []
    for date_ns, group in frame.groupby("date_ns", sort=True):
        expected = int(date_ns) + np.arange(6, dtype=np.int64) * (DAY_NS // 6)
        if len(group) != 6 or not np.array_equal(group["open_time_ns"].to_numpy(), expected):
            continue
        if int(group["close_time_ns"].max()) >= int(date_ns) + DAY_NS:
            raise ValueError("BTC constituent close escapes UTC day")
        rows.append({"date_ns": int(date_ns), "close": float(group.iloc[-1]["close"])})
    if not rows:
        raise ValueError("no complete BTC days")
    return pd.DataFrame(rows)


def validate_proxy(frame: pd.DataFrame) -> pd.DataFrame:
    """Date labels are never treated as actual close/publication timestamps."""
    if not {"ts_ns", "value"}.issubset(frame.columns) or frame.empty:
        raise ValueError("missing GLD date/price")
    dates = frame["ts_ns"]
    if dates.duplicated().any() or not dates.is_monotonic_increasing or (dates % DAY_NS != 0).any():
        raise ValueError("GLD requires unique ordered vendor midnight date labels")
    values = frame["value"].to_numpy(float)
    if not np.isfinite(values).all() or (values <= 0).any():
        raise ValueError("invalid GLD price")
    if int(dates.max()) >= DEV_END.value:
        raise ValueError("2026 GLD learner data prohibited")
    return frame.rename(columns={"ts_ns": "date_ns", "value": "close"}).copy()


def learning_table(prices: pd.DataFrame) -> tuple[pd.DataFrame, float]:
    """Dimensionless risk features with train-only scale, stale delayed targets.

    date+48h is a deliberately conservative surrogate, NOT verified availability.
    At observation i forecast return i+2 -> i+3 (one matched observed-date gap).
    Calendar gaps are sqrt-day normalized, not padded with artificial zero returns.
    """
    dates = prices["date_ns"].astype("int64")
    if len(dates) < 100 or dates.duplicated().any() or not dates.is_monotonic_increasing:
        raise ValueError("insufficient or unordered learning prices")
    gap_days = dates.diff() / DAY_NS
    returns = np.log(prices["close"].astype(float)).diff() / np.sqrt(gap_days)
    available = dates + 2 * DAY_NS
    train_returns = returns[(available < TRAIN_END.value) & (dates >= pd.Timestamp("2020-01-01", tz="UTC").value)]
    scale = float(np.sqrt(np.mean(np.square(train_returns.dropna()))))
    if not np.isfinite(scale) or scale <= 1e-12:
        raise ValueError("missing nonzero train-only return scale")
    unit_returns = returns / scale
    absolute = unit_returns.abs()
    table = pd.DataFrame({"date_ns": dates, "available_ns": available})
    table["abs_return"] = absolute
    table["mean_abs_5"] = absolute.rolling(5).mean()
    table["mean_abs_20"] = absolute.rolling(20).mean()
    table["vol_20"] = unit_returns.rolling(20).std()
    table["vol_60"] = unit_returns.rolling(60).std()
    table["abs_momentum_20"] = unit_returns.rolling(20).mean().abs()
    table["ewma"] = absolute.ewm(span=20, adjust=False, min_periods=20).mean()
    # Features from i, target increment ends at i+3 and starts at i+2.
    table["target"] = absolute.shift(-3)
    table["target_start_ns"] = available.shift(-2)
    table["target_end_ns"] = available.shift(-3)
    table["target_gap_days"] = gap_days.shift(-3)
    table = table.dropna().reset_index(drop=True)
    table["train"] = (table["available_ns"] < TRAIN_END.value) & (table["target_end_ns"] < TRAIN_END.value)
    table["dev"] = (table["available_ns"] >= TRAIN_END.value) & (table["target_end_ns"] < DEV_END.value)
    if not (table["target_start_ns"] > table["available_ns"]).all():
        raise ValueError("target overlaps feature information")
    return table, scale


def score(actual: np.ndarray, predicted: np.ndarray, scale: float) -> dict[str, Any]:
    result = prediction_metrics(actual * scale, predicted * scale)
    result["mae"] = float(np.mean(np.abs(actual - predicted)) * scale)
    result["units"] = "absolute log-return per sqrt(calendar day), not trade PnL"
    return result


def run(context: dict[str, Any]) -> dict[str, Any]:
    root = Path(context["repo_root"])
    proxy_path = root / "data/raw/macro/gold/data.parquet"
    # Predicate read prevents physical post2025 prices entering this experiment.
    raw = pd.read_parquet(proxy_path, filters=[("ts_ns", "<", DEV_END.value), ("ts_ns", ">=", pd.Timestamp("2020-01-01", tz="UTC").value)])
    gld = validate_proxy(raw)
    btc = btc_daily(load_bars(context))
    common_dates = np.intersect1d(gld["date_ns"], btc["date_ns"])
    gld = gld[gld["date_ns"].isin(common_dates)].reset_index(drop=True)
    btc = btc[btc["date_ns"].isin(common_dates)].reset_index(drop=True)
    btc_table, btc_scale = learning_table(btc)
    gld_table, gld_scale = learning_table(gld)
    btc_train = btc_table[btc_table["train"]]
    gld_train, dev = gld_table[gld_table["train"]], gld_table[gld_table["dev"]]
    if min(len(btc_train), len(gld_train), len(dev)) < 100:
        raise ValueError("insufficient matched train/dev observations")
    test_x, actual = dev[FEATURES].to_numpy(), dev["target"].to_numpy()
    predictions = {
        "btc_frozen_transfer": np.maximum(0.0, ridge_predict(btc_train[FEATURES].to_numpy(), btc_train["target"].to_numpy(), test_x, alpha=10.0)),
        "gld_own_train_ridge": np.maximum(0.0, ridge_predict(gld_train[FEATURES].to_numpy(), gld_train["target"].to_numpy(), test_x, alpha=10.0)),
        "gld_train_constant": np.full(len(dev), gld_train["target"].mean()),
        "gld_causal_ewma": dev["ewma"].to_numpy(),
    }
    scores = {name: score(actual, prediction, gld_scale) for name, prediction in predictions.items()}
    constant_rmse = scores["gld_train_constant"]["rmse"]
    for result in scores.values():
        result["rmse_skill_vs_constant"] = 1.0 - result["rmse"] / constant_rmse
    transfer = scores["btc_frozen_transfer"]
    wins = transfer["rmse"] < min(scores[name]["rmse"] for name in predictions if name != "btc_frozen_transfer")
    directory = root / "data/raw"
    # Audit only filenames; never use supplemental post2025 prices or option marks.
    contract_paths = sorted(str(path.relative_to(root)) for path in directory.rglob("*.parquet") if any(part.upper() in {"GC", "OG", "XAU", "XAUUSD"} for part in path.parts))
    timestamps = gld["date_ns"].to_numpy()
    gaps, counts = np.unique(np.diff(timestamps) // DAY_NS, return_counts=True)
    audit = {
        "instrument": "GLD ETF (SPDR Gold Shares), not GC futures, XAU spot, or OG options",
        "identity_evidence": "scripts/download_macro_data.py SERIES['gold']='GLD'",
        "physical_source_rows": pq.ParquetFile(proxy_path).metadata.num_rows,
        "learner_rows_pre2026": len(gld),
        "first_date": str(pd.Timestamp(int(timestamps[0]), tz="UTC")),
        "last_date": str(pd.Timestamp(int(timestamps[-1]), tz="UTC")),
        "vendor_timestamp_midnight_utc": True,
        "publication_timestamp_verified": False,
        "surrogate_availability_lag_hours": 48,
        "observed_calendar_gap_counts": {str(int(gap)): int(count) for gap, count in zip(gaps, counts, strict=True)},
        "weekend_rows": int((pd.to_datetime(timestamps, utc=True).dayofweek >= 5).sum()),
        "local_gc_og_xau_parquet_paths": contract_paths,
        "local_gc_og_xau_coverage_verified": False,
        "cme_acquisition": "Licensed/entitled DataMine historical GC/OG plus contract/roll, session calendars, quote/settlement timestamps, delivery/margin specifications required; no purchase or login performed.",
        "market_hours_boundary": "GLD exchange sessions differ from near-24-hour CME Globex gold; neither represents all global gold trading.",
    }
    return {
        "direction_id": "d11",
        "hypothesis": "A fixed BTC-trained dimensionless volatility model transfers incrementally to GLD without fitting GLD model weights.",
        "evidence_type": "mixed",
        "method": {
            "target": "absolute future observed-session log return / sqrt(actual calendar gap), two-observation embargo",
            "feature_availability": "vendor date +48h diagnostic surrogate; unverified publication time",
            "btc_aggregation": "complete six-bar UTC days, sampled on same GLD dates; no weekend zero padding",
            "features": FEATURES,
            "learner": "ridge alpha=10, nonnegative prediction clipping; BTC feature standardizer and coefficients fit on BTC train only",
            "asset_units": "each asset RMS return unit estimated on its own pre2024 history; no GLD-dev calibration",
            "split": "2020-2023 train with label-end purge; previously observed 2024-2025 dev; no 2026 learning/evaluation",
        },
        "metrics": {
            "data_audit": audit,
            "train_samples_btc": len(btc_train),
            "train_samples_gld": len(gld_train),
            "dev_samples_gld": len(dev),
            "btc_train_scale": btc_scale,
            "gld_train_scale": gld_scale,
            "gld_dev_prediction": scores,
            "transfer_beats_all_three_baselines": wins,
            "yearly_rmse": {str(year): {name: score(actual[mask], prediction[mask], gld_scale)["rmse"] for name, prediction in predictions.items()} for year in (2024, 2025) if (mask := (pd.to_datetime(dev["available_ns"], utc=True).dt.year.to_numpy() == year)).any()},
            "latest_train_label_end_ns": int(btc_train["target_end_ns"].max()),
            "latest_dev_label_end_ns": int(dev["target_end_ns"].max()),
        },
        "verdict": "Diagnostic transfer advantage over all baselines observed; requires fresh frozen validation." if wins else "BTC transfer does not beat all simple/local GLD baselines; no incremental transfer established.",
        "limitations": [
            "GLD ETF is a proxy with equity-session gaps and trust expenses; does not establish gold futures/spot behavior.",
            "Vendor timestamp is a date label and historical publication/adjustment provenance is unavailable; 48h lag remains a diagnostic convention, not verified live availability.",
            "BTC UTC daily close and GLD exchange close are asynchronous; matched observed-date gaps normalize duration but cannot remove differing session microstructure.",
            "Forecast errors are not tradable returns; fees, spread, financing, futures rolls/delivery/margin and options surfaces are absent.",
            "2024-25 was previously observed and 12 directions were considered; this is neither untouched OOS nor statistical alpha/promotion evidence.",
            "No verified local GC/OG contract dataset; filename audit cannot prove absence of every possible vendor alias.",
        ],
        "falsification": "Reject useful BTC transfer if frozen model cannot outperform GLD own-train ridge, constant, and EWMA on a separately frozen prospective period with verified session availability.",
        "next_experiment": "Acquire licensed timestamped GC contracts and OG quotes with calendars and roll rules, freeze a new future evaluation; retain GLD-local volatility baselines before any return strategy.",
        "input_paths": ["data/raw/macro/gold/data.parquet", "scripts/download_macro_data.py"],
        "internet_sources": SOURCES,
        "rankable": False,
        "promotion_eligible": False,
        "live_eligible": False,
    }
