"""Pure research wallet accounting, without execution/risk or venue margin proof.

Spot trades exchange principal. Linear perpetual trades exchange only fees,
realized PnL and funding, never notional principal. Actual fills may reveal a
negative wallet; it remains visible as capital insufficiency, not usable credit.
All arithmetic uses a fixed Decimal context independent of caller defaults.
"""

from __future__ import annotations

from dataclasses import dataclass, replace
from decimal import ROUND_HALF_EVEN, Context, Decimal, localcontext

ZERO = Decimal("0")
ONE = Decimal("1")
PRECISION = 50


def _decimal(value: Decimal, name: str, *, positive: bool = False, nonnegative: bool = False) -> None:
    if not isinstance(value, Decimal) or not value.is_finite():
        raise ValueError(f"{name} requires finite Decimal")
    if (positive and value <= ZERO) or (nonnegative and value < ZERO):
        raise ValueError(f"invalid {name}")


def _time(value: int, name: str) -> None:
    if type(value) is not int or value < 0:
        raise ValueError(f"invalid {name}")


@dataclass(frozen=True)
class Instrument:
    venue: str
    product: str
    base: str
    quote: str
    settlement: str
    expiry: str | None = None
    multiplier: Decimal = ONE
    quantity_unit: str = "base"

    def __post_init__(self) -> None:
        if (
            not isinstance(self.venue, str)
            or not self.venue
            or self.base not in {"BTC", "ETH"}
            or self.product not in {"spot", "linear_perp"}
            or self.quote != "USDT"
            or self.settlement != "USDT"
            or self.expiry is not None
        ):
            raise ValueError("unsupported instrument: same-venue BTC/ETH spot or linear USDT perpetual only")
        _decimal(self.multiplier, "multiplier", positive=True)
        if self.quantity_unit not in {"base", "contracts"} or (self.quantity_unit == "base" and self.multiplier != ONE):
            raise ValueError("unsupported quantity unit/multiplier combination")
        if self.product == "spot" and (self.multiplier != ONE or self.quantity_unit != "base"):
            raise ValueError("unsupported spot multiplier")


@dataclass(frozen=True)
class EventKey:
    event_id: str
    time_ns: int
    sequence: int
    source: str
    available_at_ns: int


@dataclass(frozen=True)
class Fill:
    key: EventKey
    instrument: Instrument
    signed_quantity: Decimal
    price: Decimal
    fee: Decimal
    fee_currency: str = "USDT"


@dataclass(frozen=True)
class Funding:
    key: EventKey
    instrument: Instrument
    rate: Decimal
    mark_price: Decimal
    mark_time_ns: int
    currency: str = "USDT"


@dataclass(frozen=True)
class Transfer:
    key: EventKey
    from_wallet: str
    to_wallet: str
    amount: Decimal


@dataclass(frozen=True)
class Margin:
    key: EventKey
    locked_amount: Decimal


@dataclass(frozen=True)
class Mark:
    key: EventKey
    instrument: Instrument
    price: Decimal
    source_time_ns: int


Event = Fill | Funding | Transfer | Margin | Mark


@dataclass(frozen=True)
class Position:
    instrument: Instrument
    quantity: Decimal = ZERO
    average_entry: Decimal = ZERO


@dataclass(frozen=True)
class MarkObservation:
    instrument: Instrument
    price: Decimal
    source_time_ns: int
    available_at_ns: int
    source: str


@dataclass(frozen=True)
class AccountState:
    instruments: tuple[Instrument, ...]
    initial_capital: Decimal
    spot_cash: Decimal
    derivative_cash: Decimal
    positions: tuple[Position, ...]
    marks: tuple[MarkObservation, ...] = ()
    margin_locked: Decimal = ZERO
    fees_total: Decimal = ZERO
    funding_total: Decimal = ZERO
    realized_perp_pnl: Decimal = ZERO
    receipts: tuple[Event, ...] = ()
    last_order: tuple[int, int] = (-1, -1)

    def __post_init__(self) -> None:
        if any(type(value) is not tuple for value in (self.instruments, self.positions, self.marks, self.receipts)):
            raise ValueError("state collections must be immutable tuples")
        if (
            not self.instruments
            or len(set(self.instruments)) != len(self.instruments)
            or len({i.venue for i in self.instruments}) != 1
        ):
            raise ValueError("unique same-venue instruments required")
        if tuple(p.instrument for p in self.positions) != self.instruments:
            raise ValueError("position instrument identities must match registry")
        if len({m.instrument for m in self.marks}) != len(self.marks) or any(
            m.instrument not in self.instruments for m in self.marks
        ):
            raise ValueError("mark identities must be registered and unique")
        for name in ("initial_capital", "margin_locked", "fees_total"):
            _decimal(getattr(self, name), name, nonnegative=True)
        for name in ("spot_cash", "derivative_cash", "funding_total", "realized_perp_pnl"):
            _decimal(getattr(self, name), name)
        for position in self.positions:
            _decimal(position.quantity, "position quantity")
            _decimal(position.average_entry, "average entry", nonnegative=True)
            if (position.quantity == ZERO) != (position.average_entry == ZERO):
                raise ValueError("position average-entry/quantity inconsistency")
            if position.instrument.product == "spot" and position.quantity < ZERO:
                raise ValueError("unsupported spot borrow")

    def position(self, instrument: Instrument) -> Position:
        for position in self.positions:
            if position.instrument == instrument:
                return position
        raise ValueError("unregistered instrument")

    @property
    def capital_insufficient(self) -> bool:
        return self.spot_cash < ZERO or self.derivative_cash < self.margin_locked

    def metadata(self) -> dict:
        return {
            "schema_version": "research-multi-leg-accounting.v1",
            "E3": "FAIL",
            "E4": "NOT_EVALUABLE",
            "rankable": False,
            "live_eligible": False,
            "eligible_for_paper": False,
            "optimizer_feedback_eligible": False,
            "scope": "hand-event research accounting; no gateway/risk approval/restart/venue parity proof",
            "margin_scope": "explicit free-to-locked classification only; not a venue requirement/liquidation model",
            "capital_status": "CAPITAL_INSUFFICIENT" if self.capital_insufficient else "NONNEGATIVE_RECORDED_WALLETS",
            "decimal_precision": PRECISION,
        }


@dataclass(frozen=True)
class Valuation:
    asof_ns: int
    nav: Decimal
    spot_inventory_value: Decimal
    perp_unrealized_pnl: Decimal
    derivative_free: Decimal
    capital_insufficient: bool


def initial_state(
    instruments: tuple[Instrument, ...], *, spot_cash: Decimal = ZERO, derivative_cash: Decimal = ZERO
) -> AccountState:
    if type(instruments) is not tuple or not instruments or len(set(instruments)) != len(instruments):
        raise ValueError("unique frozen instrument tuple required")
    if len({instrument.venue for instrument in instruments}) != 1:
        raise ValueError("unsupported cross-venue wallets/transfers")
    _decimal(spot_cash, "initial spot cash", nonnegative=True)
    _decimal(derivative_cash, "initial derivative cash", nonnegative=True)
    with localcontext(Context(prec=PRECISION, rounding=ROUND_HALF_EVEN)):
        return AccountState(
            instruments,
            spot_cash + derivative_cash,
            spot_cash,
            derivative_cash,
            tuple(Position(instrument) for instrument in instruments),
        )


def apply_event(state: AccountState, event: Event) -> AccountState:
    """Apply one actual event in strict (time, sequence) order, purely.

    Callers must sequence settlement-eligible funding before same-time fills.
    This ledger does not infer settlement eligibility, approve fills or create
    hedge orders. A late new event is rejected; exact prior receipts deduplicate.
    """
    if not isinstance(event, (Fill, Funding, Transfer, Margin, Mark)):
        raise ValueError("unsupported event")
    for previous in state.receipts:
        if previous.key.event_id == event.key.event_id:
            if previous == event:
                return state
            raise ValueError("conflicting duplicate event_id")
    key = event.key
    if not isinstance(key, EventKey) or not isinstance(key.event_id, str) or not key.event_id:
        raise ValueError("explicit event_id required")
    if not isinstance(key.source, str) or not key.source:
        raise ValueError("explicit event source required")
    _time(key.time_ns, "event time")
    _time(key.sequence, "sequence")
    _time(key.available_at_ns, "available-at")
    if key.available_at_ns > key.time_ns:
        raise ValueError("event not available at application time")
    order = (key.time_ns, key.sequence)
    if order <= state.last_order:
        raise ValueError("event order must strictly increase")
    with localcontext(Context(prec=PRECISION, rounding=ROUND_HALF_EVEN)):
        updated = _apply(state, event)
        return replace(updated, receipts=state.receipts + (event,), last_order=order)


def _apply(state: AccountState, event: Event) -> AccountState:
    if isinstance(event, Transfer):
        _decimal(event.amount, "transfer", positive=True)
        if {event.from_wallet, event.to_wallet} != {"spot", "derivative"}:
            raise ValueError("unsupported transfer wallets")
        available = state.spot_cash if event.from_wallet == "spot" else state.derivative_cash - state.margin_locked
        if event.amount > available:
            raise ValueError("insufficient capital for internal transfer; no credit")
        amount = event.amount if event.from_wallet == "spot" else -event.amount
        return replace(state, spot_cash=state.spot_cash - amount, derivative_cash=state.derivative_cash + amount)
    if isinstance(event, Margin):
        _decimal(event.locked_amount, "margin", nonnegative=True)
        if event.locked_amount > state.margin_locked and event.locked_amount > state.derivative_cash:
            raise ValueError("insufficient capital for margin lock")
        return replace(state, margin_locked=event.locked_amount)
    position = state.position(event.instrument)
    if isinstance(event, Mark):
        _decimal(event.price, "mark", positive=True)
        _time(event.source_time_ns, "mark source time")
        if event.source_time_ns > event.key.time_ns or event.source_time_ns > event.key.available_at_ns:
            raise ValueError("future mark source time")
        old = next((mark for mark in state.marks if mark.instrument == event.instrument), None)
        if old is not None and event.source_time_ns < old.source_time_ns:
            raise ValueError("mark source time regression")
        observation = MarkObservation(
            event.instrument, event.price, event.source_time_ns, event.key.available_at_ns, event.key.source
        )
        marks = tuple(mark for mark in state.marks if mark.instrument != event.instrument) + (observation,)
        return replace(state, marks=marks)
    if isinstance(event, Funding):
        if event.instrument.product != "linear_perp" or event.currency != "USDT":
            raise ValueError("unsupported funding product/currency")
        _decimal(event.rate, "funding rate")
        _decimal(event.mark_price, "funding mark", positive=True)
        _time(event.mark_time_ns, "funding mark time")
        if event.mark_time_ns > event.key.time_ns or event.mark_time_ns > event.key.available_at_ns:
            raise ValueError("future funding mark")
        pnl = -position.quantity * event.instrument.multiplier * event.mark_price * event.rate
        return replace(state, derivative_cash=state.derivative_cash + pnl, funding_total=state.funding_total + pnl)
    if event.fee_currency != "USDT":
        raise ValueError("unsupported fee currency; base/borrow fees require distinct accounting")
    _decimal(event.signed_quantity, "fill quantity")
    if event.signed_quantity == ZERO:
        raise ValueError("zero fill quantity")
    _decimal(event.price, "fill price", positive=True)
    _decimal(event.fee, "fee", nonnegative=True)
    old_qty, delta = position.quantity, event.signed_quantity
    qty = old_qty + delta
    if event.instrument.product == "spot" and qty < ZERO:
        raise ValueError("unsupported spot borrow/short inventory")
    realized = ZERO
    average = position.average_entry
    if old_qty == ZERO or old_qty * delta > ZERO:
        average = (abs(old_qty) * average + abs(delta) * event.price) / abs(qty)
    else:
        closed = min(abs(old_qty), abs(delta))
        realized = closed * (event.price - average) * (ONE if old_qty > ZERO else -ONE) * event.instrument.multiplier
        if qty == ZERO:
            average = ZERO
        elif old_qty * qty < ZERO:
            average = event.price
    positions = tuple(
        replace(p, quantity=qty, average_entry=average) if p.instrument == event.instrument else p
        for p in state.positions
    )
    if event.instrument.product == "spot":
        return replace(
            state,
            positions=positions,
            spot_cash=state.spot_cash - delta * event.price - event.fee,
            fees_total=state.fees_total + event.fee,
        )
    return replace(
        state,
        positions=positions,
        derivative_cash=state.derivative_cash + realized - event.fee,
        fees_total=state.fees_total + event.fee,
        realized_perp_pnl=state.realized_perp_pnl + realized,
    )


def valuation(state: AccountState, *, asof_ns: int, max_mark_age_ns: int) -> Valuation:
    """NAV from known marks; stale/missing prices never become zero PnL."""
    _time(asof_ns, "valuation asof")
    _time(max_mark_age_ns, "maximum mark age")
    if asof_ns < state.last_order[0]:
        raise ValueError("valuation asof precedes applied events")
    with localcontext(Context(prec=PRECISION, rounding=ROUND_HALF_EVEN)):
        spot_value = perp_pnl = ZERO
        for position in state.positions:
            if position.quantity == ZERO:
                continue
            mark = next((row for row in state.marks if row.instrument == position.instrument), None)
            if mark is None:
                raise ValueError("missing mark for held instrument")
            if mark.source_time_ns > asof_ns or mark.available_at_ns > asof_ns:
                raise ValueError("future mark at valuation asof")
            if asof_ns - mark.source_time_ns > max_mark_age_ns:
                raise ValueError("stale mark for held instrument")
            if position.instrument.product == "spot":
                spot_value += position.quantity * mark.price
            else:
                perp_pnl += position.quantity * position.instrument.multiplier * (mark.price - position.average_entry)
        return Valuation(
            asof_ns,
            state.spot_cash + spot_value + state.derivative_cash + perp_pnl,
            spot_value,
            perp_pnl,
            state.derivative_cash - state.margin_locked,
            state.capital_insufficient,
        )
