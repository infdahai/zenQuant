"""Create-only evaluation artifacts for imported historical evidence."""

from __future__ import annotations

import hashlib
import json
import os
from collections.abc import Sequence
from pathlib import Path
from typing import Any

from zenquant.research.evaluation import evaluate_evidence
from zenquant.research.evidence_adapter import import_repaired_run
from zenquant.research.run_evidence import TrialJournal, identity_hash


def _write(path: Path, value: Any) -> str:
    payload = (json.dumps(value, sort_keys=True, indent=2, allow_nan=False) + "\n").encode()
    with path.open("xb") as handle:
        handle.write(payload)
        handle.flush()
        os.fsync(handle.fileno())
    return hashlib.sha256(payload).hexdigest()


def _cell(value: Any) -> str:
    return str(value).replace("|", "\\|").replace("\n", " ")


def _percent(metric: dict[str, Any]) -> str:
    value = metric["value"]
    return "N/E" if value is None else f"{100 * value:.2f}%"


def _markdown(cases: list[dict[str, Any]], reports: list[dict[str, Any]]) -> str:
    lines = [
        "# Recorded historical evidence evaluation",
        "",
        "This re-evaluates existing fixed-run evidence; it is not a new backtest or fresh OOS test. "
        "All cases remain diagnostic. Completion means every requested input has a recorded outcome, not admission.",
        "",
        "| Case | Status | Net return | Sampled max drawdown | Fees / initial equity | Rankable |",
        "|---|---|---:|---:|---:|---|",
    ]
    for case, report in zip(cases, reports, strict=True):
        if case["status"] == "INVALID":
            lines.append(f"| {_cell(case['source'])} | INVALID | — | — | — | False |")
            continue
        metrics = report["metrics"]
        lines.append(
            f"| {_cell(case['case'])} | {_cell(case['status'])} | {_percent(metrics['net_return'])} | "
            f"{_percent(metrics['max_drawdown'])} | {_percent(metrics['fees_initial_equity'])} | False |"
        )
    for case, report in zip(cases, reports, strict=True):
        lines.extend(["", f"## {_cell(case['case'])}", ""])
        if case["status"] == "INVALID":
            lines.append(f"Invalid input: {_cell(case['error'])}")
            continue
        lines.extend(["| Dimension | Status | Reason |", "|---|---|---|"])
        for name, gate in report["gates"].items():
            lines.append(f"| {name} | {gate['status']} | {_cell(gate['reason'])} |")
        lines.extend(
            ["", f"Full metrics, sample sizes and provenance: [{case['report_path']}]({case['report_path']})."]
        )
    return "\n".join(lines) + "\n"


def write_recorded_reports(sources: Sequence[Path], output: Path) -> dict[str, Any]:
    """Evaluate each explicit source, retaining invalid cases and refusing overwrite.

    A missing completion marker means an interrupted batch. This exporter has no
    resume or admission operation. Each successful import binds the exact parsed
    source bytes; every outcome is chained to this batch's evaluator code identity.
    """
    if not sources:
        raise ValueError("at least one explicit source is required")
    sources = [Path(source).resolve() for source in sources]
    output = Path(output)
    output.mkdir(parents=True, exist_ok=False)
    module_root = Path(__file__).parent
    code_names = ("reporting.py", "evaluation.py", "evidence_adapter.py", "run_evidence.py")
    identity = {
        "schema_version": "recorded-report-batch.v1",
        "evidence_role": "observed_historical_diagnostic",
        "inputs": [str(source) for source in sources],
        "code_sha256": {name: hashlib.sha256((module_root / name).read_bytes()).hexdigest() for name in code_names},
    }
    journal = TrialJournal(output / "journal", identity)
    cases, reports = [], []
    for index, source in enumerate(sources):
        attempt = f"case-{index:04d}"
        case = {"attempt_id": attempt, "case": source.name, "source": str(source), "report_path": f"{attempt}.json"}
        try:
            evidence = import_repaired_run(source)
            report = evaluate_evidence(evidence)
            report.update(
                {
                    key: evidence[key]
                    for key in ("evidence_role", "valuation_basis", "sampling", "source_checks", "source_completed")
                }
            )
            case.update(status="EVALUATED_DIAGNOSTIC", evidence_identity_hash=identity_hash(evidence["identity"]))
        except (ValueError, TypeError, KeyError, OSError) as exc:
            case.update(status="INVALID", error=f"{type(exc).__name__}: {exc}")
            report = {"status": "INVALID", "error": case["error"], "rankable": False, "live_eligible": False}
        case["report_sha256"] = _write(output / case["report_path"], report)
        journal.append(case)
        cases.append(case)
        reports.append(report)
    summary = {
        "schema_version": "recorded-report-batch.v1",
        "identity_hash": identity_hash(identity),
        "complete": True,
        "live_eligible": False,
        "counts": {
            "evaluated": sum(case["status"] == "EVALUATED_DIAGNOSTIC" for case in cases),
            "invalid": sum(case["status"] == "INVALID" for case in cases),
            "rankable": sum(report["rankable"] for report in reports),
        },
        "cases": cases,
    }
    with (output / "REPORT.md").open("x") as handle:
        handle.write(_markdown(cases, reports))
        handle.flush()
        os.fsync(handle.fileno())
    _write(output / "completed.json", summary)
    return summary
