"""Frozen component-removal comparisons, without weakening system invariants."""

from __future__ import annotations

import math
from collections.abc import Mapping, Sequence
from typing import Any

from zenquant.research.run_evidence import identity_hash
from zenquant.research.suite_evaluation import COMPARISON_FIELDS, paired_daily_excess

PROTECTED = frozenset({"risk_overlay", "accounting", "pit", "fees", "audit"})
PROTECTED_PARAMETERS = frozenset(
    {
        "max_leverage",
        "target_annual_vol",
        "risk_budget",
        "kill_switch_drawdown",
        "daily_loss_limit",
        "starting_equity",
        "taker_fee_bps",
        "slippage_bps",
        "execution_timeframe",
    }
)
MIN_BOOTSTRAP_REPETITIONS = 2000
# These semantics belong to the reviewed strategy implementation, not to the
# caller's declaration that an arbitrary parameter change is a "removal".
_REMOVAL_RULES = {
    ("trend", "entry_confirmation"): ("confirmation_bars", 1),
    ("breakout", "entry_confirmation"): ("confirmation_bars", 1),
    ("breakout", "entry_gate"): ("entry_gate_enabled", False),
    ("breakout", "exit_gate"): ("exit_gate_enabled", False),
}
_COMPARISON_FIELDS = (*COMPARISON_FIELDS, "initial_equity", "boundary_policy", "warmup_contract")


class _Issue(ValueError):
    def __init__(self, status: str, reason: str):
        super().__init__(reason)
        self.status, self.reason = status, reason


def _object(value: Any) -> Mapping:
    if not isinstance(value, Mapping):
        raise _Issue("INVALID", "EXPECTED_OBJECT")
    return value


def _number(value: Any, *, nonnegative: bool = True) -> float:
    if type(value) not in (int, float) or not math.isfinite(value) or (nonnegative and value < 0):
        raise _Issue("INVALID", "INVALID_NUMERIC_CONTRACT")
    return float(value)


def _candidate(report: Mapping) -> Mapping:
    identity = _object(report.get("identity"))
    candidate = _object(identity.get("candidate"))
    _object(candidate.get("params"))
    if (
        not isinstance(candidate.get("family"), str)
        or not candidate["family"]
        or not isinstance(candidate.get("code_hash"), str)
        or not candidate["code_hash"]
        or not _object(candidate.get("risk_budget"))
        or not identity.get("policy_hash")
    ):
        raise _Issue("NOT_EVALUABLE", "INCOMPLETE_CANDIDATE_PROVENANCE")
    if identity.get("code_hash", candidate["code_hash"]) != candidate["code_hash"]:
        raise _Issue("INVALID", "CODE_IDENTITY_MISMATCH")
    if identity.get("candidate_id") != identity_hash(candidate):
        raise _Issue("INVALID", "CANDIDATE_HASH_MISMATCH")
    if "run_id" in identity and identity["run_id"] != identity_hash(
        {k: v for k, v in identity.items() if k != "run_id"}
    ):
        raise _Issue("INVALID", "RUN_HASH_MISMATCH")
    return candidate


def _safety(report: Mapping, source: str) -> list[dict[str, str]]:
    issues = []

    def add(status: str, reason: str) -> None:
        normalized = status if status in {"INVALID", "STALE", "FAIL"} else "NOT_EVALUABLE"
        issues.append({"source": source, "status": normalized, "reason": reason})

    if report.get("source_completed") is not True:
        add("NOT_EVALUABLE", "INCOMPLETE_SOURCE_RUN")
    if _object(report.get("ruin", {})).get("occurred") is not False:
        add("FAIL", "RUIN_OR_MISSING_RUIN_EVIDENCE")
    gates = _object(report.get("gates", {}))
    for name in (f"E{i}" for i in range(5)):
        status = _object(gates.get(name, {})).get("status", "NOT_EVALUABLE")
        if status != "PASS":
            add(status, f"{name}_NOT_PASS")
    for item in report.get("blockers", []):
        status = _object(item).get("status", "NOT_EVALUABLE")
        if status != "PASS":
            add(status, "UNRESOLVED_REPORT_BLOCKER")
    if report.get("diagnostic_only") is True:
        add("NOT_EVALUABLE", "DIAGNOSTIC_SOURCE_SCOPE")
    return issues


def _metric(report: Mapping, name: str) -> float:
    metric = _object(report.get("metrics", {})).get(name)
    if metric is None or _object(metric).get("status") != "PASS":
        raise _Issue("NOT_EVALUABLE", f"MISSING_OR_UNEVALUATED_{name}")
    if metric.get("unit") != "fraction":
        raise _Issue("INVALID", "INCOMPARABLE_METRIC_UNITS")
    return _number(metric.get("value"), nonnegative=name == "max_drawdown")


def _comparison(report: Mapping, candidate: Mapping) -> Mapping:
    contract = report.get("comparison_contract")
    if contract is None:
        raise _Issue("NOT_EVALUABLE", "MISSING_COMPARISON_CONTRACT")
    contract = _object(contract)
    for name in _COMPARISON_FIELDS:
        if contract.get(name) in (None, "", {}, []):
            raise _Issue("NOT_EVALUABLE", f"MISSING_COMPARISON_{name}")
    for name in ("symbol", "window", "data_hash", "accounting_hash", "execution", "boundary_policy"):
        if not isinstance(contract[name], str) or not contract[name].strip():
            raise _Issue("INVALID", f"INVALID_COMPARISON_{name}")
    if _number(contract["initial_equity"]) <= 0:
        raise _Issue("INVALID", "NONPOSITIVE_INITIAL_EQUITY")
    _object(contract["cost"])
    if contract["accounting_hash"] != candidate["code_hash"] or contract["risk_budget"] != candidate["risk_budget"]:
        raise _Issue("INVALID", "COMPARISON_CANDIDATE_IDENTITY_MISMATCH")
    warmup = _object(contract["warmup_contract"])
    for name in ("strategy_bars", "regime_bars"):
        if name not in warmup:
            raise _Issue("NOT_EVALUABLE", f"MISSING_WARMUP_{name}")
        if type(warmup[name]) is not int or warmup[name] < 0:
            raise _Issue("INVALID", f"INVALID_WARMUP_{name}")
    identity_hash(contract)  # All additional conditions must be finite JSON too.
    return contract


def evaluate_ablation_suite(full_report: Mapping, variants: Sequence[Mapping], protocol: Mapping) -> dict:
    """Require every frozen removal and report scoped keep/remove/inconclusive evidence.

    Candidate hashes bind actual parameters. Registered removals are repeated
    entry confirmation (integer above one becomes one), and one explicit
    breakout entry/exit gate (True becomes False, other gate remains True).
    Other component/family semantics remain unevaluable. Family, code, policy,
    risk and complete comparison conditions must remain equal. At least 2000
    bootstrap repetitions and positive intervals at all three frozen block
    lengths are required.
    This never proves R7 in full: robustness, operational complexity and broader
    component interactions remain separate evidence.
    """
    result = {
        "schema_version": "ablation-evaluation.v2",
        "status": "NOT_EVALUABLE",
        "decision": "INCONCLUSIVE",
        "reason": "INCOMPLETE_ABLATION_EVIDENCE",
        "components": {},
        "missing": [],
        "blockers": [],
        "diagnostic_only": True,
        "live_eligible": False,
        "R7": {"status": "NOT_EVALUABLE", "reason": "BROADER_ROBUSTNESS_AND_COMPLEXITY_EVIDENCE_REQUIRED"},
    }
    try:
        full, policy = _object(full_report), _object(protocol)
        result["protocol_hash"] = identity_hash(policy)
        if set(policy.get("protected_components", [])) != PROTECTED:
            raise _Issue("INVALID", "PROTECTED_INVARIANTS_REQUIRED")
        components = _object(policy.get("components"))
        if not components:
            raise _Issue("NOT_EVALUABLE", "MISSING_COMPONENT_INVENTORY")
        if set(components) & PROTECTED:
            raise _Issue("INVALID", "PROTECTED_COMPONENT_ABLATION")
        repetitions = policy.get("repetitions")
        if type(repetitions) is not int or repetitions < 1:
            raise _Issue("INVALID", "INVALID_BOOTSTRAP_REPETITIONS")
        if repetitions < MIN_BOOTSTRAP_REPETITIONS:
            raise _Issue("NOT_EVALUABLE", "INSUFFICIENT_BOOTSTRAP_REPETITIONS")
        original = _candidate(full)
        if identity_hash(original) != policy.get("full_candidate_hash"):
            raise _Issue("INVALID", "FULL_CANDIDATE_NOT_FROZEN")
        full_contract = _comparison(full, original)
        dd_tolerance = _number(policy.get("max_drawdown_increase"))
        cost_tolerance = _number(policy.get("max_cost_increase"))
        cost_metric = policy.get("cost_metric")
        if cost_metric not in {"fees_initial_equity", "funding_cost_initial_equity"}:
            raise _Issue("INVALID", "UNSUPPORTED_NORMALIZED_COST_METRIC")
        if not isinstance(variants, Sequence) or isinstance(variants, (str, bytes)):
            raise _Issue("INVALID", "EXPECTED_VARIANT_SEQUENCE")
        expected, observed = {}, {}
        for component, definitions in components.items():
            rule = _REMOVAL_RULES.get((original["family"], component))
            if rule is None:
                raise _Issue("NOT_EVALUABLE", "UNREGISTERED_REMOVAL_SEMANTICS")
            if not isinstance(definitions, list) or not definitions:
                raise _Issue("INVALID", "EMPTY_COMPONENT_VARIANTS")
            for definition in definitions:
                definition = _object(definition)
                key = (component, definition.get("variant_id"))
                keys = definition.get("parameter_keys")
                if not isinstance(keys, list) or not keys or not all(isinstance(k, str) for k in keys):
                    raise _Issue("INVALID", "EXPLICIT_PARAMETER_KEYS_REQUIRED")
                if set(keys) & PROTECTED_PARAMETERS:
                    raise _Issue("INVALID", "PROTECTED_PARAMETER_ABLATION")
                if keys != [rule[0]]:
                    raise _Issue("INVALID", "PARAMETERS_OUTSIDE_REGISTERED_REMOVAL")
                if key in expected or not isinstance(key[1], str) or not key[1]:
                    raise _Issue("INVALID", "DUPLICATE_OR_INVALID_VARIANT_DEFINITION")
                expected[key] = definition
        for raw in variants:
            row = _object(raw)
            key = (row.get("component_id"), row.get("variant_id"))
            if key in observed or key not in expected or row.get("kind") != "remove":
                raise _Issue("INVALID", "DUPLICATE_UNKNOWN_OR_NONREMOVAL_VARIANT")
            observed[key] = _object(row.get("report"))
        missing = [dict(component_id=k[0], variant_id=k[1]) for k in expected if k not in observed]
        if missing:
            result["missing"] = missing
            raise _Issue("NOT_EVALUABLE", "MISSING_PREDECLARED_VARIANTS")
        full_safety = _safety(full, "full")
        statuses = []
        for component in components:
            results = []
            for key, definition in expected.items():
                if key[0] != component:
                    continue
                reduced = observed[key]
                candidate = _candidate(reduced)
                if identity_hash(candidate) != definition.get("candidate_hash"):
                    raise _Issue("INVALID", "VARIANT_CANDIDATE_NOT_FROZEN")
                if {k: v for k, v in original.items() if k != "params"} != {
                    k: v for k, v in candidate.items() if k != "params"
                }:
                    raise _Issue("INVALID", "NON_COMPONENT_CANDIDATE_CHANGE")
                keys = original["params"].keys() | candidate["params"].keys()
                difference = {k for k in keys if original["params"].get(k) != candidate["params"].get(k)}
                if difference != set(definition["parameter_keys"]):
                    raise _Issue("INVALID", "UNDECLARED_OR_NOOP_PARAMETER_CHANGE")
                parameter, disabled = _REMOVAL_RULES[(original["family"], component)]
                before, after = original["params"].get(parameter), candidate["params"].get(parameter)
                if type(disabled) is bool:
                    other = "exit_gate_enabled" if parameter == "entry_gate_enabled" else "entry_gate_enabled"
                    valid = (
                        before is True
                        and after is False
                        and original["params"].get(other) is True
                        and candidate["params"].get(other) is True
                    )
                else:
                    valid = type(before) is int and before > disabled and type(after) is int and after == disabled
                if not valid:
                    raise _Issue("INVALID", "NOT_REGISTERED_COMPONENT_REMOVAL")
                if full_contract != _comparison(reduced, candidate):
                    raise _Issue("NOT_EVALUABLE", "COMPARISON_CONTRACT_MISMATCH")
                if full["identity"].get("policy_hash") != reduced["identity"].get("policy_hash"):
                    raise _Issue("INVALID", "POLICY_MISMATCH")
                observation = {"variant_id": key[1], "decision": "INCONCLUSIVE"}
                safety = full_safety + _safety(reduced, "removed")
                if safety:
                    severity = {"INVALID": 0, "STALE": 1, "FAIL": 2, "NOT_EVALUABLE": 3}
                    strongest = min(safety, key=lambda issue: severity[issue["status"]])
                    observation.update(status=strongest["status"], reason=strongest["reason"], safety_issues=safety)
                    statuses.append(strongest["status"])
                    results.append(observation)
                    continue
                full_dd, less_dd = _metric(full, "max_drawdown"), _metric(reduced, "max_drawdown")
                full_cost, less_cost = _metric(full, cost_metric), _metric(reduced, cost_metric)
                kwargs = {key: policy[key] for key in ("block_lengths", "min_samples", "repetitions", "seed")}
                keep = paired_daily_excess(full, reduced, **kwargs)
                remove = paired_daily_excess(reduced, full, **kwargs)
                if (
                    keep["status"] == "PASS"
                    and full_dd - less_dd <= dd_tolerance
                    and full_cost - less_cost <= cost_tolerance
                ):
                    observation["decision"] = "KEEP"
                elif (
                    remove["status"] == "PASS"
                    and less_dd - full_dd <= dd_tolerance
                    and less_cost - full_cost <= cost_tolerance
                ):
                    observation["decision"] = "REMOVE_CANDIDATE"
                observation.update(
                    full_minus_removed=keep,
                    removed_minus_full=remove,
                    drawdown_difference=full_dd - less_dd,
                    cost_difference=full_cost - less_cost,
                    status="PASS" if observation["decision"] != "INCONCLUSIVE" else "NOT_EVALUABLE",
                )
                statuses.append(observation["status"])
                results.append(observation)
            decisions = {row["decision"] for row in results}
            decision = next(iter(decisions)) if len(decisions) == 1 else "INCONCLUSIVE"
            result["components"][component] = {"decision": decision, "variants": results}
        decisions = {row["decision"] for row in result["components"].values()}
        if len(decisions) == 1:
            result["decision"] = next(iter(decisions))
        result["status"] = next((s for s in ("INVALID", "STALE", "FAIL", "NOT_EVALUABLE") if s in statuses), "PASS")
        if any(row["decision"] == "INCONCLUSIVE" for row in result["components"].values()):
            result["status"] = "NOT_EVALUABLE" if result["status"] == "PASS" else result["status"]
        result["reason"] = "FROZEN_ABLATION_COMPARISONS_ONLY"
    except _Issue as exc:
        result.update(status=exc.status, decision="INCONCLUSIVE", reason=exc.reason)
    except (ValueError, TypeError, KeyError, OverflowError) as exc:
        result.update(status="INVALID", decision="INCONCLUSIVE", reason=f"INVALID_ABLATION_INPUT: {exc}")
    return result
