"""Offline TradFi event marks, not fills, causal attribution or a production signal."""

from __future__ import annotations

import math
from dataclasses import asdict, dataclass
from typing import Any


@dataclass(frozen=True)
class StudyConfig:
    lookback_minutes: int = 5
    drop_pct: float = -3.0
    cooldown_minutes: int = 60
    horizons_minutes: tuple[int, ...] = (1, 5, 15, 60)
    roundtrip_cost_bps: tuple[int, ...] = (10, 30, 100)

    def __post_init__(self):
        if not math.isfinite(self.drop_pct) or not -100 < self.drop_pct < 0:
            raise ValueError("drop must be between -100 and 0 percent")
        if self.lookback_minutes < 1 or not self.horizons_minutes or min(self.horizons_minutes) < 1:
            raise ValueError("positive lookback and horizons required")
        if self.cooldown_minutes < max(self.horizons_minutes):
            raise ValueError("cooldown must cover the longest label")
        if not self.roundtrip_cost_bps or min(self.roundtrip_cost_bps) < 0:
            raise ValueError("nonnegative cost scenarios required")


def validate_bars(rows: list, as_of_ms: int) -> list[dict[str, Any]]:
    """Reject unsorted/duplicate/gapped history; ignore still-open tail bars."""
    if not isinstance(rows, list) or not rows:
        raise ValueError("empty or invalid bars")
    result = []
    previous = None
    for row in rows:
        if not isinstance(row, list) or len(row) < 11:
            raise ValueError("invalid kline")
        start, end = int(row[0]), int(row[6])
        if start % 60000 or end != start + 59999 or (previous is not None and start != previous + 60000):
            raise ValueError("invalid timestamp, gap or duplicate")
        previous = start
        op, high, low, close = map(float, row[1:5])
        volume, quote, buy = float(row[5]), float(row[7]), float(row[10])
        if not all(math.isfinite(x) for x in (op, high, low, close, volume, quote, buy)):
            raise ValueError("nonfinite kline")
        if not 0 < low <= min(op, close) <= max(op, close) <= high:
            raise ValueError("invalid OHLC")
        if min(volume, quote, buy) < 0 or buy > quote:
            raise ValueError("invalid volume")
        if end < as_of_ms:
            result.append(
                dict(
                    open_ms=start,
                    close_ms=end,
                    open=op,
                    high=high,
                    low=low,
                    close=close,
                    quote_volume=quote,
                    taker_buy_quote=buy,
                )
            )
    return result


def analyze(rows: list, *, as_of_ms: int, config: StudyConfig = StudyConfig()) -> dict[str, Any]:
    bars = validate_bars(rows, as_of_ms)
    events = []
    blocked_until = -1
    for i in range(config.lookback_minutes, len(bars)):
        drop = (bars[i]["close"] / bars[i - config.lookback_minutes]["close"] - 1) * 100
        if drop > config.drop_pct or i <= blocked_until:
            continue
        blocked_until = i + config.cooldown_minutes
        entry = bars[i + 1] if i + 1 < len(bars) else None
        event = {
            "decision_close_ms": bars[i]["close_ms"],
            "drop_pct": drop,
            "entry_reference_ms": entry["open_ms"] if entry else None,
            "entry_reference_price": entry["open"] if entry else None,
            "forward": {},
        }
        for horizon in config.horizons_minutes:
            # Decision at close i; entry mark open i+1; exit mark open i+1+h.
            exit_i = i + 1 + horizon
            label = None
            if entry is not None and exit_i < len(bars):
                price = entry["open"]
                gross = (bars[exit_i]["open"] / price - 1) * 100
                window = bars[i + 1 : exit_i]
                label = {
                    "exit_reference_ms": bars[exit_i]["open_ms"],
                    "gross_reference_return_pct": gross,
                    "net_sensitivity_pct": {str(cost): gross - cost / 100 for cost in config.roundtrip_cost_bps},
                    "adverse_excursion_pct": min(0.0, (min(b["low"] for b in window) / price - 1) * 100),
                    "favorable_excursion_pct": max(0.0, (max(b["high"] for b in window) / price - 1) * 100),
                }
            event["forward"][str(horizon)] = label
        events.append(event)
    return {
        "config": asdict(config),
        "closed_bar_count": len(bars),
        "events": events,
        "evidence_type": "selected_event_historical_diagnostic",
        "forced_liquidation_confirmed": False,
        "promotion_eligible": False,
        "limitations": [
            "Post-event hypothesis and selected window: not untouched OOS or an alpha proof.",
            "Next-open marks are not achievable fills; no order book, latency, impact, funding or margin model.",
            "Three linked equities in one crash are one event cluster, not independent successes.",
            "News absence and forced liquidations cannot be inferred from price, volume or OI alone.",
            "Weekend index prices may follow the local book and are not independent fair value.",
        ],
    }


def oi_change(rows: list, start_ms: int, end_ms: int) -> dict | None:
    """Exact endpoint contract OI: price-induced notional drops are not deleveraging."""
    indexed = {}
    for row in rows:
        timestamp = int(row["timestamp"])
        contracts = float(row["sumOpenInterest"])
        if timestamp in indexed or not math.isfinite(contracts) or contracts <= 0:
            raise ValueError("invalid OI")
        indexed[timestamp] = contracts
    if start_ms not in indexed or end_ms not in indexed:
        return None
    return {
        "start_ms": start_ms,
        "end_ms": end_ms,
        "contracts_change_pct": (indexed[end_ms] / indexed[start_ms] - 1) * 100,
        "liquidation_attribution": "unproven",
    }
