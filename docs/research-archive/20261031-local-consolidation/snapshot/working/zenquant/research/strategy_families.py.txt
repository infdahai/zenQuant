"""Small, causal BTC research hypotheses that emit exposure intents only.

Targets are long/flat signed notional exposure divided by account equity, not
orders, quantities or returns. Index i is approved after bar i is confirmed;
the caller must execute later through the shared risk/execution/accounting
kernel. No strategy here can override that downstream risk authority.
"""

from __future__ import annotations

import math
from collections.abc import Mapping, Sequence
from typing import Any

import numpy as np
from numpy.lib.stride_tricks import sliding_window_view

BAR_NS = 4 * 3_600_000_000_000
VOLATILITY_BARS = 30 * 6
ANNUALIZATION = math.sqrt(365 * 6)
_ZERO_VOL = 1e-12
_DEFAULTS = {
    "trend": {"fast_bars": 12, "slow_bars": 72, "confirmation_bars": 2},
    "breakout": {
        "entry_bars": 48,
        "exit_bars": 12,
        "confirmation_bars": 1,
        "entry_gate_enabled": True,
        "exit_gate_enabled": True,
    },
    "range_reversion": {
        "mean_bars": 24,
        "entry_z": 2.0,
        "exit_z": 0.5,
        "trend_filter_bars": 120,
        "max_trend_drop": 0.05,
    },
    "cash": {},
    "vol_target": {},
}


def _number(value: Any, name: str) -> float:
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        raise ValueError(f"{name} must be a finite number")
    try:
        result = float(value)
    except OverflowError as exc:
        raise ValueError(f"{name} must be finite") from exc
    if not math.isfinite(result):
        raise ValueError(f"{name} must be finite")
    return result


def _parameters(family: str, params: Mapping[str, Any]) -> dict[str, Any]:
    if not isinstance(family, str) or family not in _DEFAULTS:
        raise ValueError(f"unknown strategy family: {family!r}")
    if not isinstance(params, Mapping):
        raise ValueError("params must be a mapping")
    unknown = set(params) - set(_DEFAULTS[family])
    if unknown:
        raise ValueError(f"unknown or non-searchable parameters: {sorted(unknown, key=str)!r}")
    resolved = {**_DEFAULTS[family], **params}
    for name, value in resolved.items():
        if name.endswith("_gate_enabled"):
            if type(value) is not bool:
                raise ValueError(f"{name} must be a boolean")
        elif name.endswith("_bars"):
            if isinstance(value, bool) or not isinstance(value, int) or value < 1:
                raise ValueError(f"{name} must be a positive integer")
        else:
            resolved[name] = _number(value, name)
    if family == "breakout" and not resolved["entry_gate_enabled"] and not resolved["exit_gate_enabled"]:
        raise ValueError("breakout ablation must remove only a single component")
    if family == "trend" and resolved["fast_bars"] >= resolved["slow_bars"]:
        raise ValueError("fast_bars must be smaller than slow_bars")
    if family == "range_reversion":
        if resolved["mean_bars"] < 2 or resolved["trend_filter_bars"] < resolved["mean_bars"]:
            raise ValueError("range windows require trend_filter_bars >= mean_bars >= 2")
        if not 0.0 <= resolved["exit_z"] < resolved["entry_z"]:
            raise ValueError("range thresholds require 0 <= exit_z < entry_z")
        if not 0.0 < resolved["max_trend_drop"] < 1.0:
            raise ValueError("max_trend_drop must be between zero and one")
    return resolved


def _validated_bars(bars: Sequence[Mapping[str, Any]]) -> list[tuple[int, float, float, float]]:
    if not isinstance(bars, Sequence) or isinstance(bars, (str, bytes)):
        raise ValueError("bars must be a sequence")
    rows = []
    previous_open: int | None = None
    for bar in bars:
        if not isinstance(bar, Mapping):
            raise ValueError("each bar must be a mapping")
        opened, closed = bar.get("open_time_ns"), bar.get("close_time_ns")
        if any(isinstance(value, bool) or not isinstance(value, int) for value in (opened, closed)):
            raise ValueError("bar times must be integer nanoseconds")
        if previous_open is not None and opened <= previous_open:
            raise ValueError("bar open times must be strictly increasing")
        if opened % BAR_NS or closed - opened not in (BAR_NS, BAR_NS - 1_000_000):
            raise ValueError("bars must use UTC-aligned confirmed 4h intervals")
        opening, high, low, close = (_number(bar.get(field), field) for field in ("open", "high", "low", "close"))
        if low <= 0.0 or not low <= min(opening, close) <= max(opening, close) <= high:
            raise ValueError("bar OHLC must be positive and internally consistent")
        rows.append((opened, high, low, close))
        previous_open = opened
    return rows


def _rolling(values: np.ndarray, window: int, operation: str) -> np.ndarray:
    result = np.full(len(values), np.nan)
    if len(values) >= window:
        views = sliding_window_view(values, window)
        functions = {"mean": np.mean, "std": np.std, "max": np.max, "min": np.min}
        kwargs = {"ddof": 1} if operation == "std" else {}
        result[window - 1 :] = functions[operation](views, axis=1, **kwargs)
    return result


def _segment_targets(
    rows: list[tuple[int, float, float, float]],
    family: str,
    params: dict[str, Any],
    target_vol: float,
    leverage_cap: float,
) -> list[float]:
    count = len(rows)
    targets = [0.0] * count
    if family == "cash" or count <= VOLATILITY_BARS or target_vol == 0.0 or leverage_cap == 0.0:
        return targets
    prices = np.array([row[3] for row in rows], dtype=float)
    with np.errstate(over="raise", divide="raise", invalid="raise"):
        try:
            returns = prices[1:] / prices[:-1] - 1.0
            volatility = np.concatenate(([np.nan], _rolling(returns, VOLATILITY_BARS, "std"))) * ANNUALIZATION
            if family == "trend":
                fast = _rolling(prices, params["fast_bars"], "mean")
                slow = _rolling(prices, params["slow_bars"], "mean")
            elif family == "breakout":
                highs = _rolling(np.array([row[1] for row in rows]), params["entry_bars"], "max")
                lows = _rolling(np.array([row[2] for row in rows]), params["exit_bars"], "min")
            elif family == "range_reversion":
                mean = _rolling(prices, params["mean_bars"], "mean")
                deviation = _rolling(prices, params["mean_bars"], "std")
        except FloatingPointError as exc:
            raise ValueError("bar values produce non-finite derived features") from exc

    holding = False
    confirmation = 0
    for index in range(VOLATILITY_BARS, count):
        vol = float(volatility[index])
        if not math.isfinite(vol) or vol <= _ZERO_VOL:
            holding, confirmation = False, 0
            continue
        if family == "vol_target":
            holding = True
        elif family == "trend":
            bullish = fast[index] > slow[index]
            confirmation = confirmation + 1 if bullish else 0
            if not bullish:
                holding = False
            elif confirmation >= params["confirmation_bars"]:
                holding = True
        elif family == "breakout":
            if (params["entry_gate_enabled"] and not math.isfinite(highs[index - 1])) or (
                params["exit_gate_enabled"] and not math.isfinite(lows[index - 1])
            ):
                holding, confirmation = False, 0
                continue
            if holding and params["exit_gate_enabled"] and prices[index] < lows[index - 1]:
                holding, confirmation = False, 0
            elif not holding:
                breakout = not params["entry_gate_enabled"] or prices[index] > highs[index - 1]
                confirmation = confirmation + 1 if breakout else 0
                holding = confirmation >= params["confirmation_bars"]
        elif family == "range_reversion":
            lookback = params["trend_filter_bars"]
            if index < lookback or not math.isfinite(deviation[index - 1]) or deviation[index - 1] <= _ZERO_VOL:
                holding = False
                continue
            slow_change = float(prices[index]) / float(prices[index - lookback]) - 1.0
            z_score = (float(prices[index]) - float(mean[index - 1])) / float(deviation[index - 1])
            if not math.isfinite(slow_change) or not math.isfinite(z_score):
                raise ValueError("bar values produce non-finite derived features")
            if slow_change < -params["max_trend_drop"] or (holding and z_score >= -params["exit_z"]):
                holding = False
            elif not holding and z_score <= -params["entry_z"]:
                holding = True
        if holding:
            # Every confirmed bar re-applies the same risk budget. Confirmation
            # gates entries, never a risk-driven reduction or a flat exit.
            targets[index] = min(leverage_cap, target_vol / vol)
    return targets


def generate_targets(
    bars: Sequence[Mapping[str, Any]],
    family: str,
    params: Mapping[str, Any],
    *,
    target_annual_vol: float = 0.10,
    max_leverage: float = 1.5,
) -> list[float]:
    """Return one deterministic, post-close exposure intent per input 4h bar.

    The fixed volatility estimator is sample standard deviation of 180 simple
    close-to-close returns, annualized by sqrt(365 * 6). A gap resets signal
    state and requires a new contiguous 30-day warmup. Insufficient/zero vol
    means flat. Risk-budget arguments are not accepted inside search params.

    Breakout gate booleans support frozen single-component diagnostic ablations
    only; they are not optimizer dimensions. Entry removal retains confirmation
    and enabled exits, and an exit bar cannot reenter until a later close. Exit
    removal retains breakout entry and never bypasses volatility or gap resets.

    OHLC and timestamp validation rejects future-invalid input rather than
    silently correcting it. This routine proves no fill or profitability claim.
    """
    resolved = _parameters(family, params)
    target_vol = _number(target_annual_vol, "target_annual_vol")
    leverage_cap = _number(max_leverage, "max_leverage")
    if target_vol < 0.0 or leverage_cap < 0.0:
        raise ValueError("risk budget and leverage cap must be nonnegative")
    rows = _validated_bars(bars)
    targets: list[float] = []
    start = 0
    for index in range(1, len(rows) + 1):
        if index == len(rows) or rows[index][0] != rows[index - 1][0] + BAR_NS:
            targets.extend(_segment_targets(rows[start:index], family, resolved, target_vol, leverage_cap))
            start = index
    return targets
