"""Pure descriptive risk evidence; no engineering or capital-risk admission.

The daily series comes from the matching unified evaluator. Losses are signed
``-return``; a negative loss describes a gain. ES averages exactly the worst 5%
probability mass, including a fractional boundary observation, rather than all
observations tied at VaR. Twenty *full* tail observations are required for its
sample-sufficiency flag, which is not a confidence interval or a safety verdict.

Exposure uses actual account rows as a piecewise-constant, as-of approximation.
It is not an inferred intrabar path, a target exposure, or evidence of alpha.
"""

from __future__ import annotations

import math
import statistics
from collections.abc import Mapping, Sequence
from datetime import UTC, datetime
from itertools import pairwise
from typing import Any

VERSION = "research-risk-diagnostics.v1"
DAY_NS = 86_400_000_000_000


def _finite(value: Any, name: str) -> float:
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        raise ValueError(f"{name} must be a finite number")
    try:
        number = float(value)
    except OverflowError as exc:
        raise ValueError(f"{name} must be finite") from exc
    if not math.isfinite(number):
        raise ValueError(f"{name} must be finite")
    return number


def _time(value: Any, name: str) -> int:
    if isinstance(value, bool) or not isinstance(value, int):
        raise ValueError(f"{name} must be integer nanoseconds")
    return value


def _rows(value: Any, name: str) -> list[Mapping[str, Any]]:
    if not isinstance(value, Sequence) or isinstance(value, (str, bytes)):
        raise ValueError(f"{name} must be a sequence of objects")
    if not all(isinstance(row, Mapping) for row in value):
        raise ValueError(f"{name} must contain objects")
    return list(value)


def _metric(value: float | None, unit: str, count: int, sampling: str, reason: str = "") -> dict[str, Any]:
    if value is not None and not math.isfinite(value):
        value, reason = None, "NUMERICAL_OVERFLOW"
    return {
        "definition_version": VERSION,
        "value": value,
        "unit": unit,
        "sampling": sampling,
        "sample_size": count,
        "status": "NOT_EVALUABLE" if reason or value is None else "PASS",
        "reason": reason or ("MISSING_EVIDENCE" if value is None else "OBSERVED"),
        "confidence_interval": None,
    }


def _daily(
    report: Mapping[str, Any], curve: list[tuple[int, float]], ruin_time: int | None
) -> tuple[dict[str, Any], dict[str, Any]]:
    marks = dict(curve)
    curve_ruin = next((time for time, equity in curve if equity <= 0), None)
    expected_times = {
        time
        for time, _ in curve
        if time % DAY_NS == 0 and marks.get(time - DAY_NS, 0) > 0 and (curve_ruin is None or time <= curve_ruin)
    }
    daily = []
    supplied_times = []
    for row in _rows(report.get("daily_returns", []), "report.daily_returns"):
        time = _time(row.get("time_ns"), "daily_return.time_ns")
        value = _finite(row.get("value"), "daily_return.value")
        if time % DAY_NS or (supplied_times and time <= supplied_times[-1]):
            raise ValueError("daily returns must have strictly increasing UTC midnight times")
        if time not in expected_times or not math.isclose(
            value, marks[time] / marks[time - DAY_NS] - 1, rel_tol=1e-12, abs_tol=1e-14
        ):
            raise ValueError("daily returns must match the supplied curve")
        supplied_times.append(time)
        if ruin_time is None or time <= ruin_time:
            daily.append((time, value))
    if set(supplied_times) != expected_times:
        raise ValueError("daily returns must match all evaluable daily points in the curve")
    times = {time for time, _ in curve}
    missing = (
        [time for time in range(-(-curve[0][0] // DAY_NS) * DAY_NS, curve[-1][0] + 1, DAY_NS) if time not in times]
        if curve
        else []
    )
    reason = "RUIN" if ruin_time is not None else "MISSING_DAILY_VALUATIONS" if missing else ""
    losses = sorted(-value for _, value in daily)
    count = len(losses)
    weeks = []
    for index in range(6, count):
        group = daily[index - 6 : index + 1]
        if group[-1][0] - group[0][0] == 6 * DAY_NS:
            weeks.append(1.0 - math.prod(1.0 + value for _, value in group))
    full_tail, remainder = divmod(count, 20)
    mass = count / 20
    quantile = es = None
    if count:
        quantile = losses[(19 * count + 19) // 20 - 1]
        descending = losses[::-1]
        weighted_sum = math.fsum(descending[:full_tail])
        if remainder:
            weighted_sum += remainder / 20 * descending[full_tail]
        es = weighted_sum / mass
    tail_reason = reason or ("INSUFFICIENT_TAIL_OBSERVATIONS" if full_tail < 20 else "")
    metrics = {
        "worst_day_loss": _metric(max(losses) if losses else None, "fraction", count, "UTC_daily", reason),
        "worst_7d_loss": _metric(
            max(weeks) if weeks else None,
            "fraction",
            len(weeks),
            "overlapping_contiguous_7_UTC_days",
            reason or ("INSUFFICIENT_CONSECUTIVE_DAYS" if not weeks else ""),
        ),
        "daily_loss_var95": _metric(quantile, "fraction", count, "UTC_daily", tail_reason),
        "daily_loss_es95": _metric(es, "fraction", count, "UTC_daily", tail_reason),
        "annualized_daily_volatility": _metric(
            statistics.stdev(losses) * math.sqrt(365) if count > 1 else None,
            "annualized_fraction",
            count,
            "UTC_daily",
            reason or ("INSUFFICIENT_DAILY_SAMPLES" if count < 2 else ""),
        ),
    }
    return metrics, {
        "loss_definition": "negative_daily_return_without_clipping",
        "var_definition": "empirical_nearest_rank_95_percent",
        "es_definition": "worst_5_percent_mass_with_fractional_boundary_weight",
        "full_tail_observations": full_tail,
        "effective_tail_mass": mass,
        "min_full_tail_observations": 20,
        "overlapping_weeks_are_not_independent": True,
    }


def _episodes(evidence: Mapping[str, Any], net_pnl: float | None) -> dict[str, Any]:
    present = evidence.get("episodes") is not None
    rows = _rows(evidence["episodes"], "episodes") if present else []
    closed, open_realized, open_unrealized = [], [], []
    missing_unrealized = False
    close_times = []
    for row in rows:
        if not isinstance(row.get("closed"), bool):
            raise ValueError("episode.closed must be boolean")
        pnl = _finite(row.get("net_pnl"), "episode.net_pnl")
        if row["closed"]:
            closed.append(pnl)
            close_times.append(row.get("closed_at_ns"))
        else:
            open_realized.append(pnl)
            if row.get("unrealized_pnl") is None:
                missing_unrealized = True
            else:
                open_unrealized.append(_finite(row["unrealized_pnl"], "episode.unrealized_pnl"))
    ordered = all(isinstance(time, int) and not isinstance(time, bool) for time in close_times)
    if ordered and any(right < left for left, right in pairwise(close_times)):
        raise ValueError("closed episodes must be in chronological close order")
    count = len(closed)
    wins = [pnl for pnl in closed if pnl > 0]
    losses = [-pnl for pnl in closed if pnl < 0]
    total, gain = math.fsum(closed), math.fsum(wins)
    largest = max(wins) if wins else None
    win_mean, loss_mean = statistics.mean(wins) if wins else None, statistics.mean(losses) if losses else None
    streak = longest = 0
    for pnl in closed:
        streak = streak + 1 if pnl < 0 else 0
        longest = max(streak, longest)
    missing = "MISSING_EPISODES" if not present else ""
    ratio_reason = missing or ("NO_LOSS_EPISODES" if not losses else "NO_WIN_EPISODES" if not wins else "")
    unrealized = None if missing_unrealized or not present else math.fsum(open_unrealized)
    attributed = total + math.fsum(open_realized) + unrealized if unrealized is not None else None
    metrics = {
        "closed_episode_net_pnl": _metric(total if present else None, "USDT", count, "closed_episodes", missing),
        "largest_closed_episode_profit": _metric(largest, "USDT", count, "closed_episodes", missing),
        "largest_episode_share_of_gross_profit": _metric(
            largest / gain if gain > 0 and largest is not None else None,
            "fraction",
            count,
            "closed_episodes",
            missing or ("NO_PROFIT_EPISODES" if not wins else ""),
        ),
        "closed_pnl_without_best_episode": _metric(
            total - largest if largest is not None else None, "USDT", count, "closed_episodes_attribution_only", missing
        ),
        "mean_winning_episode_pnl": _metric(win_mean, "USDT", len(wins), "closed_episodes", missing),
        "mean_losing_episode_loss": _metric(loss_mean, "USDT", len(losses), "closed_episodes", missing),
        "average_win_loss_ratio": _metric(
            win_mean / loss_mean if win_mean is not None and loss_mean is not None else None,
            "ratio",
            count,
            "closed_episodes",
            ratio_reason,
        ),
        "longest_consecutive_losing_episodes": _metric(
            longest if present else None,
            "episodes",
            count,
            "supplied_closed_episode_order",
            missing or ("MISSING_EPISODE_CLOSE_TIMES" if not ordered else ""),
        ),
        "open_episode_realized_net_pnl": _metric(
            math.fsum(open_realized) if present else None, "USDT", len(open_realized), "open_episodes", missing
        ),
        "open_episode_unrealized_pnl": _metric(
            unrealized,
            "USDT",
            len(open_realized),
            "open_episodes",
            missing or ("MISSING_OPEN_UNREALIZED_PNL" if missing_unrealized else ""),
        ),
        "episode_attribution_residual": _metric(
            net_pnl - attributed if net_pnl is not None and attributed is not None else None,
            "USDT",
            len(rows),
            "terminal_NAV_less_episode_attribution",
            missing or ("MISSING_OPEN_UNREALIZED_PNL" if missing_unrealized else ""),
        ),
    }
    return metrics


def _month_start(year: int, month: int) -> int:
    return int(datetime(year, month, 1, tzinfo=UTC).timestamp()) * 1_000_000_000


def _monthly(curve: list[tuple[int, float]], ruin_time: int | None) -> list[dict[str, Any]]:
    if len(curve) < 2:
        return []
    first, last = curve[0][0], curve[-1][0]
    date = datetime.fromtimestamp(first // 1_000_000_000, tz=UTC)
    year, month = date.year, date.month
    values = dict(curve)
    result = []
    while (month_start := _month_start(year, month)) < last:
        next_year, next_month = (year + 1, 1) if month == 12 else (year, month + 1)
        next_start = _month_start(next_year, next_month)
        start, end = max(first, month_start), min(last, next_start)
        missing = [time for time in range(-(-start // DAY_NS) * DAY_NS, end + 1, DAY_NS) if time not in values]
        initial, final = values.get(start), values.get(end)
        reason = "RUIN" if ruin_time is not None and end >= ruin_time else ""
        reason = reason or ("MISSING_DAILY_VALUATIONS" if missing else "")
        valid_endpoints = initial is not None and final is not None
        pnl = final - initial if valid_endpoints else None
        ret = final / initial - 1 if valid_endpoints and initial > 0 else None
        count = (end - start) // DAY_NS
        result.append(
            {
                "month": f"{year:04d}-{month:02d}",
                "start_time_ns": start,
                "end_time_ns": end,
                "partial_start": first > month_start,
                "partial_end": last < next_start,
                "missing_midnights": missing,
                "net_return": _metric(ret, "fraction", count, "UTC_month_endpoint_valuations", reason),
                "net_pnl": _metric(pnl, "USDT", count, "UTC_month_endpoint_valuations", reason),
            }
        )
        year, month = next_year, next_month
    return result


def _exposure(
    evidence: Mapping[str, Any], curve: list[tuple[int, float]], ruin_time: int | None, net_return: float | None
) -> tuple[dict[str, Any], dict[str, Any]]:
    accounting = evidence.get("accounting")
    if accounting is not None and not isinstance(accounting, Mapping):
        raise ValueError("accounting must be an object")
    raw = accounting.get("rows") if accounting is not None else None
    rows = []
    max_exposure = None
    observed_ruin = ruin_time is not None
    account_ruin_time = account_ruin_transition = previous_equity = None
    for row in _rows(raw, "accounting.rows") if raw is not None else []:
        time = _time(row.get("time_ns"), "account.time_ns")
        equity = _finite(row.get("equity"), "account.equity")
        quantity = _finite(row.get("position_qty"), "account.position_qty")
        raw_mark = row.get("mark_price")
        # A settlement can precede the first quote at the same timestamp. The
        # flat actual position has zero notional without inventing a quote.
        mark = 0.0 if raw_mark is None and quantity == 0 else _finite(raw_mark, "account.mark_price")
        if mark < 0 or (mark == 0 and quantity != 0):
            raise ValueError("account.mark_price must be positive for a nonzero actual position")
        if rows and time < rows[-1][0]:
            raise ValueError("accounting rows must be in chronological event order")
        if curve and not curve[0][0] <= time <= curve[-1][0]:
            raise ValueError("accounting rows must lie within the evaluated curve window")
        if equity <= 0 and account_ruin_time is None:
            account_ruin_time = time
            if previous_equity is not None and previous_equity > 0:
                account_ruin_transition = equity / previous_equity - 1
        previous_equity = equity
        observed_ruin |= equity <= 0
        exposure = abs(quantity * mark) / equity if equity > 0 else None
        if exposure is not None:
            max_exposure = max(max_exposure or 0.0, exposure)
        point = (time, exposure, quantity != 0)
        if rows and time == rows[-1][0]:
            rows[-1] = point
        else:
            rows.append(point)
    complete = len(curve) >= 2 and len(rows) >= 2 and rows[0][0] == curve[0][0] and rows[-1][0] == curve[-1][0]
    elapsed = rows[-1][0] - rows[0][0] if len(rows) >= 2 else 0
    reason = "RUIN" if observed_ruin else "MISSING_ACCOUNTING_ROWS" if not rows else ""
    reason = reason or ("INCOMPLETE_ACCOUNTING_WINDOW" if not complete else "")
    exposure = participation = None
    if complete and elapsed > 0 and not observed_ruin:
        exposure = math.fsum(point[1] * (after[0] - point[0]) / elapsed for point, after in pairwise(rows))
        participation = sum(after[0] - point[0] for point, after in pairwise(rows) if point[2]) / elapsed
    ratio_reason = reason or ("ZERO_EXPOSURE" if exposure == 0 else "")
    basis = "accounting_rows_piecewise_constant_asof"
    return {
        "time_weighted_abs_exposure": _metric(exposure, "NAV_multiple", len(rows), basis, reason),
        "time_in_position_fraction": _metric(participation, "fraction", len(rows), basis, reason),
        "max_observed_abs_exposure": _metric(max_exposure, "NAV_multiple", len(rows), "all_accounting_events", reason),
        "net_return_per_abs_exposure": _metric(
            net_return / exposure if net_return is not None and exposure is not None and exposure > 0 else None,
            "fraction_per_NAV_multiple",
            len(rows),
            basis,
            ratio_reason,
        ),
    }, {
        "basis": basis,
        "duplicate_time_policy": "last_event_state_for_next_interval_all_events_for_peak",
        "complete_window": complete,
        "longest_interval_ns": max((after[0] - point[0] for point, after in pairwise(rows)), default=None),
        "not_an_intrabar_leverage_or_alpha_claim": True,
        "first_nonpositive_account_time_ns": account_ruin_time,
        "first_nonpositive_account_transition_return": account_ruin_transition,
    }


def evaluate_risk_diagnostics(evidence: Mapping[str, Any], report: Mapping[str, Any]) -> dict[str, Any]:
    """Return deterministic observations for the same evidence/evaluator report.

    ``PASS`` on an individual metric means an observation is available under its
    declared sampling rule. It never passes an E gate or the full R2 gate. The
    caller retains the unified evaluator's engineering blockers and evidence
    identity; this module does not rewrite them or grant promotion.
    """
    if not isinstance(evidence, Mapping) or not isinstance(report, Mapping):
        raise ValueError("evidence and report must be objects")
    if report.get("identity", {}) != evidence.get("identity", {}):
        raise ValueError("report identity must match evidence identity")
    curve = []
    for row in _rows(evidence.get("curve", []), "curve"):
        time = _time(row.get("time_ns"), "curve.time_ns")
        equity = _finite(row.get("equity"), "curve.equity")
        if curve and time <= curve[-1][0]:
            raise ValueError("curve times must be strictly increasing")
        curve.append((time, equity))
    ruin_index = next((index for index, (_, equity) in enumerate(curve) if equity <= 0), None)
    ruin_time = curve[ruin_index][0] if ruin_index is not None else None
    transition = (
        curve[ruin_index][1] / curve[ruin_index - 1][1] - 1 if ruin_index is not None and ruin_index > 0 else None
    )
    net_pnl = curve[-1][1] - curve[0][1] if curve else None
    net_return = net_pnl / curve[0][1] if curve and curve[0][1] > 0 else None
    exposure, exposure_sampling = _exposure(evidence, curve, ruin_time, net_return)
    account_ruin = exposure_sampling["first_nonpositive_account_time_ns"]
    if account_ruin is not None and (ruin_time is None or account_ruin <= ruin_time):
        ruin_time = account_ruin
        account_transition = exposure_sampling["first_nonpositive_account_transition_return"]
        if account_transition is not None:
            transition = account_transition
    metrics, tail = _daily(report, curve, ruin_time)
    metrics.update(_episodes(evidence, net_pnl))
    metrics.update(exposure)
    metrics["ruin_transition_return"] = _metric(
        transition,
        "fraction",
        1 if transition is not None else 0,
        "last_positive_to_first_nonpositive_NAV",
        "" if transition is not None else "NO_RUIN",
    )
    return {
        "schema_version": VERSION,
        "metrics": metrics,
        "tail_sampling": tail,
        "monthly_returns": _monthly(curve, ruin_time),
        "exposure_sampling": exposure_sampling,
        "capital_risk_gate": {
            "status": "FAIL" if ruin_time is not None else "NOT_EVALUABLE",
            "reason": "RUIN" if ruin_time is not None else "MISSING_COMPLETE_CAPITAL_RISK_EVIDENCE",
        },
        "margin_and_liquidation": {"status": "NOT_EVALUABLE", "reason": "MISSING_MARGIN_AND_LIQUIDATION_MODEL"},
        "limitations": [
            "Descriptive statistics do not grant engineering or research admission.",
            "Episode subtraction is attribution, not a counterfactual strategy replay.",
            "Open episode net_pnl is realized net; unrealized_pnl is added separately.",
            "Twenty empirical tail observations do not establish independent observations or a confidence interval.",
            "Risk-budget utilization, liquidation distance and intrabar exposure are not evaluated.",
        ],
    }
