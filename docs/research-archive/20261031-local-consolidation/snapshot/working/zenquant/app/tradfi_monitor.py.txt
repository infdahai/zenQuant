from __future__ import annotations

import fcntl
import gzip
import hashlib
import json
import math
import os
import re
import shutil
import signal
import sys
import time
from collections import Counter
from contextlib import ExitStack
from datetime import UTC, datetime
from pathlib import Path
from urllib.request import urlopen
from uuid import uuid4

import websockets
from websockets.asyncio.client import connect

from zenquant.research.tradfi_monitor import SectorMonitor, SymbolMonitor


def validate_config(config):
    symbols = config.get("symbols")
    if (
        not isinstance(symbols, list)
        or not 1 <= len(symbols) <= 30
        or any(not isinstance(s, str) or not re.fullmatch(r"[A-Z0-9]+USDT", s) for s in symbols)
        or len(set(symbols)) != len(symbols)
    ):
        raise ValueError("1-30 unique explicit USDT symbols required")
    if config.get("research_only") is not True:
        raise ValueError("research_only must remain true")
    for key in (
        "duration_hours",
        "drop_60s_pct",
        "min_quote_volume_60s",
        "warmup_ms",
        "max_event_delay_ms",
        "depth_stale_ms",
        "trade_stale_ms",
        "cooldown_ms",
        "reconnect_seconds",
        "max_consecutive_failures",
        "max_raw_bytes",
        "min_free_bytes",
    ):
        value = config.get(key)
        if isinstance(value, bool) or not isinstance(value, (float, int)) or not math.isfinite(value):
            raise ValueError(f"invalid {key}")
        if key not in ("duration_hours", "drop_60s_pct") and value <= 0:
            raise ValueError(f"positive {key} required")
    if not 0 <= config["duration_hours"] <= 48 or not -100 < config["drop_60s_pct"] < 0:
        raise ValueError("invalid duration/threshold")
    if config["warmup_ms"] < 60000 or config["cooldown_ms"] < 120000:
        raise ValueError("warmup/cooldown too short")
    sectors = config.get("sectors", {})
    if not isinstance(sectors, dict):
        raise ValueError("sectors must be a mapping")
    for name, sector in sectors.items():
        if not isinstance(name, str) or not isinstance(sector, dict):
            raise ValueError("invalid sector")
        members = sector.get("members", {})
        if (
            not isinstance(members, dict)
            or not members
            or not set(members).issubset(symbols)
            or any(not isinstance(v, str) or not v for v in members.values())
        ):
            raise ValueError("sector members must map subscribed symbols to issuer IDs")
        for key in ("min_issuers", "window_ms", "episode_ms"):
            if type(sector.get(key)) is not int or sector[key] <= 0:
                raise ValueError(f"invalid sector {key}")
        if not 2 <= sector["min_issuers"] <= len(set(members.values())) or sector["episode_ms"] < sector["window_ms"]:
            raise ValueError("invalid sector breadth/window")


def validate_instruments(config, info):
    catalogue = {row["symbol"]: row for row in info["symbols"]}
    admitted = []
    for symbol in config["symbols"]:
        row = catalogue.get(symbol, {})
        if (
            row.get("status") != "TRADING"
            or row.get("contractType") != "TRADIFI_PERPETUAL"
            or row.get("underlyingType") not in ("EQUITY", "KR_EQUITY")
        ):
            raise ValueError(f"{symbol}: active supported TradFi equity required; refusing ticker substitution")
        admitted.append(row)
    return admitted


def now_ms():
    return time.time_ns() // 1_000_000


def write_json(path, data):
    temp = path.with_suffix(".tmp")
    temp.write_text(json.dumps(data, ensure_ascii=False, indent=2, allow_nan=False) + "\n")
    temp.replace(path)


class Recorder:
    def __init__(self, output, config):
        self.output, self.config = output, config
        self.monitors = {s: SymbolMonitor(s, config) for s in config["symbols"]}
        self.sector_monitor = SectorMonitor(config.get("sectors", {}))
        self.started = now_ms()
        self.deadline = self.started + int(config["duration_hours"] * 3600000) if config["duration_hours"] else None
        self.sequence = 0
        self.counts = Counter()
        self.alert_counts = Counter()
        self.stream_received = {}
        self.connections = {}
        self.raw_bytes = 0
        self.raw = None
        self.raw_context = ExitStack()
        self.raw_hour = None
        self.closed_segments = []
        self.stop = asyncio.Event()
        self.lifecycle = "starting"
        self.reason = None
        self.reconnections = 0
        self.alerts = (output / "alerts.jsonl").open("a", buffering=1)
        self.observations = (output / "observations.jsonl").open("a", buffering=1)

    def record(self, kind, payload):
        received = now_ms()
        hour = datetime.fromtimestamp(received / 1000, UTC).strftime("%Y%m%dT%H")
        if hour != self.raw_hour:
            self.close_segment()
            self.raw_hour = hour
            self.raw = self.raw_context.enter_context(
                gzip.open(self.output / f"raw-{hour}.jsonl.gz", "at", compresslevel=3)  # noqa: SIM115 - rotated ExitStack
            )
        self.sequence += 1
        envelope = {
            "sequence": self.sequence,
            "received_ms": received,
            "received_monotonic_ns": time.monotonic_ns(),
            "channel": kind,
            "payload": payload,
        }
        line = json.dumps(envelope, separators=(",", ":"), allow_nan=False) + "\n"
        self.raw_bytes += len(line.encode())
        self.raw.write(line)
        return received

    def close_segment(self):
        if self.raw:
            self.raw_context.close()
            path = self.output / f"raw-{self.raw_hour}.jsonl.gz"
            with path.open("rb") as file:
                digest = hashlib.file_digest(file, "sha256").hexdigest()
            self.closed_segments.append({"file": path.name, "sha256": digest, "bytes": path.stat().st_size})
            write_json(self.output / "segments.json", self.closed_segments)
            self.raw = None

    def alert(self, value):
        self.alert_counts[value["kind"]] += 1
        self.alerts.write(json.dumps(value, allow_nan=False) + "\n")
        for sector_event in self.sector_monitor.ingest(value):
            self.alert_counts[sector_event["kind"]] += 1
            self.alerts.write(json.dumps(sector_event, allow_nan=False) + "\n")

    async def stream(self, channel, suffixes):
        streams = [s.lower() + "@" + suffix for s in self.config["symbols"] for suffix in suffixes]
        url = f"wss://fstream.binance.com/{channel}/stream?streams=" + "/".join(streams)
        self.connections[channel] = "connecting"
        async with connect(
            url, proxy=None, open_timeout=20, ping_interval=20, ping_timeout=20, max_queue=4096, max_size=2**20
        ) as ws:
            self.connections[channel] = "connected"
            self.record("lifecycle", {"channel": channel, "state": "connected", "url": url})
            async for raw in ws:
                packet = json.loads(raw)
                received = self.record(channel, packet)
                if "data" not in packet or "stream" not in packet:
                    raise ValueError(f"unexpected server message: {packet}")
                data = packet["data"]
                stream = packet["stream"]
                self.counts[stream] += 1
                self.stream_received[stream] = received
                symbol = data.get("s", data.get("o", {}).get("s"))
                if symbol in self.monitors:
                    for alert in self.monitors[symbol].ingest(data, received):
                        alert["raw_sequence"] = self.sequence
                        self.alert(alert)
            raise ConnectionError(f"{channel} stream ended")

    def status(self):
        now = now_ms()
        return {
            "pid": os.getpid(),
            "lifecycle": self.lifecycle,
            "reason": self.reason,
            "heartbeat_ms": now,
            "started_ms": self.started,
            "stop_at_ms": self.deadline,
            "config": self.config,
            "connections": self.connections,
            "reconnections": self.reconnections,
            "messages": dict(self.counts),
            "stream_received_ms": self.stream_received,
            "alerts": dict(self.alert_counts),
            "raw_uncompressed_bytes": self.raw_bytes,
            "symbols": {s: m.status(now) for s, m in self.monitors.items()},
            "promotion_eligible": False,
            "notifications": "local_files_only",
            "limitations": [
                "Top20 snapshots, not a full order-level book; absorption is unconfirmed.",
                "Forced liquidation stream is sampled, not complete.",
                "No independent equity fair value, news or executable size model.",
                "Deployment host must remain running and connected.",
            ],
        }

    async def heartbeat(self):
        while not self.stop.is_set():
            status = self.status()
            write_json(self.output / "status.json", status)
            self.observations.write(json.dumps(status, separators=(",", ":"), allow_nan=False) + "\n")
            if self.raw:
                self.raw.flush()
            if self.deadline is not None and now_ms() >= self.deadline:
                self.reason = "duration_complete"
                return
            if self.raw_bytes >= self.config["max_raw_bytes"]:
                self.reason = "raw_budget_reached"
                return
            if shutil.disk_usage(self.output).free < self.config["min_free_bytes"]:
                self.reason = "low_disk_space"
                return
            try:
                await asyncio.wait_for(self.stop.wait(), timeout=5)
            except TimeoutError:
                pass

    async def run(self):
        loop = asyncio.get_running_loop()
        for sig in (signal.SIGTERM, signal.SIGINT):
            loop.add_signal_handler(sig, self.stop.set)
        failures = 0
        try:
            while not self.stop.is_set() and (self.deadline is None or now_ms() < self.deadline):
                self.lifecycle = "collecting"
                connected_at = time.monotonic()
                tasks = [
                    asyncio.create_task(self.stream("public", ["depth20@100ms"])),
                    asyncio.create_task(self.stream("market", ["aggTrade", "kline_1m", "markPrice@1s", "forceOrder"])),
                    asyncio.create_task(self.heartbeat()),
                ]
                done, pending = await asyncio.wait(tasks, return_when=asyncio.FIRST_COMPLETED)
                errors = [task.exception() for task in done if not task.cancelled() and task.exception()]
                for task in pending:
                    task.cancel()
                await asyncio.gather(*pending, return_exceptions=True)
                if not errors:
                    break
                self.reason = "; ".join(str(error) for error in errors)
                self.connections = {key: "disconnected" for key in ("public", "market")}
                self.lifecycle = "reconnecting"
                for monitor in self.monitors.values():
                    self.alert(monitor.gap(now_ms(), "connection_lost"))
                failures = 1 if time.monotonic() - connected_at > 60 else failures + 1
                self.reconnections += 1
                self.record("lifecycle", {"error": self.reason, "consecutive_failures": failures})
                write_json(self.output / "status.json", self.status())
                if failures >= self.config["max_consecutive_failures"]:
                    self.lifecycle = "failed"
                    break
                try:
                    await asyncio.wait_for(self.stop.wait(), timeout=self.config["reconnect_seconds"])
                except TimeoutError:
                    pass
                self.reason = None
        finally:
            self.lifecycle = "failed" if self.lifecycle == "failed" else "stopped"
            self.reason = self.reason or ("user_stop" if self.stop.is_set() else "duration_complete")
            for monitor in self.monitors.values():
                monitor.reset(now_ms())
            self.connections = {key: "closed" for key in ("public", "market")}
            self.close_segment()
            write_json(self.output / "status.json", self.status())
            self.alerts.close()
            self.observations.close()


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--config", type=Path, required=True)
    parser.add_argument("--data-dir", type=Path, default=Path("data/tradfi-monitor"))
    parser.add_argument("--output", type=Path, help="optional new run directory; refuses overwrite")
    parser.add_argument("--continuous", action="store_true", help="no time limit; storage/failure limits still apply")
    parser.add_argument("--check-config", action="store_true", help="offline validation only; no files or collection")
    parser.add_argument(
        "--check-market", action="store_true", help="verify configured exchange instruments; no collection"
    )
    args = parser.parse_args(argv)
    config = json.loads(args.config.read_text())
    if args.continuous:
        config["duration_hours"] = 0
    validate_config(config)
    if args.check_config:
        print(json.dumps({"valid": True, "symbols": config["symbols"], "collection_started": False}))
        return 0
    # Startup admission is read-only and fails closed; no ticker guessing/fallback.
    with urlopen("https://fapi.binance.com/fapi/v1/exchangeInfo", timeout=20) as response:
        instruments = validate_instruments(config, json.load(response))
    if args.check_market:
        print(json.dumps({"symbols": [s["symbol"] for s in instruments], "collection_started": False}))
        return 0
    base = args.data_dir.resolve()
    base.mkdir(parents=True, exist_ok=True)
    with (base / ".collector.lock").open("a") as lock:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        output = (
            args.output.resolve()
            if args.output
            else base / (datetime.now(UTC).strftime("%Y%m%dT%H%M%SZ-") + uuid4().hex[:8])
        )
        output.mkdir(parents=True, exist_ok=False)
        files = [Path(__file__), Path(__file__).parents[1] / "research/tradfi_monitor.py", args.config]
        write_json(
            output / "manifest.json",
            {
                "created_ms": now_ms(),
                "config": config,
                "python": sys.version,
                "websockets": websockets.__version__,
                "sources": {p.name: hashlib.sha256(p.read_bytes()).hexdigest() for p in files},
                "instruments": instruments,
                "public_api_only": True,
                "orders_enabled": False,
            },
        )
        write_json(base / "latest_run.json", {"output": str(output)})
        recorder = Recorder(output, config)
        asyncio.run(recorder.run())
        return 1 if recorder.lifecycle == "failed" else 0


if __name__ == "__main__":
    raise SystemExit(main())
