#!/usr/bin/env python3
"""Frozen single-component breakout diagnostics; no search or promotion."""

from __future__ import annotations

import argparse
import importlib.util
import json
from pathlib import Path

try:
    from scripts import run_research_stress as common
except ModuleNotFoundError as exc:
    if exc.name != "scripts":
        raise
    # Direct script invocation does not place repository root on sys.path.
    # Load the sibling by exact path without mutating the interpreter path.
    spec = importlib.util.spec_from_file_location(
        "research_partition_executor", Path(__file__).with_name("run_research_stress.py")
    )
    common = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(common)


def build_cases() -> list[dict]:
    cases = []
    for symbol in ("BTCUSDT", "ETHUSDT"):
        for variant in ("full", "without_entry", "without_exit"):
            for scenario, seed in [("identity", None), *[("price_noise", x) for x in common.SEEDS]]:
                for cost in (0, 5, 15):
                    cases.append(
                        {
                            "case_id": f"{symbol}-breakout-{variant}-{scenario}-{seed}-cost{cost}",
                            "symbol": symbol,
                            "family": "breakout",
                            "variant": variant,
                            "params": {
                                "entry_bars": 48,
                                "exit_bars": 12,
                                "confirmation_bars": 1,
                                "entry_gate_enabled": variant != "without_entry",
                                "exit_gate_enabled": variant != "without_exit",
                            },
                            "risk_budget": {"target_annual_vol": 0.10, "max_leverage": 1.5},
                            "scenario": scenario,
                            "seed": seed,
                            "extra_cost_bps": cost,
                        }
                    )
    return cases


def validate_plan(plan: dict) -> None:
    # Reuse the exact dataset identity contract, independently freeze recipes.
    for case in plan.get("cases", []):
        params = case.get("params", {})
        if any(type(params.get(key)) is not bool for key in ("entry_gate_enabled", "exit_gate_enabled")):
            raise ValueError("ablation gates must be strict booleans")
    surrogate = {**plan, "schema_version": "fixed-price-stress.v1", "cases": common.build_cases()}
    common.validate_plan(surrogate)
    if plan.get("schema_version") != "fixed-breakout-ablation.v1" or plan.get("cases") != build_cases():
        raise ValueError("plan does not match frozen single-component ablation matrix")


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--plan", type=Path, required=True)
    parser.add_argument("--symbol", choices=("BTCUSDT", "ETHUSDT"), required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args(argv)
    plan = json.loads(args.plan.read_text())
    validate_plan(plan)
    selected = [case for case in plan["cases"] if case["symbol"] == args.symbol]
    if len(selected) != 54:
        raise ValueError("exact 54-case partition required")
    return common.execute_partition(plan, symbol=args.symbol, target_vol=0.10, output=args.output, selected=selected)


if __name__ == "__main__":
    raise SystemExit(main())
