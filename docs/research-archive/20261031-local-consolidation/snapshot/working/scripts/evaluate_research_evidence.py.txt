#!/usr/bin/env python3
"""Re-evaluate explicit saved runs without replaying or promoting strategies."""

from __future__ import annotations

import argparse
import json
from pathlib import Path

from zenquant.research.reporting import write_recorded_reports


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--input", type=Path, action="append", required=True, help="Recorded repaired case directory; repeatable"
    )
    parser.add_argument(
        "--output", type=Path, required=True, help="New output directory; existing evidence is never overwritten"
    )
    args = parser.parse_args()
    try:
        summary = write_recorded_reports(args.input, args.output)
    except (OSError, ValueError) as exc:
        parser.error(str(exc))
    print(json.dumps({"output": str(args.output), "counts": summary["counts"], "live_eligible": False}))
    return 2 if summary["counts"]["invalid"] else 0


if __name__ == "__main__":
    raise SystemExit(main())
