import { expect, test } from "@playwright/test";

function panel(page, title) {
  return page.getByRole("article").filter({ has: page.getByText(title, { exact: true }) });
}

function metric(container, label) {
  return container.locator(".metric-cell").filter({ hasText: label });
}

async function usePositionSnapshot(page, request, overrides) {
  const response = await request.get("/api/state?symbol=BTCUSDT_PERP&tf=4h");
  expect(response.ok()).toBeTruthy();
  const snapshot = { ...(await response.json()), ...overrides };
  await page.route("**/api/state?**", (route) => route.fulfill({ json: snapshot }));
  // Keep position fixtures stable while the transport test uses real sockets.
  await page.routeWebSocket((url) => url.pathname === "/ws", (socket) => {
    socket.send(JSON.stringify(snapshot));
  });
}

function observeStateSockets(page) {
  const sockets = [];
  const frames = [];
  page.on("websocket", (socket) => {
    sockets.push(socket);
    if (new URL(socket.url()).pathname !== "/ws") return;
    socket.on("framereceived", ({ payload }) => frames.push(JSON.parse(String(payload))));
  });
  return { sockets, frames };
}

test("dashboard exposes operator decisions, risk and incomplete paper-live evidence", async ({ page }) => {
  await page.goto("/static/");

  for (const title of [
    "Current Thesis", "Live Overview", "Decision Path", "Exit & Position",
    "Sizing Breakdown", "4D State", "Risk Guards & Alpha", "Paper-Live Scorecard",
    "Price Curve", "Equity Curve", "Alert Panel", "Execution Log",
  ]) {
    await expect(panel(page, title)).toBeVisible();
  }
  await expect(panel(page, "Current Thesis").getByRole("heading")).not.toHaveText("");
  await expect(metric(panel(page, "Decision Path"), "1D bias").locator(".metric-value")).not.toHaveText("");
  await expect(metric(panel(page, "Decision Path"), "Risk flags")).toContainText("Leverage cap");
  await expect(metric(panel(page, "Live Overview"), "Total PnL").locator(".metric-value")).toHaveText(/\$/);
  await expect(panel(page, "Paper-Live Scorecard")).toContainText("runtime_not_live");
  await expect(panel(page, "Paper-Live Scorecard").getByText("Still Required", { exact: true })).toBeVisible();
  await expect(page.getByRole("button", { name: "Expand All", exact: true })).toBeVisible();
});

test("position, exit and chart overlays expose filled-position values", async ({ page, request }) => {
  await usePositionSnapshot(page, request, {
    last_price: 66000,
    position_state: { direction: 1, current_qty: 0.25, entry_price: 60000, trailing_stop: 57000, bars_held: 3 },
    latest_exit: {
      exit_type: "TRAILING_STOP", exit_urgency: "IMMEDIATE", stop_price: 57000,
      target_reduction_pct: 0.5, reason_code: "TRAILING_STOP", reason_codes: ["TRAILING_STOP"],
    },
  });
  await page.goto("/static/");

  await expect(page.getByTestId("position-status")).toHaveText("Long 0.2500 BTC");
  await expect(page.getByTestId("position-pnl")).toHaveText("Unrealized +10.00%");
  await expect(page.getByTestId("exit-decision")).toHaveText("Trailing Stop · Immediate");
  await expect(page.getByTestId("target-reduction")).toHaveText("50%");
  const legend = page.getByTestId("price-overlay-legend");
  await expect(metric(legend, "Long entry").locator(".metric-value")).toHaveText("$60,000.00");
  await expect(legend.getByTestId("trailing-stop")).toHaveText("$57,000.00");
  await expect(metric(panel(page, "Exit & Position"), "Entry price").locator(".metric-value")).toHaveText("$60,000.00");
});

test("flat positions show no active overlays or invented unrealized PnL", async ({ page, request }) => {
  await usePositionSnapshot(page, request, { position_state: null, latest_exit: null });
  await page.goto("/static/");

  await expect(page.getByTestId("position-status")).toHaveText("Flat");
  await expect(page.getByTestId("position-pnl")).toHaveCount(0);
  await expect(page.getByTestId("trailing-stop")).toHaveCount(0);
  await expect(page.getByTestId("price-overlay-legend")).toContainText("No active overlays");
  await expect(page.getByTestId("exit-decision")).toHaveText("No active exit");
  await expect(page.getByTestId("target-reduction")).toHaveText("0%");
});

test("dashboard receives continuing real demo websocket updates", async ({ page }) => {
  const { sockets, frames } = observeStateSockets(page);
  await page.goto("/static/");

  await expect.poll(() => frames.length).toBeGreaterThan(1);
  expect(frames.at(-1)).toMatchObject({ symbol: "BTCUSDT_PERP", tf: "4h", runtime_status: { mode: "demo" } });
  await expect.poll(() => new Set(frames.map((frame) => frame.runtime_status.last_market_event_ns)).size).toBeGreaterThan(1);
  expect(sockets.map((socket) => new URL(socket.url()).pathname)).toContain("/ws/indicators");
  await expect(page.getByTestId("stream-status")).toHaveText("LIVE");
  await expect(page.getByTestId("stream-update")).toHaveText("LIVE");
  await expect(metric(panel(page, "Decision Path"), "Data stream")).toContainText("Simulated feed active");
});

test("symbol and timeframe selection updates REST and websocket context", async ({ page }) => {
  const { sockets, frames } = observeStateSockets(page);
  await page.goto("/static/");
  await expect.poll(() => frames.length).toBeGreaterThan(0);

  const selectors = page.locator(".status-right .n-select");
  await selectors.nth(0).click();
  await page.getByText("ETHUSDT_PERP", { exact: true }).click();
  await expect(selectors.nth(0)).toHaveText("ETHUSDT_PERP");

  const selectedState = page.waitForResponse((response) => {
    const url = new URL(response.url());
    return url.pathname === "/api/state" && url.searchParams.get("symbol") === "ETHUSDT_PERP" && url.searchParams.get("tf") === "1d";
  });
  await selectors.nth(1).click();
  await page.getByText("1D", { exact: true }).click();
  expect((await selectedState).ok()).toBeTruthy();
  await expect(selectors.nth(1)).toHaveText("1D");
  await expect.poll(() => frames.at(-1)?.symbol + ":" + frames.at(-1)?.tf).toBe("ETHUSDT_PERP:1d");
  await expect.poll(() => sockets.filter((socket) => {
    const url = new URL(socket.url());
    return url.searchParams.get("symbol") === "ETHUSDT_PERP" && url.searchParams.get("tf") === "1d" && !socket.isClosed();
  }).length).toBe(2);
  await expect.poll(() => sockets.filter((socket) => new URL(socket.url()).searchParams.get("symbol") === "BTCUSDT_PERP").every((socket) => socket.isClosed())).toBe(true);
  await expect(panel(page, "Decision Path")).toBeVisible();
});
