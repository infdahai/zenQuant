"""Research wiring preserves causal warmup, cost feedback and evidence scope."""

import math
from copy import deepcopy

import pytest

from zenquant.research.research_runner import evaluate_candidate
from zenquant.research.run_evidence import identity_hash
from zenquant.research.suite_evaluation import paired_daily_excess

STEP = 14_400_000_000_000


def bundle():
    prices = [100 + 0.07 * i + math.sin(i) * 0.4 for i in range(240)]
    bars = [
        {
            "open_time_ns": i * STEP,
            "close_time_ns": (i + 1) * STEP - 1_000_000,
            "open": price,
            "high": price + 1,
            "low": price - 1,
            "close": price,
        }
        for i, price in enumerate(prices)
    ]
    return {
        "bars": bars,
        "warmup_count": 180,
        "funding": [],
        "manifest": {"window_data_hash": "fixture", "start_ns": 180 * STEP, "end_ns_exclusive": 240 * STEP},
    }


def test_runner_keeps_warmup_out_of_pnl_and_marks_historical_diagnostic():
    evidence, report = evaluate_candidate(bundle(), "trend", {}, window="dev", code_hash="fixture")
    assert evidence["curve"][0] == {"time_ns": 180 * STEP, "equity": 100000.0}
    assert len(evidence["audit"]["intents"]) == 60
    assert evidence["fills"][0]["time_ns"] > evidence["audit"]["intents"][0]["cutoff_time_ns"]
    assert report["evidence_role"] == "development_selection"
    assert report["rankable"] is False
    assert report["live_eligible"] is False
    assert report["identity"]["candidate_id"] == evidence["identity"]["candidate_id"]


def test_extra_cost_enters_each_fill_and_changes_nav_and_later_targets():
    a, ra = evaluate_candidate(bundle(), "vol_target", {}, window="train", code_hash="fixture")
    b, rb = evaluate_candidate(bundle(), "vol_target", {}, window="train", code_hash="fixture", extra_cost_bps=15)
    assert ra["metrics"]["net_return"]["value"] > rb["metrics"]["net_return"]["value"]
    assert b["fills"][0]["fee"] > a["fills"][0]["fee"]
    assert b["audit"]["intents"][1]["target_qty"] != a["audit"]["intents"][1]["target_qty"]
    assert ra["comparison_contract"] != rb["comparison_contract"]


def test_candidate_identity_ignores_cost_but_run_identity_binds_it():
    a, _ = evaluate_candidate(bundle(), "trend", {}, window="train", code_hash="fixture")
    b, _ = evaluate_candidate(bundle(), "trend", {}, window="train", code_hash="fixture", extra_cost_bps=5)
    assert a["identity"]["candidate_id"] == b["identity"]["candidate_id"]
    assert a["identity"]["run_id"] != b["identity"]["run_id"]


def test_observed_2026_is_never_labelled_fresh_oos():
    _, report = evaluate_candidate(bundle(), "cash", {}, window="observed_2026", code_hash="fixture")
    assert report["evidence_role"] == "observed_historical"
    with pytest.raises(ValueError, match="window"):
        evaluate_candidate(bundle(), "cash", {}, window="fresh_test", code_hash="fixture")


def test_constant_1x_reference_cannot_enter_risk_matched_incremental_comparison():
    _, vol = evaluate_candidate(bundle(), "vol_target", {}, window="train", code_hash="fixture")
    _, hold = evaluate_candidate(bundle(), "buy_hold_1x", {}, window="train", code_hash="fixture")
    result = paired_daily_excess(vol, hold, block_lengths=[1, 2, 3], repetitions=10, min_samples=3)
    assert result["status"] == "NOT_EVALUABLE"
    assert result["reason"] == "CONTRACT_MISMATCH"
    assert vol["comparison_contract"]["exposure_policy"] != hold["comparison_contract"]["exposure_policy"]


def test_rule_family_and_volatility_baseline_keep_the_same_risk_budget_contract():
    _, trend = evaluate_candidate(bundle(), "trend", {}, window="dev", code_hash="fixture")
    _, vol = evaluate_candidate(bundle(), "vol_target", {}, window="dev", code_hash="fixture")
    assert trend["comparison_contract"] == vol["comparison_contract"]


def test_comparison_binds_actual_warmup_history_and_keeps_it_out_of_pnl():
    source = regime_bundle()
    evidence, report = evaluate_candidate(source, "cash", {}, window="dev", code_hash="fixture")
    contract = report["comparison_contract"]["warmup_contract"]
    assert contract == {
        "strategy_bars": 180,
        "regime_bars": 720,
        "actual_strategy_bars": 180,
        "actual_regime_bars": 720,
        "regime_input_hash": identity_hash({"bars": source["regime_bars"]}),
    }
    assert evidence["identity"]["warmup_contract"] == contract
    report["comparison_contract"]["warmup_contract"]["actual_strategy_bars"] = 0
    assert evidence["identity"]["warmup_contract"]["actual_strategy_bars"] == 180
    reduced = deepcopy(source)
    reduced["bars"] = reduced["bars"][1:]
    reduced["warmup_count"] -= 1
    _, second = evaluate_candidate(reduced, "cash", {}, window="dev", code_hash="fixture")
    assert second["comparison_contract"] != report["comparison_contract"]
    assert second["comparison_contract"]["warmup_contract"]["actual_strategy_bars"] == 179


def test_mutating_input_manifest_cannot_change_issued_evidence_identity():
    source = bundle()
    source["manifest"]["nested"] = {"revision": "original"}
    evidence, report = evaluate_candidate(source, "trend", {}, window="train", code_hash="fixture")
    original = deepcopy(evidence["identity"])
    source["manifest"]["window_data_hash"] = "changed"
    source["manifest"]["nested"]["revision"] = "changed"
    assert evidence["identity"] == original
    assert report["identity"] == original
    unhashed = {key: value for key, value in original.items() if key != "run_id"}
    assert original["run_id"] == identity_hash(unhashed)


def test_mutating_returned_config_cannot_change_next_candidate_or_other_returned_artifact():
    before, before_report = evaluate_candidate(bundle(), "trend", {}, window="train", code_hash="fixture")
    pristine_evidence, pristine_report = deepcopy(before), deepcopy(before_report)
    before_report["comparison_contract"]["risk_budget"]["max_leverage"] = 9.0
    before_report["comparison_contract"]["cost"]["slippage_bps"] = 99.0
    assert before == pristine_evidence
    before["identity"]["candidate"]["risk_budget"]["target_annual_vol"] = 0.9
    before["policy"]["max_drawdown"] = 0.9
    after, after_report = evaluate_candidate(bundle(), "trend", {}, window="train", code_hash="fixture")
    assert after == pristine_evidence
    assert after_report == pristine_report


def test_runner_adds_risk_observations_without_overwriting_primary_metrics_or_passing_gates():
    evidence, report = evaluate_candidate(bundle(), "trend", {}, window="dev", code_hash="fixture")
    assert report["risk_diagnostics"]["capital_risk_gate"]["status"] == "NOT_EVALUABLE"
    assert report["metrics"]["net_return"]["value"] == pytest.approx(evidence["curve"][-1]["equity"] / 100000 - 1)
    assert report["metrics"]["time_weighted_abs_exposure"]["value"] > 0.0
    assert report["gates"]["R2"]["status"] != "PASS"
    assert report["candidate_status"]["status"] in {"DIAGNOSTIC", "BLOCKED"}
    assert report["metrics"]["max_drawdown"]["source_hash"] == report["identity"]["run_id"]
    assert report["metrics"]["max_drawdown"]["policy_threshold"] == {"operator": "<=", "value": 0.15}
    assert report["metrics"]["net_return"]["window"]["start_ns"] == 180 * STEP


def ruin_bundle(recover_at_open):
    prices = (100.0, 100.0, 300.0 if recover_at_open else 100.0)
    return {
        "bars": [
            {
                "open_time_ns": index * STEP,
                "close_time_ns": (index + 1) * STEP - 1_000_000,
                "open": price,
                "high": price,
                "low": price,
                "close": price,
            }
            for index, price in enumerate(prices)
        ],
        "warmup_count": 0,
        "funding": [{"funding_time_ns": 2 * STEP, "funding_rate": 1.1, "mark_price": 100.0}],
        "manifest": {"window_data_hash": "genuine-funding-ruin", "start_ns": 0, "end_ns_exclusive": 3 * STEP},
    }


@pytest.mark.parametrize("recover_at_open", [False, True])
def test_actual_funding_ruin_blocks_candidate_even_when_same_time_open_recovers(recover_at_open):
    evidence, report = evaluate_candidate(
        ruin_bundle(recover_at_open), "buy_hold_1x", {}, window="dev", code_hash="fixture", initial_equity=100
    )
    assert any(row["equity"] < 0 for row in evidence["accounting"]["rows"])
    if recover_at_open:
        assert all(row["equity"] > 0 for row in evidence["curve"])
        assert report["metrics"]["net_return"]["value"] > 0
    assert report["ruin"] == {"occurred": True, "first_time_ns": 2 * STEP}
    assert report["gates"]["R2"] == {"status": "FAIL", "reason": "RUIN"}
    assert {"gate": "R2", "status": "FAIL", "reason": "RUIN"} in report["blockers"]
    assert sum(row.get("gate") == "R2" for row in report["blockers"]) == 1
    assert report["candidate_status"]["status"] == "BLOCKED"
    assert report["development_screen_passed"] is False
    assert report["rankable"] is False
    assert report["live_eligible"] is False
    assert report["metrics"]["net_return"]["status"] == "NOT_EVALUABLE"
    assert report["metrics"]["daily_sharpe"]["reason"] == "RUIN"
    assert report["regime_diagnostics"]["attribution"]["status"] == "FAIL"


def regime_bundle():
    bars = []
    for index in range(750):
        price = 100.0 + index * 0.1
        bars.append(
            {
                "open_time_ns": index * STEP,
                "close_time_ns": (index + 1) * STEP - 1_000_000,
                "open": price,
                "high": price,
                "low": price,
                "close": price,
            }
        )
    return {
        "bars": bars[540:],
        "regime_bars": bars,
        "warmup_count": 180,
        "funding": [],
        "manifest": {"window_data_hash": "regime-fixture", "start_ns": 720 * STEP, "end_ns_exclusive": 750 * STEP},
    }


def test_regime_labels_use_day_start_and_additional_history_without_extending_trading_window():
    evidence, report = evaluate_candidate(regime_bundle(), "cash", {}, window="dev", code_hash="fixture")
    assert evidence["curve"][0]["time_ns"] == 720 * STEP
    diagnostic = report["regime_diagnostics"]
    assert [row["day_start_ns"] for row in diagnostic["labels"]["labels"]] == [
        row["time_ns"] - 6 * STEP for row in report["daily_returns"]
    ]
    assert all(row["regime"] == "bull" for row in diagnostic["labels"]["labels"])
    assert diagnostic["attribution"]["coverage"]["unknown_days"] == 0
    assert report["gates"]["R3"]["status"] != "PASS"


def test_short_regime_history_stays_unknown_instead_of_using_future_days():
    data = regime_bundle()
    del data["regime_bars"]
    _, report = evaluate_candidate(data, "cash", {}, window="dev", code_hash="fixture")
    assert report["regime_diagnostics"]["attribution"]["coverage"]["unknown_days"] == len(report["daily_returns"])
    assert report["regime_diagnostics"]["labels"]["status"] == "NOT_EVALUABLE"


def test_label_only_input_change_changes_evidence_identity_without_changing_trade_targets():
    data = regime_bundle()
    first, _ = evaluate_candidate(data, "cash", {}, window="dev", code_hash="fixture")
    data["regime_bars"][0] = {**data["regime_bars"][0], "available_at_ns": 999 * STEP}
    second, report = evaluate_candidate(data, "cash", {}, window="dev", code_hash="fixture")
    assert first["audit"]["intents"] == second["audit"]["intents"]
    assert first["identity"]["run_id"] != second["identity"]["run_id"]
    assert report["regime_diagnostics"]["labels"]["labels"][0]["regime"] == "UNKNOWN"


def test_symbol_policy_and_account_scope_bind_candidate_run_and_comparison():
    source = bundle()
    source.update(symbol="ETHUSDT")
    source["manifest"]["symbol"] = "ETHUSDT_PERP"
    actual, report = evaluate_candidate(
        source, "trend", {}, window="dev", code_hash="fixture", inventory_policy="actual_filled_inventory_v1"
    )
    shared, shared_report = evaluate_candidate(source, "trend", {}, window="dev", code_hash="fixture")
    btc, _ = evaluate_candidate(bundle(), "trend", {}, window="dev", code_hash="fixture")
    assert all(fill["symbol"] == "ETHUSDT" for fill in actual["fills"])
    for identity in (actual["identity"], actual["identity"]["candidate"], report["comparison_contract"]):
        assert identity["symbol"] == "ETHUSDT"
        assert identity["inventory_policy"] == "actual_filled_inventory_v1"
        assert identity["account_scope"] == "independent_single_symbol_account"
    assert actual["identity"]["candidate_id"] != shared["identity"]["candidate_id"]
    assert actual["identity"]["candidate_id"] != btc["identity"]["candidate_id"]
    assert actual["identity"]["run_id"] != shared["identity"]["run_id"]
    assert report["comparison_contract"] != shared_report["comparison_contract"]
    assert report["gates"]["E3"]["status"] == "FAIL"
    assert shared_report["gates"]["E3"]["status"] == "FAIL"
    assert report["optimizer_feedback_eligible"] is False
    assert report["rankable"] is False


@pytest.mark.parametrize("manifest_symbol", ["BTCUSDT_PERP", None])

    source = bundle()

def test_optional_research_budget_defaults_and_lower_budget_are_explicit_and_detached():
    default, default_report = evaluate_candidate(
        bundle(), "vol_target", {}, window="dev", code_hash="fixture", inventory_policy="actual_filled_inventory_v1"
    )
    explicit, _ = evaluate_candidate(
        bundle(),
        "vol_target",
        {},
        window="dev",
        code_hash="fixture",
        inventory_policy="actual_filled_inventory_v1",
        risk_budget={"target_annual_vol": 0.10, "max_leverage": 1.5},
    )
    assert explicit == default
    budget = {"target_annual_vol": 0.05, "max_leverage": 1.5}
    low, report = evaluate_candidate(
        bundle(),
        "vol_target",
        {},
        window="dev",
        code_hash="fixture",
        inventory_policy="actual_filled_inventory_v1",
        risk_budget=budget,
    )
    budget["target_annual_vol"] = 0.09
    assert low["identity"]["candidate"]["risk_budget"] == {"target_annual_vol": 0.05, "max_leverage": 1.5}
    assert low["identity"]["research_intent_risk_budget"] == report["comparison_contract"]["risk_budget"]
    assert low["identity"]["candidate_id"] != default["identity"]["candidate_id"]
    assert low["identity"]["run_id"] != default["identity"]["run_id"]
    assert low["identity"]["engine_identity"]["targets_hash"] != default["identity"]["engine_identity"]["targets_hash"]
    assert report["comparison_contract"] != default_report["comparison_contract"]
    assert low["resolved_config"]["risk"]["max_leverage"] == default["resolved_config"]["risk"]["max_leverage"] == 1.5
    assert report["gates"]["E3"]["status"] == "FAIL" and report["optimizer_feedback_eligible"] is False

@pytest.mark.parametrize("family", ["vol_target", "buy_hold_1x"])
def test_lower_intent_cap_leaves_shared_risk_config_unchanged(family):
    evidence, _ = evaluate_candidate(
        bundle(),
        family,
        {},
        window="dev",
        code_hash="fixture",
        risk_budget={"target_annual_vol": 0.05, "max_leverage": 0.25},
    )
    assert max(row["target_leverage"] for row in evidence["audit"]["intents"]) <= 0.25
    assert evidence["resolved_config"]["risk"]["max_leverage"] == 1.5
