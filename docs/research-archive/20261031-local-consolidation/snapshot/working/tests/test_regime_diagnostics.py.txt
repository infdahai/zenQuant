"""Causal labeling and hand-calculated regime attribution contracts."""

from __future__ import annotations

import copy
import json

import pytest

BAR = 14_400_000_000_000
DAY = 6 * BAR
MS = 1_000_000


def call(name, *args):
    from zenquant.research import regime_diagnostics

    return getattr(regime_diagnostics, name)(*args)


def bars(count=720, direction=1):
    return [
        {
            "open_time_ns": index * BAR,
            "close_time_ns": (index + 1) * BAR - MS,
            "close": 1000.0 + direction * index,
        }
        for index in range(count)
    ]


@pytest.mark.parametrize("direction,expected", [(1, "bull"), (-1, "bear"), (0, "neutral")])
def test_fixed_720_bar_level_and_180_bar_momentum_direction(direction, expected):
    rows = bars(direction=direction)
    result = call("assign_daily_regimes", rows, [120 * DAY])
    label = result["labels"][0]
    assert label["regime"] == expected
    assert label["source_cutoff_ns"] == 120 * DAY - MS
    assert label["availability_assumed"] is True
    assert result["method"]["sma_bars"] == 720
    assert result["method"]["momentum_bars"] == 180
    assert result["live_eligible"] is False


def test_conflicting_level_and_momentum_are_neutral():
    rows = bars(direction=0)
    rows[-181]["close"] = 2000.0
    rows[-1]["close"] = 1500.0
    assert call("assign_daily_regimes", rows, [120 * DAY])["labels"][0]["regime"] == "neutral"


def test_future_mutation_and_prefix_do_not_change_past_labels_or_inputs():
    rows = bars(count=738)
    days = [120 * DAY, 121 * DAY, 122 * DAY]
    before = copy.deepcopy(rows)
    labels = call("assign_daily_regimes", rows, days)["labels"]
    assert rows == before
    for row in rows[726:]:
        row["close"] *= 10
    assert call("assign_daily_regimes", rows, days)["labels"][:2] == labels[:2]
    assert call("assign_daily_regimes", rows[:726], days[:2])["labels"] == labels[:2]


def test_explicit_availability_at_cutoff_is_allowed_but_later_is_unknown():
    rows = bars()
    for row in rows:
        row["available_at_ns"] = row["close_time_ns"] + MS
    result = call("assign_daily_regimes", rows, [120 * DAY])
    assert result["labels"][0]["regime"] == "bull"
    assert result["labels"][0]["availability_assumed"] is False
    rows[-1]["available_at_ns"] = 120 * DAY + 1
    result = call("assign_daily_regimes", rows, [120 * DAY])
    assert result["labels"][0]["regime"] == "UNKNOWN"
    assert "AVAILABLE" in result["labels"][0]["reason"]


def test_close_exactly_at_cutoff_is_excluded_even_if_available():
    rows = bars(count=726)
    for row in rows:
        row["close_time_ns"] += MS
        row["available_at_ns"] = row["close_time_ns"]
    rows[719]["close"] = 1.0
    labels = call("assign_daily_regimes", rows, [120 * DAY, 121 * DAY])["labels"]
    assert labels[0]["regime"] == "UNKNOWN"  # Only 719 strictly earlier bars.
    assert labels[1]["regime"] == "bull"
    assert labels[1]["source_cutoff_ns"] == 121 * DAY - BAR


@pytest.mark.parametrize("damage", ["too_short", "internal_gap", "missing_latest", "unconfirmed"])
def test_incomplete_required_bar_window_is_unknown(damage):
    rows = bars()
    if damage == "too_short":
        rows = rows[1:]
    elif damage == "internal_gap":
        del rows[300]
    elif damage == "missing_latest":
        rows.pop()
    else:
        rows[-1]["confirmed"] = False
    result = call("assign_daily_regimes", rows, [120 * DAY])
    assert result["status"] == "NOT_EVALUABLE"
    assert result["labels"][0]["regime"] == "UNKNOWN"
    assert result["missing"]


def test_late_historical_gap_does_not_fall_back_to_older_contiguous_window():
    rows = bars(count=1440, direction=0)
    rows[-20]["available_at_ns"] = 241 * DAY
    result = call("assign_daily_regimes", rows, [240 * DAY])
    assert result["labels"][0]["regime"] == "UNKNOWN"


@pytest.mark.parametrize("damage", ["duplicate", "unordered", "nan", "bad_interval", "available_before_close"])
def test_invalid_bars_return_json_safe_invalid(damage):
    rows = bars()
    if damage == "duplicate":
        rows.insert(10, copy.deepcopy(rows[10]))
    elif damage == "unordered":
        rows[0], rows[1] = rows[1], rows[0]
    elif damage == "nan":
        rows[0]["close"] = float("nan")
    elif damage == "bad_interval":
        rows[0]["close_time_ns"] -= 1
    else:
        rows[0]["available_at_ns"] = rows[0]["close_time_ns"] - 1
    result = call("assign_daily_regimes", rows, [120 * DAY])
    assert result["status"] == "INVALID"
    json.dumps(result, allow_nan=False)


@pytest.mark.parametrize("days", [[DAY + 1], [DAY, DAY], [2 * DAY, DAY]])
def test_invalid_day_boundaries_or_order_are_rejected(days):
    assert call("assign_daily_regimes", [], days)["status"] == "INVALID"


def report(values):
    return {
        "identity": {"candidate_id": "candidate"},
        "daily_returns": [{"time_ns": (index + 1) * DAY, "value": value} for index, value in enumerate(values)],
    }


def labels(regimes):
    return [
        {
            "day_start_ns": index * DAY,
            "regime": regime,
            "source_cutoff_ns": index * DAY - MS,
            "availability_assumed": False,
            "reason": "RULE",
        }
        for index, regime in enumerate(regimes)
    ]


def test_return_end_t_uses_start_t_minus_one_day_and_hand_known_attribution():
    result = call("attribute_regimes", report([0.1, 0.2, -0.1, -0.05]), labels(["bull", "bull", "bear", "bull"]))
    bull = result["by_regime"]["bull"]
    assert bull["days"] == 3
    assert bull["compound_return"] == pytest.approx(1.1 * 1.2 * 0.95 - 1)
    assert bull["arithmetic_contribution"] == pytest.approx(0.25)
    assert bull["worst_contiguous_segment_return"] == pytest.approx(-0.05)
    assert bull["contiguous_segments"] == 2
    assert result["by_regime"]["bear"]["compound_return"] == pytest.approx(-0.1)
    assert result["by_regime"]["neutral"]["compound_return"] is None
    assert result["by_regime"]["neutral"]["status"] == "NOT_EVALUABLE"
    assert result["compound_return_is_tradable_nav"] is False
    assert result["live_eligible"] is False
    assert "max_drawdown" not in bull


def test_worst_segment_is_full_contiguous_run_not_drawdown_or_worst_day():
    result = call("attribute_regimes", report([-0.1, 0.2, -0.02]), labels(["bull", "bull", "bear"]))
    assert result["by_regime"]["bull"]["worst_contiguous_segment_return"] == pytest.approx(0.08)


def test_missing_labels_keep_returns_in_unknown_with_specific_missing_days():
    result = call("attribute_regimes", report([0.1, -0.2]), labels(["bull"]))
    assert result["status"] == "NOT_EVALUABLE"
    assert result["by_regime"]["UNKNOWN"]["days"] == 1
    assert result["by_regime"]["UNKNOWN"]["compound_return"] == pytest.approx(-0.2)
    assert any(item.get("day_start_ns") == DAY for item in result["missing"])


def test_assignment_object_and_label_sequence_are_equivalent():
    sequence = labels(["bull", "UNKNOWN"])
    wrapped = {"status": "NOT_EVALUABLE", "labels": sequence}
    assert call("attribute_regimes", report([0.1, -0.2]), sequence) == call(
        "attribute_regimes", report([0.1, -0.2]), wrapped
    )


def test_gap_in_daily_returns_is_not_evaluable_and_does_not_join_segments():
    evidence = report([0.1, -0.9, 0.2])
    del evidence["daily_returns"][1]
    result = call("attribute_regimes", evidence, labels(["bull", "bull", "bull"]))
    assert result["status"] == "NOT_EVALUABLE"
    assert result["by_regime"]["bull"]["contiguous_segments"] == 2
    assert result["by_regime"]["bull"]["worst_contiguous_segment_return"] == pytest.approx(0.1)
    assert result["daily_sampling_complete"] is False
    assert result["missing"]


@pytest.mark.parametrize(
    "damage", ["duplicate_return", "nan_return", "duplicate_label", "future_cutoff", "unknown_regime"]
)
def test_invalid_attribution_evidence_does_not_produce_pass(damage):
    evidence, sequence = report([0.1, 0.2]), labels(["bull", "bear"])
    if damage == "duplicate_return":
        evidence["daily_returns"][1]["time_ns"] = DAY
    elif damage == "nan_return":
        evidence["daily_returns"][0]["value"] = float("nan")
    elif damage == "duplicate_label":
        sequence.append(copy.deepcopy(sequence[0]))
    elif damage == "future_cutoff":
        sequence[1]["source_cutoff_ns"] = DAY
    else:
        sequence[1]["regime"] = "optimistic"
    result = call("attribute_regimes", evidence, sequence)
    assert result["status"] == "INVALID"
    json.dumps(result, allow_nan=False)


def test_ruin_cannot_be_rendered_as_normal_regime_profit_and_empty_is_missing():
    assert call("attribute_regimes", report([-1.0]), labels(["bear"]))["status"] == "FAIL"
    result = call("attribute_regimes", report([]), [])
    assert result["status"] == "NOT_EVALUABLE"
    assert result["missing"]


def test_source_sampling_gap_is_retained_when_remaining_returns_look_contiguous():
    evidence = report([0.1, 0.2])
    evidence["daily_sampling"] = {"missing_midnights": [0], "valid_return_count": 2}
    result = call("attribute_regimes", evidence, labels(["bull", "bull"]))
    assert result["status"] == "NOT_EVALUABLE"
    assert result["daily_sampling_complete"] is False
    assert any(item.get("time_ns") == 0 for item in result["missing"])


def test_ruin_flag_cannot_be_hidden_by_pre_ruin_only_return_series():
    evidence = report([0.1])
    evidence["ruin"] = {"occurred": True}
    assert call("attribute_regimes", evidence, labels(["bull"]))["status"] == "FAIL"
