"""Protect causality and evidence boundaries of the independent exploration lab."""

from __future__ import annotations

import copy

import numpy as np
import pandas as pd
import pytest

from zenquant.research.frontier_lab.common import (
    causal_features,
    next_open_labels,
    ridge_predict,
    split_masks,
    validate_bars,
    validate_result,
)


def bars(n: int = 80, start: str = "2023-12-20") -> pd.DataFrame:
    times = pd.date_range(start, periods=n, freq="4h", tz="UTC").as_unit("ns").asi8
    prices = np.exp(np.arange(n) * 0.001) * 100
    return pd.DataFrame(
        {
            "open_time_ns": times,
            "close_time_ns": times + 4 * 3600 * 10**9 - 10**6,
            "open": prices,
            "high": prices * 1.01,
            "low": prices * 0.99,
            "close": prices * 1.001,
            "volume": np.arange(n) + 1,
            "taker_buy_base": (np.arange(n) + 1) * 0.6,
        }
    )


def result() -> dict:
    return {
        "direction_id": "d01",
        "hypothesis": "h",
        "evidence_type": "synthetic_mechanism",
        "method": "m",
        "metrics": {},
        "verdict": "inconclusive",
        "limitations": ["synthetic"],
        "falsification": "f",
        "next_experiment": "n",
    }


def test_features_do_not_change_when_future_prices_change() -> None:
    before = bars()
    after = before.copy()
    after.loc[60:, ["close", "open", "high", "low"]] *= 3
    pd.testing.assert_frame_equal(causal_features(before).iloc[:60], causal_features(after).iloc[:60])


def test_no_same_close_fill_and_training_label_purging() -> None:
    frame = bars()
    labels = next_open_labels(frame)
    assert labels["return"].iloc[0] == pytest.approx(frame["open"].iloc[2] / frame["open"].iloc[1] - 1)
    train, dev = split_masks(frame, labels["label_end_ns"])
    boundary = pd.Timestamp("2024-01-01", tz="UTC").value
    assert (labels.loc[train, "label_end_ns"] < boundary).all()
    crossing = (frame["close_time_ns"] < boundary) & (labels["label_end_ns"] >= boundary)
    assert crossing.any() and not train[crossing].any() and not dev[crossing].any()


@pytest.mark.parametrize("problem", ["duplicate", "gap", "future", "nan", "negative", "unordered"])
def test_bad_or_acceptance_inputs_fail_closed(problem: str) -> None:
    frame = bars()
    if problem == "duplicate":
        frame.loc[1, "open_time_ns"] = frame.loc[0, "open_time_ns"]
    elif problem == "gap":
        frame = frame.drop(index=10)
    elif problem == "future":
        frame = bars(start="2026-01-01")
    elif problem == "nan":
        frame.loc[5, "close"] = np.nan
    elif problem == "negative":
        frame.loc[5, "open"] = -1
    elif problem == "unordered":
        frame = frame.iloc[::-1]
    with pytest.raises(ValueError):
        validate_bars(frame)


def test_ridge_preprocessing_fits_only_training_rows() -> None:
    x = np.arange(20, dtype=float).reshape(-1, 1)
    y = x[:, 0] * 0.5
    first = ridge_predict(x, y, np.array([[21.0]]))
    with_outlier = ridge_predict(x, y, np.array([[21.0], [1e9]]))
    assert first[0] == with_outlier[0]


def test_unknown_evidence_type_or_eligibility_is_rejected() -> None:
    good = result()
    validate_result(good, "d01")
    for key, value in [
        ("evidence_type", "proven_alpha"),
        ("live_eligible", True),
        ("rankable", True),
        ("promotion_eligible", True),
        ("limitations", []),
        ("direction_id", "d02"),
    ]:
        bad = {**copy.deepcopy(good), key: value}
        with pytest.raises(ValueError):
            validate_result(bad, "d01")


def test_nonfinite_evidence_cannot_be_saved() -> None:
    bad = {**result(), "metrics": {"return": float("nan")}}
    with pytest.raises(ValueError):
        validate_result(bad, "d01")
