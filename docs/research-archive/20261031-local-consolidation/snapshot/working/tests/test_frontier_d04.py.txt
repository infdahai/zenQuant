"""Guard causal boundaries, target gradients and representation collapse."""

import numpy as np
import pandas as pd
import torch

from zenquant.research.frontier_lab.experiments import d04_jepa as experiment


def _bars():
    times = pd.date_range("2023-12-05", periods=400, freq="4h", tz="UTC").as_unit("ns").asi8
    close = 100 * np.exp(np.sin(np.arange(400) / 13) * 0.02 + np.arange(400) * 0.0001)
    return pd.DataFrame(
        {
            "open_time_ns": times,
            "close_time_ns": times + 4 * 3600 * 10**9 - 1,
            "open": close * 0.999,
            "close": close,
            "high": close * 1.01,
            "low": close * 0.99,
            "volume": 100 + np.arange(400) % 17,
        }
    )


def test_windows_and_future_labels_are_purged_at_train_boundary():
    frame = _bars()
    data = experiment.build_samples(frame)
    boundary = pd.Timestamp("2024-01-01", tz="UTC").value
    assert data["train"].sum() > 10
    assert data["dev"].sum() > 10
    assert (data["training_end_ns"][data["train"]] < boundary).all()
    assert (data["window_start_ns"][data["dev"]] >= boundary).all()
    changed = frame.copy()
    changed.loc[changed.close_time_ns >= boundary, "close"] *= 1.5
    revised = experiment.build_samples(changed)
    np.testing.assert_array_equal(data["x"][data["train"]], revised["x"][revised["train"]])
    np.testing.assert_array_equal(data["future"][data["train"]], revised["future"][revised["train"]])
    np.testing.assert_array_equal(data["y"][data["train"]], revised["y"][revised["train"]])


def test_dev_purge_includes_rolling_feature_lookbacks():
    frame = _bars()
    data = experiment.build_samples(frame)
    boundary = pd.Timestamp("2024-01-01", tz="UTC").value
    changed = frame.copy()
    changed.loc[changed.close_time_ns < boundary, "close"] *= 1.5
    revised = experiment.build_samples(changed)
    np.testing.assert_array_equal(data["x"][data["dev"]], revised["x"][revised["dev"]])


def test_jepa_target_has_no_gradients_and_encoder_freezes():
    torch.set_num_threads(1)
    model = experiment.JEPA(8, 16)
    x = torch.arange(64, dtype=torch.float32).reshape(8, 8) / 64
    loss = experiment.jepa_loss(model, x, x + 0.1)
    loss.backward()
    assert all(parameter.grad is None for parameter in model.target.parameters())
    assert any(parameter.grad is not None for parameter in model.encoder.parameters())
    before = [parameter.detach().clone() for parameter in model.encoder.parameters()]
    embedded = experiment.freeze_encode(model.encoder, x.numpy())
    assert embedded.shape == (8, 16)
    assert all(not parameter.requires_grad for parameter in model.encoder.parameters())
    for previous, current in zip(before, model.encoder.parameters(), strict=True):
        torch.testing.assert_close(previous, current)


def test_collapse_detector_rejects_constant_and_accepts_variable_control():
    assert experiment.collapse_diagnostics(np.ones((100, 16)))["collapsed"]
    positive = np.random.default_rng(17).normal(size=(100, 16))
    diagnostic = experiment.collapse_diagnostics(positive)
    assert not diagnostic["collapsed"]
    assert diagnostic["effective_rank"] > 10


def test_variance_regularizer_penalizes_constant_representation():
    constant = torch.zeros((32, 16))
    variable = torch.from_numpy(np.random.default_rng(17).normal(size=(32, 16))).float() * 2
    assert experiment.variance_penalty(constant) > experiment.variance_penalty(variable)
