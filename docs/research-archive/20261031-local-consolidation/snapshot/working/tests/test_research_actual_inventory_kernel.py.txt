"""Research inventory policy preserves execution rules and exposes production divergence."""

from dataclasses import replace

import pytest

import test_execution_router as router_tests
from zenquant.execution import ExecutionKernel, ExecutionKernelConfig, SimulatedGateway
from zenquant.infra import InMemoryJetStream
from zenquant.research.actual_inventory_kernel import ResearchActualInventoryKernel, inventory_policy_metadata
from zenquant.risk import RiskOverlay


def _run(kernel_type, targets, *, partial_dust=False):
    bus = InMemoryJetStream()
    risk = RiskOverlay(bus)
    gateway = SimulatedGateway()
    original = gateway.submit_order
    if partial_dust:

        def submit(order, **kwargs):
            return original(replace(order, quantity=order.quantity - 5e-7), **kwargs)

        gateway.submit_order = submit
    config = ExecutionKernelConfig(execution_authority_mode="target_tracking_v2")
    kernel = kernel_type(bus, gateway, risk, config)
    kernel._last_price["BTCUSDT_PERP"] = 100.0
    events = []
    for subject in ("exec.order.submit", "exec.order.report", "exec.order.audit"):
        bus.subscribe(subject, lambda topic, payload: events.append((topic, payload)))
    for target in targets:
        router_tests.TargetTrackingV2Tests._publish_approved(bus, target)
    return kernel, risk, events


def test_only_target_tracking_v2_is_allowed():
    bus = InMemoryJetStream()
    with pytest.raises(ValueError, match="target_tracking_v2"):
        ResearchActualInventoryKernel(bus, SimulatedGateway(), RiskOverlay(bus))


def test_no_dust_exact_event_and_execution_state_parity():
    targets = [
        router_tests.TargetTrackingV2Tests._target(qty=q, cutoff_ns=i + 2)
        for i, q in enumerate([0.75, 1.25, 0.5, -0.5, 0])
    ]
    shared, shared_risk, shared_events = _run(ExecutionKernel, targets)
    research, research_risk, research_events = _run(ResearchActualInventoryKernel, targets)
    assert research_events == shared_events
    assert research.positions == shared.positions
    assert research.purity_counters == shared.purity_counters
    assert research._completed_decisions == shared._completed_decisions
    assert research._remaining_target_delta == shared._remaining_target_delta
    assert research_risk._cash == shared_risk._cash
    assert research_risk._actual_positions == shared_risk._actual_positions


@pytest.mark.parametrize("sign", [1, -1])
def test_partial_fill_dust_is_retained_in_kernel_risk_and_audit(sign):
    targets = [
        router_tests.TargetTrackingV2Tests._target(qty=sign * 0.5, cutoff_ns=2),
        router_tests.TargetTrackingV2Tests._target(qty=0, cutoff_ns=3),
    ]
    shared, _, _ = _run(ExecutionKernel, targets, partial_dust=True)
    research, risk, events = _run(ResearchActualInventoryKernel, targets, partial_dust=True)
    reports = [payload for topic, payload in events if topic == "exec.order.report"]
    actual = sum(report.side * report.filled_qty for report in reports)
    assert 0 < abs(actual) <= 1e-6
    assert shared.positions[targets[0].symbol] == 0
    assert research.positions[targets[0].symbol] == actual
    assert risk._actual_positions[targets[0].symbol] == actual
    assert events[-1][1].position_after_fill == actual
    assert research.quantity_epsilon == 1e-6


def test_policy_guard_fails_closed_on_shared_source_drift(monkeypatch):
    metadata = inventory_policy_metadata()
    assert metadata["production_parity"] == "FAIL"
    assert metadata["inventory_policy"] == "actual_filled_inventory_v1"
    assert len(metadata["shared_method_source_sha256"]) == 64
    monkeypatch.setattr(ExecutionKernel, "_on_target_tracking_v2", lambda self, payload, **kwargs: None)
    with pytest.raises(RuntimeError, match="source drift"):
        inventory_policy_metadata()
    bus = InMemoryJetStream()
    with pytest.raises(RuntimeError, match="source drift"):
        ResearchActualInventoryKernel(
            bus,
            SimulatedGateway(),
            RiskOverlay(bus),
            ExecutionKernelConfig(execution_authority_mode="target_tracking_v2"),
        )
