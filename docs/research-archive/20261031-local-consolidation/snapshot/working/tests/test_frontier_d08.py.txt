"""Options mechanisms and data-boundary regressions; no synthetic alpha claims."""

import json

import pyarrow as pa
import pyarrow.parquet as pq
import pytest

from zenquant.research.frontier_lab.experiments import d08_options as options


def test_gamma_is_long_positive_for_both_option_sides_and_oi_scales_once():
    chain = options.synthetic_chain(17)
    summary = options.summarize_chain(chain, options.SYNTHETIC_CUTOFF_NS)
    doubled = options.summarize_chain(
        [{**row, "open_interest": 2 * row["open_interest"]} for row in chain],
        options.SYNTHETIC_CUTOFF_NS,
    )
    assert summary["gross_gamma_usd_per_1pct_move"] > 0
    assert doubled["gross_gamma_usd_per_1pct_move"] == pytest.approx(2 * summary["gross_gamma_usd_per_1pct_move"])
    scenarios = summary["dealer_scenarios"]
    assert scenarios["all_long_gamma_usd_per_1pct_move"] > 0
    assert scenarios["all_short_gamma_usd_per_1pct_move"] < 0
    assert scenarios["long_dealer_hedge_btc_for_1pct_up"] < 0
    assert scenarios["short_dealer_hedge_btc_for_1pct_up"] > 0


def test_expiry_gamma_concentrates_atm_but_not_every_otm_strike():
    sensitivity = options.gamma_sensitivity()
    assert sensitivity["atm_1d_to_30d_gamma_ratio"] == pytest.approx(5.5, rel=0.01)
    assert sensitivity["strike_130pct_1d_to_30d_gamma_ratio"] < 0.001
    assert sensitivity["atm_to_130pct_gamma_ratio_1d"] > 1000


def test_rr25_uses_delta_coordinates_not_fixed_strike_distance():
    summary = options.summarize_chain(options.synthetic_chain(17), options.SYNTHETIC_CUTOFF_NS)
    assert summary["rr25_30d_call_minus_put_decimal"] < 0
    assert abs(summary["fixed_95_105_skew_error_vol_points"]) > 0.5
    assert summary["term_30d_minus_7d_decimal"] == pytest.approx(-0.08, abs=0.005)


def test_missing_or_stale_quotes_are_unknown_not_zero_market_risk():
    chain = options.synthetic_chain(29)
    missing = [{**row, "bid_price": None} for row in chain]
    stale = [{**row, "quote_ts_ns": options.SYNTHETIC_CUTOFF_NS - 5 * 3600 * 10**9} for row in chain]
    for rows in (missing, stale):
        summary = options.summarize_chain(rows, options.SYNTHETIC_CUTOFF_NS)
        assert summary["status"] == "UNKNOWN"
        assert summary["rr25_30d_call_minus_put_decimal"] is None
        assert summary["gross_gamma_usd_per_1pct_move"] is None
        assert summary["valid_quote_count"] == 0


def test_future_availability_excluded_and_partial_surface_flagged():
    chain = options.synthetic_chain(43)
    rows = [{**row, "available_at_ns": options.SYNTHETIC_CUTOFF_NS + 1} for row in chain]
    assert options.summarize_chain(rows, options.SYNTHETIC_CUTOFF_NS)["status"] == "UNKNOWN"
    partial = [{**row, "ask_price": None} if row["option_type"] == "put" else row for row in chain]
    summary = options.summarize_chain(partial, options.SYNTHETIC_CUTOFF_NS)
    assert summary["status"] == "PARTIAL"
    assert summary["rr25_30d_call_minus_put_decimal"] is None
    assert summary["gross_gamma_is_lower_bound"]


def test_metadata_audit_never_reads_post2025_quote_values(tmp_path):
    root = tmp_path / "data/raw/deribit_options/BTC/2026-03"
    root.mkdir(parents=True)
    path = root / "options_chain_20260317T042229Z.parquet"
    pq.write_table(pa.table({"snapshot_ts_ns": [1773721349000000000], "mark_iv": [99999.0]}), path)
    audit, inputs = options.audit_local_coverage(tmp_path)
    assert audit["pre2026_snapshot_count"] == 0
    assert audit["excluded_post2025_snapshot_count"] == 1
    assert audit["total_option_rows_metadata_only"] == 1
    assert audit["historical_status"] == "data_blocked"
    assert audit["option_value_pages_read"] == 0
    assert inputs == [str(path.relative_to(tmp_path))]
    json.dumps(audit, allow_nan=False)


def test_misdated_pre2026_filename_cannot_admit_future_snapshot(tmp_path):
    root = tmp_path / "data/raw/deribit_options/BTC/2025-12"
    root.mkdir(parents=True)
    pq.write_table(
        pa.table({"snapshot_ts_ns": [1773721349000000000], "available_at_ns": [1773721349000000000]}),
        root / "options_chain_20251201T000000Z.parquet",
    )
    audit, _ = options.audit_local_coverage(tmp_path)
    assert audit["pre2026_snapshot_count"] == 0
    assert audit["timestamp_filename_conflict_count"] == 1
    assert audit["historical_status"] == "data_blocked"


def test_run_is_deterministic_finite_and_never_trading_eligible(tmp_path):
    first = options.run({"repo_root": str(tmp_path)})
    second = options.run({"repo_root": str(tmp_path)})
    assert first == second
    assert first["verdict"] == "data_blocked"
    assert len(first["metrics"]["synthetic_seed_results"]) == 5
    assert first["live_eligible"] is False
    assert first["promotion_eligible"] is False
    json.dumps(first, allow_nan=False)
