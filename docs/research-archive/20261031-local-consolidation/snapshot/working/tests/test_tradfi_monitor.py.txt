"""Live-observation failure boundaries and timestamp causality."""

import json
from pathlib import Path

from zenquant.research.tradfi_monitor import SymbolMonitor

CONFIG = json.loads((Path(__file__).parents[1] / "config/research/tradfi_monitor_v1.json").read_text())


def event(kind, t, **values):
    return {"e": kind, "E": t, "s": "MUUSDT", **values}


def depth(t, price=100, uid=None, pu=None):
    uid = t if uid is None else uid
    return event(
        "depthUpdate",
        t,
        u=uid,
        pu=t - 1000 if pu is None else pu,
        b=[[str(price - 0.01), "100"]],
        a=[[str(price + 0.01), "100"]],
    )


def ready():
    m = SymbolMonitor("MUUSDT", CONFIG)
    for t in range(0, 61000, 1000):
        m.ingest(event("markPriceUpdate", t, p="100", i="100"), t)
        m.ingest(event("aggTrade", t, a=t // 1000, p="100", q="10", m=True), t)
        assert m.ingest(depth(t), t) == []
    return m


def test_shock_detected_in_seconds_but_not_a_buy_signal():
    m = ready()
    alerts = m.ingest(depth(61000, 98), 61000)
    assert [a["kind"] for a in alerts] == ["SHOCK"]
    assert alerts[0]["promotion_eligible"] is False
    assert alerts[0]["observed_ms"] == 61000
    assert alerts[0]["metrics"]["return_60s_pct"] < -1


def test_partial_bar_cannot_make_followup_and_closed_bar_is_deduplicated():
    m = ready()
    m.ingest(depth(61000, 98), 61000)
    m.ingest(event("aggTrade", 61000, a=61, p="98", q="10", m=True), 61000)
    k = dict(t=60000, T=119999, x=False, o="100", c="98", h="100", l="97", q="10000", Q="5000")
    assert m.ingest(event("kline", 62000, k=k), 62000) == []
    # Keep inputs fresh up to the close, while price remains flat.
    for t in range(62000, 121000, 1000):
        m.ingest(event("markPriceUpdate", t, p="98", i="100"), t)
        m.ingest(event("aggTrade", t, a=t // 1000, p="98", q="10", m=True), t)
        m.ingest(depth(t, 98), t)
    k["x"] = True
    result = m.ingest(event("kline", 120001, k=k), 120001)
    assert len(result) == 1
    assert result[0]["kind"] == "MINUTE_1"
    assert m.ingest(event("kline", 120002, k=k), 120002) == []


def test_gap_cancels_event_and_requires_new_warmup():
    m = ready()
    m.ingest(depth(61000, 98), 61000)
    alerts = m.ingest(depth(62000, 96, pu=123), 62000)
    assert alerts[0]["kind"] == "DATA_GAP"
    assert not m.status(62000)["ready"]
    assert m.status(62000)["active_event"] is None


def test_stale_data_never_triggers_and_health_expires():
    m = ready()
    assert not m.status(70001)["ready"]
    result = m.ingest(depth(61000, 90), 90000)
    assert all(a["kind"] != "SHOCK" for a in result)
    assert not m.status(90000)["ready"]


def test_duplicate_trade_is_not_counted_twice():
    m = ready()
    before = m.status(60000)["metrics"]["quote_volume_60s"]
    m.ingest(event("aggTrade", 60000, a=60, p="100", q="10", m=True), 60001)
    assert m.status(60001)["metrics"]["quote_volume_60s"] <= before


def test_future_timestamp_and_crossed_book_fail_closed():
    m = ready()
    assert all(a["kind"] != "SHOCK" for a in m.ingest(depth(90000, 90), 61000))
    bad = depth(62000, 90)
    bad["b"] = [["95", "1"]]
    bad["a"] = [["90", "1"]]
    assert all(a["kind"] != "SHOCK" for a in m.ingest(bad, 62000))
    assert not m.status(62000)["ready"]
