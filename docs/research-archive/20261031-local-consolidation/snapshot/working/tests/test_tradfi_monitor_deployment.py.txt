"""Deployment portability and explicit instrument admission."""

import json
from pathlib import Path

import pytest

from zenquant.app import tradfi_monitor as app

CONFIG = json.loads((Path(__file__).parents[1] / "config/research/tradfi_monitor_v1.json").read_text())


def test_coin_ticker_collision_is_rejected():
    config = {**CONFIG, "symbols": ["STXUSDT"]}
    info = {
        "symbols": [{"symbol": "STXUSDT", "status": "TRADING", "contractType": "PERPETUAL", "underlyingType": "COIN"}]
    }
    assert hasattr(app, "validate_instruments")
    with pytest.raises(ValueError, match="TradFi"):
        app.validate_instruments(config, info)


def test_equity_universe_is_configured_and_not_hardcoded():
    info = {
        "symbols": [
            {"symbol": s, "status": "TRADING", "contractType": "TRADIFI_PERPETUAL", "underlyingType": "KR_EQUITY"}
            for s in CONFIG["symbols"]
        ]
    }
    assert hasattr(app, "validate_instruments")
    assert len(app.validate_instruments(CONFIG, info)) == len(CONFIG["symbols"])
    info["symbols"][0]["status"] = "SETTLING"
    with pytest.raises(ValueError):
        app.validate_instruments(CONFIG, info)


@pytest.mark.parametrize(
    "changes",
    [
        {"symbols": []},
        {"symbols": ["MUUSDT", "MUUSDT"]},
        {"drop_60s_pct": float("nan")},
        {"min_quote_volume_60s": -1},
        {"duration_hours": -1},
        {"research_only": False},
    ],
)
def test_invalid_deployment_config_fails_before_running(changes):
    assert hasattr(app, "validate_config")
    with pytest.raises(ValueError):
        app.validate_config({**CONFIG, **changes})


def test_validate_cli_works_outside_repository_without_starting(tmp_path, monkeypatch):
    config = tmp_path / "monitor.json"
    config.write_text(json.dumps(CONFIG))
    monkeypatch.chdir(tmp_path)
    assert app.main(["--config", str(config), "--check-config"]) == 0
    assert list(tmp_path.iterdir()) == [config]


def test_continuous_keeps_storage_budget_and_stops_without_network(tmp_path):
    import asyncio

    config = {**CONFIG, "duration_hours": 0, "max_raw_bytes": 1}
    app.validate_config(config)
    recorder = app.Recorder(tmp_path, config)
    assert recorder.deadline is None
    recorder.raw_bytes = 1
    try:
        asyncio.run(recorder.heartbeat())
        assert recorder.reason == "raw_budget_reached"
    finally:
        recorder.alerts.close()
        recorder.observations.close()


def test_recorder_writes_sector_alert_alongside_member_alerts(tmp_path):
    recorder = app.Recorder(tmp_path, CONFIG)
    try:
        for i, symbol in enumerate(("MUUSDT", "SNDKUSDT", "SKHYNIXUSDT")):
            recorder.alert({"kind": "SHOCK", "symbol": symbol, "observed_ms": i, "event_id": f"{symbol}-{i}"})
        records = [json.loads(line) for line in (tmp_path / "alerts.jsonl").read_text().splitlines()]
        assert [r["kind"] for r in records] == ["SHOCK", "SHOCK", "SHOCK", "SECTOR_SHOCK"]
        assert recorder.alert_counts["SECTOR_SHOCK"] == 1
    finally:
        recorder.alerts.close()
        recorder.observations.close()
