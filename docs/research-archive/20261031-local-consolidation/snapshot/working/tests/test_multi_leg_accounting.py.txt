"""Independent Decimal hand calculations for research wallets, not trading proof."""

from decimal import Decimal as D

import pytest

from zenquant.research.multi_leg_accounting import (
    AccountState,
    EventKey,
    Fill,
    Funding,
    Instrument,
    Margin,
    Mark,
    Transfer,
    apply_event,
    initial_state,
    valuation,
)

SPOT = Instrument("BINANCE", "spot", "BTC", "USDT", "USDT")
PERP = Instrument("BINANCE", "linear_perp", "BTC", "USDT", "USDT")


def key(n, *, time=None, event_id=None):
    return EventKey(event_id or str(n), n if time is None else time, n, "fixture", n if time is None else time)


def mark(n, instrument, price):
    return Mark(key(n), instrument, D(price), n)


def nav(state):
    return valuation(state, asof_ns=state.last_order[0], max_mark_age_ns=100).nav


def test_exact_contract_sequence_and_final_pnl():
    state = initial_state((SPOT, PERP), spot_cash=D("10000"))
    events = [
        mark(1, SPOT, "50000"),
        mark(2, PERP, "50100"),
        Transfer(key(3), "spot", "derivative", D("2000")),
        Fill(key(4), SPOT, D(".1"), D("50000"), D("2")),
        Fill(key(5), PERP, D("-.06"), D("50100"), D("1.2024")),
        Fill(key(6), PERP, D("-.04"), D("50120"), D(".80192")),
        mark(7, PERP, "50120"),
        mark(8, SPOT, "51000"),
        mark(9, PERP, "51200"),
        Funding(key(10), PERP, D(".0001"), D("51200"), 10),
        Fill(key(11), SPOT, D("-.1"), D("51000"), D("2.04")),
        Fill(key(12), PERP, D(".1"), D("51200"), D("2.048")),
    ]
    expected = {
        3: "10000",
        4: "9998",
        5: "9996.7976",
        7: "9994.79568",
        9: "9986.79568",
        10: "9987.30768",
        11: "9985.26768",
        12: "9983.21968",
    }
    for event in events:
        state = apply_event(state, event)
        if event.key.sequence in expected:
            assert nav(state) == D(expected[event.key.sequence])
    assert state.spot_cash == D("8095.96")
    assert state.derivative_cash == D("1887.25968")
    assert nav(state) - state.initial_capital == D("-16.78032")
    assert state.fees_total == D("8.09232")
    assert state.funding_total == D(".512")
    assert state.realized_perp_pnl == D("-109.2")
    assert state.metadata()["E3"] == "FAIL" and state.metadata()["E4"] == "NOT_EVALUABLE"


def test_partial_exit_and_reversal_avg_entry_do_not_book_nominal_cash():
    state = initial_state((PERP,), derivative_cash=D("1000"))
    state = apply_event(state, Fill(key(1), PERP, D("-2"), D("100"), D("1")))
    assert state.derivative_cash == D("999")
    state = apply_event(state, Fill(key(2), PERP, D("1"), D("90"), D("1")))
    assert state.position(PERP).quantity == D("-1") and state.position(PERP).average_entry == D("100")
    assert state.derivative_cash == D("1008")
    state = apply_event(state, Fill(key(3), PERP, D("2"), D("80"), D("1")))
    assert state.position(PERP).quantity == D("1") and state.position(PERP).average_entry == D("80")
    assert state.derivative_cash == D("1027")


def test_funding_old_inventory_same_time_sequence_and_duplicate_is_idempotent():
    state = initial_state((PERP,), derivative_cash=D("100"))
    state = apply_event(state, Fill(key(1), PERP, D("-1"), D("100"), D("0")))
    funding = Funding(key(2, time=3), PERP, D(".01"), D("100"), 3)
    state = apply_event(state, funding)
    assert state.derivative_cash == D("101")
    state = apply_event(state, Fill(key(3, time=3), PERP, D("1"), D("100"), D("0")))
    assert apply_event(state, funding) is state
    with pytest.raises(ValueError, match="conflict"):
        apply_event(state, Funding(funding.key, PERP, D("-.01"), D("100"), 3))
    with pytest.raises(ValueError, match="order"):
        apply_event(state, Transfer(key(4, time=2), "spot", "derivative", D("1")))


def test_transfer_and_margin_reclassification_nav_neutral():
    state = initial_state((SPOT, PERP), spot_cash=D("1000"))
    state = apply_event(state, Transfer(key(1), "spot", "derivative", D("400")))
    state = apply_event(state, Margin(key(2), D("300")))
    value = valuation(state, asof_ns=2, max_mark_age_ns=0)
    assert value.nav == D("1000") and value.derivative_free == D("100")
    assert state.margin_locked == D("300")
    with pytest.raises(ValueError, match="capital"):
        apply_event(state, Transfer(key(3), "derivative", "spot", D("101")))
    with pytest.raises(ValueError, match="capital"):
        apply_event(state, Margin(key(3), D("401")))


def test_actual_fee_or_spot_fill_preserves_negative_wallet_and_capital_status():
    state = initial_state((SPOT, PERP), spot_cash=D("1"))
    state = apply_event(state, Fill(key(1), SPOT, D("1"), D("100"), D("2")))
    assert state.spot_cash == D("-101") and state.capital_insufficient
    state = apply_event(state, Fill(key(2), PERP, D("-1"), D("100"), D("1")))
    assert state.derivative_cash == D("-1") and state.capital_insufficient
    with pytest.raises(ValueError, match="borrow"):
        apply_event(state, Fill(key(3), SPOT, D("-2"), D("100"), D("0")))


def test_future_stale_unknown_source_and_unsupported_fee_fail_closed():
    state = initial_state((SPOT,), spot_cash=D("1000"))
    with pytest.raises(ValueError, match="future"):
        apply_event(state, Mark(key(1), SPOT, D("100"), 2))
    with pytest.raises(ValueError, match="source"):
        apply_event(state, Mark(EventKey("x", 1, 1, "", 1), SPOT, D("100"), 1))
    with pytest.raises(ValueError, match="available"):
        apply_event(state, Mark(EventKey("x", 1, 1, "fixture", 2), SPOT, D("100"), 1))
    state = apply_event(state, mark(1, SPOT, "100"))
    state = apply_event(state, Fill(key(2), SPOT, D("1"), D("100"), D("0")))
    with pytest.raises(ValueError, match="stale"):
        valuation(state, asof_ns=3, max_mark_age_ns=1)
    with pytest.raises(ValueError, match="asof"):
        valuation(state, asof_ns=1, max_mark_age_ns=10)
    with pytest.raises(ValueError, match="fee"):
        apply_event(state, Fill(key(3), SPOT, D("1"), D("100"), D(".01"), "BTC"))


def test_instrument_identity_same_base_does_not_net_spot_and_perp_and_inverse_rejected():
    state = initial_state((SPOT, PERP), spot_cash=D("1000"))
    state = apply_event(state, Fill(key(1), SPOT, D("1"), D("100"), D("0")))
    state = apply_event(state, Fill(key(2), PERP, D("-1"), D("100"), D("0")))
    assert state.position(SPOT).quantity == D("1") and state.position(PERP).quantity == D("-1")
    with pytest.raises(ValueError, match="unsupported"):
        initial_state((Instrument("BINANCE", "inverse", "BTC", "USD", "BTC"),))
    assert isinstance(state, AccountState)


def test_decimal_context_determinism_and_frozen_state():
    from dataclasses import FrozenInstanceError
    from decimal import ROUND_FLOOR, localcontext

    state = initial_state((PERP,), derivative_cash=D("1000"))
    events = [Fill(key(1), PERP, D("1"), D("100"), D(".01")), Fill(key(2), PERP, D("2"), D("101"), D(".02"))]
    expected = state
    for event in events:
        expected = apply_event(expected, event)
    with localcontext() as context:
        context.prec = 2
        context.rounding = ROUND_FLOOR
        for event in events:
            state = apply_event(state, event)
    assert state == expected
    with pytest.raises(FrozenInstanceError):
        state.spot_cash = D("100")


def test_missing_mark_negative_funding_and_explicit_contract_multiplier():
    instrument = Instrument(
        "BINANCE", "linear_perp", "ETH", "USDT", "USDT", multiplier=D("2"), quantity_unit="contracts"
    )
    state = initial_state((instrument,), derivative_cash=D("10"))
    state = apply_event(state, Fill(key(1), instrument, D("-1"), D("100"), D("0")))
    with pytest.raises(ValueError, match="missing mark"):
        valuation(state, asof_ns=1, max_mark_age_ns=10)
    state = apply_event(state, Funding(key(2), instrument, D("-.01"), D("100"), 2))
    assert state.derivative_cash == D("8")
    state = apply_event(state, Fill(key(3), instrument, D("1"), D("90"), D("0")))
    assert state.derivative_cash == D("28")


@pytest.mark.parametrize("value", [D("NaN"), D("Infinity"), 1.0])
def test_nonfinite_or_float_amounts_fail_closed(value):
    state = initial_state((SPOT,), spot_cash=D("1000"))
    with pytest.raises(ValueError, match="Decimal"):
        apply_event(state, Fill(key(1), SPOT, D("1"), D("100"), value))


def test_margin_can_release_after_actual_fees_exhaust_wallet_without_hiding_loss():
    state = initial_state((PERP,), derivative_cash=D("1"))
    state = apply_event(state, Margin(key(1), D("1")))
    state = apply_event(state, Fill(key(2), PERP, D("1"), D("100"), D("2")))
    assert state.derivative_cash == D("-1") and state.capital_insufficient
    state = apply_event(state, Margin(key(3), D("0")))
    state = apply_event(state, mark(4, PERP, "100"))
    assert nav(state) == D("-1") and state.capital_insufficient
