"""Clock, coarse-bar alignment and two-leg economic failure controls."""

import numpy as np
import pandas as pd
import pytest

from zenquant.research.frontier_lab.experiments.d09_crossvenue import (
    HOUR_NS,
    aggregate_complete_4h,
    audited_hourly,
    clock_experiment,
    lag_correlations,
    two_leg_cost,
)


def hourly_frame(hours):
    stamps = np.asarray(hours, dtype=np.int64) * HOUR_NS
    return pd.DataFrame({"timestamp": stamps // 10**6, "timestamp_ns": stamps, "close": 100.0 + np.asarray(hours)})


def test_aggregation_uses_last_hour_close_and_drops_incomplete_window():
    aggregate = aggregate_complete_4h(hourly_frame([0, 1, 2, 3, 4, 5, 7]))
    assert aggregate["open_time_ns"].tolist() == [0]
    assert aggregate["deribit_close"].tolist() == [103]
    assert aggregate["naive_first_hour_close"].tolist() == [100]
    assert aggregate["close_time_ns"].tolist() == [4 * HOUR_NS - 10**6]


def test_aggregation_rejects_unit_confusion():
    frame = hourly_frame([0, 1, 2, 3])
    frame.loc[2, "timestamp_ns"] = frame.loc[2, "timestamp"]
    with pytest.raises(ValueError, match="millisecond"):
        aggregate_complete_4h(frame)


def test_aggregation_rejects_duplicate_hour():
    with pytest.raises(ValueError, match="duplicate"):
        aggregate_complete_4h(hourly_frame([0, 1, 1, 2, 3]))


def test_identical_duplicates_are_counted_and_conflicting_hour_is_excluded():
    frame = hourly_frame([0, 1, 1, 2, 3, 4, 5, 6, 7, 7])
    frame.loc[9, "close"] = 999
    clean, audit = audited_hourly(frame)
    assert audit["exact_duplicate_rows_removed"] == 1
    assert audit["conflicting_timestamp_rows_excluded"] == 2
    assert clean.timestamp_ns.tolist() == [i * HOUR_NS for i in range(7)]
    assert len(aggregate_complete_4h(clean)) == 1


def test_clock_offset_fabricates_lead_lag_and_receive_correction_removes_it():
    result = clock_experiment(17)
    assert result["event_clock_peak_lag_steps"] == 3
    assert result["uncorrected_receive_peak_lag_steps"] == 2
    assert result["corrected_receive_peak_lag_steps"] == 0
    assert result["corrected_receive_peak_correlation"] > 0.99


def test_lag_sign_means_a_precedes_b():
    values = np.random.default_rng(29).normal(size=500)
    delayed = np.concatenate([np.zeros(2), values[:-2]])
    assert max(lag_correlations(values, delayed, 3), key=lambda x: x["correlation"])["lag_steps"] == 2


def test_two_leg_cost_includes_entry_and_exit_on_both_venues():
    cheap = two_leg_cost(20, 2, 0.5, 1, 0)
    expensive = two_leg_cost(20, 5, 0.5, 8, 4)
    assert cheap["fee_bps"] == 8
    assert cheap["net_bps_per_paired_notional"] == 9
    assert expensive["net_bps_per_paired_notional"] == -14
    assert expensive["required_collateral_usd"] == [100, 100]


def test_collateral_cannot_be_netted_across_venues():
    result = two_leg_cost(20, 2, 0.5, 1, 0, collateral_by_venue_usd=(200, 50))
    assert result["execution_feasible"] is False
    assert result["net_bps_per_paired_notional"] is None
    assert result["collateral_shortfall_usd"] == [0, 50]
