"""Shock-event leakage guards and liquidity-mechanism counterexamples."""

import numpy as np
import pandas as pd

from zenquant.research.frontier_lab.common import TRAIN_END
from zenquant.research.frontier_lab.experiments.d07_liquidity import (
    event_masks,
    nonoverlapping_events,
    response_features,
    shock_candidates,
    synthetic_process,
)


def example_features(n=300):
    rng = np.random.default_rng(17)
    return pd.DataFrame(
        {
            "return_1": rng.normal(0, 0.01, n),
            "vol_42": np.full(n, 0.01),
            "range": rng.uniform(0.005, 0.04, n),
            "volume_z": rng.normal(size=n),
            "imbalance": rng.uniform(-0.5, 0.5, n),
        }
    )


def test_shock_thresholds_ignore_development_and_use_prior_volatility():
    f = example_features()
    train = np.arange(len(f)) < 200
    candidates, thresholds = shock_candidates(f, train)
    changed = f.copy()
    changed.loc[200:, ["return_1", "range", "volume_z"]] = 1e6
    _, other = shock_candidates(changed, train)
    assert thresholds == other
    f.loc[100, ["return_1", "range", "volume_z"]] = [0.1, 0.3, 20]
    f.loc[100, "vol_42"] = 100
    selected, _ = shock_candidates(f, train)
    assert selected[100]
    assert candidates.dtype == bool


def test_events_never_overlap_labels_and_tail_is_excluded():
    candidates = np.zeros(30, dtype=bool)
    candidates[[2, 3, 9, 10, 18, 25]] = True
    assert nonoverlapping_events(candidates).tolist() == [2, 10, 18]


def test_training_label_end_is_purged_even_when_event_is_before_cutoff():
    timestamps = pd.date_range("2023-12-29", periods=40, freq="4h", tz="UTC").as_unit("ns").astype("int64")
    frame = pd.DataFrame({"close_time_ns": timestamps})
    events = np.array([0, 14, 24])
    train, dev = event_masks(frame, events)
    assert train.tolist() == [True, False, False]
    assert dev.tolist() == [False, False, True]
    assert frame["close_time_ns"].iloc[events[train] + 7].max() < TRAIN_END.value


def test_response_features_cannot_see_future_beyond_observation_bar():
    f = example_features()
    original = response_features(f, np.array([100]))
    f.loc[102:, :] = 999
    assert np.array_equal(original, response_features(f, np.array([100])))
    f.loc[101, "imbalance"] += 0.1
    assert not np.array_equal(original, response_features(f, np.array([100])))


def test_same_price_shock_has_different_response_with_persistent_liquidity():
    for seed in [17, 29, 43, 71, 101]:
        result = synthetic_process(seed)
        assert result["initial_price_shock"] == -0.02
        assert result["healthy_mean_future_return"] > 0
        assert result["withdrawal_mean_future_return"] < 0
        assert result["models"]["plus_depth_response"]["mse_reduction_vs_current_fraction"] > 0.7
        assert result == synthetic_process(seed)


def test_redrawing_regime_destroys_observed_depth_predictive_value():
    gains = [
        synthetic_process(seed, stable=False)["models"]["plus_depth_response"]["mse_reduction_vs_current_fraction"]
        for seed in [17, 29, 43, 71, 101]
    ]
    assert np.mean(gains) < 0.01
    assert max(gains) < 0.02
