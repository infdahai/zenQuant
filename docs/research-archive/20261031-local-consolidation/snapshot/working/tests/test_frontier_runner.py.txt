"""Integration evidence must retain failures and reject tampered attempts."""

from __future__ import annotations

import importlib.util
import json
import sys
from pathlib import Path

import pytest

from zenquant.research.frontier_lab.common import file_hash, write_json
from zenquant.research.run_evidence import TrialJournal, identity_hash

SPEC = importlib.util.spec_from_file_location("frontier_cli_for_tests", Path("scripts/run_market_exploration.py"))
assert SPEC is not None and SPEC.loader is not None
CLI = importlib.util.module_from_spec(SPEC)
sys.modules[SPEC.name] = CLI
SPEC.loader.exec_module(CLI)


def test_missing_directions_are_counted_not_dropped(tmp_path, monkeypatch) -> None:
    monkeypatch.setattr(CLI, "prepare", lambda output: {})
    assert CLI.summarize(tmp_path, "initial") == 1
    summary = json.loads((tmp_path / "summary_initial.json").read_text())
    assert summary["registered_directions"] == 12
    assert summary["completed"] == 0
    assert all(row["status"] == "missing" for row in summary["directions"])


def test_failure_evidence_stays_visible(tmp_path, monkeypatch) -> None:
    monkeypatch.setattr(CLI, "prepare", lambda output: {})
    write_json(
        tmp_path / "d01/initial/failure.json", {"direction_id": "d01", "status": "failed", "error": "probe failed"}
    )
    assert CLI.summarize(tmp_path, "initial") == 1
    summary = json.loads((tmp_path / "summary_initial.json").read_text())
    assert summary["directions"][0]["error"] == "probe failed"
    assert summary["registered_directions"] == 12


def _committed_result(target: Path) -> None:
    identity = {"direction": "d01", "bound_files": {}}
    result = {
        "direction_id": "d01",
        "hypothesis": "h",
        "evidence_type": "synthetic_mechanism",
        "method": "m",
        "metrics": {"observations": 3},
        "verdict": "inconclusive",
        "limitations": ["synthetic"],
        "falsification": "f",
        "next_experiment": "n",
        "status": "completed",
        "rankable": False,
        "live_eligible": False,
        "promotion_eligible": False,
        "bound_files": {},
        "supplementary_input_hashes": {},
        "identity_hash": identity_hash(identity),
    }
    write_json(target / "result.json", result)
    TrialJournal(target / "journal", identity).append(
        {
            "attempt_id": "initial",
            "status": "completed",
            "result_sha256": file_hash(target / "result.json"),
        }
    )


def test_tampered_result_cannot_count_as_completed(tmp_path, monkeypatch) -> None:
    monkeypatch.setattr(CLI, "prepare", lambda output: {})
    target = tmp_path / "d01/initial"
    _committed_result(target)
    result_path = target / "result.json"
    result = json.loads(result_path.read_text())
    result["metrics"]["observations"] = 99999
    write_json(result_path, result)
    assert CLI.summarize(tmp_path, "initial") == 1
    summary = json.loads((tmp_path / "summary_initial.json").read_text())
    assert summary["completed"] == 0
    assert summary["directions"][0]["status"] == "invalid"


def test_result_without_committed_journal_is_invalid(tmp_path, monkeypatch) -> None:
    monkeypatch.setattr(CLI, "prepare", lambda output: {})
    target = tmp_path / "d01/initial"
    _committed_result(target)
    (target / "journal/journal_state.json").unlink()
    assert CLI.summarize(tmp_path, "initial") == 1
    summary = json.loads((tmp_path / "summary_initial.json").read_text())
    assert summary["directions"][0]["status"] == "invalid"


@pytest.mark.parametrize("violation", ["duplicate", "over_budget", "acceptance", "threads", "boundary"])
def test_campaign_budget_and_temporal_contract_are_enforced(tmp_path, monkeypatch, violation) -> None:
    registry = json.loads(CLI.REGISTRY.read_text())
    if violation == "duplicate":
        registry["directions"].append(registry["directions"][0])
    elif violation == "over_budget":
        registry["directions"] = [{"id": f"d{i}", "module": f"d{i}_module"} for i in range(21)]
    elif violation == "acceptance":
        registry["acceptance_data_allowed"] = True
    elif violation == "threads":
        registry["max_threads_per_experiment"] = 14
    elif violation == "boundary":
        registry["train_end"] = "2026-01-01T00:00:00Z"
    path = tmp_path / "registry.json"
    write_json(path, registry)
    monkeypatch.setattr(CLI, "REGISTRY", path)
    with pytest.raises(ValueError):
        CLI.load_registry()
