"""Hand-derived contracts for the independent research evaluator."""

from __future__ import annotations

import copy
import json
from math import sqrt

import pytest

DAY = 86_400_000_000_000


def evaluate(evidence):
    # Import inside the call so the first TDD run exercises the missing entrypoint.
    from zenquant.research.evaluation import evaluate_evidence

    return evaluate_evidence(evidence)


def evidence(values=(100.0, 110.0, 99.0, 108.9)):
    return {
        "initial_equity": 100.0,
        "curve": [{"time_ns": day * DAY, "equity": value} for day, value in enumerate(values)],
        "fills": [],
        "funding": [],
        "episodes": [],
        "identity": {"run_id": "hand-calculated", "evidence_role": "observed_historical"},
    }


def value(report, name):
    return report["metrics"][name]["value"]


def test_daily_ratios_use_daily_returns_and_full_sample_downside_deviation():
    report = evaluate(evidence())
    assert [row["value"] for row in report["daily_returns"]] == pytest.approx([0.1, -0.1, 0.1])
    assert value(report, "net_return") == pytest.approx(0.089)
    assert value(report, "daily_sharpe") == pytest.approx(sqrt(365 / 12))
    assert value(report, "daily_sortino") == pytest.approx(sqrt(365 / 3))
    assert report["metrics"]["daily_sortino"]["sample_size"] == 3
    assert report["metrics"]["daily_sortino"]["status"] == "NOT_EVALUABLE"
    assert report["metrics"]["daily_sortino"]["reason"] == "INSUFFICIENT_DAILY_SAMPLES"


def test_exact_midnight_sampling_does_not_use_intraday_extremes_for_daily_ratios():
    data = evidence((100.0, 110.0, 99.0))
    data["curve"].insert(1, {"time_ns": DAY // 2, "equity": 50.0})
    report = evaluate(data)
    assert [row["value"] for row in report["daily_returns"]] == pytest.approx([0.1, -0.1])
    assert value(report, "max_drawdown") == pytest.approx(0.5)
    assert report["metrics"]["max_drawdown"]["sampling"] == "provided_valuation_points"


def test_missing_midnight_never_bridges_gap_or_inserts_flat_return():
    data = evidence()
    data["curve"] = [
        {"time_ns": 0, "equity": 100.0},
        {"time_ns": DAY, "equity": 110.0},
        {"time_ns": 3 * DAY, "equity": 99.0},
        {"time_ns": 4 * DAY, "equity": 108.9},
    ]
    report = evaluate(data)
    assert [row["value"] for row in report["daily_returns"]] == pytest.approx([0.1, 0.1])
    assert report["daily_sampling"]["missing_midnights"] == [2 * DAY]
    assert value(report, "daily_sharpe") is None
    assert report["metrics"]["daily_sharpe"]["reason"] == "MISSING_DAILY_VALUATIONS"
    assert not report["rankable"]


def test_quote_after_midnight_is_not_backfilled_to_missing_midnight():
    data = evidence((100.0, 110.0, 99.0))
    data["curve"][1]["time_ns"] += 1
    report = evaluate(data)
    assert report["daily_sampling"]["missing_midnights"] == [DAY]
    assert report["daily_returns"] == []


def test_drawdown_recovery_reports_elapsed_time_from_peak_and_trough():
    report = evaluate(evidence((100.0, 120.0, 90.0, 100.0, 120.0, 110.0)))
    assert value(report, "max_drawdown") == pytest.approx(0.25)
    assert value(report, "longest_underwater_days") == 3.0
    assert value(report, "peak_to_trough_days") == 1.0
    assert value(report, "recovery_days") == 2.0
    assert report["drawdown"]["max_drawdown_censored"] is False
    assert report["drawdown"]["currently_underwater"] is True


def test_unrecovered_drawdown_keeps_duration_and_censored_recovery():
    report = evaluate(evidence((100.0, 120.0, 90.0, 100.0)))
    assert value(report, "longest_underwater_days") == 2.0
    assert value(report, "recovery_days") is None
    assert report["metrics"]["recovery_days"]["reason"] == "CENSORED"
    assert report["drawdown"]["max_drawdown_censored"] is True


@pytest.mark.parametrize("ruined_equity, expected_drawdown", [(0.0, 1.0), (-10.0, 1.1)])
def test_intraday_ruin_cannot_be_hidden_by_recovery(ruined_equity, expected_drawdown):
    data = evidence((100.0, 120.0))
    data["curve"].insert(1, {"time_ns": DAY // 2, "equity": ruined_equity})
    report = evaluate(data)
    assert report["ruin"]["occurred"] is True
    assert report["ruin"]["first_time_ns"] == DAY // 2
    assert value(report, "max_drawdown") == expected_drawdown
    assert value(report, "net_return") == pytest.approx(0.2)
    assert value(report, "daily_sharpe") is None
    assert value(report, "daily_sortino") is None
    assert report["metrics"]["daily_sharpe"]["reason"] == "RUIN"
    assert report["gates"]["R2"]["status"] == "FAIL"
    assert not report["rankable"]


def test_costs_and_turnover_distinguish_cash_amounts_from_equity_multiples():
    data = evidence((100.0, 119.0, 137.0))
    data["fills"] = [
        {"time_ns": DAY, "signed_qty": 2.0, "price": 10.0, "fee": 1.0},
        {"time_ns": 2 * DAY, "signed_qty": -2.0, "price": 20.0, "fee": 2.0},
    ]
    data["funding"] = [{"time_ns": DAY, "pnl": 3.0}, {"time_ns": 2 * DAY, "pnl": -1.0}]
    report = evaluate(data)
    assert value(report, "fees_total") == 3.0
    assert value(report, "funding_pnl_total") == 2.0
    assert value(report, "turnover_notional_total") == 60.0
    assert value(report, "fees_initial_equity") == 0.03
    assert value(report, "funding_cost_initial_equity") == -0.02
    assert value(report, "turnover_initial_equity") == 0.6
    assert value(report, "turnover_usdt_per_day") == 30.0
    assert value(report, "fees_gross_price_pnl") == pytest.approx(3 / 38)
    assert report["metrics"]["turnover_initial_equity"]["unit"] == "initial_equity_multiple"
    assert report["metrics"]["turnover_usdt_per_day"]["unit"] == "USDT/day"


def test_missing_cashflow_records_are_not_treated_as_no_cashflows():
    data = evidence()
    del data["fills"]
    del data["funding"]
    report = evaluate(data)
    for name in ("fees_total", "funding_pnl_total", "turnover_notional_total"):
        assert value(report, name) is None
        assert report["metrics"][name]["status"] == "NOT_EVALUABLE"


def test_nonpositive_gross_pnl_does_not_report_zero_fee_drag():
    data = evidence((100.0, 90.0))
    data["fills"] = [{"time_ns": DAY, "signed_qty": 1.0, "price": 10.0, "fee": 1.0}]
    report = evaluate(data)
    assert value(report, "fees_gross_price_pnl") is None
    assert report["metrics"]["fees_gross_price_pnl"]["reason"] == "NONPOSITIVE_GROSS_PRICE_PNL"


def test_episode_statistics_exclude_open_episode_and_count_closed_episodes():
    data = evidence()
    data["episodes"] = [{"closed": True, "net_pnl": pnl} for pnl in (4.0, -2.0)] * 15
    data["episodes"].append({"closed": False, "net_pnl": 1_000_000.0})
    report = evaluate(data)
    assert value(report, "episode_profit_factor") == 2.0
    assert value(report, "win_rate") == 0.5
    assert report["metrics"]["episode_profit_factor"]["sample_size"] == 30
    assert report["metrics"]["episode_profit_factor"]["status"] == "PASS"


def test_too_few_closed_episodes_keep_diagnostic_value_without_sample_pass():
    data = evidence()
    data["episodes"] = [{"closed": True, "net_pnl": 4.0}, {"closed": True, "net_pnl": -2.0}]
    report = evaluate(data)
    assert value(report, "episode_profit_factor") == 2.0
    assert report["metrics"]["episode_profit_factor"]["status"] == "NOT_EVALUABLE"
    assert report["metrics"]["episode_profit_factor"]["reason"] == "INSUFFICIENT_CLOSED_EPISODES"


def test_no_loss_episodes_do_not_receive_infinity_or_zero_profit_factor():
    data = evidence()
    data["episodes"] = [{"closed": True, "net_pnl": 1.0}] * 30
    report = evaluate(data)
    assert value(report, "episode_profit_factor") is None
    assert report["metrics"]["episode_profit_factor"]["reason"] == "NO_LOSS_EPISODES"
    json.dumps(report, allow_nan=False)


def test_engineering_failure_cannot_be_compensated_by_large_return():
    data = evidence((100.0, 1_000_000.0))
    data["engineering"] = {f"E{i}": {"status": "PASS", "reason": "fixture_proof"} for i in range(5)}
    data["engineering"]["E2"] = {"status": "FAIL", "reason": "cash_created_without_source"}
    report = evaluate(data)
    assert report["gates"]["E2"]["status"] == "FAIL"
    assert any(item["gate"] == "E2" for item in report["blockers"])
    assert not report["rankable"]
    assert report["diagnostic_only"]
    assert report["live_eligible"] is False


def test_missing_engineering_and_research_dimensions_remain_explicit():
    report = evaluate(evidence())
    assert set(report["gates"]) == {*(f"E{i}" for i in range(5)), *(f"R{i}" for i in range(8)), "O0", "O1"}
    assert all(gate["status"] == "NOT_EVALUABLE" for gate in report["gates"].values())
    assert report["diagnostic_only"]
    assert not report["rankable"]


@pytest.mark.parametrize("bad_value", [float("nan"), float("inf"), float("-inf")])
def test_nonfinite_equity_is_rejected_instead_of_json_nan(bad_value):
    data = evidence()
    data["curve"][1]["equity"] = bad_value
    with pytest.raises(ValueError, match="finite"):
        evaluate(data)


@pytest.mark.parametrize("times", [(0, 0), (DAY, 0)])
def test_duplicate_or_out_of_order_curve_points_are_rejected(times):
    data = evidence((100.0, 101.0))
    for row, timestamp in zip(data["curve"], times, strict=True):
        row["time_ns"] = timestamp
    with pytest.raises(ValueError, match="strictly increasing"):
        evaluate(data)


def test_empty_curve_retains_missing_evidence_and_does_not_manufacture_returns():
    data = evidence(())
    report = evaluate(data)
    assert value(report, "net_return") is None
    assert value(report, "max_drawdown") is None
    assert not report["rankable"]


def test_report_is_json_safe_deterministic_and_does_not_mutate_input():
    data = evidence()
    before = copy.deepcopy(data)
    first = evaluate(data)
    second = evaluate(data)
    assert first == second
    assert data == before
    assert json.loads(json.dumps(first, allow_nan=False)) == first
    assert first["schema_version"] == "research-evaluation.v1"
    assert all(item["definition_version"] for item in first["metrics"].values())


def test_zero_volatility_and_no_downside_ratios_are_undefined():
    flat = evaluate(evidence((100.0, 100.0, 100.0)))
    assert value(flat, "daily_sharpe") is None
    assert flat["metrics"]["daily_sharpe"]["reason"] == "ZERO_VOLATILITY"
    rising = evaluate(evidence((100.0, 110.0, 132.0)))
    assert value(rising, "daily_sortino") is None
    assert rising["metrics"]["daily_sortino"]["reason"] == "NO_DOWNSIDE_RETURNS"


@pytest.mark.parametrize("field", ["fills", "funding"])
def test_cashflows_outside_evaluated_curve_are_rejected(field):
    data = evidence()
    data[field] = (
        [{"time_ns": 4 * DAY, "signed_qty": 1.0, "price": 10.0, "fee": 1.0}]
        if field == "fills"
        else [{"time_ns": -1, "pnl": 5.0}]
    )
    with pytest.raises(ValueError, match="outside"):
        evaluate(data)


def test_known_net_return_failure_is_not_hidden_by_small_sample():
    data = evidence((100.0, 90.0))
    data["policy"] = {"min_net_return": 0.0, "max_drawdown": 0.15}
    report = evaluate(data)
    assert report["gates"]["R0"]["status"] == "FAIL"
    assert not report["development_screen_passed"]


def test_missing_identity_cannot_pass_supplied_identity_attestation():
    data = evidence()
    data["identity"] = {}
    data["engineering"] = {"E0": {"status": "PASS"}}
    report = evaluate(data)
    assert report["gates"]["E0"]["status"] == "NOT_EVALUABLE"


def test_policy_cannot_remove_missing_research_gates_from_rankability():
    data = evidence()
    data["policy"] = {"required_gates": [], "basis": "production_approved"}
    report = evaluate(data)
    assert report["rankable"] is False
    assert report["live_eligible"] is False
    assert report["policy"]["basis"] == "design_v1"


def test_basic_development_screen_does_not_claim_complete_research_quality():
    data = evidence()
    data["policy"] = {"min_net_return": 0.0, "max_drawdown": 0.15, "min_daily_returns": 2}
    data["episodes"] = [{"closed": True, "net_pnl": pnl} for pnl in (4.0, -2.0)] * 15
    data["engineering"] = {f"E{i}": {"status": "PASS", "reason": "fixture_proof"} for i in range(5)}
    report = evaluate(data)
    assert report["development_screen_passed"] is True
    assert report["development_screens"]["net_return"]["status"] == "PASS"
    assert report["development_screens"]["max_drawdown"]["status"] == "PASS"
    assert report["gates"]["R0"]["status"] == "NOT_EVALUABLE"
    assert report["gates"]["R2"]["status"] == "NOT_EVALUABLE"
    assert report["rankable"] is False
    assert report["diagnostic_only"] is True


def test_future_quote_cannot_enter_a_valuation_even_with_supplied_pass():
    data = evidence()
    data["curve"][1]["source_time_ns"] = 2 * DAY
    data["engineering"] = {f"E{i}": {"status": "PASS"} for i in range(5)}
    with pytest.raises(ValueError, match="source_time"):
        evaluate(data)


@pytest.mark.parametrize("flag", ["diagnostic_only", "source_completed"])
def test_known_incomplete_or_diagnostic_input_cannot_pass_development_screen(flag):
    data = evidence((100.0, 110.0, 105.0, 115.0))
    data["episodes"] = [{"closed": True, "net_pnl": 2.0}, {"closed": True, "net_pnl": -1.0}]
    data["engineering"] = {f"E{i}": {"status": "PASS"} for i in range(5)}
    data["policy"] = {"min_daily_returns": 2, "min_closed_episodes": 2, "min_net_return": 0.0, "max_drawdown": 0.15}
    data[flag] = flag == "diagnostic_only"
    report = evaluate(data)
    assert report["development_screen_passed"] is False
    assert report["diagnostic_only"] is True
