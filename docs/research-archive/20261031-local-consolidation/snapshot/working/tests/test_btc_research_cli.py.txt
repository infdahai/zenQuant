def test_fixed_default_dev_replays_all_frozen_candidates_and_costs(cli, tmp_path):
    module, calls = cli
    output = tmp_path / "fixed"
    assert module.main(["--output", str(output)]) == 0
    assert calls["loads"] == [(1704067200000000000, 1767225600000000000, 180)]
    assert len(calls["evaluations"]) == 18
    assert {row[0] for row in calls["evaluations"]} == {
        "trend",
        "breakout",
        "range_reversion",
        "cash",
        "vol_target",
        "buy_hold_1x",
    }
    assert {row[2]["extra_cost_bps"] for row in calls["evaluations"]} == {0, 5, 15}
    assert all(row[2]["window"] == "dev" for row in calls["evaluations"])
    summary = json.loads((output / "summary.json").read_text())
    assert summary["complete"] is True
    assert summary["rankable"] is False
    assert summary["live_eligible"] is False
    assert (output / "completed.json").exists()
    assert (output / "REPORT.md").exists()
    case = summary["cases"][0]
    compressed = (output / case["evidence_path"]).read_bytes()
    assert compressed[4:8] == b"\x00" * 4
    assert json.loads(gzip.decompress(compressed))["fixture"] == case["family"]
    assert case["evidence_sha256"] and case["report_sha256"]
    assert json.loads((output / "run_manifest.json").read_text())["code"]["files"]["fixture.py"] == "content-hash"

def test_failed_fixed_attempt_is_recorded_and_does_not_skip_the_rest(cli, tmp_path, monkeypatch):
    module, calls = cli
    original = module.evaluate_candidate

    def fail_one(bundle, family, params, **kwargs):

            raise ValueError("fixture-failure")
        return original(bundle, family, params, **kwargs)

    monkeypatch.setattr(module, "evaluate_candidate", fail_one)
    output = tmp_path / "failed"
    assert module.main(["--output", str(output)]) == 2
    summary = json.loads((output / "summary.json").read_text())
    assert len(summary["cases"]) == 18
    failed = [row for row in summary["cases"] if row["status"] == "FAILED"]
    assert len(failed) == 1
    assert "fixture-failure" in failed[0]["error"]
    assert (output / failed[0]["report_path"]).exists()
    assert summary["complete"] is True
    assert (output / "completed.json").exists()

def test_search_retains_known_accounting_failures_without_optimizer_feedback(cli, tmp_path, monkeypatch):
    module, calls = cli
    original = module.evaluate_candidate

    def broken_accounting(bundle, family, params, **kwargs):
        evidence, report = original(bundle, family, params, **kwargs)
        if family == "trend":
            report["gates"]["E2"] = {"status": "FAIL", "reason": "cash mismatch"}
        return evidence, report

    monkeypatch.setattr(module, "evaluate_candidate", broken_accounting)
    output = tmp_path / "failed-search"
    assert (
        module.main(
            ["--mode", "search", "--family", "trend", "--sampler", "tpe", "--budget", "2", "--output", str(output)]

        == 2
    )
    summary = json.loads((output / "summary.json").read_text())
    search = json.loads((output / "search_summary.json").read_text())
    assert len(calls["evaluations"]) == 3
    assert summary["failed_attempts"] == 2
    assert all(row["status"] == "FAILED" and row["objectives"] is None for row in search["attempts"])
    assert all((output / row["evidence_path"]).exists() for row in summary["cases"])
    assert (output / "completed.json").exists()


@pytest.mark.parametrize("failure", [ValueError("missing-data"), KeyboardInterrupt()])
def test_unfinished_run_preserves_error_and_has_no_completion(cli, tmp_path, monkeypatch, failure):
    module, _calls = cli

    def fail(*args, **kwargs):
        raise failure

    monkeypatch.setattr(module, "load_window" if isinstance(failure, ValueError) else "evaluate_candidate", fail)
    output = tmp_path / "unfinished"
    assert module.main(["--output", str(output)]) == 2
    summary = json.loads((output / "summary.json").read_text())
    manifest = json.loads((output / "run_manifest.json").read_text())
    assert summary["manifest_hash"] == module.identity_hash(manifest)
    assert summary["complete"] is False
    assert summary["error"].startswith(type(failure).__name__)
    assert not (output / "completed.json").exists()


def test_code_identity_tracks_uncommitted_file_bytes(cli, tmp_path, monkeypatch):
    module, _calls = cli
    # Reload the actual hashing helper replaced by the orchestration fixture.
    spec = importlib.util.spec_from_file_location("btc_research_identity", module.__file__)
    fresh = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(fresh)
    script = tmp_path / "scripts/run_btc_research.py"
    script.parent.mkdir()
    script.write_text("# fixture script")
    source = tmp_path / "zenquant/untracked.py"
    source.parent.mkdir()
    source.write_text("VALUE = 1")
    catalog = tmp_path / "catalog.json"
    catalog.write_text("{}")
    monkeypatch.setattr(fresh, "ROOT", tmp_path)
    monkeypatch.setattr(fresh, "CATALOG", catalog)
    monkeypatch.setattr(fresh, "__file__", str(script))
    before = fresh._code_identity()
    source.write_text("VALUE = 2")
    after = fresh._code_identity()
    assert before["git_head"] == after["git_head"]
    assert before["source_content_hash"] != after["source_content_hash"]
    assert before["files"]["zenquant/untracked.py"] != after["files"]["zenquant/untracked.py"]


def test_search_report_write_failures_are_counted_for_the_full_budget(cli, tmp_path, monkeypatch):
    module, calls = cli
    original = module._write

    def fail_trial_report(path, value, **kwargs):
        if path.name.startswith("trial-") and path.name.endswith(".report.json"):
            raise OSError("fixture report write failure")
        return original(path, value, **kwargs)

    monkeypatch.setattr(module, "_write", fail_trial_report)
    output = tmp_path / "report-failures"
    result = module.main(
        ["--mode", "search", "--family", "trend", "--sampler", "random", "--budget", "8", "--output", str(output)]
    )
    assert result == 2
    summary = json.loads((output / "summary.json").read_text())
    search = json.loads((output / "search_summary.json").read_text())
    assert len(calls["evaluations"]) == 9
    assert len(search["attempts"]) == 8
    assert summary["status"] == "COMPLETE_WITH_FAILURES"
    assert summary["complete"] is True
    assert summary["failed_attempts"] == 8
    assert summary["compute"]["finished_evaluations"] == 9
    assert len(summary["cases"]) == 9
    for case in summary["cases"][1:]:
        assert case["status"] == "FAILED"
        assert "fixture report write failure" in case["error"]
        assert case["family"] == "trend"
        assert case["window"] == "train"
        assert case["params"]
        assert not {"metrics", "report_path", "report_sha256", "evidence_path", "evidence_sha256"} & case.keys()
    assert (output / "completed.json").exists()
    assert "[Full report](trial-" not in (output / "REPORT.md").read_text()
