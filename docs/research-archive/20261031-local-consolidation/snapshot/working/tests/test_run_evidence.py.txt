import hashlib
import json
from concurrent.futures import ThreadPoolExecutor
from copy import deepcopy
from pathlib import Path
from types import MappingProxyType

import pytest

from zenquant.research.run_evidence import TrialJournal, identity_hash


def _identity() -> dict:
    return {
        "code": "accounting-v2",
        "data": "data-sha256",
        "config": {"params": {"threshold": 0.5}, "cost_bps": 5},
        "policy": "evaluation-v1",
        "seed": 42,
        "window": {"start_ns": 100, "end_ns": 200},
    }


def test_canonical_hash_preserves_full_content_and_mapping_order() -> None:
    # Canonical bytes are specified independently of the implementation.
    expected = hashlib.sha256(b'{"a":{"x":true},"b":[1,null]}').hexdigest()
    assert identity_hash({"b": [1, None], "a": {"x": True}}) == expected
    assert identity_hash(MappingProxyType({"a": {"x": True}, "b": [1, None]})) == expected
    assert identity_hash({"value": 1}) != identity_hash({"value": True})
    assert identity_hash({"value": 1}) != identity_hash({"value": 1.0})


@pytest.mark.parametrize("field", ["code", "data", "config", "policy", "seed", "window"])
def test_every_run_identity_dimension_affects_hash(field: str) -> None:
    original = _identity()
    changed = deepcopy(original)
    changed[field] = "changed"
    assert identity_hash(original) != identity_hash(changed)


@pytest.mark.parametrize("invalid", [float("nan"), float("inf"), -float("inf"), {1, 2}, (1, 2), Path("x"), object()])
def test_identity_rejects_non_json_and_nonfinite_nested_values(invalid: object) -> None:
    with pytest.raises(ValueError):
        identity_hash({"nested": [invalid]})


def test_identity_rejects_non_string_keys_and_cycles() -> None:
    with pytest.raises(ValueError):
        identity_hash({1: "cannot-coerce"})
    circular: dict = {}
    circular["self"] = circular
    with pytest.raises(ValueError):
        identity_hash(circular)


def test_same_params_distinct_attempts_survive_restart(tmp_path: Path) -> None:
    journal = TrialJournal(tmp_path, _identity())
    failed = {"attempt_id": "one", "params": {"x": 1}, "status": "failed", "error": "timeout"}
    completed = {"attempt_id": "two", "params": {"x": 1}, "status": "completed", "net_pnl": -3}
    assert journal.append(failed) == failed
    assert journal.append(completed) == completed
    assert TrialJournal(tmp_path, _identity()).records() == [failed, completed]


def test_mutating_inputs_or_return_values_cannot_change_persisted_evidence(tmp_path: Path) -> None:
    identity = _identity()
    journal = TrialJournal(tmp_path, identity)
    identity["seed"] = 100
    record = {"attempt_id": "one", "params": {"x": 1}}
    returned = journal.append(record)
    record["params"]["x"] = 2
    returned["params"]["x"] = 3
    read_back = journal.records()
    read_back[0]["params"]["x"] = 4
    assert journal.records() == [{"attempt_id": "one", "params": {"x": 1}}]


@pytest.mark.parametrize("changed", [False, True])
def test_duplicate_attempt_is_rejected_without_appending(tmp_path: Path, changed: bool) -> None:
    journal = TrialJournal(tmp_path, _identity())
    journal.append({"attempt_id": "one", "status": "failed"})
    before = (tmp_path / "trials.jsonl").read_bytes()
    with pytest.raises(ValueError, match="attempt_id"):
        journal.append({"attempt_id": "one", "status": "completed" if changed else "failed"})
    assert (tmp_path / "trials.jsonl").read_bytes() == before


@pytest.mark.parametrize(
    "record",
    [{}, {"attempt_id": ""}, {"attempt_id": "  "}, {"attempt_id": 1}, {"attempt_id": "a", "pnl": float("nan")}],
)
def test_invalid_record_is_rejected_before_any_append(tmp_path: Path, record: dict) -> None:
    journal = TrialJournal(tmp_path, _identity())
    with pytest.raises(ValueError):
        journal.append(record)
    assert journal.records() == []


@pytest.mark.parametrize("field", ["code", "data", "config", "policy", "seed", "window"])
def test_resume_requires_exact_identity_and_does_not_overwrite_manifest(tmp_path: Path, field: str) -> None:
    TrialJournal(tmp_path, _identity()).append({"attempt_id": "one"})
    before = {name: (tmp_path / name).read_bytes() for name in ("identity.json", "trials.jsonl", "journal_state.json")}
    changed = _identity()
    changed[field] = "changed"
    with pytest.raises(ValueError, match="identity"):
        TrialJournal(tmp_path, changed)
    assert {name: (tmp_path / name).read_bytes() for name in before} == before


def test_same_parameters_with_distinct_seeds_are_separate_runs(tmp_path: Path) -> None:
    first = _identity()
    second = {**first, "seed": 123}
    record = {"attempt_id": "one", "params": {"threshold": 0.5}}
    TrialJournal(tmp_path / "first", first).append(record)
    TrialJournal(tmp_path / "second", second).append(record)
    assert TrialJournal(tmp_path / "first", first).records() == [record]
    assert TrialJournal(tmp_path / "second", second).records() == [record]


@pytest.mark.parametrize(
    "damage", ["partial_line", "whole_line", "invalid_json", "changed_payload", "missing_journal", "missing_state"]
)
def test_corrupt_or_truncated_journal_fails_closed(tmp_path: Path, damage: str) -> None:
    journal = TrialJournal(tmp_path, _identity())
    journal.append({"attempt_id": "one", "status": "failed"})
    journal.append({"attempt_id": "two", "status": "completed"})
    path = tmp_path / "trials.jsonl"
    content = path.read_bytes()
    if damage == "partial_line":
        path.write_bytes(content[:-2])
    elif damage == "whole_line":
        path.write_bytes(content.splitlines(keepends=True)[0])
    elif damage == "invalid_json":
        path.write_bytes(b"not-json\n")
    elif damage == "changed_payload":
        path.write_bytes(content.replace(b'"failed"', b'"passed"'))
    else:
        (tmp_path / ("trials.jsonl" if damage == "missing_journal" else "journal_state.json")).unlink()
    with pytest.raises(ValueError):
        TrialJournal(tmp_path, _identity())
    with pytest.raises(ValueError):
        journal.records()
    with pytest.raises(ValueError):
        journal.append({"attempt_id": "three"})
    if path.exists():
        assert b'"three"' not in path.read_bytes()


def test_append_before_state_commit_cannot_appear_complete(tmp_path: Path) -> None:
    journal = TrialJournal(tmp_path, _identity())
    journal.append({"attempt_id": "one"})
    committed_state = (tmp_path / "journal_state.json").read_bytes()
    journal.append({"attempt_id": "two", "status": "completed"})
    # Model a crash after journal fsync but before atomic state publication.
    (tmp_path / "journal_state.json").write_bytes(committed_state)
    with pytest.raises(ValueError):
        TrialJournal(tmp_path, _identity())


def test_manifest_corruption_and_missing_identity_are_never_repaired(tmp_path: Path) -> None:
    journal = TrialJournal(tmp_path, _identity())
    path = tmp_path / "identity.json"
    original = path.read_bytes()
    path.write_bytes(original.replace(b'"seed":42', b'"seed":43'))
    with pytest.raises(ValueError):
        journal.append({"attempt_id": "one"})
    path.unlink()
    with pytest.raises(ValueError):
        TrialJournal(tmp_path, _identity())
    assert not path.exists()


def test_duplicate_json_keys_in_manifest_are_not_silently_collapsed(tmp_path: Path) -> None:
    TrialJournal(tmp_path, _identity())
    path = tmp_path / "identity.json"
    original = path.read_bytes()
    assert json.loads(original)
    path.write_bytes(original.replace(b'"seed":42', b'"seed":42,"seed":42'))
    with pytest.raises(ValueError):
        TrialJournal(tmp_path, _identity())


def test_local_concurrent_writers_serialize_and_keep_every_attempt(tmp_path: Path) -> None:
    def append_one(index: int) -> None:
        TrialJournal(tmp_path, _identity()).append({"attempt_id": str(index), "params": {"x": 1}})

    with ThreadPoolExecutor(max_workers=4) as pool:
        list(pool.map(append_one, range(12)))
    records = TrialJournal(tmp_path, _identity()).records()
    assert len(records) == 12
    assert {record["attempt_id"] for record in records} == {str(index) for index in range(12)}
