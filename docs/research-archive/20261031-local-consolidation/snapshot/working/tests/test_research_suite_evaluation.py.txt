"""Adversarial contracts for multi-run historical research conclusions."""

from __future__ import annotations

import copy
import json

import pytest

DAY_NS = 86_400_000_000_000
GATES = [*(f"E{i}" for i in range(5)), *(f"R{i}" for i in range(8)), "O0", "O1"]


def call(name, *args, **kwargs):
    from zenquant.research import suite_evaluation

    return getattr(suite_evaluation, name)(*args, **kwargs)


def paired_report(candidate_id="candidate", daily=0.002, count=200):
    return {
        "identity": {"candidate_id": candidate_id},
        "comparison_contract": {
            "symbol": "BTCUSDT_PERP",
            "window": {"start_ns": 0, "end_ns": count * DAY_NS},
            "data_hash": "data-v1",
            "accounting_hash": "accounting-v2",
            "execution": {"timing": "next_available_open"},
            "cost": {"fee_bps": 4.0},
            "risk_budget": {"target_annual_vol": 0.1, "max_leverage": 1.5},
        },
        "daily_returns": [{"time_ns": (index + 1) * DAY_NS, "value": daily} for index in range(count)],
    }


def test_paired_constant_excess_has_hand_known_confidence_interval_and_is_reproducible():
    candidate, baseline = paired_report(), paired_report("baseline", daily=0.001)
    before = copy.deepcopy(candidate)
    result = call("paired_daily_excess", candidate, baseline, block_lengths=[2, 5, 10], repetitions=200, seed=7)
    assert result["status"] == "PASS"
    assert result["mean_daily_excess"] == pytest.approx(0.001)
    assert result["sample_size"] == 200
    assert result["sensitive"] is False
    for block in result["blocks"]:
        assert block["confidence_interval"] == pytest.approx([0.001, 0.001])
    assert result == call("paired_daily_excess", candidate, baseline, block_lengths=[2, 5, 10], repetitions=200, seed=7)
    assert candidate == before
    assert result["live_eligible"] is False


def test_default_two_thousand_bootstrap_repetitions_are_executed_for_each_block():
    result = call(
        "paired_daily_excess", paired_report(), paired_report("baseline", daily=0.001), block_lengths=[2, 5, 10]
    )
    assert result["status"] == "PASS"
    assert result["repetitions"] == 2000
    assert len(result["blocks"]) == 3


def test_paired_block_sensitivity_cannot_choose_the_most_favorable_interval():
    candidate, baseline = paired_report(), paired_report("baseline", daily=0.001)
    for row in candidate["daily_returns"][100:]:
        row["value"] = 0.0005
    result = call("paired_daily_excess", candidate, baseline, block_lengths=[1, 20, 80], repetitions=500, seed=7)
    assert result["status"] == "NOT_EVALUABLE"
    assert result["sensitive"] is True
    assert result["blocks"][0]["confidence_interval"][0] > 0.0
    assert result["blocks"][-1]["confidence_interval"][0] < 0.0


@pytest.mark.parametrize("damage", ["missing_contract", "cost_mismatch", "missing_identity", "timestamp_mismatch"])
def test_paired_comparison_requires_complete_matching_contracts_and_dates(damage):
    candidate, baseline = paired_report(), paired_report("baseline", daily=0.001)
    if damage == "missing_contract":
        del candidate["comparison_contract"]["execution"]
    elif damage == "cost_mismatch":
        baseline["comparison_contract"]["cost"]["fee_bps"] = 0.0
    elif damage == "missing_identity":
        candidate["identity"] = {}
    else:
        baseline["daily_returns"][-1]["time_ns"] += DAY_NS
    result = call("paired_daily_excess", candidate, baseline, block_lengths=[2, 5, 10], repetitions=10)
    assert result["status"] == "NOT_EVALUABLE"
    assert result["missing"]


@pytest.mark.parametrize("damage", ["nan", "duplicate_time"])
def test_paired_invalid_returns_are_rejected(damage):
    candidate, baseline = paired_report(), paired_report("baseline", daily=0.001)
    if damage == "nan":
        candidate["daily_returns"][2]["value"] = float("nan")
    else:
        candidate["daily_returns"][2]["time_ns"] = candidate["daily_returns"][1]["time_ns"]
    result = call("paired_daily_excess", candidate, baseline, block_lengths=[2, 5, 10])
    assert result["status"] == "INVALID"
    json.dumps(result, allow_nan=False)


def test_matching_missing_days_are_not_treated_as_consecutive_daily_observations():
    candidate, baseline = paired_report(), paired_report("baseline", daily=0.001)
    del candidate["daily_returns"][10]
    del baseline["daily_returns"][10]
    result = call("paired_daily_excess", candidate, baseline, block_lengths=[2, 5, 10])
    assert result["status"] == "NOT_EVALUABLE"
    assert result["missing"]


def test_insufficient_paired_samples_and_invalid_block_protocol_never_pass():
    result = call(
        "paired_daily_excess", paired_report(count=20), paired_report("base", count=20), block_lengths=[2, 5, 10]
    )
    assert result["status"] == "NOT_EVALUABLE"

def test_paired_positive_difference_cannot_override_explicit_engineering_failure():

def test_return_that_implies_ruin_cannot_produce_increment_pass():

    candidate["daily_returns"][10]["value"] = -1.0

def protocol():
    return {
        "seeds": [1, 2, 3, 4, 5],
        "windows": ["dev", "bear"],
        "costs": [0, 5, 15],
        "aggregate_window": "dev",
        "required_positive_costs": [0, 5],
        "max_drawdown": 0.15,
        "min_pass_seeds": 4,
        "candidate_id": "candidate",
    }


def cases():
    result = []
    for seed in (1, 2, 3, 4, 5):
        for window in ("dev", "bear"):
            for cost in (0, 5, 15):
                result.append(
                    {
                        "seed": seed,
                        "window": window,
                        "extra_cost_bps": cost,
                        "report": {
                            "identity": {"candidate_id": "candidate", "seed": seed, "window": window},
                            "gates": {f"E{i}": {"status": "PASS"} for i in range(5)},
                            "metrics": {
                                "net_return": {"value": 0.1 if window == "dev" else -0.02, "status": "PASS"},
                                "max_drawdown": {"value": 0.1, "status": "PASS"},
                            },
                        },
                    }
                )
    return result


def test_complete_matrix_can_pass_economic_screen_without_15bp_promotion_authority():
    result = call("evaluate_robustness", cases(), protocol())
    assert result["status"] == "PASS"
    assert result["passing_seeds"] == [1, 2, 3, 4, 5]
    assert result["expected_case_count"] == 30
    assert result["promotion_ready"] is False
    assert result["promotion_threshold_status"] == "NOT_EVALUABLE"
    assert result["live_eligible"] is False


def test_same_four_seeds_must_pass_all_costs_instead_of_costwise_cherry_picking():
    matrix = cases()
    for cell in matrix:
        if cell["window"] == "dev" and (cell["seed"], cell["extra_cost_bps"]) in {(1, 0), (2, 5)}:
            cell["report"]["metrics"]["net_return"]["value"] = -0.01
    result = call("evaluate_robustness", matrix, protocol())
    assert result["status"] == "FAIL"
    assert result["passing_seeds"] == [3, 4, 5]


def test_missing_case_prevents_reducing_seed_or_matrix_denominator():
    matrix = cases()
    missing = matrix.pop()
    result = call("evaluate_robustness", matrix, protocol())
    assert result["status"] == "NOT_EVALUABLE"
    assert result["expected_case_count"] == 30
    assert {key: missing[key] for key in ("seed", "window", "extra_cost_bps")} in result["missing_cases"]


def test_duplicate_case_invalidates_even_identical_duplicate():
    matrix = cases()
    matrix.append(copy.deepcopy(matrix[0]))
    assert call("evaluate_robustness", matrix, protocol())["status"] == "INVALID"


@pytest.mark.parametrize(
    "engineering_status,expected", [("FAIL", "FAIL"), ("INVALID", "INVALID"), ("NOT_RUN", "NOT_EVALUABLE")]
)
def test_fifth_seed_engineering_problem_cannot_be_voted_away(engineering_status, expected):
    matrix = cases()
    matrix[-1]["report"]["gates"]["E2"] = {"status": engineering_status, "reason": "accounting_issue"}
    result = call("evaluate_robustness", matrix, protocol())
    assert result["status"] == expected
    assert result["blockers"]


def test_all_required_windows_and_15bp_drawdown_are_checked():
    matrix = cases()
    for cell in matrix:
        if cell["window"] == "bear" and cell["extra_cost_bps"] == 15 and cell["seed"] in (1, 2):
            cell["report"]["metrics"]["max_drawdown"]["value"] = 0.16
    result = call("evaluate_robustness", matrix, protocol())
    assert result["status"] == "FAIL"
    assert result["passing_seeds"] == [3, 4, 5]


@pytest.mark.parametrize("damage", ["empty_identity", "mixed_candidate", "nan_metric", "missing_metric"])
def test_matrix_invalid_or_missing_identity_and_metrics_do_not_pass(damage):
    matrix = cases()
    report = matrix[0]["report"]
    if damage == "empty_identity":
        report["identity"] = {}
    elif damage == "mixed_candidate":
        report["identity"]["candidate_id"] = "other"
    elif damage == "nan_metric":
        report["metrics"]["net_return"]["value"] = float("nan")
    else:
        del report["metrics"]["max_drawdown"]
    result = call("evaluate_robustness", matrix, protocol())
    assert result["status"] in {"INVALID", "NOT_EVALUABLE"}
    assert result["blockers"] or result["missing"]
    json.dumps(result, allow_nan=False)


def test_frozen_15bp_threshold_is_separate_from_basic_seed_survival():
    config = protocol()
    config["promotion_min_net_return_by_cost"] = {"15": 0.0}
    matrix = cases()
    assert call("evaluate_robustness", matrix, config)["promotion_ready"] is True
    for cell in matrix:
        if cell["window"] == "dev" and cell["extra_cost_bps"] == 15 and cell["seed"] in (1, 2):
            cell["report"]["metrics"]["net_return"]["value"] = -0.01
    result = call("evaluate_robustness", matrix, config)
    assert result["status"] == "PASS"
    assert result["promotion_ready"] is False


def test_robustness_cost_cell_must_match_cost_declared_by_report_identity():
    matrix = cases()
    matrix[0]["report"]["identity"]["extra_cost_bps"] = 15
    assert call("evaluate_robustness", matrix, protocol())["status"] == "INVALID"


def test_malformed_matrix_metrics_return_invalid_instead_of_crashing():

def candidate(identifier, excess, drawdown, cost, rankable=True):
    return {
        "candidate_id": identifier,
        "identity": {"candidate_id": identifier},
        "rankable": rankable,
        "axes": {"net_excess": excess, "max_drawdown": drawdown, "cost": cost},
    }


def test_pareto_respects_axis_direction_excludes_blocked_and_keeps_stable_ties():
    items = [
        candidate("tie_b", 0.10, 0.10, 0.02),
        candidate("dominated", 0.08, 0.11, 0.03),
        candidate("cheap", 0.09, 0.10, 0.01),
        candidate("tie_a", 0.10, 0.10, 0.02),
        candidate("unsafe_winner", 5.0, 0.01, 0.001, rankable=False),
    ]
    result = call("pareto_candidates", items)
    assert result["frontier_ids"] == ["cheap", "tie_a", "tie_b"]
    assert result["dominated_ids"] == ["dominated"]
    assert result["excluded"][0]["candidate_id"] == "unsafe_winner"
    assert call("pareto_candidates", list(reversed(items))) == result


@pytest.mark.parametrize("damage", ["nonfinite", "missing_axis", "empty_identity", "truthy_rankable"])
def test_pareto_rejects_invalid_values_and_non_boolean_rankability(damage):
    item = candidate("bad", 0.1, 0.1, 0.1)
    if damage == "nonfinite":
        item["axes"]["net_excess"] = float("inf")
    elif damage == "missing_axis":
        del item["axes"]["cost"]
    elif damage == "empty_identity":
        item["identity"] = {}
    else:
        item["rankable"] = "yes"
    result = call("pareto_candidates", [item])
    assert result["frontier_ids"] == []
    assert result["excluded"][0]["reason"]
    json.dumps(result, allow_nan=False)


def test_duplicate_pareto_ids_are_invalid():
    item = candidate("same", 0.1, 0.1, 0.1)
    assert call("pareto_candidates", [item, item])["status"] == "INVALID"


def test_pareto_preserves_actual_blockers_and_rejects_inconsistent_rankable_flag():
    item = candidate("blocked", 5.0, 0.01, 0.001)
    item["blockers"] = [{"gate": "E2", "status": "FAIL", "reason": "cash_created"}]
    result = call("pareto_candidates", [item])
    assert result["frontier_ids"] == []
    assert result["excluded"][0]["blockers"] == item["blockers"]


def test_pareto_candidate_id_must_match_identity_binding():
    item = candidate("one", 0.1, 0.1, 0.1)
    item["identity"]["candidate_id"] = "other"
    assert call("pareto_candidates", [item])["frontier_ids"] == []


@pytest.mark.parametrize(
    "override,expected",
    [
        ({"E2": "INVALID", "E3": "STALE"}, "INVALID"),
        ({"E2": "FAIL", "E3": "STALE"}, "STALE"),
        ({"E2": "FAIL"}, "BLOCKED"),
        ({"R6": "NOT_RUN"}, "DIAGNOSTIC"),
        ({"O0": "NOT_RUN", "O1": "NOT_EVALUABLE"}, "HISTORICAL_ONLY"),
        ({}, "REVIEW_REQUIRED"),
    ],
)
def test_status_projection_separates_invalid_blocked_missing_history_and_review(override, expected):
    report = {"identity": {"candidate_id": "candidate"}, "gates": {name: {"status": "PASS"} for name in GATES}}
    for name, status in override.items():
        report["gates"][name]["status"] = status
    report["live_eligible"] = True
    result = call("project_candidate_status", report)
    assert result["status"] == expected
    assert result["live_eligible"] is False


def test_status_projection_cannot_pass_empty_identity_or_missing_gate():
    report = {"identity": {}, "gates": {name: {"status": "PASS"} for name in GATES}}
    result = call("project_candidate_status", report)
    assert result["status"] == "DIAGNOSTIC"
    assert result["missing"]
    report["identity"] = {"candidate_id": "candidate"}
    del report["gates"]["E3"]
    result = call("project_candidate_status", report)
    assert result["status"] == "DIAGNOSTIC"
    assert any(item["gate"] == "E3" for item in result["missing"])


def test_status_projection_preserves_external_hard_blocker_despite_all_pass_gates():
    report = {
        "identity": {"candidate_id": "candidate"},
        "gates": {name: {"status": "PASS"} for name in GATES},
        "blockers": [{"status": "INVALID", "reason": "evidence_hash_mismatch"}],
    }
    result = call("project_candidate_status", report)
    assert result["status"] == "INVALID"

def test_malformed_status_is_invalid_instead_of_unhashable_type_error():

@pytest.mark.parametrize("function", ["paired_daily_excess", "evaluate_robustness"])
def test_external_hard_blocker_cannot_be_hidden_by_passing_metrics(function):
    blocker = {"status": "INVALID", "reason": "evidence_hash_mismatch"}
    if function == "paired_daily_excess":
        candidate, baseline = paired_report(), paired_report("baseline", daily=0.001)
        candidate["blockers"] = [blocker]
        result = call(function, candidate, baseline, block_lengths=[2, 5, 10])
    else:
        matrix = cases()
        matrix[-1]["report"]["blockers"] = [blocker]
        result = call(function, matrix, protocol())
    assert result["status"] == "INVALID"
    assert any(item["reason"] == blocker["reason"] for item in result["blockers"])


def test_status_projection_missing_gate_container_is_diagnostic_with_missing_path():
    result = call("project_candidate_status", {"identity": {"run_id": "run"}, "gates": None})
    assert result["status"] == "DIAGNOSTIC"
    assert result["missing"][0]["path"] == "report.gates"


def test_fifth_seed_ruin_implied_by_metrics_cannot_be_voted_away():
    matrix = cases()
    matrix[-1]["report"]["metrics"]["net_return"]["value"] = -1.0
    matrix[-1]["report"]["metrics"]["max_drawdown"]["value"] = 1.0

    assert result["status"] == "FAIL"
    assert any(item["reason"] == "RUIN" for item in result["blockers"])


def real_evaluation_matrix(fifth_values):
    from zenquant.research.evaluation import evaluate_evidence

    matrix = cases()
    for cell in matrix:
        values = fifth_values if cell["seed"] == 5 else [100.0, 110.0]
        cell["report"] = evaluate_evidence(
            {
                "identity": {"candidate_id": "candidate", "seed": cell["seed"], "window": cell["window"]},
                "initial_equity": 100.0,
                "curve": [{"time_ns": index * DAY_NS, "equity": value} for index, value in enumerate(values)],
                "fills": [],
                "funding": [],
                "episodes": [],
                "engineering": {
                    f"E{i}": {"status": "PASS", "reason": "INDEPENDENT_ACCOUNTING_FIXTURE"} for i in range(5)
                },
                "policy": {"min_net_return": 0.0, "max_drawdown": 0.15},
            }
        )
        # Other dimensions are unproven. This test isolates the robustness
        # economic screen; it does not grant overall report rankability.
        cell["report"]["blockers"] = [item for item in cell["report"]["blockers"] if item["status"] == "FAIL"]
    return matrix


@pytest.mark.parametrize("fifth_values", [[100.0, 90.0], [100.0, 80.0, 110.0]])
def test_real_evaluator_fifth_seed_economic_screen_failure_can_use_four_of_five_rule(fifth_values):
    matrix = real_evaluation_matrix(fifth_values)
    assert matrix[-1]["report"]["blockers"]
    result = call("evaluate_robustness", matrix, protocol())
    assert result["status"] == "PASS"
    assert result["passing_seeds"] == [1, 2, 3, 4]
    assert result["seed_results"][-1]["economic_screen_passed"] is False
    assert result["promotion_ready"] is False


@pytest.mark.parametrize(
    "blocker",
    [
        {"gate": "E2", "status": "FAIL", "reason": "FROZEN_NET_RETURN_SCREEN"},
        {"gate": "R2", "status": "FAIL", "reason": "RUIN"},
        {"gate": "R2", "status": "FAIL", "reason": "UNKNOWN_RISK_FAILURE"},
        {"gate": "R0", "status": "FAIL", "reason": "FROZEN_NET_RETURN_SCREEN", "scope": "global"},
        {"gate": "GLOBAL", "status": "INVALID", "reason": "IDENTITY_MISMATCH"},
    ],
)
def test_real_evaluator_economic_vote_does_not_waive_engineering_ruin_or_global_blockers(blocker):
    matrix = real_evaluation_matrix([100.0, 90.0])
    matrix[-1]["report"]["blockers"].append(blocker)
    result = call("evaluate_robustness", matrix, protocol())
    assert result["status"] == blocker["status"]
    assert any(item["reason"] == blocker["reason"] for item in result["blockers"])
