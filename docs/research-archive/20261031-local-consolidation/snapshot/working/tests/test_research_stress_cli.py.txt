"""Failure-path checks for the fixed diagnostic stress executor."""

import importlib.util
import json
from pathlib import Path

import pytest


@pytest.fixture
def cli():
    path = Path(__file__).parents[1] / "scripts/run_research_stress.py"
    spec = importlib.util.spec_from_file_location("research_stress_cli", path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def plan(cli):
    return {
        "schema_version": "fixed-price-stress.v1",
        "cases": cli.build_cases(),
        "inventory_policy": "actual_filled_inventory_v1",
        "initial_equity": 100_000.0,
        "start_ns": 1704067200000000000,
        "end_ns_exclusive": 1767225600000000000,
        "search_trials": 0,
        "code_hash": "test-frozen-code",
        "data_files_sha256": {
            path: "0" * 64
            for path in (
                "klines/BTCUSDT/4h/data.parquet",
                "klines/ETHUSDT/4h/data.parquet",
                "funding/BTCUSDT/data.parquet",
                "funding/ETHUSDT/data.parquet",
                "snapshot_manifest.json",
            )
        },
    }


def test_duplicate_case_cannot_replace_a_required_seed(cli):
    value = plan(cli)
    value["cases"][-1] = value["cases"][0]
    with pytest.raises(ValueError, match="frozen matrix"):
        cli.validate_plan(value)


@pytest.mark.parametrize("missing", ["all", "funding/ETHUSDT/data.parquet"])
def test_incomplete_source_set_is_rejected_before_output(cli, missing):
    value = plan(cli)
    if missing == "all":
        value["data_files_sha256"] = {}
    else:
        del value["data_files_sha256"][missing]
    with pytest.raises(ValueError, match="frozen data set"):
        cli.validate_plan(value)


def setup_job(cli, monkeypatch, tmp_path):
    value = plan(cli)
    value["data_root"] = str(tmp_path)
    source = tmp_path / "plan.json"
    source.write_text(json.dumps(value))
    monkeypatch.setattr(cli, "verify_sources", lambda value: None)
    monkeypatch.setattr(cli, "load_window", lambda *a, **kw: {"manifest": {}, "symbol": kw["symbol"]})
    monkeypatch.setattr(cli, "perturb_bundle", lambda bundle, **kw: bundle)
    return source


def test_failed_attempt_is_retained_and_remaining_matrix_finishes(cli, monkeypatch, tmp_path):
    source = setup_job(cli, monkeypatch, tmp_path)
    calls = []

    def evaluate(*a, **kw):
        calls.append(kw)
        if len(calls) == 1:
            raise ValueError("real failed attempt")
        return {"fills": [], "curve": []}, {
            "gates": {"E3": {"status": "FAIL", "reason": "known production difference"}},
            "metrics": {},
            "optimizer_feedback_eligible": False,
            "rankable": False,
            "live_eligible": False,
        }

    monkeypatch.setattr(cli, "evaluate_candidate", evaluate)
    output = tmp_path / "run"
    result = cli.main(["--plan", str(source), "--symbol", "BTCUSDT", "--target-vol", "0.1", "--output", str(output)])
    summary = json.loads((output / "summary.json").read_text())
    assert result == 1
    assert len(calls) == len(summary["cases"]) == 36
    assert summary["cases"][0]["status"] == "FAILED"
    assert summary["status"] == "COMPLETE_WITH_FAILURES"
    assert summary["failed_attempts"] == 1
    assert summary["rankable"] is False
    assert (output / "completed.json").exists()


def test_source_drift_never_creates_completion(cli, monkeypatch, tmp_path):
    source = setup_job(cli, monkeypatch, tmp_path)
    checks = []

    def verify(value):
        checks.append(True)
        if len(checks) == 2:
            raise ValueError("source drift")

    monkeypatch.setattr(cli, "verify_sources", verify)
    monkeypatch.setattr(
        cli,
        "evaluate_candidate",
        lambda *a, **kw: (
            {},
            {
                "gates": {"E3": {"status": "FAIL"}},
                "metrics": {},
                "optimizer_feedback_eligible": False,
                "rankable": False,
                "live_eligible": False,
            },
        ),
    )
    output = tmp_path / "run"
    assert (
        cli.main(["--plan", str(source), "--symbol", "ETHUSDT", "--target-vol", "0.05", "--output", str(output)]) == 1
    )
    summary = json.loads((output / "summary.json").read_text())
    assert summary["status"] == "STALE"
    assert len(summary["cases"]) == 36
    assert not (output / "completed.json").exists()


def test_invalid_sources_rejected_before_output_creation(cli, monkeypatch, tmp_path):
    source = setup_job(cli, monkeypatch, tmp_path)
    monkeypatch.setattr(cli, "verify_sources", lambda value: (_ for _ in ()).throw(ValueError("source mismatch")))
    output = tmp_path / "run"
    with pytest.raises(ValueError, match="source mismatch"):
        cli.main(["--plan", str(source), "--symbol", "BTCUSDT", "--target-vol", "0.1", "--output", str(output)])
    assert not output.exists()


def test_started_interrupt_is_not_reported_as_unstarted(cli, monkeypatch, tmp_path):
    source = setup_job(cli, monkeypatch, tmp_path)
    monkeypatch.setattr(cli, "evaluate_candidate", lambda *a, **kw: (_ for _ in ()).throw(KeyboardInterrupt()))
    output = tmp_path / "run"
    assert cli.main(["--plan", str(source), "--symbol", "BTCUSDT", "--target-vol", "0.1", "--output", str(output)]) == 1
    summary = json.loads((output / "summary.json").read_text())
    assert summary["status"] == "INCOMPLETE"
    assert summary["started_attempts"] == 1
    assert summary["unstarted_attempts"] == 35
    assert len(summary["cases"]) == 1
    assert summary["cases"][0]["status"] == "INTERRUPTED"
    assert (output / summary["cases"][0]["report_path"]).exists()
    assert not (output / "completed.json").exists()
