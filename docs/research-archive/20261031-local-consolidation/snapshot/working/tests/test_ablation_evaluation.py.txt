"""Frozen, single-component ablations cannot relax system safety contracts."""

from __future__ import annotations

from copy import deepcopy

import pytest

from zenquant.research.run_evidence import identity_hash

DAY = 86_400_000_000_000
PROTECTED = ["risk_overlay", "accounting", "pit", "fees", "audit"]


def report(enabled, daily, *, dd=0.1, cost=0.01):
    candidate = {
        "family": "trend",
        "params": {"confirmation_bars": 2 if enabled else 1, "slow_bars": 72},
        "code_hash": "code",
        "risk_budget": {"target_annual_vol": 0.1, "max_leverage": 1.5},
    }
    return {
        "identity": {
            "candidate": candidate,
            "candidate_id": identity_hash(candidate),
            "code_hash": "code",
            "policy_hash": "policy",
        },
        "source_completed": True,
        "comparison_contract": {
            "symbol": "BTCUSDT_PERP",
            "window": "dev",
            "data_hash": "data",
            "accounting_hash": "code",
            "execution": "next_open",
            "cost": {"fee_bps": 4.0},
            "risk_budget": candidate["risk_budget"],
            "initial_equity": 100000.0,
            "boundary_policy": "reset_flat_no_warmup_pnl",
            "warmup_contract": {"strategy_bars": 180, "regime_bars": 720},
        },
        "daily_returns": [{"time_ns": (i + 1) * DAY, "value": daily} for i in range(200)],
        "gates": {f"E{i}": {"status": "PASS", "reason": "fixture_proof"} for i in range(5)},
        "ruin": {"occurred": False},
        "blockers": [],
        "metrics": {
            "max_drawdown": {"value": dd, "status": "PASS", "unit": "fraction"},
            "fees_initial_equity": {"value": cost, "status": "PASS", "unit": "fraction"},
        },
    }


def setup(full_daily=0.002, removed_daily=0.001):
    full, removed = report(True, full_daily), report(False, removed_daily)
    variants = [
        {
            "component_id": "entry_confirmation",
            "variant_id": "without_confirmation",
            "kind": "remove",
            "report": removed,
        }
    ]
    protocol = {
        "full_candidate_hash": identity_hash(full["identity"]["candidate"]),
        "components": {
            "entry_confirmation": [
                {
                    "variant_id": "without_confirmation",
                    "candidate_hash": identity_hash(removed["identity"]["candidate"]),
                    "parameter_keys": ["confirmation_bars"],
                }
            ]
        },
        "protected_components": PROTECTED,
        "min_samples": 180,
        "block_lengths": [2, 5, 10],
        "repetitions": 2000,
        "seed": 7,
        "max_drawdown_increase": 0.0,
        "max_cost_increase": 0.0,
        "cost_metric": "fees_initial_equity",
    }
    return full, variants, protocol


def run(*args):
    from zenquant.research.ablation_evaluation import evaluate_ablation_suite

    return evaluate_ablation_suite(*args)


def refresh_binding(full, variants, protocol):
    full["identity"]["candidate_id"] = identity_hash(full["identity"]["candidate"])
    protocol["full_candidate_hash"] = full["identity"]["candidate_id"]
    for row in variants:
        row["report"]["identity"]["candidate_id"] = identity_hash(row["report"]["identity"]["candidate"])
        protocol["components"][row["component_id"]][0]["candidate_hash"] = row["report"]["identity"]["candidate_id"]


def breakout_gate_setup(component):
    full, variants, protocol = setup()
    parameter = {"entry_gate": "entry_gate_enabled", "exit_gate": "exit_gate_enabled"}[component]
    for item in (full, variants[0]["report"]):
        item["identity"]["candidate"].update(
            family="breakout",
            params={
                "entry_bars": 48,
                "exit_bars": 12,
                "confirmation_bars": 1,
                "entry_gate_enabled": True,
                "exit_gate_enabled": True,
            },
        )
    variants[0].update(component_id=component, variant_id="without_gate")
    variants[0]["report"]["identity"]["candidate"]["params"][parameter] = False
    protocol["components"] = {component: [{"variant_id": "without_gate", "parameter_keys": [parameter]}]}
    refresh_binding(full, variants, protocol)
    return full, variants, protocol


@pytest.mark.parametrize("component", ["entry_gate", "exit_gate"])
def test_registered_single_breakout_gate_removal(component):
    full, variants, protocol = breakout_gate_setup(component)
    result = run(full, variants, protocol)
    assert result["decision"] == "KEEP"
    assert result["components"][component]["decision"] == "KEEP"


@pytest.mark.parametrize("disabled", [0, 0.0, None, "false"])
def test_gate_removal_requires_exact_boolean_not_false_like_values(disabled):
    full, variants, protocol = breakout_gate_setup("entry_gate")
    variants[0]["report"]["identity"]["candidate"]["params"]["entry_gate_enabled"] = disabled
    refresh_binding(full, variants, protocol)
    assert run(full, variants, protocol)["status"] == "INVALID"


def test_non_full_source_cannot_silently_remove_second_gate():
    full, variants, protocol = breakout_gate_setup("entry_gate")
    for item in (full, variants[0]["report"]):
        item["identity"]["candidate"]["params"]["exit_gate_enabled"] = False
    refresh_binding(full, variants, protocol)
    assert run(full, variants, protocol)["status"] == "INVALID"


def test_gate_removal_does_not_override_real_engineering_failure():
    full, variants, protocol = breakout_gate_setup("exit_gate")
    variants[0]["report"]["gates"]["E3"] = {"status": "FAIL", "reason": "known production difference"}
    result = run(full, variants, protocol)
    assert result["status"] == "FAIL"
    assert result["decision"] == "INCONCLUSIVE"


@pytest.mark.parametrize(
    "full_daily,removed_daily,want",
    [(0.002, 0.001, "KEEP"), (0.001, 0.002, "REMOVE_CANDIDATE"), (0.001, 0.001, "INCONCLUSIVE")],
)
def test_direction_requires_significant_net_increment_in_all_frozen_blocks(full_daily, removed_daily, want):
    inputs = setup(full_daily, removed_daily)
    result = run(*inputs)
    assert result["components"]["entry_confirmation"]["decision"] == want
    assert result["live_eligible"] is False
    assert result["R7"]["status"] == "NOT_EVALUABLE"
    assert result["protocol_hash"] == identity_hash(inputs[2])


def test_positive_return_does_not_justify_worse_drawdown_or_cost():
    full, variants, policy = setup()
    full["metrics"]["max_drawdown"]["value"] = 0.11
    assert run(full, variants, policy)["components"]["entry_confirmation"]["decision"] == "INCONCLUSIVE"
    policy["max_drawdown_increase"] = 0.02
    assert run(full, variants, policy)["components"]["entry_confirmation"]["decision"] == "KEEP"
    full["metrics"]["fees_initial_equity"]["value"] = 0.02
    assert run(full, variants, policy)["components"]["entry_confirmation"]["decision"] == "INCONCLUSIVE"


def test_simpler_variant_must_also_respect_drawdown_and_cost_tolerances():
    full, variants, policy = setup(0.001, 0.002)
    variants[0]["report"]["metrics"]["max_drawdown"]["value"] = 0.12
    assert run(full, variants, policy)["components"]["entry_confirmation"]["decision"] == "INCONCLUSIVE"


@pytest.mark.parametrize("protected", PROTECTED)
def test_protected_system_components_cannot_be_ablated(protected):
    full, variants, policy = setup()
    policy["components"][protected] = policy["components"].pop("entry_confirmation")
    variants[0]["component_id"] = protected
    result = run(full, variants, policy)
    assert result["status"] == "INVALID"
    assert result["decision"] == "INCONCLUSIVE"
    assert result["reason"] == "PROTECTED_COMPONENT_ABLATION"


def test_renaming_risk_ablation_as_signal_does_not_allow_leverage_removal():
    full, variants, policy = setup()
    full["identity"]["candidate"]["params"] = {"max_leverage": 1.5}
    variants[0]["report"]["identity"]["candidate"]["params"] = {"max_leverage": 9.0}
    policy["components"]["entry_confirmation"][0]["parameter_keys"] = ["max_leverage"]
    refresh_binding(full, variants, policy)
    assert run(full, variants, policy)["reason"] == "PROTECTED_PARAMETER_ABLATION"


@pytest.mark.parametrize("damage", ["missing", "duplicate", "unknown", "kind"])
def test_missing_duplicate_unknown_and_nonremoval_variants_never_pass(damage):
    full, variants, policy = setup()
    if damage == "missing":
        variants = []
    elif damage == "duplicate":
        variants.append(deepcopy(variants[0]))
    elif damage == "unknown":
        variants[0]["variant_id"] = "not_frozen"
    else:
        variants[0]["kind"] = "add"
    result = run(full, variants, policy)
    assert result["status"] != "PASS"
    assert result["decision"] == "INCONCLUSIVE"


@pytest.mark.parametrize(
    "damage", ["wrong_binding", "extra_parameter", "code", "family", "cost", "risk", "data", "execution"]
)
def test_identity_and_comparison_contract_cannot_change_beyond_declared_component(damage):
    full, variants, policy = setup()
    reduced = variants[0]["report"]
    if damage == "wrong_binding":
        policy["components"]["entry_confirmation"][0]["candidate_hash"] = "unbound"
    elif damage == "extra_parameter":
        reduced["identity"]["candidate"]["params"]["slow_bars"] = 24
        refresh_binding(full, variants, policy)
    elif damage in {"code", "family"}:
        reduced["identity"]["candidate"]["code_hash" if damage == "code" else "family"] = "changed"
        refresh_binding(full, variants, policy)
    else:
        reduced["comparison_contract"][
            {"cost": "cost", "risk": "risk_budget", "data": "data_hash", "execution": "execution"}[damage]
        ] = "changed"
    result = run(full, variants, policy)
    assert result["status"] in {"INVALID", "NOT_EVALUABLE"}
    assert result["decision"] == "INCONCLUSIVE"


@pytest.mark.parametrize("damage", ["safety_fail", "ruin", "incomplete", "missing_engineering"])
def test_engineering_and_completion_evidence_cannot_be_overridden_by_return(damage):

    assert result["status"] != "PASS"
    assert result["R7"]["status"] == "NOT_EVALUABLE"
    assert result["diagnostic_only"] is True

def test_all_predeclared_variants_are_required_and_cannot_skip_missing_evidence():
    full, variants, policy = setup()
    second = deepcopy(variants[0])
    second["variant_id"] = "without_confirmation_alternative"
    del second["report"]["gates"]["E1"]
    policy["components"]["entry_confirmation"].append(
        {
            "variant_id": second["variant_id"],
            "candidate_hash": second["report"]["identity"]["candidate_id"],
            "parameter_keys": ["confirmation_bars"],
        }
    )
    missing = run(full, variants, policy)
    assert missing["status"] == "NOT_EVALUABLE"
    assert missing["missing"] == [{"component_id": "entry_confirmation", "variant_id": second["variant_id"]}]
    variants.append(second)
    result = run(full, variants, policy)
    assert result["components"]["entry_confirmation"]["decision"] == "INCONCLUSIVE"
    assert len(result["components"]["entry_confirmation"]["variants"]) == 2

def test_one_bootstrap_draw_cannot_turn_negative_mean_increment_into_keep():
    full, variants, policy = setup(0.001, 0.0)
    full["daily_returns"][100]["value"] = -0.5
    for value in (full, variants[0]["report"]):
        value["metrics"]["max_drawdown"]["value"] = 0.5
    policy.update(repetitions=1, seed=18)
    assert sum(row["value"] for row in full["daily_returns"]) < 0
    result = run(full, variants, policy)
    assert result["status"] == "NOT_EVALUABLE"
    assert result["decision"] == "INCONCLUSIVE"
    assert result["reason"] == "INSUFFICIENT_BOOTSTRAP_REPETITIONS"

def test_missing_early_gate_does_not_hide_later_accounting_failure():
    full, variants, policy = setup()
    full["gates"]["E0"] = {"status": "NOT_EVALUABLE", "reason": "missing provenance"}
    full["gates"]["E2"] = {"status": "FAIL", "reason": "cash mismatch"}
    result = run(full, variants, policy)
    assert result["status"] == "FAIL"
    observation = result["components"]["entry_confirmation"]["variants"][0]
    assert observation["reason"] == "E2_NOT_PASS"
    assert {row["status"] for row in observation["safety_issues"]} == {"NOT_EVALUABLE", "FAIL"}
    assert result["decision"] == "INCONCLUSIVE"


@pytest.mark.parametrize("status", ["FAIL", "STALE", "INVALID"])
def test_full_missing_evidence_does_not_hide_removed_variant_hard_blocker(status):
    full, variants, policy = setup()
    full["gates"]["E0"] = {"status": "NOT_EVALUABLE"}

    observation = result["components"]["entry_confirmation"]["variants"][0]

def test_incomplete_source_does_not_hide_ruin_or_later_blocker_severity():
    full, variants, policy = setup()
    full["source_completed"] = False
    full["ruin"]["occurred"] = True
    assert run(full, variants, policy)["status"] == "FAIL"
    full["blockers"] = [{"status": "NOT_EVALUABLE"}, {"status": "STALE"}]
    assert run(full, variants, policy)["status"] == "STALE"
    variants[0]["report"]["blockers"] = [{"status": "NOT_EVALUABLE"}, {"status": "INVALID"}]
    assert run(full, variants, policy)["status"] == "INVALID"
