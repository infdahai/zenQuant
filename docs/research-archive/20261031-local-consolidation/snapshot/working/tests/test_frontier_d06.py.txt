"""Failure controls for synthetic latency evidence, not a venue fill calibration."""

import pandas as pd
import pytest

from zenquant.research.frontier_lab.experiments.d06_hft import (
    audit_sidecar,
    break_even_taker_edge,
    simulate_episode,
)


def test_taker_latency_can_destroy_the_same_signal():
    instant = simulate_episode("taker", 17, latency_ms=0, half_life_ms=100)
    delayed = simulate_episode("taker", 17, latency_ms=200, half_life_ms=100)
    assert instant["net_bps"] > 0 > delayed["net_bps"]
    assert instant["ending_inventory"] == delayed["ending_inventory"] == 0
    assert max(instant["ledger_error_usd"], delayed["ledger_error_usd"]) < 1e-9


def test_zero_queue_probability_has_no_phantom_position_or_fee():
    episode = simulate_episode("maker", 29, queue_probability=0)
    assert episode["fill_fraction"] == episode["net_bps"] == episode["fees_usd"] == 0
    assert episode["missed"] and episode["ending_inventory"] == 0


def test_cancel_ack_blocks_future_fills_but_request_does_not():
    immediate = simulate_episode("maker", 43, queue_probability=1, cancel_delay_ms=0, adverse_bps=8, trace=True)
    delayed = simulate_episode("maker", 43, queue_probability=1, cancel_delay_ms=200, adverse_bps=8, trace=True)
    assert immediate["late_fill_fraction"] == 0
    assert delayed["late_fill_fraction"] > 0
    assert all(fill["time_ms"] < immediate["cancel_ack_ms"] for fill in immediate["entry_fills"])
    assert delayed["ledger_error_usd"] < 1e-9


def test_cancel_before_order_arrival_and_partial_fills_are_accounted():
    no_arrival = simulate_episode("maker", 17, latency_ms=500, half_life_ms=20)
    assert no_arrival["missed"] and no_arrival["ending_inventory"] == 0
    assert no_arrival["cancel_unknown_before_arrival"]
    cancel_races = [
        simulate_episode("maker", seed, latency_ms=200, cancel_delay_ms=0, queue_probability=1) for seed in range(20)
    ]
    assert all(row["cancel_unknown_before_arrival"] for row in cancel_races)
    assert any(row["fill_fraction"] > 0 for row in cancel_races)
    partials = [simulate_episode("maker", seed, queue_probability=0.5, cancel_delay_ms=0) for seed in range(20)]
    assert any(row["partial"] for row in partials)
    assert all(row["ledger_error_usd"] < 1e-9 for row in partials)


def test_break_even_requires_more_edge_as_latency_increases():
    assert break_even_taker_edge(200, 100, 0) > break_even_taker_edge(1, 100, 0) > 0


def test_scenario_is_reproducible_and_audit_cannot_grant_replay_eligibility(tmp_path):
    assert simulate_episode("maker", 101) == simulate_episode("maker", 101)
    path = tmp_path / "sidecar.parquet"
    cutoff = pd.Timestamp("2026-04-22", tz="UTC").value
    pd.DataFrame({"cutoff_time_ns": [cutoff, cutoff + 10**9], "mid_price": [60_000, 60_001]}).to_parquet(path)
    audit = audit_sidecar(path)
    assert audit["pre2026_rows"] == 0 and audit["median_step_ms"] == 1000
    assert not audit["usable_for_hft_execution_replay"]
    assert "recv_time_ns" in audit["missing_fields_by_contract"]["exchange_and_local_receive_timestamps"]


@pytest.mark.parametrize("invalid", [{"latency_ms": -1}, {"queue_probability": 2}, {"half_life_ms": 0}])
def test_reject_invalid_scenario(invalid):
    with pytest.raises(ValueError):
        simulate_episode("maker", 17, **invalid)
