+def map_perturbed_flips(
+    base: Sequence[TimeSpaceFlipObservation],
+    perturbed: Sequence[TimeSpaceFlipObservation],
+) -> tuple[PerturbedFlipMapping, ...]:
+    """Map identity-compatible events within one retained 4h bar.
+
+    Chronological ordered matching preserves maximum event cardinality. Caller
+    order cannot change the result, neither side can be reused, and unmatched
+    events are censored rather than relabelled.
+    """
+
+    base_rows = _validated_unique_flips(base, name="base")
+    perturbed_rows = _validated_unique_flips(perturbed, name="perturbed")
+    base_by_signature: dict[tuple[FlipKind, str, int, int], list[TimeSpaceFlipObservation]] = {}
+    perturbed_by_signature: dict[tuple[FlipKind, str, int, int], list[TimeSpaceFlipObservation]] = {}
+    for base_flip in base_rows:
+        base_by_signature.setdefault(_perturbation_signature(base_flip), []).append(base_flip)
+    for perturbed_flip in perturbed_rows:
+        perturbed_by_signature.setdefault(_perturbation_signature(perturbed_flip), []).append(perturbed_flip)
+
+    matches: list[PerturbedFlipMapping] = []
+    for signature in sorted(base_by_signature, key=_perturbation_signature_order):
+        base_group = base_by_signature[signature]
+        perturbed_group = perturbed_by_signature.get(signature, [])
+        base_index = 0
+        perturbed_index = 0
+        while base_index < len(base_group) and perturbed_index < len(perturbed_group):
+            base_flip = base_group[base_index]
+            perturbed_flip = perturbed_group[perturbed_index]
+            delta = perturbed_flip.cutoff_time_ns - base_flip.cutoff_time_ns
+            if delta < -FOUR_HOURS_NS:
+                perturbed_index += 1
+            elif delta > FOUR_HOURS_NS:
+                base_index += 1
+            else:
+                matches.append(
+                    PerturbedFlipMapping(
+                        base_flip_id=base_flip.base_flip_id,
+                        perturbed_flip_id=perturbed_flip.base_flip_id,
+                    )
+                )
+                base_index += 1
+                perturbed_index += 1
+    matches.sort(key=lambda item: (item.base_flip_id, item.perturbed_flip_id))
+    return tuple(matches)

+def _month_block_lower_bound(
+    labels: tuple[tuple[TimeSpaceFlipObservation, TimeSpaceLabel], ...],
+    *,
+    flip_by_id: dict[str, TimeSpaceFlipObservation],
+    seed: int,
+) -> float | None:
+    if not labels:
+        return None
+    values_by_month: dict[str, list[float]] = {}
+    for flip, label in labels:
+        if flip_by_id.get(label.base_flip_id) != flip:
+            raise ValueError("bootstrap label is not bound to a primary flip")
+        month = datetime.fromtimestamp(flip.cutoff_time_ns // 1_000_000_000, tz=UTC).strftime("%Y-%m")
+        values_by_month.setdefault(month, []).append(float(label.hold_minus_flip))
+    blocks = tuple(tuple(values_by_month[month]) for month in sorted(values_by_month))
+    rng = random.Random(seed)
+    statistics: list[float] = []
+    for _ in range(_BOOTSTRAP_REPLICATIONS):
+        sample: list[float] = []
+        for _ in blocks:
+            sample.extend(blocks[rng.randrange(len(blocks))])
+        statistics.append(fmean(sample))
+    statistics.sort()
+    lower_index = max(0, math.ceil((1.0 - 0.95) * len(statistics)) - 1)
+    return float(statistics[lower_index])
