+def _normalize_forward_funding(frame: pd.DataFrame, *, reused_end_ns: int, as_of_ns: int) -> pd.DataFrame:
+    if "funding_time_ns" not in frame.columns:
+        raise ValueError("forward funding source lacks funding_time_ns")
+    interval = 8 * 60 * 60 * 1_000_000_000
+    values = frame.loc[frame["funding_time_ns"].astype("int64") <= as_of_ns].copy()
+    raw = values["funding_time_ns"].astype("int64")
+    anchors = ((raw + interval // 2) // interval) * interval
+    if ((raw - anchors).abs() > 60 * 1_000_000_000).any():
+        raise ValueError("forward funding timestamp exceeds the known 60s exchange jitter rule")
+    values["raw_funding_time_ns"] = raw
+    values["funding_time_ns"] = anchors
+    _validate_forward_grid(
+        values,
+        time_column="funding_time_ns",
+        first_ns=reused_end_ns + interval,
+        last_ns=as_of_ns - (as_of_ns % interval),
+        step_ns=interval,
+        label="8h funding",
+    )
+    return values

+def _validate_forward_grid(
+    frame: pd.DataFrame,
+    *,
+    time_column: str,
+    first_ns: int,
+    last_ns: int,
+    step_ns: int,
+    label: str,
+) -> None:
+    if time_column not in frame.columns or first_ns > last_ns:
+        raise ValueError(f"forward {label} coverage is unavailable")
+    observed = sorted(
+        int(value)
+        for value in frame.loc[(frame[time_column] >= first_ns) & (frame[time_column] <= last_ns), time_column]
+    )
+    expected = list(range(first_ns, last_ns + 1, step_ns))
+    if observed != expected:
+        raise ValueError(f"forward {label} coverage is not exact, continuous, and duplicate-free")
