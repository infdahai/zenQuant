+def _ns(value: str) -> int:
+    return int(datetime.fromisoformat(value).replace(tzinfo=UTC).timestamp() * 1_000_000_000)

+    interval_start_ns: int | None = None,

+    prior_short_boundary: float | None = None,
+    current_long_pivot: float | None = None,
+    current_short_pivot: float | None = None,
+    prior_long_invalidation: float | None = 90.0,
+    prior_short_invalidation: float | None = 110.0,
+    alpha_id: str | None = None,
+) -> TimeSpaceInputRow:
+    return TimeSpaceInputRow(
+        alpha_id=alpha_id or f"alpha-{cutoff_time_ns}",
+        cutoff_time_ns=cutoff_time_ns,

+        prior_long_boundary=prior_long_boundary,
+        prior_short_boundary=prior_short_boundary,
+        current_long_pivot=current_long_pivot,
+        current_short_pivot=current_short_pivot,
+        prior_long_invalidation=prior_long_invalidation,
+        prior_short_invalidation=prior_short_invalidation,
+        formula_hash="sha256:formula",
+        config_hash="sha256:config",
+        source_data_hash="sha256:data",
+        decision_id_by_stage=(("A1_SIGN", f"decision-{cutoff_time_ns}"),),

+"""Point-in-time, research-only time-space state construction.
+
+This module deliberately has no strategy, risk, execution, or domain-model
+imports.  It turns already-confirmed 4h research rows into immutable diagnostic
+state and has no authority to block or create orders.
+"""
+
+from __future__ import annotations
+
+import hashlib
+import json
+import math
+from collections.abc import Sequence
+from dataclasses import asdict, dataclass, field
+from enum import StrEnum
+
+FOUR_HOURS_NS = 4 * 60 * 60 * 1_000_000_000
+_DAY_NS = 24 * 60 * 60 * 1_000_000_000
+_WEEK_NS = 7 * _DAY_NS
+# 1970-01-05T00:00:00Z, the first Monday after the Unix epoch.
+_MONDAY_EPOCH_NS = 4 * _DAY_NS
+
+
+class StructureIntegrity(StrEnum):
+    """Point-in-time integrity of a retained directional breakout."""
+
+    INTACT = "INTACT"
+    BROKEN = "BROKEN"
+    UNKNOWN = "UNKNOWN"
+
+
+@dataclass(frozen=True, slots=True)
+class TimeSpaceInputRow:
+    """Confirmed 4h research input visible at ``cutoff_time_ns``.
+
+    ``prior_*_boundary`` values were visible before the covered interval.
+    ``current_*_pivot`` values are confirmed by this interval and therefore
+    become eligible only for later rows.
+    """
+
+    alpha_id: str
+    cutoff_time_ns: int
+    interval_start_ns: int
+    close: float
+    structure_side: int
+    prior_long_boundary: float | None
+    prior_short_boundary: float | None
+    current_long_pivot: float | None
+    current_short_pivot: float | None
+    prior_long_invalidation: float | None
+    prior_short_invalidation: float | None
+    formula_hash: str
+    config_hash: str
+    source_data_hash: str
+    decision_id_by_stage: tuple[tuple[str, str], ...] = ()
+
+    def __post_init__(self) -> None:
+        if not self.alpha_id:
+            raise ValueError("alpha_id must be non-empty")
+        if self.cutoff_time_ns <= self.interval_start_ns:
+            raise ValueError("cutoff_time_ns must be after interval_start_ns")
+        if self.cutoff_time_ns - self.interval_start_ns != FOUR_HOURS_NS:
+            raise ValueError("input row must cover exactly one 4h interval")
+        if self.structure_side not in (-1, 0, 1):
+            raise ValueError("structure_side must be -1, 0, or 1")
+        if not math.isfinite(self.close) or self.close <= 0:
+            raise ValueError("close must be positive and finite")
+        for name in (
+            "prior_long_boundary",
+            "prior_short_boundary",
+            "current_long_pivot",
+            "current_short_pivot",
+            "prior_long_invalidation",
+            "prior_short_invalidation",
+        ):
+            value = getattr(self, name)
+            if value is not None and (not math.isfinite(value) or value <= 0):
+                raise ValueError(f"{name} must be positive and finite when present")
+        for name in ("formula_hash", "config_hash", "source_data_hash"):
+            if not getattr(self, name):
+                raise ValueError(f"{name} must be non-empty")
+
+        ordered_decisions = tuple(sorted(self.decision_id_by_stage))
+        if any(not stage or not decision_id for stage, decision_id in ordered_decisions):
+            raise ValueError("decision stage and ID must be non-empty")
+        if len({stage for stage, _ in ordered_decisions}) != len(ordered_decisions):
+            raise ValueError("decision stages must be unique")
+        object.__setattr__(self, "decision_id_by_stage", ordered_decisions)
+
+
+@dataclass(frozen=True, slots=True)
+class TimeSpaceState:
+    """Immutable time-space state at one confirmed cutoff."""
+
+    state_id: str
+    alpha_id: str
+    cutoff_time_ns: int
+    feature_available_at_ns: int
+    decision_id_by_stage: tuple[tuple[str, str], ...]
+    formula_hash: str
+    config_hash: str
+    source_data_hash: str
+    market_structure_side: int
+    trend_age_bars: int | None
+    trend_age_ns: int | None
+    long_breakout_age_bars: int | None
+    long_breakout_age_ns: int | None
+    short_breakout_age_bars: int | None
+    short_breakout_age_ns: int | None
+    utc_week_start_ns: int
+    weekly_completion_fraction: float
+    long_structure_integrity: StructureIntegrity
+    short_structure_integrity: StructureIntegrity
+    source_gap_bars: int
+    quality_flags: tuple[str, ...]
+    blocks_order: bool = field(default=False, init=False)
+
+
+@dataclass(frozen=True, slots=True)
+class _Breakout:
+    cutoff_time_ns: int
+    retained_index: int
+
+
+def build_time_space_states(rows: Sequence[TimeSpaceInputRow]) -> tuple[TimeSpaceState, ...]:
+    """Build deterministic PIT state from strictly ordered confirmed 4h rows."""
+
+    states: list[TimeSpaceState] = []
+    previous_row: TimeSpaceInputRow | None = None
+    trend_side = 0
+    trend_started_at_ns: int | None = None
+    trend_started_index: int | None = None
+    long_breakout: _Breakout | None = None
+    short_breakout: _Breakout | None = None
+    carried_long_boundary: float | None = None
+    carried_short_boundary: float | None = None
+
+    for retained_index, row in enumerate(rows):
+        source_gap_bars = _source_gap_bars(previous_row, row)
+        if source_gap_bars:
+            long_breakout = None
+            short_breakout = None
+
+        if row.structure_side == 0:
+            trend_side = 0
+            trend_started_at_ns = None
+            trend_started_index = None
+        elif row.structure_side != trend_side:
+            trend_side = row.structure_side
+            trend_started_at_ns = row.cutoff_time_ns
+            trend_started_index = retained_index
+
+        visible_long_boundary = (
+            row.prior_long_boundary if row.prior_long_boundary is not None else carried_long_boundary
+        )
+        visible_short_boundary = (
+            row.prior_short_boundary if row.prior_short_boundary is not None else carried_short_boundary
+        )
+
+        long_broken = False
+        short_broken = False
+        if long_breakout is not None and _crossed_long_invalidation(row):
+            long_breakout = None
+            long_broken = True
+        if short_breakout is not None and _crossed_short_invalidation(row):
+            short_breakout = None
+            short_broken = True
+
+        long_crossed = _crossed_long_boundary(
+            previous_row,
+            row,
+            visible_long_boundary,
+            source_gap_bars,
+        )
+        short_crossed = _crossed_short_boundary(
+            previous_row,
+            row,
+            visible_short_boundary,
+            source_gap_bars,
+        )
+        if long_crossed and short_crossed:
+            raise ValueError("one row cannot create both long and short breakouts")
+        if long_crossed:
+            short_breakout = None
+            short_broken = False
+            long_breakout = _Breakout(row.cutoff_time_ns, retained_index)
+            long_broken = False
+        elif short_crossed:
+            long_breakout = None
+            long_broken = False
+            short_breakout = _Breakout(row.cutoff_time_ns, retained_index)
+            short_broken = False
+
+        if long_breakout is not None and row.structure_side == -1:
+            long_breakout = None
+            long_broken = True
+        if short_breakout is not None and row.structure_side == 1:
+            short_breakout = None
+            short_broken = True
+        if row.structure_side == 0:
+            long_breakout = None
+            short_breakout = None
+
+        week_start_ns = _utc_week_start(row.interval_start_ns)
+        weekly_completion = min(
+            1.0,
+            max(0.0, (row.cutoff_time_ns - week_start_ns) / _WEEK_NS),
+        )
+        quality_flags = ("SOURCE_GAP",) if source_gap_bars else ()
+
+        state_values = {
+            "alpha_id": row.alpha_id,
+            "cutoff_time_ns": row.cutoff_time_ns,
+            "feature_available_at_ns": row.cutoff_time_ns,
+            "decision_id_by_stage": row.decision_id_by_stage,
+            "formula_hash": row.formula_hash,
+            "config_hash": row.config_hash,
+            "source_data_hash": row.source_data_hash,
+            "market_structure_side": row.structure_side,
+            "trend_age_bars": _age_bars(trend_started_index, retained_index),
+            "trend_age_ns": _age_ns(trend_started_at_ns, row.cutoff_time_ns),
+            "long_breakout_age_bars": _breakout_age_bars(long_breakout, retained_index),
+            "long_breakout_age_ns": _breakout_age_ns(long_breakout, row.cutoff_time_ns),
+            "short_breakout_age_bars": _breakout_age_bars(short_breakout, retained_index),
+            "short_breakout_age_ns": _breakout_age_ns(short_breakout, row.cutoff_time_ns),
+            "utc_week_start_ns": week_start_ns,
+            "weekly_completion_fraction": weekly_completion,
+            "long_structure_integrity": _integrity(
+                breakout=long_breakout,
+                broken=long_broken,
+                structure_side=row.structure_side,
+                expected_side=1,
+                invalidation=row.prior_long_invalidation,
+            ),
+            "short_structure_integrity": _integrity(
+                breakout=short_breakout,
+                broken=short_broken,
+                structure_side=row.structure_side,
+                expected_side=-1,
+                invalidation=row.prior_short_invalidation,
+            ),
+            "source_gap_bars": source_gap_bars,
+            "quality_flags": quality_flags,
+        }
+        state_id = _state_id(row, state_values)
+        states.append(TimeSpaceState(state_id=state_id, **state_values))
+
+        if long_broken:
+            long_breakout = None
+        if short_broken:
+            short_breakout = None
+        if row.current_long_pivot is not None:
+            carried_long_boundary = row.current_long_pivot
+        elif row.prior_long_boundary is not None:
+            carried_long_boundary = row.prior_long_boundary
+        if row.current_short_pivot is not None:
+            carried_short_boundary = row.current_short_pivot
+        elif row.prior_short_boundary is not None:
+            carried_short_boundary = row.prior_short_boundary
+        previous_row = row
+
+    return tuple(states)
+
+
+def _source_gap_bars(previous: TimeSpaceInputRow | None, current: TimeSpaceInputRow) -> int:
+    if previous is None:
+        return 0
+    delta = current.cutoff_time_ns - previous.cutoff_time_ns
+    if delta <= 0:
+        raise ValueError("cutoff_time_ns values must be strictly increasing")
+    if delta % FOUR_HOURS_NS:
+        raise ValueError("cutoff_time_ns values must remain on one 4h grid")
+    return delta // FOUR_HOURS_NS - 1
+
+
+def _crossed_long_boundary(
+    previous: TimeSpaceInputRow | None,
+    current: TimeSpaceInputRow,
+    boundary: float | None,
+    source_gap_bars: int,
+) -> bool:
+    return bool(
+        previous is not None
+        and not source_gap_bars
+        and boundary is not None
+        and previous.close <= boundary < current.close
+    )
+
+
+def _crossed_short_boundary(
+    previous: TimeSpaceInputRow | None,
+    current: TimeSpaceInputRow,
+    boundary: float | None,
+    source_gap_bars: int,
+) -> bool:
+    return bool(
+        previous is not None
+        and not source_gap_bars
+        and boundary is not None
+        and previous.close >= boundary > current.close
+    )
+
+
+def _crossed_long_invalidation(row: TimeSpaceInputRow) -> bool:
+    return row.prior_long_invalidation is not None and row.close <= row.prior_long_invalidation
+
+
+def _crossed_short_invalidation(row: TimeSpaceInputRow) -> bool:
+    return row.prior_short_invalidation is not None and row.close >= row.prior_short_invalidation
+
+
+def _age_bars(started_index: int | None, retained_index: int) -> int | None:
+    return None if started_index is None else retained_index - started_index
+
+
+def _age_ns(started_at_ns: int | None, cutoff_time_ns: int) -> int | None:
+    return None if started_at_ns is None else cutoff_time_ns - started_at_ns
+
+
+def _breakout_age_bars(breakout: _Breakout | None, retained_index: int) -> int | None:
+    return None if breakout is None else retained_index - breakout.retained_index
+
+
+def _breakout_age_ns(breakout: _Breakout | None, cutoff_time_ns: int) -> int | None:
+    return None if breakout is None else cutoff_time_ns - breakout.cutoff_time_ns
+
+
+def _integrity(
+    *,
+    breakout: _Breakout | None,
+    broken: bool,
+    structure_side: int,
+    expected_side: int,
+    invalidation: float | None,
+) -> StructureIntegrity:
+    if broken:
+        return StructureIntegrity.BROKEN
+    if breakout is None or structure_side == 0 or invalidation is None:
+        return StructureIntegrity.UNKNOWN
+    if structure_side == expected_side:
+        return StructureIntegrity.INTACT
+    return StructureIntegrity.BROKEN
+
+
+def _utc_week_start(interval_start_ns: int) -> int:
+    return _MONDAY_EPOCH_NS + ((interval_start_ns - _MONDAY_EPOCH_NS) // _WEEK_NS) * _WEEK_NS
+
+
+def _state_id(row: TimeSpaceInputRow, values: dict[str, object]) -> str:
+    payload = {
+        "input": asdict(row),
+        "state": {key: value.value if isinstance(value, StrEnum) else value for key, value in values.items()},
+    }
+    encoded = json.dumps(payload, sort_keys=True, separators=(",", ":"), ensure_ascii=True).encode()
+    return hashlib.sha256(encoded).hexdigest()
+
+
+__all__ = [
+    "FOUR_HOURS_NS",
+    "StructureIntegrity",
+    "TimeSpaceInputRow",
+    "TimeSpaceState",
+    "build_time_space_states",
+]
