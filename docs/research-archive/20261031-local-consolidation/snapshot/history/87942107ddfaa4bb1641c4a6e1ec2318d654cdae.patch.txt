+def test_export_repository_identity_uses_origin_and_checked_in_pin_only(
+    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
+) -> None:
+    module = _load_script_module()
+    repository = tmp_path / "repository"
+    repository.mkdir()
+    subprocess.run(["git", "init", "-q", str(repository)], check=True)
+    (repository / "tracked.txt").write_text("clean\n")
+    subprocess.run(["git", "-C", str(repository), "add", "tracked.txt"], check=True)
+    subprocess.run(
+        [
+            "git",
+            "-C",
+            str(repository),
+            "-c",
+            "user.name=TimeSpace Test",
+            "-c",
+            "user.email=time-space@example.invalid",
+            "commit",
+            "-q",
+            "-m",
+            "baseline",
+        ],
+        check=True,
+    )
+    subprocess.run(
+        [
+            "git",
+            "-C",
+            str(repository),
+            "remote",
+            "add",
+            "origin",
+            "https://github.com/infdahai/zenQuant.git",
+        ],
+        check=True,
+    )
+    trust = repository / "config/research/time_space_export_registration_trust_v1.json"
+    trust.parent.mkdir(parents=True)
+    trust.write_text(json.dumps({"remote_identity": "infdahai/zenQuant"}))
+    subprocess.run(["git", "-C", str(repository), "add", trust.relative_to(repository)], check=True)
+    subprocess.run(
+        [
+            "git",
+            "-C",
+            str(repository),
+            "-c",
+            "user.name=TimeSpace Test",
+            "-c",
+            "user.email=time-space@example.invalid",
+            "commit",
+            "-q",
+            "-m",
+            "pin repository",
+        ],
+        check=True,
+    )
+    monkeypatch.setattr(module, "_ROOT", repository)
+
+    identity, commit = module._resolve_repository()
+
+    assert identity == "infdahai/zenQuant"
+    assert (
+        commit
+        == subprocess.run(
+            ["git", "-C", str(repository), "rev-parse", "HEAD"],
+            check=True,
+            capture_output=True,
+            text=True,
+        ).stdout.strip()
+    )
+
