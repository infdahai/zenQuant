+def build_flip_observations(
+    *,
+    states: Sequence[TimeSpaceState],
+    alpha_signs: Sequence[int],
+    approved_targets: Sequence[float],
+    filled_positions: Sequence[float],
+) -> tuple[TimeSpaceFlipObservation, ...]:
+    """Join aligned strategy, risk, and execution sign crossings to PIT state."""
+
+    expected_length = len(states)
+    populations: tuple[tuple[FlipKind, Sequence[int | float]], ...] = (
+        (FlipKind.ALPHA, alpha_signs),
+        (FlipKind.APPROVED, approved_targets),
+        (FlipKind.FILLED, filled_positions),
+    )
+    if any(len(values) != expected_length for _, values in populations):
+        raise ValueError("states and flip population inputs must have the same length")
+
+    state_by_cutoff: dict[int, TimeSpaceState] = {}
+    previous_cutoff: int | None = None
+    for state in states:
+        if previous_cutoff is not None and state.cutoff_time_ns <= previous_cutoff:
+            raise ValueError("state cutoff_time_ns values must be strictly increasing")
+        state_by_cutoff[state.cutoff_time_ns] = state
+        previous_cutoff = state.cutoff_time_ns
+
+    observations: list[TimeSpaceFlipObservation] = []
+    prior_sides: dict[FlipKind, int | None] = {kind: None for kind, _ in populations}
+    for index, state in enumerate(states):
+        for kind, values in populations:
+            new_side = _population_side(values[index], kind)
+            old_side = prior_sides[kind]
+            if old_side is not None and old_side == -new_side and old_side != 0:
+                identity = {
+                    "kind": kind.value,
+                    "alpha_id": state.alpha_id,
+                    "cutoff_time_ns": state.cutoff_time_ns,
+                    "old_side": old_side,
+                    "new_side": new_side,
+                    "state_id": state.state_id,
+                }
+                observations.append(
+                    TimeSpaceFlipObservation(
+                        base_flip_id=_canonical_id(identity),
+                        kind=kind,
+                        alpha_id=state.alpha_id,
+                        cutoff_time_ns=state.cutoff_time_ns,
+                        old_side=old_side,
+                        new_side=new_side,
+                        state_id=state.state_id,
+                        primary_population=(
+                            kind is FlipKind.ALPHA
+                            and _is_primary_alpha_flip(
+                                state=state,
+                                old_side=old_side,
+                                state_by_cutoff=state_by_cutoff,
+                            )
+                        ),
+                    )
+                )
+            prior_sides[kind] = new_side
+
+    return tuple(observations)

+def _population_side(value: object, kind: FlipKind) -> int:
+    if kind is FlipKind.ALPHA:
+        if type(value) is not int or value not in (-1, 0, 1):
+            raise ValueError("alpha_signs must contain exact ints in {-1, 0, 1}")
+        return value
+    if type(value) not in (int, float):
+        raise TypeError(f"{kind.value} values must be finite numbers")
+    numeric = float(value)
+    if not math.isfinite(numeric):
+        raise ValueError(f"{kind.value} values must be finite numbers")
+    return (numeric > 0) - (numeric < 0)

+def _is_primary_alpha_flip(
+    *,
+    state: TimeSpaceState,
+    old_side: int,
+    state_by_cutoff: dict[int, TimeSpaceState],
+) -> bool:
+    if state.source_gap_bars != 0:
+        return False
+    if old_side == 1:
+        breakout_age_ns = state.long_breakout_age_ns
+        integrity = state.long_structure_integrity
+    else:
+        breakout_age_ns = state.short_breakout_age_ns
+        integrity = state.short_structure_integrity

+        return False

+    return breakout_state is not None and breakout_state.utc_week_start_ns == state.utc_week_start_ns
