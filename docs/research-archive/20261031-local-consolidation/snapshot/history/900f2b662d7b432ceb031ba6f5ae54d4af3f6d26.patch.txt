+#!/usr/bin/env python3
+"""Fail closed until independent GitHub workflow protection is pinned."""
+
+from __future__ import annotations
+
+import argparse
+import json
+import sys
+from collections.abc import Sequence
+from pathlib import Path
+from typing import Any
+
+CANONICAL_REPOSITORY = "infdahai/zenQuant"
+DEFAULT_BRANCH = "main"
+VERIFIED = "VERIFIED"
+PROTECTION_KEYS = {
+    "schema_version",
+    "layer",
+    "registration_status",
+    "remote_identity",
+    "default_branch",
+    "authority_type",
+    "authority_id",
+    "required_check",
+    "evidence_sha256",
+    "provider_observed_at_ns",
+}
+
+
+def validate_external_protection(path: Path) -> None:
+    """Require one canonical pin for externally enforced workflow authority."""
+
+    payload = _canonical_object(path)
+    if set(payload) != PROTECTION_KEYS:
+        raise ValueError("external workflow protection schema is incomplete")
+    if (
+        payload.get("schema_version") != "time-space-export-external-protection.v1"
+        or payload.get("layer") != "time_space_shadow"
+    ):
+        raise ValueError("external workflow protection identity is invalid")
+    if payload.get("remote_identity") != CANONICAL_REPOSITORY:
+        raise ValueError("external workflow protection is not pinned to the canonical repository")
+    if payload.get("default_branch") != DEFAULT_BRANCH:
+        raise ValueError("external workflow protection is not pinned to the default branch")
+    if payload.get("registration_status") != VERIFIED:
+        raise ValueError("external workflow protection is not VERIFIED")
+    if payload.get("authority_type") not in {"required_workflow", "branch_ruleset"}:
+        raise ValueError("external workflow protection authority type is invalid")
+    _positive_int(payload.get("authority_id"), "authority id")
+    required_check = payload.get("required_check")
+    if not isinstance(required_check, str) or not required_check or required_check.strip() != required_check:
+        raise ValueError("external workflow protection required check is invalid")
+    evidence_sha256 = payload.get("evidence_sha256")
+    if (
+        not isinstance(evidence_sha256, str)
+        or len(evidence_sha256) != 64
+        or any(character not in "0123456789abcdef" for character in evidence_sha256)
+    ):
+        raise ValueError("external workflow protection evidence digest is invalid")
+    _positive_int(payload.get("provider_observed_at_ns"), "provider observed timestamp")
+
+
+def _canonical_object(path: Path) -> dict[str, Any]:
+    try:
+        content = Path(path).read_bytes()
+        payload = json.loads(content.decode("utf-8"))
+    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
+        raise ValueError("external workflow protection evidence is not JSON") from exc
+    if not isinstance(payload, dict):
+        raise ValueError("external workflow protection evidence must be an object")
+    canonical = (json.dumps(payload, sort_keys=True, separators=(",", ":"), allow_nan=False) + "\n").encode()
+    if content != canonical:
+        raise ValueError("external workflow protection evidence bytes are not canonical")
+    return payload
+
+
+def _positive_int(value: Any, name: str) -> int:
+    if isinstance(value, bool) or not isinstance(value, int) or value <= 0:
+        raise ValueError(f"external workflow protection {name} is invalid")
+    return value
+
+
+def _main(argv: Sequence[str] | None = None) -> int:
+    parser = argparse.ArgumentParser(description=__doc__)
+    parser.add_argument("--config", required=True, type=Path)
+    args = parser.parse_args(argv)
+    try:
+        validate_external_protection(args.config)
+    except (OSError, TypeError, ValueError) as exc:
+        print(f"external workflow protection validation failed: {exc}", file=sys.stderr)
+        return 1
+    return 0
+
+
+if __name__ == "__main__":
+    raise SystemExit(_main())
