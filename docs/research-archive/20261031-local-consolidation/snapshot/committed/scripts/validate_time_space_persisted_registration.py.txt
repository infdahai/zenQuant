"""Validate one canonical, durable TimeSpace export registration transaction."""

from __future__ import annotations

import argparse
import hashlib
import json
import sys
from collections.abc import Sequence
from datetime import UTC, datetime
from pathlib import Path
from typing import Any

CHECKPOINT_WORKFLOW_PATH = ".github/workflows/time-space-export-checkpoint.yml"
CHECKPOINT_BRANCH = "feature/trend-registration-time-space-export"
GENERATOR_WORKFLOW_PATH = ".github/workflows/time-space-export-generate.yml"
GENERATOR_EVENT = "workflow_dispatch"
CANONICAL_REPOSITORY = "infdahai/zenQuant"
TRUST_KEYS = {
    "schema_version",
    "layer",
    "registration_status",
    "remote_identity",
    "checkpoint_workflow_id",
    "checkpoint_workflow_path",
    "checkpoint_branch",
    "provider_run_id",
    "commit_sha",
    "checkpoint_manifest_sha256",
    "provider_created_at_ns",
    "checkpoint_artifact_id",
    "checkpoint_artifact_name",
    "receipt_artifact_sha256",
    "generator_commit_sha",
    "generator_provider_run_id",
    "generator_workflow_id",
    "generator_workflow_path",
    "generator_event",
    "generator_branch",
    "generator_created_at_ns",
    "generator_artifact_id",
    "generator_artifact_name",
}
RECEIPT_KEYS = {
    "schema_version",
    "manifest_path",
    "manifest_sha256",
    "export_sha256",
    "provenance_sha256",
    "provider_run_id",
    "provider_created_at_ns",
    "commit_sha",
    "remote_identity",
    "workflow",
    "event",
    "ref",
    "artifact",
    "generator_artifact",
}
CHECKPOINT_KEYS = {
    "schema_version",
    "layer",
    "registration_status",
    "export_sha256",
    "provenance_sha256",
    "receipt",
    "as_of_ns",
    "generator_provider_run_id",
    "source_slice_sha256",
}


def validate_persisted_registration(
    *,
    checkpoint_path: Path,
    receipt_path: Path,
    trust_path: Path,
    export_path: Path,
    provenance_path: Path,
    provider: dict[str, Any],
    generator_provider: dict[str, Any],
) -> None:
    checkpoint, checkpoint_bytes = _canonical_object(checkpoint_path, "checkpoint")
    receipt, receipt_bytes = _canonical_object(receipt_path, "receipt")
    trust, _trust_bytes = _canonical_object(trust_path, "trust")
    export, export_bytes = _canonical_object(export_path, "export")
    provenance, provenance_bytes = _canonical_object(provenance_path, "provenance")
    if set(checkpoint) != CHECKPOINT_KEYS or set(receipt) != RECEIPT_KEYS or set(trust) != TRUST_KEYS:
        raise ValueError("checkpoint, receipt, or trust schema is incomplete")
    if (
        checkpoint.get("schema_version") != "time-space-export-checkpoint.v1"
        or checkpoint.get("layer") != "time_space_shadow"
        or checkpoint.get("registration_status") != "PENDING"
        or checkpoint.get("receipt") is not None
        or receipt.get("schema_version") != "time-space-export-registration-receipt.v1"
        or trust.get("schema_version") != "time-space-export-registration-trust.v1"
        or trust.get("layer") != "time_space_shadow"
        or trust.get("registration_status") != "VERIFIED"
        or export.get("schema_version") != "time-space-shadow-export.v1"
        or export.get("layer") != "time_space_shadow"
        or provenance.get("schema_version") != "time-space-shadow-export-provenance.v1"
        or provenance.get("layer") != "time_space_shadow"
    ):
        raise ValueError("persisted registration identity is invalid")
    if (
        trust.get("remote_identity") != CANONICAL_REPOSITORY
        or receipt.get("remote_identity") != CANONICAL_REPOSITORY
        or provenance.get("repository_identity") != CANONICAL_REPOSITORY
    ):
        raise ValueError("persisted registration is not pinned to the canonical repository")
    as_of_ns = _positive_int(checkpoint.get("as_of_ns"), "checkpoint as_of")
    if export.get("as_of_ns") != as_of_ns or provenance.get("as_of_ns") != as_of_ns:
        raise ValueError("persisted export cutoff does not match checkpoint")
    export_sha = _payload_sha(export_bytes)
    provenance_sha = _payload_sha(provenance_bytes)
    source_slice_sha = _canonical_sha(provenance.get("source_files"))
    manifest_sha = hashlib.sha256(checkpoint_bytes).hexdigest()
    receipt_sha = hashlib.sha256(receipt_bytes).hexdigest()
    if (
        checkpoint.get("export_sha256") != export_sha
        or checkpoint.get("provenance_sha256") != provenance_sha
        or checkpoint.get("source_slice_sha256") != source_slice_sha
        or receipt.get("manifest_sha256") != manifest_sha
        or receipt.get("export_sha256") != export_sha
        or receipt.get("provenance_sha256") != provenance_sha
        or trust.get("checkpoint_manifest_sha256") != manifest_sha
        or trust.get("receipt_artifact_sha256") != receipt_sha
    ):
        raise ValueError("persisted registration digest binding is invalid")
    _validate_checkpoint_authority(receipt, trust, provider, as_of_ns)
    _validate_generator_authority(receipt, trust, provenance, checkpoint, generator_provider, as_of_ns)


def _validate_checkpoint_authority(
    receipt: dict[str, Any], trust: dict[str, Any], provider: dict[str, Any], as_of_ns: int
) -> None:
    workflow_id = _positive_int(trust.get("checkpoint_workflow_id"), "checkpoint workflow id")
    artifact_id = _positive_int(trust.get("checkpoint_artifact_id"), "checkpoint artifact id")
    provider_created = _nonnegative_int(trust.get("provider_created_at_ns"), "provider created at")
    provider_run = _digits(trust.get("provider_run_id"), "provider run id")
    commit = _sha(trust.get("commit_sha"), 40, "checkpoint commit")
    expected_artifact_name = f"time-space-export-checkpoint-{as_of_ns}-{provider_run}"
    if (
        trust.get("remote_identity") != receipt.get("remote_identity")
        or trust.get("checkpoint_workflow_path") != CHECKPOINT_WORKFLOW_PATH
        or trust.get("checkpoint_branch") != CHECKPOINT_BRANCH
        or trust.get("checkpoint_artifact_name") != expected_artifact_name
        or receipt.get("provider_run_id") != provider_run
        or receipt.get("provider_created_at_ns") != provider_created
        or receipt.get("commit_sha") != commit
        or receipt.get("workflow") != {"id": workflow_id, "path": CHECKPOINT_WORKFLOW_PATH}
        or receipt.get("event") != "push"
        or receipt.get("ref") != f"refs/heads/{CHECKPOINT_BRANCH}"
        or receipt.get("artifact") != {"id": artifact_id, "name": expected_artifact_name}
        or provider.get("id") != int(provider_run)
        or provider.get("head_sha") != commit
        or provider.get("conclusion") != "success"
        or provider.get("event") != "push"
        or provider.get("head_branch") != CHECKPOINT_BRANCH
        or provider.get("path") != CHECKPOINT_WORKFLOW_PATH
        or provider.get("workflow_id") != workflow_id
        or _provider_created_at_ns(provider.get("created_at")) != provider_created
        or not isinstance(provider.get("repository"), dict)
        or provider["repository"].get("full_name") != CANONICAL_REPOSITORY
    ):
        raise ValueError("persisted trust checkpoint provider fields are not fully bound")


def _validate_generator_authority(
    receipt: dict[str, Any],
    trust: dict[str, Any],
    provenance: dict[str, Any],
    checkpoint: dict[str, Any],
    provider: dict[str, Any],
    as_of_ns: int,
) -> None:
    workflow_id = _positive_int(trust.get("generator_workflow_id"), "generator workflow id")
    artifact_id = _positive_int(trust.get("generator_artifact_id"), "generator artifact id")
    _nonnegative_int(trust.get("generator_created_at_ns"), "generator created at")
    generator_run = _digits(trust.get("generator_provider_run_id"), "generator provider run id")
    generator_commit = _sha(trust.get("generator_commit_sha"), 40, "generator commit")
    generator_branch = trust.get("generator_branch")
    if not isinstance(generator_branch, str) or not generator_branch or generator_branch.strip() != generator_branch:
        raise ValueError("persisted trust generator branch is invalid")
    expected_name = f"time-space-export-{as_of_ns}-{generator_run}"
    generator_artifact = receipt.get("generator_artifact")
    if (
        trust.get("generator_workflow_path") != GENERATOR_WORKFLOW_PATH
        or trust.get("generator_event") != GENERATOR_EVENT
        or trust.get("generator_artifact_name") != expected_name
        or provenance.get("repository_identity") != trust.get("remote_identity")
        or provenance.get("commit_sha") != generator_commit
        or provenance.get("provider_run_id") != generator_run
        or checkpoint.get("generator_provider_run_id") != generator_run
        or not isinstance(generator_artifact, dict)
        or generator_artifact.get("id") != artifact_id
        or generator_artifact.get("name") != expected_name
        or generator_artifact.get("export_sha256") != checkpoint.get("export_sha256")
        or generator_artifact.get("provenance_sha256") != checkpoint.get("provenance_sha256")
        or generator_artifact.get("source_slice_sha256") != checkpoint.get("source_slice_sha256")
        or provider.get("id") != int(generator_run)
        or provider.get("head_sha") != generator_commit
        or provider.get("conclusion") != "success"
        or provider.get("event") != GENERATOR_EVENT
        or provider.get("head_branch") != generator_branch
        or provider.get("path") != GENERATOR_WORKFLOW_PATH
        or provider.get("workflow_id") != workflow_id
        or _provider_created_at_ns(provider.get("created_at")) != trust.get("generator_created_at_ns")
        or not isinstance(provider.get("repository"), dict)
        or provider["repository"].get("full_name") != CANONICAL_REPOSITORY
    ):
        raise ValueError("persisted trust generator provider fields are not fully bound")


def _canonical_object(path: Path, name: str) -> tuple[dict[str, Any], bytes]:
    content = Path(path).read_bytes()
    try:
        value = json.loads(content.decode("utf-8"))
    except (UnicodeError, json.JSONDecodeError) as exc:
        raise ValueError(f"persisted {name} is not JSON") from exc
    if not isinstance(value, dict):
        raise ValueError(f"persisted {name} must be an object")
    canonical = (json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False) + "\n").encode()
    if content != canonical:
        raise ValueError(f"persisted {name} bytes are not canonical")
    return value, content


def _json_object(path: Path, name: str) -> dict[str, Any]:
    try:
        value = json.loads(Path(path).read_text(encoding="utf-8"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise ValueError(f"{name} provider payload is not JSON") from exc
    if not isinstance(value, dict):
        raise ValueError(f"{name} provider payload must be an object")
    return value


def _provider_created_at_ns(value: Any) -> int:
    if not isinstance(value, str):
        raise ValueError("provider created_at is invalid")
    try:
        parsed = datetime.strptime(value, "%Y-%m-%dT%H:%M:%SZ").replace(tzinfo=UTC)
    except ValueError as exc:
        raise ValueError("provider created_at is invalid") from exc
    return int(parsed.timestamp()) * 1_000_000_000


def _payload_sha(content: bytes) -> str:
    return hashlib.sha256(content[:-1]).hexdigest()


def _canonical_sha(value: Any) -> str:
    encoded = json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()
    return hashlib.sha256(encoded).hexdigest()


def _positive_int(value: Any, name: str) -> int:
    if isinstance(value, bool) or not isinstance(value, int) or value <= 0:
        raise ValueError(f"persisted trust {name} is invalid")
    return value


def _nonnegative_int(value: Any, name: str) -> int:
    if isinstance(value, bool) or not isinstance(value, int) or value < 0:
        raise ValueError(f"persisted trust {name} is invalid")
    return value


def _digits(value: Any, name: str) -> str:
    if not isinstance(value, str) or not value.isdigit() or value.startswith("0"):
        raise ValueError(f"persisted trust {name} is invalid")
    return value


def _sha(value: Any, length: int, name: str) -> str:
    if (
        not isinstance(value, str)
        or len(value) != length
        or any(character not in "0123456789abcdef" for character in value)
    ):
        raise ValueError(f"persisted trust {name} is invalid")
    return value


def _main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--checkpoint", required=True, type=Path)
    parser.add_argument("--receipt", required=True, type=Path)
    parser.add_argument("--trust", required=True, type=Path)
    parser.add_argument("--export", required=True, type=Path)
    parser.add_argument("--provenance", required=True, type=Path)
    parser.add_argument("--provider-run-json", required=True, type=Path)
    parser.add_argument("--generator-provider-run-json", required=True, type=Path)
    args = parser.parse_args(argv)
    try:
        validate_persisted_registration(
            checkpoint_path=args.checkpoint,
            receipt_path=args.receipt,
            trust_path=args.trust,
            export_path=args.export,
            provenance_path=args.provenance,
            provider=_json_object(args.provider_run_json, "checkpoint"),
            generator_provider=_json_object(args.generator_provider_run_json, "generator"),
        )
    except (OSError, TypeError, ValueError) as exc:
        print(f"persisted registration trust validation failed: {exc}", file=sys.stderr)
        return 1
    return 0


if __name__ == "__main__":
    raise SystemExit(_main())
