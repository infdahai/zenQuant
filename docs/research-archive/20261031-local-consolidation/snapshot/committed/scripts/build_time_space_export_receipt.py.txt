#!/usr/bin/env python3
"""Build and verify the independent TimeSpace export checkpoint receipt."""

from __future__ import annotations

import argparse
import hashlib
import json
from collections.abc import Mapping, Sequence
from datetime import UTC, datetime
from pathlib import Path
from typing import Any

ROOT = Path(__file__).resolve().parent.parent
MANIFEST_ROOT = "config/research/registrations/time_space_exports"
TRUST_PATH = ROOT / "config/research/time_space_export_registration_trust_v1.json"
WORKFLOW_PATH = ".github/workflows/time-space-export-checkpoint.yml"
BRANCH = "feature/trend-registration-time-space-export"
REPOSITORY = "infdahai/zenQuant"
MANIFEST_KEYS = {
    "schema_version",
    "layer",
    "registration_status",
    "export_sha256",
    "provenance_sha256",
    "receipt",
    "as_of_ns",
    "generator_provider_run_id",
    "source_slice_sha256",
}


def build_receipt(
    *,
    manifest_path: str,
    manifest_bytes: bytes,
    provider: Mapping[str, Any],
    artifact: Mapping[str, Any],
    generator_export_bytes: bytes,
    generator_provenance_bytes: bytes,
    generator_artifact: Mapping[str, Any],
) -> dict[str, Any]:
    """Build a receipt using only the checked-in trust root."""

    return _build_receipt(
        manifest_path=manifest_path,
        manifest_bytes=manifest_bytes,
        provider=provider,
        artifact=artifact,
        generator_export_bytes=generator_export_bytes,
        generator_provenance_bytes=generator_provenance_bytes,
        generator_artifact=generator_artifact,
        trust=_load_json(TRUST_PATH),
    )


def _build_receipt(
    *,
    manifest_path: str,
    manifest_bytes: bytes,
    provider: Mapping[str, Any],
    artifact: Mapping[str, Any],
    generator_export_bytes: bytes,
    generator_provenance_bytes: bytes,
    generator_artifact: Mapping[str, Any],
    trust: Mapping[str, Any],
) -> dict[str, Any]:
    manifest = _validate_manifest(manifest_path, manifest_bytes)
    as_of_ns = _manifest_as_of(manifest_path)
    expected_workflow_id = trust.get("checkpoint_workflow_id")
    if isinstance(expected_workflow_id, bool) or not isinstance(expected_workflow_id, int) or expected_workflow_id <= 0:
        raise ValueError("checkpoint workflow id is not independently pinned")
    expected = {
        "id": provider.get("id"),
        "head_sha": provider.get("head_sha"),
        "created_at": provider.get("created_at"),
        "conclusion": provider.get("conclusion"),
        "event": provider.get("event"),
        "head_branch": provider.get("head_branch"),
        "path": provider.get("path"),
        "workflow_id": provider.get("workflow_id"),
        "repository": provider.get("repository"),
    }
    if expected["conclusion"] != "success" or expected["event"] != "push":
        raise ValueError("provider checkpoint did not complete a successful push")
    if expected["head_branch"] != BRANCH or expected["path"] != WORKFLOW_PATH:
        raise ValueError("provider checkpoint branch or workflow path changed")
    if expected["workflow_id"] != expected_workflow_id:
        raise ValueError("provider checkpoint workflow id changed")
    repository = expected["repository"]
    if not isinstance(repository, Mapping) or repository.get("full_name") != REPOSITORY:
        raise ValueError("provider checkpoint repository changed")
    run_id = expected["id"]
    if isinstance(run_id, bool) or not isinstance(run_id, int) or run_id <= 0:
        raise ValueError("provider run id is invalid")
    head_sha = _sha(expected["head_sha"], 40, "provider head sha")
    created_at_ns = _timestamp_ns(expected["created_at"])
    expected_artifact_name = f"time-space-export-checkpoint-{as_of_ns}-{run_id}"
    generator_export = _canonical_payload(generator_export_bytes, "generator export")
    generator_provenance = _canonical_payload(generator_provenance_bytes, "generator provenance")
    if (
        _canonical_sha256(generator_export) != manifest["export_sha256"]
        or _canonical_sha256(generator_provenance) != manifest["provenance_sha256"]
        or generator_export.get("as_of_ns") != as_of_ns
        or generator_provenance.get("as_of_ns") != as_of_ns
        or _canonical_sha256(generator_provenance.get("source_files")) != manifest["source_slice_sha256"]
    ):
        raise ValueError("downloaded generator artifact content does not match checkpoint")
    generator_run_id = int(manifest["generator_provider_run_id"])
    expected_generator_name = f"time-space-export-{as_of_ns}-{generator_run_id}"
    if (
        generator_artifact.get("name") != expected_generator_name
        or generator_artifact.get("expired") is not False
        or isinstance(generator_artifact.get("id"), bool)
        or not isinstance(generator_artifact.get("id"), int)
        or generator_artifact["id"] <= 0
    ):
        raise ValueError("generator artifact identity is invalid")
    workflow_run = artifact.get("workflow_run")
    if (
        artifact.get("expired") is not False
        or artifact.get("name") != expected_artifact_name
        or isinstance(artifact.get("id"), bool)
        or not isinstance(artifact.get("id"), int)
        or not isinstance(workflow_run, Mapping)
        or workflow_run.get("id") != run_id
        or workflow_run.get("head_sha") != head_sha
    ):
        raise ValueError("checkpoint artifact identity changed")
    return {
        "schema_version": "time-space-export-registration-receipt.v1",
        "manifest_path": manifest_path,
        "manifest_sha256": hashlib.sha256(manifest_bytes).hexdigest(),
        "export_sha256": manifest["export_sha256"],
        "provenance_sha256": manifest["provenance_sha256"],
        "provider_run_id": str(run_id),
        "provider_created_at_ns": created_at_ns,
        "commit_sha": head_sha,
        "remote_identity": REPOSITORY,
        "workflow": {"id": expected_workflow_id, "path": WORKFLOW_PATH},
        "event": "push",
        "ref": f"refs/heads/{BRANCH}",
        "artifact": {"id": artifact["id"], "name": expected_artifact_name},
        "generator_artifact": {
            "id": generator_artifact["id"],
            "name": expected_generator_name,
            "export_sha256": manifest["export_sha256"],
            "provenance_sha256": manifest["provenance_sha256"],
            "source_slice_sha256": manifest["source_slice_sha256"],
        },
    }


def _validate_manifest(path: str, content: bytes) -> dict[str, Any]:
    _manifest_as_of(path)
    try:
        manifest = json.loads(content.decode("utf-8"))
    except (UnicodeError, json.JSONDecodeError) as exc:
        raise ValueError("checkpoint manifest is not JSON") from exc
    if not isinstance(manifest, dict) or set(manifest) != MANIFEST_KEYS:
        raise ValueError("checkpoint manifest schema changed")
    canonical = (json.dumps(manifest, sort_keys=True, separators=(",", ":")) + "\n").encode()
    if content != canonical:
        raise ValueError("checkpoint manifest bytes are not canonical")
    if (
        manifest.get("schema_version") != "time-space-export-checkpoint.v1"
        or manifest.get("layer") != "time_space_shadow"
        or manifest.get("registration_status") != "PENDING"
        or manifest.get("receipt") is not None
    ):
        raise ValueError("checkpoint manifest must remain the pending exporter output")
    _sha(manifest.get("export_sha256"), 64, "export sha")
    _sha(manifest.get("provenance_sha256"), 64, "provenance sha")
    _sha(manifest.get("source_slice_sha256"), 64, "source slice sha")
    if manifest.get("as_of_ns") != _manifest_as_of(path):
        raise ValueError("checkpoint manifest cutoff does not match path")
    if (
        not isinstance(manifest.get("generator_provider_run_id"), str)
        or not manifest["generator_provider_run_id"].isdigit()
    ):
        raise ValueError("checkpoint generator provider run id is invalid")
    return manifest


def _canonical_payload(content: bytes, label: str) -> dict[str, Any]:
    try:
        payload = json.loads(content.decode("utf-8"))
    except (UnicodeError, json.JSONDecodeError) as exc:
        raise ValueError(f"{label} is not JSON") from exc
    if not isinstance(payload, dict):
        raise ValueError(f"{label} must be an object")
    canonical = (json.dumps(payload, sort_keys=True, separators=(",", ":")) + "\n").encode()
    if content != canonical:
        raise ValueError(f"{label} bytes are not canonical")
    return payload


def _canonical_sha256(value: Any) -> str:
    encoded = json.dumps(value, sort_keys=True, separators=(",", ":")).encode()
    return hashlib.sha256(encoded).hexdigest()


def _manifest_as_of(path: str) -> int:
    prefix = f"{MANIFEST_ROOT}/"
    suffix = "/time_space_shadow_export_checkpoint_v1.json"
    if not path.startswith(prefix) or not path.endswith(suffix):
        raise ValueError("checkpoint manifest path changed")
    value = path[len(prefix) : -len(suffix)]
    if not value.isdigit() or value.startswith("0"):
        raise ValueError("checkpoint manifest path cutoff is invalid")
    as_of_ns = int(value)
    if as_of_ns % (4 * 60 * 60 * 1_000_000_000):
        raise ValueError("checkpoint manifest cutoff is not on the 4h grid")
    return as_of_ns


def _sha(value: Any, length: int, name: str) -> str:
    if not isinstance(value, str) or len(value) != length or any(c not in "0123456789abcdef" for c in value):
        raise ValueError(f"{name} is invalid")
    return value


def _timestamp_ns(value: Any) -> int:
    if not isinstance(value, str):
        raise ValueError("provider created_at is invalid")
    try:
        parsed = datetime.strptime(value, "%Y-%m-%dT%H:%M:%SZ").replace(tzinfo=UTC)
    except ValueError as exc:
        raise ValueError("provider created_at is invalid") from exc
    return int(parsed.timestamp()) * 1_000_000_000


def _load_json(path: Path) -> dict[str, Any]:
    try:
        payload = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise ValueError(f"cannot load JSON: {path}") from exc
    if not isinstance(payload, dict):
        raise ValueError(f"JSON payload must be an object: {path}")
    return payload


def _main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--manifest", type=Path, required=True)
    parser.add_argument("--manifest-path", required=True)
    parser.add_argument("--provider-run-json", type=Path, required=True)
    parser.add_argument("--artifact-json", type=Path, required=True)
    parser.add_argument("--generator-export", type=Path, required=True)
    parser.add_argument("--generator-provenance", type=Path, required=True)
    parser.add_argument("--generator-artifact-json", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args(argv)
    receipt = build_receipt(
        manifest_path=args.manifest_path,
        manifest_bytes=args.manifest.read_bytes(),
        provider=_load_json(args.provider_run_json),
        artifact=_load_json(args.artifact_json),
        generator_export_bytes=args.generator_export.read_bytes(),
        generator_provenance_bytes=args.generator_provenance.read_bytes(),
        generator_artifact=_load_json(args.generator_artifact_json),
    )
    args.output.write_text(json.dumps(receipt, sort_keys=True, separators=(",", ":")) + "\n", encoding="utf-8")
    return 0


if __name__ == "__main__":
    raise SystemExit(_main())
