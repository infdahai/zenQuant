def _ns(value: str) -> int:
    return int(datetime.fromisoformat(value).replace(tzinfo=UTC).timestamp() * 1_000_000_000)


def _row(
    cutoff_time_ns: int,
    *,
    interval_start_ns: int | None = None,
    close: float = 100.0,
    structure_side: int = 1,
    prior_long_boundary: float | None = None,
    prior_short_boundary: float | None = None,
    current_long_pivot: float | None = None,
    current_short_pivot: float | None = None,
    prior_long_invalidation: float | None = 90.0,
    prior_short_invalidation: float | None = 110.0,
    alpha_id: str | None = None,
) -> TimeSpaceInputRow:
    return TimeSpaceInputRow(
        alpha_id=alpha_id or f"alpha-{cutoff_time_ns}",
        cutoff_time_ns=cutoff_time_ns,
        interval_start_ns=(cutoff_time_ns - FOUR_HOURS_NS if interval_start_ns is None else interval_start_ns),
        close=close,
        structure_side=structure_side,
        prior_long_boundary=prior_long_boundary,
        prior_short_boundary=prior_short_boundary,
        current_long_pivot=current_long_pivot,
        current_short_pivot=current_short_pivot,
        prior_long_invalidation=prior_long_invalidation,
        prior_short_invalidation=prior_short_invalidation,
        formula_hash="sha256:formula",
        config_hash="sha256:config",
        source_data_hash="sha256:data",
        decision_id_by_stage=(
            ("A1_SIGN", f"alpha-decision-{cutoff_time_ns}"),
            ("A5_SHADOW", f"approved-decision-{cutoff_time_ns}"),
            (FILLED_POSITION_STAGE, f"execution-position-{cutoff_time_ns}"),
        ),
    )


def _population_rows(
    states: tuple[TimeSpaceState, ...],
    kind: FlipKind,
    values: tuple[int | float, ...],
) -> tuple[FlipPopulationInputRow, ...]:
    rows: list[FlipPopulationInputRow] = []
    for state, value in zip(states, values, strict=True):
        decisions = dict(state.decision_id_by_stage)
        if kind is FlipKind.ALPHA:
            source_identity = decisions["A1_SIGN"]
        elif kind is FlipKind.APPROVED:
            source_identity = decisions["A5_SHADOW"]
        else:
            source_identity = decisions[FILLED_POSITION_STAGE]
        rows.append(
            FlipPopulationInputRow(
                kind=kind,
                alpha_id=state.alpha_id,
                cutoff_time_ns=state.cutoff_time_ns,
                value=value,
                source_identity=source_identity,
            )
        )
    return tuple(rows)


def _flip_and_states(
    closes: tuple[float, ...],
    *,
    cutoffs: tuple[int, ...] | None = None,
) -> tuple[TimeSpaceFlipObservation, tuple[TimeSpaceState, ...]]:
    resolved_cutoffs = cutoffs or tuple(index * FOUR_HOURS_NS for index in range(1, len(closes) + 1))
    states = build_time_space_states(
        tuple(
            _row(cutoff, close=close, alpha_id=f"alpha-{index}")
            for index, (cutoff, close) in enumerate(zip(resolved_cutoffs, closes, strict=True))
        )
    )
    alpha_values = (1, -1, *((-1,) * (len(states) - 2)))
    observations = build_flip_observations(
        states=states,
        alpha_rows=_population_rows(states, FlipKind.ALPHA, alpha_values),
        approved_rows=_population_rows(states, FlipKind.APPROVED, (0.0,) * len(states)),
        filled_rows=_population_rows(states, FlipKind.FILLED, (0.0,) * len(states)),
    )
    return observations[0], states


def _canonical_test_id(values: dict[str, object]) -> str:
    return hashlib.sha256(
        json.dumps(values, sort_keys=True, separators=(",", ":"), ensure_ascii=True).encode()
    ).hexdigest()


def _rehash_flip(flip: TimeSpaceFlipObservation) -> TimeSpaceFlipObservation:
    return replace(
        flip,
        base_flip_id=_canonical_test_id(
            {
                "kind": flip.kind.value,
                "alpha_id": flip.alpha_id,
                "cutoff_time_ns": flip.cutoff_time_ns,
                "old_side": flip.old_side,
                "new_side": flip.new_side,
                "state_id": flip.state_id,
                "prior_source_identity": flip.prior_source_identity,
                "source_identity": flip.source_identity,
            }
        ),
    )


def _rehash_state(state: TimeSpaceState) -> TimeSpaceState:
    return replace(
        state,
        state_id=_canonical_test_id(
            {
                item.name: getattr(state, item.name)
                for item in fields(TimeSpaceState)
                if item.name not in {"state_id", "blocks_order"}
            }
        ),
    )


def mature_flip_labels(
    flips: tuple[TimeSpaceFlipObservation, ...],
    closes: tuple[TimeSpaceState, ...],
    *,
    as_of_ns: int,
    all_populations: bool = False,
    provenance_states: tuple[TimeSpaceState, ...] | None = None,
) -> tuple[TimeSpaceLabel, ...]:
    source_states = closes if provenance_states is None else provenance_states
    directional_values = (1, -1, *((-1,) * (len(source_states) - 2)))
    neutral_values = (0.0,) * len(source_states)
    approved_values = directional_values if all_populations else neutral_values
    filled_values = directional_values if all_populations else neutral_values
    return _raw_mature_flip_labels(
        flips,
        closes,
        alpha_rows=_population_rows(source_states, FlipKind.ALPHA, directional_values),
        approved_rows=_population_rows(source_states, FlipKind.APPROVED, approved_values),
        filled_rows=_population_rows(source_states, FlipKind.FILLED, filled_values),
        as_of_ns=as_of_ns,
    )


def test_monday_close_belongs_to_prior_week_and_next_bar_resets_week() -> None:
    monday_00_ns = _ns("2026-08-24T00:00:00")
    monday_04_ns = monday_00_ns + FOUR_HOURS_NS

    states = build_time_space_states(
        (
            _row(monday_00_ns, interval_start_ns=monday_00_ns - FOUR_HOURS_NS),
            _row(monday_04_ns),
        )
    )

    assert states[0].utc_week_start_ns == _ns("2026-08-17T00:00:00")
    assert states[0].weekly_completion_fraction == 1.0
    assert states[1].utc_week_start_ns == monday_00_ns
    assert states[1].weekly_completion_fraction == pytest.approx(4 / (7 * 24))

    repeated = build_time_space_states((row,))[0]

        first.source_gap_bars = 2  # type: ignore[misc]

def test_alpha_approved_and_filled_flips_do_not_collapse() -> None:
    states = build_time_space_states(
        (
            _row(FOUR_HOURS_NS, alpha_id="alpha-before"),
            _row(2 * FOUR_HOURS_NS, alpha_id="alpha-after"),
        )
    )

    observations = build_flip_observations(
        states=states,
        alpha_rows=_population_rows(states, FlipKind.ALPHA, (1, -1)),
        approved_rows=_population_rows(states, FlipKind.APPROVED, (1.0, 0.0)),
        filled_rows=_population_rows(states, FlipKind.FILLED, (1.0, -0.25)),
    )

    assert [item.kind for item in observations] == [FlipKind.ALPHA, FlipKind.FILLED]
    assert len({item.base_flip_id for item in observations}) == 2
    assert all(item.alpha_id == "alpha-after" for item in observations)
    assert all(item.state_id == states[-1].state_id for item in observations)
    assert all(item.cutoff_time_ns == states[-1].cutoff_time_ns for item in observations)
    assert observations[-1].prior_source_identity == dict(states[0].decision_id_by_stage)[FILLED_POSITION_STAGE]
    assert observations[-1].source_identity == dict(states[-1].decision_id_by_stage)[FILLED_POSITION_STAGE]

def test_flip_observations_are_immutable_canonical_and_require_exact_joins() -> None:
    states = build_time_space_states(
        (
            _row(FOUR_HOURS_NS, alpha_id="alpha-before"),
            _row(2 * FOUR_HOURS_NS, alpha_id="alpha-after"),
        )
    )
    first = build_flip_observations(
        states=states,
        alpha_rows=_population_rows(states, FlipKind.ALPHA, (1, -1)),
        approved_rows=_population_rows(states, FlipKind.APPROVED, (1.0, -1.0)),
        filled_rows=_population_rows(states, FlipKind.FILLED, (1.0, -1.0)),
    )
    repeated = build_flip_observations(
        states=states,
        alpha_rows=_population_rows(states, FlipKind.ALPHA, (1, -1)),
        approved_rows=_population_rows(states, FlipKind.APPROVED, (1.0, -1.0)),
        filled_rows=_population_rows(states, FlipKind.FILLED, (1.0, -1.0)),
    )

    assert first == repeated
    assert all(len(item.base_flip_id) == 64 for item in first)
    input_row = _population_rows(states, FlipKind.ALPHA, (1, -1))[0]
    with pytest.raises(FrozenInstanceError):
        input_row.value = -1  # type: ignore[misc]
    with pytest.raises(FrozenInstanceError):
        first[0].old_side = -1  # type: ignore[misc]
    with pytest.raises(ValueError, match="exactly join"):
        build_flip_observations(
            states=states,
            alpha_rows=_population_rows(states[:1], FlipKind.ALPHA, (1,)),
            approved_rows=_population_rows(states, FlipKind.APPROVED, (1.0, -1.0)),

def test_identity_bearing_population_rows_fail_closed_on_misalignment(mutation: str) -> None:
    states = build_time_space_states(
        (
            _row(FOUR_HOURS_NS, alpha_id="alpha-before"),
            _row(2 * FOUR_HOURS_NS, alpha_id="alpha-after"),
        )
    )
    alpha_rows = _population_rows(states, FlipKind.ALPHA, (1, -1))
    if mutation == "swapped":
        alpha_rows = tuple(reversed(alpha_rows))
    elif mutation == "shifted_alpha":
        alpha_rows = (alpha_rows[0], replace(alpha_rows[1], alpha_id=states[0].alpha_id))
    elif mutation == "wrong_cutoff":
        alpha_rows = (alpha_rows[0], replace(alpha_rows[1], cutoff_time_ns=3 * FOUR_HOURS_NS))
    elif mutation == "wrong_stage":
        alpha_rows = (alpha_rows[0], replace(alpha_rows[1], source_identity="approved-decision-wrong"))
    elif mutation == "duplicate":
        alpha_rows = (alpha_rows[0], alpha_rows[0])
    elif mutation == "extra":
        alpha_rows = (*alpha_rows, alpha_rows[-1])
    else:
        alpha_rows = (alpha_rows[0], replace(alpha_rows[1], kind=FlipKind.APPROVED))

    with pytest.raises(ValueError, match="exactly join"):
        build_flip_observations(
            states=states,
            alpha_rows=alpha_rows,
            approved_rows=_population_rows(states, FlipKind.APPROVED, (1.0, -1.0)),
            filled_rows=_population_rows(states, FlipKind.FILLED, (1.0, -1.0)),

def _acceptance_flip(index: int, *, cutoff_time_ns: int, primary: bool = True) -> TimeSpaceFlipObservation:
    flip = TimeSpaceFlipObservation(
        base_flip_id="pending",
        kind=FlipKind.ALPHA,
        alpha_id=f"acceptance-alpha-{index}",
        cutoff_time_ns=cutoff_time_ns,
        old_side=1,
        new_side=-1,
        state_id=f"acceptance-state-{index}",
        prior_source_identity=f"acceptance-prior-{index}",
        source_identity=f"acceptance-source-{index}",
        primary_population=primary,
    )
    return _rehash_flip(flip)


def _acceptance_label(flip_id: str, value: float, *, available_at_ns: int) -> TimeSpaceLabel:
    return TimeSpaceLabel(
        base_flip_id=flip_id,
        horizon_bars=6,
        flip_unit_return=-value / 2,
        hold_old_unit_return=value / 2,
        hold_minus_flip=value,
        label_available_at_ns=available_at_ns,
        right_censored=False,
    )


def _acceptance_inputs(
    *,
    values: tuple[float, ...] = (0.02,) * 30,
    elapsed_days: int = 180,
    transition_count: int = 1,
    positive_paths: int = 5,
    manifest_id: str = "a" * 64,
) -> tuple[tuple[TimeSpaceLabel, ...], TimeSpaceAcceptanceManifest]:
    start_ns = _ns("2027-01-01T00:00:00")
    flips = tuple(
        _acceptance_flip(index, cutoff_time_ns=start_ns + index * 36 * FOUR_HOURS_NS) for index in range(len(values))
    )
    labels = tuple(
        _acceptance_label(
            flip.base_flip_id,
            value,
            available_at_ns=flip.cutoff_time_ns + 6 * FOUR_HOURS_NS,
        )
        for flip, value in zip(flips, values, strict=True)
    )
    paths: list[PerturbationPathEvidence] = []
    for path_index, seed in enumerate((42, 123, 456, 789, 1024)):
        perturbed = tuple(
            _rehash_flip(
                replace(
                    flip,
                    cutoff_time_ns=flip.cutoff_time_ns + (path_index % 2) * FOUR_HOURS_NS,
                    state_id=f"perturbed-state-{seed}-{index}",
                    prior_source_identity=f"perturbed-prior-{seed}-{index}",
                    source_identity=f"perturbed-source-{seed}-{index}",
                )
            )
            for index, flip in enumerate(flips)
        )
        sign = 1.0 if path_index < positive_paths else -1.0
        path_labels = tuple(
            _acceptance_label(
                flip.base_flip_id,
                sign * 0.01,
                available_at_ns=flip.cutoff_time_ns + 6 * FOUR_HOURS_NS,
            )
            for flip in perturbed
        )
        paths.append(PerturbationPathEvidence(seed=seed, flips=perturbed, labels=path_labels))
    manifest = TimeSpaceAcceptanceManifest(
        manifest_id=manifest_id,
        hypothesis_sha256=TIME_SPACE_HYPOTHESIS_SHA256,
        observation_start_ns=start_ns,
        observation_end_ns=start_ns + elapsed_days * 24 * 60 * 60 * 1_000_000_000,
        market_state_transition_count=transition_count,
        primary_flips=flips,
        perturbation_paths=tuple(paths),
    )
    return labels, manifest

    assert matches[0].perturbed_flip_id == within_one_bar.base_flip_id
    assert map_perturbed_flips((base,), (too_late,)) == ()
    assert map_perturbed_flips((base,), (wrong_alpha,)) == ()


def test_perturbation_mapping_is_one_to_one_and_input_order_independent() -> None:
    base_one = _acceptance_flip(1, cutoff_time_ns=10 * FOUR_HOURS_NS)
    base_two = _acceptance_flip(2, cutoff_time_ns=11 * FOUR_HOURS_NS)
    perturbed_one = _rehash_flip(
        replace(
            base_one,
            cutoff_time_ns=11 * FOUR_HOURS_NS,
            state_id="p-state-1",
            prior_source_identity="p-prior-1",
            source_identity="p-source-1",
        )
    )
    perturbed_two = _rehash_flip(
        replace(
            base_two,
            cutoff_time_ns=12 * FOUR_HOURS_NS,
            state_id="p-state-2",
            prior_source_identity="p-prior-2",
            source_identity="p-source-2",
        )
    )

    canonical = map_perturbed_flips((base_one, base_two), (perturbed_one, perturbed_two))
    permuted = map_perturbed_flips((base_two, base_one), (perturbed_two, perturbed_one))

    assert permuted == canonical
    assert len({item.base_flip_id for item in canonical}) == len(canonical)
    assert len({item.perturbed_flip_id for item in canonical}) == len(canonical)
