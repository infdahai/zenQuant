"""Gate B: Import boundary enforcement.

Architectural boundaries prevent entropy. Plugins must depend only on
protocols.py — never on domain types, strategy internals, or infra.
Strategy must not reach into app or execution.

These rules are mechanical: they cannot rot like documentation.
"""

from __future__ import annotations

import ast
import unittest
from pathlib import Path

import pytest

PROJECT_ROOT = Path(__file__).parent.parent
PLUGIN_IMPL_DIRS = [
    PROJECT_ROOT / "zenquant" / "plugins" / "signals",
    PROJECT_ROOT / "zenquant" / "plugins" / "filters",
    PROJECT_ROOT / "zenquant" / "plugins" / "sizers",
    PROJECT_ROOT / "zenquant" / "plugins" / "exits",
]

FORBIDDEN_PLUGIN_IMPORTS = [
    "zenquant.domain",
    "zenquant.strategy",
    "zenquant.infra",
    "zenquant.state",
    "zenquant.indicator",
    "zenquant.execution",
    "zenquant.risk",
    "zenquant.app",
    "zenquant.historical",
    "zenquant.research",
]

STRATEGY_DIR = PROJECT_ROOT / "zenquant" / "strategy"
LATE_DEV_ATTRIBUTION_MODULE = PROJECT_ROOT / "zenquant" / "research" / "late_dev_cost_attribution.py"

FORBIDDEN_STRATEGY_IMPORTS = [
    "zenquant.app",
    "zenquant.execution",
]


def _collect_imports(filepath: Path, *, project_root: Path = PROJECT_ROOT) -> list[str]:
    """Extract all import module paths from a Python file."""
    try:
        tree = ast.parse(filepath.read_text(), filename=str(filepath))
    except SyntaxError:
        return []

    modules: list[str] = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for alias in node.names:
                modules.append(alias.name)
        elif isinstance(node, ast.ImportFrom):
            modules.extend(_resolve_from_import_targets(filepath, node, project_root=project_root))
    return modules


def _resolve_from_import_targets(filepath: Path, node: ast.ImportFrom, *, project_root: Path) -> list[str]:
    """Return absolute module targets for an AST ``from`` import and its aliases."""
    try:
        relative_path = filepath.relative_to(project_root).with_suffix("")
    except ValueError:
        return []
    package_parts = list(relative_path.parts[:-1])
    if node.level:
        prefix_length = len(package_parts) - node.level + 1
        if prefix_length < 1:
            return []
        target_parts = package_parts[:prefix_length]
    else:
        target_parts = []
    if node.module:
        target_parts.extend(node.module.split("."))
    targets = [".".join(target_parts)] if target_parts else []
    targets.extend(".".join([*target_parts, alias.name]) for alias in node.names if alias.name != "*")
    return targets


def _check_boundary(
    source_dirs: list[Path],
    forbidden_prefixes: list[str],
    label: str,
) -> list[str]:
    """Return list of violation descriptions."""
    violations: list[str] = []
    for src_dir in source_dirs:
        if not src_dir.exists():
            continue
        for py_file in src_dir.glob("*.py"):
            if py_file.name == "__init__.py":
                continue
            imports = _collect_imports(py_file)
            for imp in imports:
                for forbidden in forbidden_prefixes:
                    if imp == forbidden or imp.startswith(forbidden + "."):
                        rel = py_file.relative_to(PROJECT_ROOT)
                        violations.append(f"{label}: {rel} imports '{imp}' (forbidden: {forbidden})")
    return violations


class TestImportBoundaries(unittest.TestCase):
    """Mechanical enforcement of module boundaries."""

    def test_plugin_implementations_only_import_protocols(self):
        """Plugin signal/filter/sizer/exit files must not import zenquant internals."""
        violations = _check_boundary(
            PLUGIN_IMPL_DIRS,
            FORBIDDEN_PLUGIN_IMPORTS,
            "PLUGIN",
        )
        if violations:
            msg = "\n".join(["Import boundary violations:"] + violations)
            self.fail(msg)

    def test_strategy_does_not_import_app_or_execution(self):
        """Strategy module must not reach into app or execution."""
        violations = _check_boundary(
            [STRATEGY_DIR],
            FORBIDDEN_STRATEGY_IMPORTS,
            "STRATEGY",
        )
        if violations:
            msg = "\n".join(["Import boundary violations:"] + violations)
            self.fail(msg)

    def test_indicator_does_not_import_strategy(self):
        """Indicator engine must not depend on strategy."""
        indicator_dir = PROJECT_ROOT / "zenquant" / "indicator"
        violations = _check_boundary(
            [indicator_dir],
            ["zenquant.strategy"],
            "INDICATOR",
        )
        if violations:
            msg = "\n".join(["Import boundary violations:"] + violations)
            self.fail(msg)

    def test_state_does_not_import_strategy(self):
        """State engine must not depend on strategy."""
        state_dir = PROJECT_ROOT / "zenquant" / "state"
        violations = _check_boundary(
            [state_dir],
            ["zenquant.strategy"],
            "STATE",
        )
        if violations:
            msg = "\n".join(["Import boundary violations:"] + violations)
            self.fail(msg)

    def test_late_dev_attribution_does_not_import_live_strategy_layers(self):
        """The diagnostic envelope stays isolated from strategy and operational layers."""
        violations = _check_boundary(
            [LATE_DEV_ATTRIBUTION_MODULE.parent],
            ["zenquant.strategy", "zenquant.execution", "zenquant.risk", "zenquant.app"],
            "LATE_DEV_ATTRIBUTION",
        )
        violations = [violation for violation in violations if "late_dev_cost_attribution.py" in violation]
        if violations:
            msg = "\n".join(["Import boundary violations:"] + violations)
            self.fail(msg)

    def test_operational_layers_do_not_import_time_space_shadow_or_exporter(self):
        """Research-only TimeSpace modules cannot become operational dependencies."""
        violations = _check_boundary(
            [
                PROJECT_ROOT / "zenquant" / "strategy",
                PROJECT_ROOT / "zenquant" / "risk",
                PROJECT_ROOT / "zenquant" / "execution",
                PROJECT_ROOT / "zenquant" / "app",
                PROJECT_ROOT / "zenquant" / "infra",
                PROJECT_ROOT / "zenquant" / "ingestor",
            ],
            [
                "zenquant.research.time_space_shadow",
                "zenquant.research.time_space_export",
                "zenquant.research.time_space_replay",
            ],
            "TIME_SPACE_ISOLATION",
        )
        if violations:
            self.fail("\n".join(["Import boundary violations:"] + violations))


@pytest.mark.parametrize(
    ("source", "forbidden_import"),
    [
        ("from ..strategy import PlaybookEngine\n", "zenquant.strategy"),
        ("from .. import execution\n", "zenquant.execution"),
        ("from zenquant import app, risk\n", "zenquant.app"),
    ],
)
def test_collect_imports_resolves_relative_and_alias_import_targets(tmp_path: Path, source: str, forbidden_import: str):
    """AST import collection must expose strategy/live-layer targets in every syntax form."""
    filepath = tmp_path / "zenquant" / "research" / "probe.py"
    filepath.parent.mkdir(parents=True)
    filepath.write_text(source)

    imports = _collect_imports(filepath, project_root=tmp_path)

    assert forbidden_import in imports


if __name__ == "__main__":
    unittest.main()
