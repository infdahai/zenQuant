def _module():
    spec = importlib.util.spec_from_file_location("build_time_space_export_receipt", SCRIPT)
    assert spec is not None and spec.loader is not None
    module = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = module
    spec.loader.exec_module(module)
    return module


def _manifest() -> dict[str, object]:
    export, provenance = _generator_payloads()
    return {
        "schema_version": "time-space-export-checkpoint.v1",
        "layer": "time_space_shadow",
        "registration_status": "PENDING",
        "export_sha256": _canonical_sha256(export),
        "provenance_sha256": _canonical_sha256(provenance),
        "receipt": None,
        "as_of_ns": AS_OF_NS,
        "generator_provider_run_id": "54321",
        "source_slice_sha256": _canonical_sha256(provenance["source_files"]),
    }


def _generator_payloads() -> tuple[dict[str, object], dict[str, object]]:
    return {
        "schema_version": "time-space-shadow-export.v1",
        "layer": "time_space_shadow",
        "as_of_ns": AS_OF_NS,
        "rows": [],
    }, {
        "schema_version": "time-space-shadow-export-provenance.v1",
        "layer": "time_space_shadow",
        "as_of_ns": AS_OF_NS,
        "repository_identity": "infdahai/zenQuant",
        "commit_sha": "d" * 40,
        "provider_run_id": "54321",
        "source_files": [{"path": "frozen.parquet", "sha256": "a" * 64}],
    }


def _canonical_sha256(value: object) -> str:
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(",", ":")).encode()).hexdigest()


def _generator_args() -> dict[str, object]:
    export, provenance = _generator_payloads()
    return {
        "generator_export_bytes": (json.dumps(export, sort_keys=True, separators=(",", ":")) + "\n").encode(),
        "generator_provenance_bytes": (json.dumps(provenance, sort_keys=True, separators=(",", ":")) + "\n").encode(),
        "generator_artifact": {
            "id": 778,
            "name": f"time-space-export-{AS_OF_NS}-54321",
            "expired": False,
        },
    }


def _provider(**changes):
    payload = {
        "id": 12345,
        "head_sha": "c" * 40,
        "created_at": "2026-08-24T08:00:00Z",
        "conclusion": "success",
        "event": "push",
        "head_branch": BRANCH,
        "path": WORKFLOW_PATH,
        "workflow_id": WORKFLOW_ID,
        "repository": {"full_name": "infdahai/zenQuant"},
    }
    payload.update(changes)
    return payload


def _artifact(**changes):
    payload = {
        "id": 777,
        "name": f"time-space-export-checkpoint-{AS_OF_NS}-12345",
        "expired": False,
        "workflow_run": {"id": 12345, "head_sha": "c" * 40},
    }
    payload.update(changes)
    return payload


def _trust() -> dict[str, object]:
    return {"checkpoint_workflow_id": WORKFLOW_ID}


def _git(repository: Path, *args: str) -> str:
    result = subprocess.run(
        ["git", "-C", str(repository), *args],
        check=True,
        capture_output=True,
        text=True,
    )
    return result.stdout.strip()


def _commit(repository: Path, message: str) -> str:
    _git(repository, "add", ".")
    _git(
        repository,
        "-c",
        "user.name=TimeSpace Test",
        "-c",
        "user.email=time-space@example.invalid",
        "commit",
        "-m",
        message,
    )
    return _git(repository, "rev-parse", "HEAD")


def test_receipt_builder_binds_completed_push_workflow_and_uploaded_manifest() -> None:
    module = _module()
    manifest_bytes = (json.dumps(_manifest(), sort_keys=True, separators=(",", ":")) + "\n").encode()

    receipt = module._build_receipt(
        manifest_path=MANIFEST_PATH,
        manifest_bytes=manifest_bytes,
        provider=_provider(),
        artifact=_artifact(),
        trust=_trust(),
        **_generator_args(),
    )

    assert receipt["schema_version"] == "time-space-export-registration-receipt.v1"
    assert receipt["manifest_sha256"] == hashlib.sha256(manifest_bytes).hexdigest()
    assert receipt["provider_run_id"] == "12345"
    assert receipt["provider_created_at_ns"] == 1787558400000000000
    assert receipt["workflow"] == {"id": WORKFLOW_ID, "path": WORKFLOW_PATH}
    assert receipt["artifact"] == {
        "id": 777,
        "name": f"time-space-export-checkpoint-{AS_OF_NS}-12345",
    }

        ({}, {"workflow_run": {"id": 9, "head_sha": "c" * 40}}),

)
def test_receipt_builder_rejects_provider_or_artifact_tampering(provider_change, artifact_change) -> None:
    module = _module()
    manifest_bytes = (json.dumps(_manifest(), sort_keys=True, separators=(",", ":")) + "\n").encode()
    with pytest.raises(ValueError):
        module._build_receipt(
            manifest_path=MANIFEST_PATH,

            provider=_provider(**provider_change),
            artifact=_artifact(**artifact_change),
            trust=_trust(),
            **_generator_args(),
        )


def test_receipt_builder_rejects_wrong_manifest_path_schema_or_noncanonical_bytes() -> None:
    module = _module()
    canonical = (json.dumps(_manifest(), sort_keys=True, separators=(",", ":")) + "\n").encode()
    with pytest.raises(ValueError, match="path"):
        module._build_receipt(
            manifest_path="checkpoint.json",
            manifest_bytes=canonical,
            provider=_provider(),
            artifact=_artifact(),
            trust=_trust(),
            **_generator_args(),
        )
    altered = _manifest()
    altered["receipt"] = {}
    with pytest.raises(ValueError, match="manifest"):
        module._build_receipt(
            manifest_path=MANIFEST_PATH,
            manifest_bytes=(json.dumps(altered, sort_keys=True, separators=(",", ":")) + "\n").encode(),
            provider=_provider(),
            artifact=_artifact(),
            trust=_trust(),
            **_generator_args(),
        )
    with pytest.raises(ValueError, match="canonical"):
        module._build_receipt(
            manifest_path=MANIFEST_PATH,
            manifest_bytes=json.dumps(_manifest(), indent=2).encode(),
            provider=_provider(),
            artifact=_artifact(),
            trust=_trust(),
            **_generator_args(),
        )

def _persisted_validator_command(paths: tuple[Path, ...]) -> list[str]:
    return [
        sys.executable,
        str(PERSISTED_VALIDATOR),
        "--checkpoint",
        str(paths[0]),
        "--receipt",
        str(paths[1]),
        "--trust",
        str(paths[2]),
        "--export",
        str(paths[3]),
        "--provenance",
        str(paths[4]),
        "--provider-run-json",
        str(paths[5]),
        "--generator-provider-run-json",
        str(paths[6]),
    ]


def test_persisted_registration_validator_rejects_malformed_irreversible_trust(tmp_path: Path) -> None:
    paths = _durable_registration(tmp_path)
    command = _persisted_validator_command(paths)
    valid = subprocess.run(command, check=False, capture_output=True, text=True)
    assert valid.returncode == 0, valid.stderr

    trust = json.loads(paths[2].read_text())
    trust.pop("generator_workflow_id")
    paths[2].write_text(json.dumps(trust, sort_keys=True, separators=(",", ":")) + "\n")
    malformed = subprocess.run(command, check=False, capture_output=True, text=True)

    assert malformed.returncode != 0
    assert "trust" in malformed.stderr


def test_persisted_registration_rejects_correlated_attacker_repository(tmp_path: Path) -> None:
    paths = _durable_registration(tmp_path)
    attacker = "attacker/forged-zenquant"
    receipt = json.loads(paths[1].read_text())
    trust = json.loads(paths[2].read_text())
    checkpoint = json.loads(paths[0].read_text())
    provenance = json.loads(paths[4].read_text())
    receipt["remote_identity"] = attacker
    provenance["repository_identity"] = attacker
    trust["remote_identity"] = attacker
    paths[4].write_text(json.dumps(provenance, sort_keys=True, separators=(",", ":")) + "\n")
    provenance_sha = _canonical_sha256(provenance)
    checkpoint["provenance_sha256"] = provenance_sha
    paths[0].write_text(json.dumps(checkpoint, sort_keys=True, separators=(",", ":")) + "\n")
    manifest_sha = hashlib.sha256(paths[0].read_bytes()).hexdigest()
    receipt["manifest_sha256"] = manifest_sha
