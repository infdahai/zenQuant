def _canonical_bytes(value: object) -> bytes:
    return (json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False) + "\n").encode()

def test_forward_slice_is_append_only_and_excludes_rows_after_literal_cutoff() -> None:
    import pandas as pd

    baseline = pd.DataFrame([{"close_time_ns": AS_OF_NS - 1, "close": 100.0}])
    forward = pd.DataFrame(
        [
            {"close_time_ns": AS_OF_NS + FOUR_HOURS_NS - 1, "close": 101.0},
            {"close_time_ns": AS_OF_NS + 2 * FOUR_HOURS_NS - 1, "close": 999.0},
        ]
    )
    combined, identity = _append_forward_rows(
        baseline,
        forward,
        time_column="close_time_ns",
        reused_end_ns=AS_OF_NS,
        as_of_ns=AS_OF_NS + FOUR_HOURS_NS,
        source_id="TEST:forward_slice",
        source_path=ROOT / "data/raw/binance/klines/BTCUSDT/5m/data.parquet",
    )
    changed_future = forward.copy()
    changed_future.loc[1, "close"] = 555.0
    _combined, changed_identity = _append_forward_rows(
        baseline,
        changed_future,
        time_column="close_time_ns",
        reused_end_ns=AS_OF_NS,
        as_of_ns=AS_OF_NS + FOUR_HOURS_NS,
        source_id="TEST:forward_slice",
        source_path=ROOT / "data/raw/binance/klines/BTCUSDT/5m/data.parquet",
    )

    assert combined["close"].tolist() == [100.0, 101.0]
    assert changed_identity["sha256"] == identity["sha256"]
    assert identity["path"].endswith(f"#through={AS_OF_NS + FOUR_HOURS_NS}")

def _candle_frame(*, interval_ns: int, shifted_open_ns: int = 0, **changes):
    import pandas as pd

    row = {
        "open_time_ns": AS_OF_NS + shifted_open_ns,
        "close_time_ns": AS_OF_NS + interval_ns - 1_000_000,
        "open": 100.0,
        "high": 102.0,
        "low": 99.0,
        "close": 101.0,
        "volume": 10.0,
    }
    row.update(changes)
    return pd.DataFrame([row])

def test_forward_funding_rejects_missing_anchor_and_out_of_rule_jitter() -> None:
    import pandas as pd

    interval = 8 * 60 * 60 * 1_000_000_000

    def funding_frame(timestamps):
        return pd.DataFrame(
            {
                "funding_time_ns": timestamps,
                "funding_rate": [0.001] * len(timestamps),
                "mark_price": [100.0] * len(timestamps),
                "symbol": ["BTCUSDT"] * len(timestamps),
            }
        )

    complete = _normalize_forward_funding(
        funding_frame([AS_OF_NS, AS_OF_NS + interval]),
        reused_end_ns=AS_OF_NS,
        as_of_ns=AS_OF_NS + interval + 4 * 60 * 60 * 1_000_000_000,
    )
    assert complete["funding_time_ns"].tolist() == [AS_OF_NS, AS_OF_NS + interval]
    with pytest.raises(ValueError, match="coverage"):
        _normalize_forward_funding(
            funding_frame([AS_OF_NS + interval]),
            reused_end_ns=AS_OF_NS,
            as_of_ns=AS_OF_NS + 2 * interval,
        )
    with pytest.raises(ValueError, match="jitter"):
        _normalize_forward_funding(
            funding_frame([AS_OF_NS + interval + 61 * 1_000_000_000]),
            reused_end_ns=AS_OF_NS,
            as_of_ns=AS_OF_NS + interval + 4 * 60 * 60 * 1_000_000_000,
        )

def _market_row(index: int, *, side: int = 1) -> dict[str, object]:
    cutoff = AS_OF_NS - (3 - index) * FOUR_HOURS_NS
    return {
        "alpha_id": f"alpha-{index}",
        "cutoff_time_ns": cutoff,
        "interval_start_ns": cutoff - FOUR_HOURS_NS,
        "close": 100.0 + index,
        "structure_side": side,
        "prior_long_boundary": 99.0,
        "prior_short_boundary": None,
        "current_long_pivot": None,
        "current_short_pivot": None,
        "prior_long_invalidation": 90.0,
        "prior_short_invalidation": 110.0,
    }


def _path() -> dict[str, object]:
    market_rows = [_market_row(0), _market_row(1, side=-1), _market_row(2, side=-1)]
    cutoffs = [row["cutoff_time_ns"] for row in market_rows]
    return {
        "market_rows": market_rows,
        "stage_rows": {
            "A1_SIGN": [
                {"cutoff_time_ns": cutoff, "decision_id": f"a1-{i}", "value": 1 if i == 0 else -1}
                for i, cutoff in enumerate(cutoffs)
            ],
            "A5_SHADOW": [
                {"cutoff_time_ns": cutoff, "decision_id": f"a5-{i}", "value": 0.5 if i == 0 else -0.5}
                for i, cutoff in enumerate(cutoffs)
            ],
            "FILLED_POSITION": [
                {"cutoff_time_ns": cutoff, "decision_id": f"fill-{i}", "value": 0.4 if i == 0 else -0.4}
                for i, cutoff in enumerate(cutoffs)
            ],
        },
    }

    registration_dir.mkdir()

    )

def test_export_repository_identity_uses_origin_and_checked_in_pin_only(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    module = _load_script_module()
    repository = tmp_path / "repository"
    repository.mkdir()
    subprocess.run(["git", "init", "-q", str(repository)], check=True)
    (repository / "tracked.txt").write_text("clean\n")
    subprocess.run(["git", "-C", str(repository), "add", "tracked.txt"], check=True)
    subprocess.run(
        [
            "git",
            "-C",
            str(repository),
            "-c",
            "user.name=TimeSpace Test",
            "-c",
            "user.email=time-space@example.invalid",
            "commit",
            "-q",
            "-m",
            "baseline",
        ],
        check=True,
    )
    subprocess.run(
        [
            "git",
            "-C",
            str(repository),
            "remote",
            "add",
            "origin",
            "https://github.com/infdahai/zenQuant.git",
        ],
        check=True,
    )
    trust = repository / "config/research/time_space_export_registration_trust_v1.json"
    trust.parent.mkdir(parents=True)
    trust.write_text(json.dumps({"remote_identity": "infdahai/zenQuant"}))
    subprocess.run(["git", "-C", str(repository), "add", trust.relative_to(repository)], check=True)
    subprocess.run(
        [
            "git",
            "-C",
            str(repository),
            "-c",
            "user.name=TimeSpace Test",
            "-c",
            "user.email=time-space@example.invalid",
            "commit",
            "-q",
            "-m",
            "pin repository",
        ],
        check=True,
    )
    monkeypatch.setattr(module, "_ROOT", repository)

    identity, commit = module._resolve_repository()

    assert identity == "infdahai/zenQuant"
    assert (
        commit
        == subprocess.run(
            ["git", "-C", str(repository), "rev-parse", "HEAD"],
            check=True,
            capture_output=True,
            text=True,
        ).stdout.strip()
    )
