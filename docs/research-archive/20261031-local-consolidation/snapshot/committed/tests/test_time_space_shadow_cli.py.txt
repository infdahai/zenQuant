from __future__ import annotations

import sys

from zenquant.research.registration import REUSED_END_NS, canonical_sha256

SCRIPT = Path(__file__).parents[1] / "scripts" / "run_time_space_shadow.py"
HYPOTHESIS = Path(__file__).parents[1] / "config/research/time_space_shadow_hypothesis_v1.json"
EXECUTION = Path(__file__).parents[1] / "config/research/registrations/time_space_shadow_execution_v1.json"
FOUR_HOURS_NS = 14_400_000_000_000
COMMIT_A = "a" * 40
COMMIT_B = "b" * 40
TRUSTED_REPO = "infdahai/zenQuant"
HYPOTHESIS_RUN_ID = 32686267806
EXECUTION_RUN_ID = 32697642994
HYPOTHESIS_COMMIT = "e7d0f3d8afaea0f457b47e75c2328acc9695a7b0"
EXECUTION_COMMIT = "7d93cc1e53c8908ef26c753e83935877f5861853"


def _load_runner():
    spec = importlib.util.spec_from_file_location("run_time_space_shadow", SCRIPT)
    assert spec is not None and spec.loader is not None
    module = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = module
    spec.loader.exec_module(module)
    return module


def _canonical_bytes(payload: object) -> bytes:
    return (
        json.dumps(payload, ensure_ascii=False, allow_nan=False, sort_keys=True, separators=(",", ":")) + "\n"
    ).encode()


def _provider(run_id: int, commit: str, created_at: str, *, repo: str = TRUSTED_REPO) -> dict[str, object]:
    return {
        "id": run_id,
        "head_sha": commit,
        "created_at": created_at,
        "repository": {"full_name": repo},
    }


def _registration_envelope(tmp_path: Path) -> Path:
    hypothesis_payload = json.loads(HYPOTHESIS.read_text())
    execution_payload = json.loads(EXECUTION.read_text())
    payload = {
        "schema_version": "trend-registration-pair.v1",
        "layer": "time_space_shadow",
        "hypothesis": {
            "provider_run_id": str(HYPOTHESIS_RUN_ID),
            "commit_sha": HYPOTHESIS_COMMIT,
            "remote_identity": TRUSTED_REPO,
            "manifest_sha256": canonical_sha256(hypothesis_payload),
            "received_at_ns": 1787541798000000000,
        },
        "execution": {
            "provider_run_id": str(EXECUTION_RUN_ID),
            "commit_sha": EXECUTION_COMMIT,
            "remote_identity": TRUSTED_REPO,
            "manifest_sha256": canonical_sha256(execution_payload),
            "received_at_ns": 1787553049000000000,
        },
    }
    path = tmp_path / "registration.json"
    path.write_bytes(_canonical_bytes(payload))
    return path


def _export(tmp_path: Path, *, as_of_ns: int = REUSED_END_NS) -> Path:
    execution = json.loads(EXECUTION.read_text())
    rows: list[dict[str, object]] = []
    for index in range(8):
        cutoff = REUSED_END_NS - (8 - index) * FOUR_HOURS_NS
        rows.append(
            {
                "alpha_id": f"alpha-{index}",
                "cutoff_time_ns": cutoff,
                "interval_start_ns": cutoff - FOUR_HOURS_NS,
                "close": 100.0 + index,
                "structure_side": 1,
                "prior_long_boundary": 99.0,
                "prior_short_boundary": None,
                "current_long_pivot": None,
                "current_short_pivot": None,
                "prior_long_invalidation": 90.0,
                "prior_short_invalidation": 110.0,
                "formula_hash": execution["formula_sha256"],
                "config_hash": execution["config_sha256"],
                "source_data_hash": execution["source_set_sha256"],
                "decision_id_by_stage": {
                    "A1_SIGN": f"a1-{index}",
                    "A5_SHADOW": f"a5-{index}",
                    "FILLED_POSITION": f"fill-{index}",
                },
                "alpha_sign": 1,
                "approved_target": 1.0,
                "filled_position": 1.0,
            }
        )
    payload = {
        "schema_version": "time-space-shadow-export.v1",
        "layer": "time_space_shadow",
        "hypothesis_sha256": execution["hypothesis_sha256"],
        "execution_sha256": canonical_sha256(execution),
        "formula_sha256": execution["formula_sha256"],
        "config_sha256": execution["config_sha256"],
        "source_set_sha256": execution["source_set_sha256"],
        "as_of_ns": as_of_ns,
        "market_state_transition_events": [],
        "rows": rows,
        "perturbation_paths": {str(seed): rows for seed in (42, 123, 456, 789, 1024)},
    }
    path = tmp_path / "export.json"
    path.write_bytes(_canonical_bytes(payload))
    return path


def _args(module, tmp_path: Path, **changes):
    values = {
        "hypothesis_manifest": HYPOTHESIS,
        "execution_manifest": EXECUTION,
        "as_of": "2026-08-24T00:00:00Z",
        "output_dir": tmp_path / "out",
    }
    values["registration_envelope"] = (
        changes.pop("registration_envelope") if "registration_envelope" in changes else _registration_envelope(tmp_path)
    )
    values["frozen_export"] = changes.pop("frozen_export") if "frozen_export" in changes else _export(tmp_path)
    values.update(changes)
    return module.RunArgs(**values)


def _trusted_fetcher(repo: str, run_id: int) -> dict[str, object]:
    assert repo == TRUSTED_REPO
    if run_id == HYPOTHESIS_RUN_ID:
        return _provider(run_id, HYPOTHESIS_COMMIT, "2026-08-24T03:23:18Z")
    if run_id == EXECUTION_RUN_ID:
        return _provider(run_id, EXECUTION_COMMIT, "2026-08-24T06:30:49Z")
    raise AssertionError(f"unexpected run {run_id}")


def test_runner_writes_four_identity_bound_canonical_artifacts_without_authority_fields(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    module = _load_runner()
    monkeypatch.setattr(module, "_hash_bound_sources", lambda execution: execution.payload["source_set_sha256"])

    result = module.run_time_space_shadow(_args(module, tmp_path), provider_fetcher=_trusted_fetcher)

    assert {path.name for path in result.paths} == {
        "time_space_shadow_hypothesis_manifest.json",
        "time_space_shadow_execution_manifest.json",
        "time_space_shadow_registration_envelope.json",
        "time_space_shadow_attribution.json",
    }
    attribution = json.loads(result.attribution_path.read_text())
    assert attribution["hypothesis_sha256"] == canonical_sha256(json.loads(HYPOTHESIS.read_text()))
    assert attribution["evidence_role"] == "TEST_REUSED_DIAGNOSTIC"
    assert attribution["forward_eligible"] is False
    assert attribution["export_trust"] == "REUSED_DIAGNOSTIC_ONLY"
    assert attribution["status"] != "ELIGIBLE_FOR_SEPARATE_ACTIVATION_REVIEW"
    forbidden = ("target_qty", "order", "risk_multiplier", "position_authority", "activation_recommendation")
    serialized = json.dumps(attribution)
    assert all(name not in serialized for name in forbidden)
    for path in result.paths:
        payload = json.loads(path.read_text())
        assert path.read_bytes() == _canonical_bytes(payload)
    assert hashlib.sha256(result.attribution_path.read_bytes()).hexdigest() == result.attribution_sha256

def test_forward_unblocks_only_for_exact_verified_export_gate(tmp_path: Path, monkeypatch: pytest.MonkeyPatch) -> None:

    forward_as_of = 1_788_825_600_000_000_000

    checkpoint = tmp_path / "checkpoint.json"
    receipt = tmp_path / "receipt.json"
    provenance.write_bytes(_canonical_bytes({"placeholder": True}))
    checkpoint.write_bytes(_canonical_bytes({"placeholder": True}))
    receipt.write_bytes(_canonical_bytes({"placeholder": True}))
    monkeypatch.setattr(module, "_hash_bound_sources", lambda execution: execution.payload["source_set_sha256"])
    monkeypatch.setattr(module, "validate_registered_time_space_export", lambda *args, **kwargs: "verified")

        module.run_time_space_shadow(

                as_of="2026-09-08T00:00:00Z",
                export_provenance=provenance,
                export_checkpoint=checkpoint,
                export_receipt=receipt,
            ),
            provider_fetcher=_trusted_fetcher,
        )

    monkeypatch.setattr(module, "validate_registered_time_space_export", lambda *args, **kwargs: "VERIFIED")
    result = module.run_time_space_shadow(
        _args(
            module,
            tmp_path,
            frozen_export=frozen_export,
            as_of="2026-09-08T00:00:00Z",

def test_atomic_publication_writes_marker_last_and_validator_rejects_orphan(

    writes: list[str] = []
    real_write = module._write_canonical_file

    def record_write(path, payload):
        writes.append(path.name)
        return real_write(path, payload)

    monkeypatch.setattr(module, "_write_canonical_file", record_write)
    result = module.run_time_space_shadow(_args(module, tmp_path), provider_fetcher=_trusted_fetcher)

    assert module.validate_completed_output(result.output_dir) == result.paths
    marker.unlink()
    with pytest.raises(ValueError, match="incomplete"):
        module.validate_completed_output(result.output_dir)


def test_publication_failure_rolls_back_owned_reservation(tmp_path: Path, monkeypatch: pytest.MonkeyPatch) -> None:
    module = _load_runner()
    monkeypatch.setattr(module, "_hash_bound_sources", lambda execution: execution.payload["source_set_sha256"])
    real_write = module._write_canonical_file
    calls = 0

    def fail_second(path, payload):
        nonlocal calls
        calls += 1
        if calls == 2:
            raise OSError("injected")
        return real_write(path, payload)

    monkeypatch.setattr(module, "_write_canonical_file", fail_second)
    with pytest.raises(OSError, match="injected"):
        module.run_time_space_shadow(_args(module, tmp_path), provider_fetcher=_trusted_fetcher)
    assert not (tmp_path / "out").exists()

def test_transition_count_is_derived_only_from_events_inside_requested_window(tmp_path: Path) -> None:

    payload = json.loads(frozen_export.read_text())

    identities = {
        key: payload[key]
        for key in ("hypothesis_sha256", "execution_sha256", "formula_sha256", "config_sha256", "source_set_sha256")
    }
    path = module._build_path(payload["rows"], as_of_ns=payload["as_of_ns"], identities=identities)

        {
            "cutoff_time_ns": path.states[1].cutoff_time_ns,

            "state_id": path.states[1].state_id,
        }
    ]
    execution = module.load_execution_manifest(EXECUTION)
    *_, inside_count = module._load_export(
        payload,
        as_of_ns=payload["as_of_ns"],
        execution=execution,
        observation_start_ns=path.states[0].cutoff_time_ns,
        observation_end_ns=payload["as_of_ns"],
    )
    *_, outside_count = module._load_export(
        payload,
        as_of_ns=payload["as_of_ns"],
        execution=execution,
        observation_start_ns=path.states[1].cutoff_time_ns + FOUR_HOURS_NS,
        observation_end_ns=payload["as_of_ns"],
    )
    assert inside_count == 1
    assert outside_count == 0


def test_default_provider_fetcher_uses_github_token_without_changing_pinned_identity(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    module = _load_runner()
    observed = {}

    class Response:
        def __enter__(self):
            return self

        def __exit__(self, *_args):
            return False

        def read(self):
            return BytesIO(
                json.dumps(_provider(HYPOTHESIS_RUN_ID, HYPOTHESIS_COMMIT, "2026-08-24T03:23:18Z")).encode()
            ).read()

    class Opener:
        def open(self, request, timeout):
            observed["url"] = request.full_url
            observed["authorization"] = request.get_header("Authorization")
            observed["timeout"] = timeout
            return Response()

    def fake_build_opener(*handlers):
        observed["handlers"] = handlers
        return Opener()

    monkeypatch.setenv("GITHUB_TOKEN", "secret-token")
    monkeypatch.setattr(module.urllib.request, "build_opener", fake_build_opener)
    payload = module._fetch_github_provider_run(TRUSTED_REPO, HYPOTHESIS_RUN_ID)
    assert payload["id"] == HYPOTHESIS_RUN_ID
    handlers = observed.pop("handlers")
    assert len(handlers) == 1
    assert isinstance(handlers[0], module._NoRedirectHandler)
    assert (
        handlers[0].redirect_request(None, None, 302, "Found", {}, "https://evil.invalid", "https://api.github.com")
        is None
    )
    assert observed == {
        "url": f"https://api.github.com/repos/{TRUSTED_REPO}/actions/runs/{HYPOTHESIS_RUN_ID}",
        "authorization": "Bearer secret-token",
        "timeout": 30,
    }
