@dataclass(frozen=True, slots=True)
class TimeSpaceExportBundle:
    export: dict[str, Any]
    provenance: dict[str, Any]
    checkpoint: dict[str, Any]


def recompute_time_space_export(
    request: ReplayRequest,
    *,
    repository_identity: str,
    commit_sha: str,
    provider_run_id: str,
    replay_producer=produce_frozen_architecture_replay,
) -> TimeSpaceExportBundle:
    """Recompute canonical evidence through the authoritative replay producer."""

    if type(request) is not ReplayRequest:
        raise TypeError("request must be ReplayRequest")
    source = copy.deepcopy(_plain_mapping(replay_producer(request), "source replay"))
    _validate_source(source)
    execution = load_execution_manifest(Path(request.execution_manifest))
    hypothesis = load_hypothesis_manifest(Path(request.hypothesis_manifest))
    if execution.layer != "time_space_shadow" or hypothesis.layer != "time_space_shadow":
        raise ValueError("TimeSpace exporter accepts only time_space_shadow manifests")
    if execution.hypothesis_sha256 != hypothesis.sha256:
        raise ValueError("execution and hypothesis manifests are not bound")
    if source["windows"]["reused"] != [hypothesis.reused_start_ns, hypothesis.reused_end_ns]:
        raise ValueError("source reused window does not match frozen hypothesis")
    if source["as_of_ns"] != source["windows"]["reused"][1] and source["as_of_ns"] < source["windows"]["forward"][0]:
        raise ValueError("source as_of is outside a permitted evidence window")
    identities = {
        "hypothesis_sha256": hypothesis.sha256,
        "execution_sha256": execution.sha256,
        "formula_sha256": execution.payload["formula_sha256"],
        "config_sha256": execution.payload["config_sha256"],
        "source_set_sha256": execution.payload["source_set_sha256"],
    }
    rows = _build_rows(source["canonical"], identities)
    perturbations = {str(seed): _build_rows(source["perturbations"][str(seed)], identities) for seed in FROZEN_SEEDS}
    transitions = _build_transitions(rows)
    export = {
        "schema_version": "time-space-shadow-export.v1",
        "layer": "time_space_shadow",
        **identities,
        "as_of_ns": source["as_of_ns"],
        "windows": source["windows"],
        "seeds": list(FROZEN_SEEDS),
        "market_state_transition_events": transitions,
        "rows": rows,
        "perturbation_paths": perturbations,
    }
    export_sha256 = canonical_sha256(export)
    generator_code_sha256 = _sha256_file(_GENERATOR_PATH)
    dependencies, generator_tree_sha256 = _generator_dependency_identity()
    generator_semantic_sha256 = canonical_sha256(
        {
            "schema_version": "time-space-export-generator.v1",
            "source_schema": "architecture-time-space-replay.v2",
            "export_schema": "time-space-shadow-export.v1",
            "decision_stages": list(DECISION_STAGES),
            "seeds": list(FROZEN_SEEDS),
            "row_keys": sorted(_ROW_KEYS),
        }
    )
    provenance = {
        "schema_version": "time-space-shadow-export-provenance.v1",
        "layer": "time_space_shadow",
        "generator_semantic_sha256": generator_semantic_sha256,
        "generator_code_sha256": generator_code_sha256,
        "generator_dependencies": dependencies,
        "generator_tree_sha256": generator_tree_sha256,
        "repository_identity": _required_string(repository_identity, "repository_identity"),
        "commit_sha": _sha_string(commit_sha, 40, "commit_sha"),
        "provider_run_id": _digits(provider_run_id, "provider_run_id"),
        **identities,
        "source_replay_sha256": canonical_sha256(source),
        "source_files": source["source_files"],
        "as_of_ns": source["as_of_ns"],
        "windows": source["windows"],
        "seeds": list(FROZEN_SEEDS),
        "ordered_row_ids": [row["row_id"] for row in rows],
        "ordered_perturbation_row_ids": {
            str(seed): [row["row_id"] for row in perturbations[str(seed)]] for seed in FROZEN_SEEDS
        },
        "ordered_transition_ids": [event["transition_id"] for event in transitions],
        "material_sha256": hash_execution_materials(source["execution_materials"]),
        "export_sha256": export_sha256,
    }
    checkpoint = {
        "schema_version": "time-space-export-checkpoint.v1",
        "layer": "time_space_shadow",
        "registration_status": "PENDING",
        "export_sha256": export_sha256,
        "provenance_sha256": canonical_sha256(provenance),
        "receipt": None,
        "as_of_ns": source["as_of_ns"],
        "generator_provider_run_id": provenance["provider_run_id"],
        "source_slice_sha256": _canonical_sha256(source["source_files"]),
    }
    return TimeSpaceExportBundle(export=export, provenance=provenance, checkpoint=checkpoint)


def hash_execution_materials(materials: Mapping[str, Any]) -> dict[str, str]:
    """Hash each production-material stream independently in canonical order."""

    payload = _plain_mapping(materials, "execution materials")
    if set(payload) != set(MATERIAL_NAMES):
        raise ValueError("execution materials must contain exactly snapshots, targets, orders, fills, equity, episodes")
    return {name: _canonical_sha256(payload[name]) for name in MATERIAL_NAMES}


def validate_registered_time_space_export(
    export: Mapping[str, Any],
    provenance: Mapping[str, Any],
    checkpoint: Mapping[str, Any],
    receipt: Mapping[str, Any] | None = None,
    *,
    provider_fetcher=None,
) -> str:
    """Return VERIFIED only through the checked-in independent trust root."""

    try:
        as_of_ns = int(export["as_of_ns"])
        trust, persisted_receipt, persisted_export, persisted_provenance = _load_persisted_registration(
            as_of_ns,
            receipt,
            export,
            provenance,
        )
        fetcher = provider_fetcher or _fetch_provider
        provider = fetcher(trust["remote_identity"], int(trust["provider_run_id"]))
        generator_provider = fetcher(trust["remote_identity"], int(trust["generator_provider_run_id"]))
        recomputed = recompute_time_space_export(
            ReplayRequest(
                execution_manifest=_PROJECT_ROOT / "config/research/registrations/time_space_shadow_execution_v1.json",
                hypothesis_manifest=_PROJECT_ROOT / "config/research/time_space_shadow_hypothesis_v1.json",
                as_of_ns=as_of_ns,
            ),
            repository_identity=str(provenance["repository_identity"]),
            commit_sha=str(provenance["commit_sha"]),
            provider_run_id=str(provenance["provider_run_id"]),
        )
    except (KeyError, TypeError, ValueError, OSError):
        return EXPORT_REGISTRATION_REQUIRED
    return _validate_registered_time_space_export_payloads(
        persisted_export,
        persisted_provenance,
        checkpoint,
        persisted_receipt,
        trust_payload=trust,
        provider_payload=provider,
        generator_provider_payload=generator_provider,
        generator_path=_GENERATOR_PATH,
        recomputed_bundle=recomputed,
    )


def _load_persisted_registration(
    as_of_ns: int,
    supplied_receipt: Mapping[str, Any] | None,
    supplied_export: Mapping[str, Any] | None,
    supplied_provenance: Mapping[str, Any] | None,
    *,
    registration_root: Path = _REGISTRATION_ROOT,
) -> tuple[dict[str, Any], dict[str, Any], dict[str, Any], dict[str, Any]]:
    """Load immutable in-repo evidence; expiring provider artifacts are not a runtime dependency."""

    registration_dir = Path(registration_root) / str(as_of_ns)
    trust = _load_canonical_json(
        registration_dir / "time_space_export_registration_trust_v1.json",
        "export registration trust root",
    )
    persisted_receipt = _load_canonical_json(
        registration_dir / _PERSISTED_RECEIPT_NAME,
        "persisted export registration receipt",
    )
    persisted_export = _load_canonical_json(
        registration_dir / _PERSISTED_EXPORT_NAME,
        "persisted canonical export",
    )
    persisted_provenance = _load_canonical_json(
        registration_dir / _PERSISTED_PROVENANCE_NAME,
        "persisted canonical provenance",
    )
    if (
        trust.get("registration_status") != VERIFIED
        or trust.get("remote_identity") != _CANONICAL_REPOSITORY
        or supplied_receipt is None
        or supplied_export is None
        or supplied_provenance is None
    ):
        raise ValueError("export registration is not verified")
    if _plain_mapping(supplied_receipt, "receipt") != persisted_receipt:
        raise ValueError("supplied receipt is not the immutable checked-in receipt")
    if _plain_mapping(supplied_export, "export") != persisted_export:
        raise ValueError("supplied export is not the immutable checked-in export")
    if _plain_mapping(supplied_provenance, "provenance") != persisted_provenance:
        raise ValueError("supplied provenance is not the immutable checked-in provenance")
    return trust, persisted_receipt, persisted_export, persisted_provenance


def _validate_registered_time_space_export_payloads(
    export: Mapping[str, Any],
    provenance: Mapping[str, Any],
    checkpoint: Mapping[str, Any],
    receipt: Mapping[str, Any],
    *,
    trust_payload: Mapping[str, Any],
    provider_payload: Mapping[str, Any] | None,
    generator_provider_payload: Mapping[str, Any] | None,
    generator_path: Path,
    recomputed_bundle: TimeSpaceExportBundle,
) -> str:
    """Pure validator used behind the pinned public gate."""

    try:
        _validate_registered(
            _plain_mapping(export, "export"),
            _plain_mapping(provenance, "provenance"),
            _plain_mapping(checkpoint, "checkpoint"),
            _plain_mapping(receipt, "receipt"),
            trust=_plain_mapping(trust_payload, "trust root"),
            provider=None if provider_payload is None else _plain_mapping(provider_payload, "provider receipt"),
            generator_provider=(
                None
                if generator_provider_payload is None
                else _plain_mapping(generator_provider_payload, "generator provider receipt")
            ),
            generator_path=Path(generator_path),
            recomputed_bundle=recomputed_bundle,
        )
    except (KeyError, TypeError, ValueError, OSError):
        return EXPORT_REGISTRATION_REQUIRED
    return VERIFIED


def _fetch_provider(remote_identity: str, provider_run_id: int) -> dict[str, Any]:
    if remote_identity != _CANONICAL_REPOSITORY:
        raise ValueError("provider lookup is restricted to the canonical repository")
    request = urllib.request.Request(
        f"https://api.github.com/repos/{_CANONICAL_REPOSITORY}/actions/runs/{provider_run_id}",
        headers={"Accept": "application/vnd.github+json", "User-Agent": "zenquant-time-space-export/1"},
    )
    with urllib.request.urlopen(request, timeout=30) as response:
        payload = json.loads(response.read().decode("utf-8"))
    if not isinstance(payload, dict):
        raise ValueError("provider checkpoint response must be an object")
    return payload


def _validate_registered(
    export: dict[str, Any],
    provenance: dict[str, Any],
    checkpoint: dict[str, Any],
    receipt: dict[str, Any],
    *,
    trust: dict[str, Any],
    provider: dict[str, Any] | None,
    generator_provider: dict[str, Any] | None,
    generator_path: Path,
    recomputed_bundle: TimeSpaceExportBundle,
) -> None:
    if set(export) != _EXPORT_KEYS or set(provenance) != _PROVENANCE_KEYS or set(checkpoint) != _CHECKPOINT_KEYS:
        raise ValueError("export envelope schema is invalid")
    if set(trust) != _TRUST_KEYS or trust.get("registration_status") != VERIFIED:
        raise ValueError("checkpoint trust root is not verified")
    if trust.get("remote_identity") != _CANONICAL_REPOSITORY:
        raise ValueError("checkpoint trust root is not pinned to the canonical repository")
    if (
        checkpoint.get("registration_status") != "PENDING"
        or checkpoint.get("receipt") is not None
        or provider is None
        or generator_provider is None
    ):
        raise ValueError("checkpoint is not independently registered")
    if export.get("schema_version") != "time-space-shadow-export.v1" or export.get("layer") != "time_space_shadow":
        raise ValueError("export identity is invalid")
    if provenance.get("schema_version") != "time-space-shadow-export-provenance.v1":
        raise ValueError("provenance identity is invalid")
    if checkpoint.get("schema_version") != "time-space-export-checkpoint.v1":
        raise ValueError("checkpoint identity is invalid")
    if trust.get("schema_version") != "time-space-export-registration-trust.v1":
        raise ValueError("trust identity is invalid")
    if export != recomputed_bundle.export or provenance != recomputed_bundle.provenance:
        raise ValueError("registered export does not match independent replay recomputation")
    if export.get("seeds") != list(FROZEN_SEEDS) or provenance.get("seeds") != list(FROZEN_SEEDS):
        raise ValueError("frozen seeds changed")
    _validate_windows(export.get("windows"))
    if provenance.get("windows") != export["windows"] or provenance.get("as_of_ns") != export.get("as_of_ns"):
        raise ValueError("window or as_of provenance mismatch")
    identity_names = (
        "hypothesis_sha256",
        "execution_sha256",
        "formula_sha256",
        "config_sha256",
        "source_set_sha256",
    )
    if any(provenance.get(name) != export.get(name) for name in identity_names):
        raise ValueError("registered identities diverged")
    if provenance.get("generator_code_sha256") != _sha256_file(generator_path):
        raise ValueError("generator code changed")
    dependencies, tree_sha256 = _generator_dependency_identity()
    if (
        provenance.get("generator_dependencies") != dependencies
        or provenance.get("generator_tree_sha256") != tree_sha256
    ):
        raise ValueError("generator dependency identity changed")
    expected_semantic = canonical_sha256(
        {
            "schema_version": "time-space-export-generator.v1",
            "source_schema": "architecture-time-space-replay.v2",
            "export_schema": "time-space-shadow-export.v1",
            "decision_stages": list(DECISION_STAGES),
            "seeds": list(FROZEN_SEEDS),
            "row_keys": sorted(_ROW_KEYS),
        }
    )
    if provenance.get("generator_semantic_sha256") != expected_semantic:
        raise ValueError("generator semantic identity changed")
    rows = export.get("rows")
    if not isinstance(rows, list) or not rows:
        raise ValueError("export rows are invalid")
    _validate_export_rows(rows, export)
    perturbations = export.get("perturbation_paths")
    if not isinstance(perturbations, dict) or set(perturbations) != {str(seed) for seed in FROZEN_SEEDS}:
        raise ValueError("perturbation paths are invalid")
    for seed in FROZEN_SEEDS:
        _validate_export_rows(perturbations[str(seed)], export)
    if provenance.get("ordered_row_ids") != [row["row_id"] for row in rows]:
        raise ValueError("ordered row identities changed")
    expected_perturbation_ids = {
        str(seed): [row["row_id"] for row in perturbations[str(seed)]] for seed in FROZEN_SEEDS
    }
    if provenance.get("ordered_perturbation_row_ids") != expected_perturbation_ids:
        raise ValueError("perturbation row identities changed")
    transitions = _build_transitions(rows)
    if export.get("market_state_transition_events") != transitions:
        raise ValueError("market-state transitions changed")
    if provenance.get("ordered_transition_ids") != [item["transition_id"] for item in transitions]:
        raise ValueError("transition identities changed")
    export_sha256 = canonical_sha256(export)
    if provenance.get("export_sha256") != export_sha256 or checkpoint.get("export_sha256") != export_sha256:
        raise ValueError("export digest mismatch")
    provenance_sha256 = canonical_sha256(provenance)
    if checkpoint.get("provenance_sha256") != provenance_sha256:
        raise ValueError("provenance digest mismatch")
    if set(receipt) != _RECEIPT_KEYS:
        raise ValueError("checkpoint receipt is invalid")
    checkpoint_bytes = (json.dumps(checkpoint, sort_keys=True, separators=(",", ":")) + "\n").encode()
    manifest_sha256 = hashlib.sha256(checkpoint_bytes).hexdigest()
    if (
        receipt.get("schema_version") != "time-space-export-registration-receipt.v1"
        or receipt.get("manifest_path") != _checkpoint_manifest_path(int(export["as_of_ns"]))
        or receipt.get("manifest_sha256") != manifest_sha256
        or receipt.get("export_sha256") != export_sha256
        or receipt.get("provenance_sha256") != provenance_sha256
        or receipt.get("generator_artifact", {}).get("export_sha256") != export_sha256
        or receipt.get("generator_artifact", {}).get("provenance_sha256") != provenance_sha256
        or receipt.get("generator_artifact", {}).get("source_slice_sha256") != checkpoint["source_slice_sha256"]
        or receipt.get("generator_artifact", {}).get("id") != trust["generator_artifact_id"]
        or receipt.get("generator_artifact", {}).get("name") != trust["generator_artifact_name"]
        or receipt.get("provider_run_id") != trust["provider_run_id"]
        or receipt.get("commit_sha") != trust["commit_sha"]
        or receipt.get("remote_identity") != _CANONICAL_REPOSITORY
        or receipt.get("provider_created_at_ns") != trust["provider_created_at_ns"]
        or receipt.get("workflow") != {"id": trust["checkpoint_workflow_id"], "path": trust["checkpoint_workflow_path"]}
        or receipt.get("event") != "push"
        or receipt.get("ref") != f"refs/heads/{trust['checkpoint_branch']}"
        or receipt.get("artifact") != {"id": trust["checkpoint_artifact_id"], "name": trust["checkpoint_artifact_name"]}
        or trust.get("checkpoint_manifest_sha256") != manifest_sha256
        or trust.get("receipt_artifact_sha256") != _artifact_sha256(receipt)
    ):
        raise ValueError("checkpoint receipt is not pinned")
    if (
        provider.get("id") != int(receipt["provider_run_id"])
        or provider.get("head_sha") != receipt["commit_sha"]
        or provider.get("conclusion") != "success"
        or provider.get("event") != "push"
        or provider.get("head_branch") != trust["checkpoint_branch"]
        or provider.get("path") != trust["checkpoint_workflow_path"]
        or provider.get("workflow_id") != trust["checkpoint_workflow_id"]
        or _provider_created_at_ns(provider.get("created_at")) != receipt["provider_created_at_ns"]
        or not isinstance(provider.get("repository"), dict)
        or provider["repository"].get("full_name") != _CANONICAL_REPOSITORY
    ):
        raise ValueError("provider receipt does not match checkpoint")
    if provenance.get("repository_identity") != _CANONICAL_REPOSITORY:
        raise ValueError("repository identity mismatch")
    if provenance.get("commit_sha") != trust.get("generator_commit_sha") or provenance.get(
        "provider_run_id"
    ) != trust.get("generator_provider_run_id"):
        raise ValueError("generator checkout or provider identity is not pinned")
    if (
        generator_provider.get("id") != int(provenance["provider_run_id"])
        or generator_provider.get("head_sha") != provenance["commit_sha"]
        or generator_provider.get("conclusion") != "success"
        or generator_provider.get("workflow_id") != trust["generator_workflow_id"]
        or generator_provider.get("path") != trust["generator_workflow_path"]
        or generator_provider.get("event") != trust["generator_event"]
        or generator_provider.get("head_branch") != trust["generator_branch"]
        or _provider_created_at_ns(generator_provider.get("created_at")) != trust["generator_created_at_ns"]
        or not isinstance(generator_provider.get("repository"), dict)
        or generator_provider["repository"].get("full_name") != _CANONICAL_REPOSITORY
    ):
        raise ValueError("generator provider does not attest the pinned checkout")


def _provider_created_at_ns(value: Any) -> int:
    if not isinstance(value, str):
        raise ValueError("provider created_at is invalid")
    try:
        parsed = datetime.strptime(value, "%Y-%m-%dT%H:%M:%SZ").replace(tzinfo=UTC)
    except ValueError as exc:
        raise ValueError("provider created_at is invalid") from exc
    return int(parsed.timestamp()) * 1_000_000_000


def _checkpoint_manifest_path(as_of_ns: int) -> str:
    if isinstance(as_of_ns, bool) or not isinstance(as_of_ns, int) or as_of_ns % FOUR_HOURS_NS:
        raise ValueError("checkpoint as_of is not a confirmed 4h cutoff")
    return f"config/research/registrations/time_space_exports/{as_of_ns}/time_space_shadow_export_checkpoint_v1.json"


def _validate_source(source: dict[str, Any]) -> None:
    if set(source) != {
        "schema_version",
        "as_of_ns",
        "windows",
        "seeds",
        "canonical",
        "perturbations",
        "execution_materials",
        "source_files",
    }:
        raise ValueError("source replay schema is invalid")
    if source["schema_version"] != "architecture-time-space-replay.v2":
        raise ValueError("source replay identity is invalid")
    if source["seeds"] != list(FROZEN_SEEDS):
        raise ValueError("source replay seeds are not frozen")
    _validate_windows(source["windows"])
    as_of = source["as_of_ns"]
    if isinstance(as_of, bool) or not isinstance(as_of, int) or as_of % FOUR_HOURS_NS:
        raise ValueError("source as_of must be an exact 4h cutoff")
    perturbations = source["perturbations"]
    if not isinstance(perturbations, dict) or set(perturbations) != {str(seed) for seed in FROZEN_SEEDS}:
        raise ValueError("source perturbation paths are invalid")
    hash_execution_materials(source["execution_materials"])
    source_files = source["source_files"]
    if not isinstance(source_files, list) or not source_files:
        raise ValueError("source replay must bind registered source files")
    if any(not isinstance(item, dict) or set(item) != {"source_id", "path", "sha256"} for item in source_files):
        raise ValueError("source file identity is invalid")


def _validate_windows(windows: Any) -> None:
    if not isinstance(windows, dict) or set(windows) != {"reused", "embargo", "forward"}:
        raise ValueError("evidence windows are invalid")
    reused, embargo, forward = windows["reused"], windows["embargo"], windows["forward"]
    if not (
        isinstance(reused, list)
        and isinstance(embargo, list)
        and isinstance(forward, list)
        and len(reused) == len(embargo) == len(forward) == 2
        and all(isinstance(value, int) and not isinstance(value, bool) for value in (*reused, *embargo))
        and isinstance(forward[0], int)
        and not isinstance(forward[0], bool)
        and forward[1] is None
        and reused[0] < reused[1] == embargo[0] < embargo[1] == forward[0]
    ):
        raise ValueError("evidence windows overlap or are malformed")


def _build_rows(path: Any, identities: dict[str, str]) -> list[dict[str, Any]]:
    if not isinstance(path, dict) or set(path) != {"market_rows", "stage_rows"}:
        raise ValueError("replay path schema is invalid")
    market_rows = path["market_rows"]
    stages = path["stage_rows"]
    if not isinstance(market_rows, list) or not market_rows:
        raise ValueError("market rows are empty")
    if not isinstance(stages, dict) or tuple(stages) != DECISION_STAGES:
        raise ValueError("decision stages must be exactly A1_SIGN, A5_SHADOW, FILLED_POSITION")
    indexed: dict[str, dict[int, dict[str, Any]]] = {}
    for stage in DECISION_STAGES:
        stage_rows = stages[stage]
        if not isinstance(stage_rows, list):
            raise ValueError(f"{stage} rows must be a list")
        stage_index: dict[int, dict[str, Any]] = {}
        for row in stage_rows:
            if not isinstance(row, dict) or set(row) != _STAGE_ROW_KEYS:
                raise ValueError(f"{stage} row schema is invalid")
            cutoff = row["cutoff_time_ns"]
            if cutoff in stage_index:
                raise ValueError(f"{stage} contains duplicate cutoffs")
            stage_index[cutoff] = row
        indexed[stage] = stage_index
    rows: list[dict[str, Any]] = []
    previous: int | None = None
    for market in market_rows:
        if not isinstance(market, dict) or set(market) != _MARKET_KEYS:
            raise ValueError("market row schema is invalid")
        cutoff = market["cutoff_time_ns"]
        if isinstance(cutoff, bool) or not isinstance(cutoff, int) or cutoff % FOUR_HOURS_NS:
            raise ValueError("market cutoff is not on the 4h grid")
        if previous is not None and cutoff <= previous:
            raise ValueError("market cutoffs must be strictly increasing")
        previous = cutoff
        if any(cutoff not in indexed[stage] for stage in DECISION_STAGES):
            raise ValueError("stage join is incomplete")
        decisions = {stage: indexed[stage][cutoff]["decision_id"] for stage in DECISION_STAGES}
        values = {stage: indexed[stage][cutoff]["value"] for stage in DECISION_STAGES}
        row = {
            **market,
            "formula_hash": identities["formula_sha256"],
            "config_hash": identities["config_sha256"],
            "source_data_hash": identities["source_set_sha256"],
            "decision_id_by_stage": decisions,
            "alpha_sign": values["A1_SIGN"],
            "approved_target": values["A5_SHADOW"],
            "filled_position": values["FILLED_POSITION"],
        }
        _validate_numeric_row(row)
        row["row_id"] = canonical_sha256(row)
        rows.append(row)
    cutoffs = {row["cutoff_time_ns"] for row in market_rows}
    if any(set(indexed[stage]) != cutoffs for stage in DECISION_STAGES):
        raise ValueError("stage rows contain cutoffs outside the market path")
    return rows


def _validate_export_rows(rows: Any, export: dict[str, Any]) -> None:
    if not isinstance(rows, list) or not rows:
        raise ValueError("export path is empty")
    previous = None
    for row in rows:
        if not isinstance(row, dict) or set(row) != _ROW_KEYS:
            raise ValueError("export row schema is invalid")
        cutoff = row["cutoff_time_ns"]
        if previous is not None and cutoff <= previous:
            raise ValueError("export rows are not ordered")
        previous = cutoff
        if set(row["decision_id_by_stage"]) != set(DECISION_STAGES):
            raise ValueError("export decision stages changed")
        if (
            row["formula_hash"] != export["formula_sha256"]
            or row["config_hash"] != export["config_sha256"]
            or row["source_data_hash"] != export["source_set_sha256"]
        ):
            raise ValueError("export row identity changed")
        expected = canonical_sha256({key: value for key, value in row.items() if key != "row_id"})
        if row["row_id"] != expected:
            raise ValueError("export row digest changed")
        _validate_numeric_row(row)


def _validate_numeric_row(row: dict[str, Any]) -> None:
    for name in ("close", "alpha_sign", "approved_target", "filled_position"):
        value = row[name]
        if isinstance(value, bool) or not isinstance(value, int | float) or not math.isfinite(float(value)):
            raise ValueError(f"{name} must be finite")
    if row["alpha_sign"] not in (-1, 0, 1):
        raise ValueError("A1_SIGN value must be -1, 0, or 1")


def _build_transitions(rows: list[dict[str, Any]]) -> list[dict[str, Any]]:
    states = build_time_space_states(tuple(_state_input(row) for row in rows))
    events = []
    for prior, current in pairwise(states):
        if prior.market_structure_side == current.market_structure_side:
            continue
        event = {
            "cutoff_time_ns": current.cutoff_time_ns,
            "prior_state_id": prior.state_id,
            "state_id": current.state_id,
        }
        event["transition_id"] = canonical_sha256(event)
        events.append(event)
    return events


def _state_input(row: dict[str, Any]) -> TimeSpaceInputRow:
    values = {key: row[key] for key in _MARKET_KEYS}
    return TimeSpaceInputRow(
        **values,
        formula_hash=row["formula_hash"],
        config_hash=row["config_hash"],
        source_data_hash=row["source_data_hash"],
        decision_id_by_stage=tuple(sorted(row["decision_id_by_stage"].items())),
    )


def _plain_mapping(value: Mapping[str, Any], name: str) -> dict[str, Any]:
    if not isinstance(value, Mapping):
        raise TypeError(f"{name} must be a mapping")
    return copy.deepcopy(dict(value))


def _load_json(path: Path, name: str) -> dict[str, Any]:
    try:
        payload = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise ValueError(f"cannot load {name}") from exc
    if not isinstance(payload, dict):
        raise ValueError(f"{name} must be an object")
    return payload


def _load_canonical_json(path: Path, name: str) -> dict[str, Any]:
    try:
        content = path.read_bytes()
        payload = json.loads(content.decode("utf-8"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise ValueError(f"cannot load {name}") from exc
    if not isinstance(payload, dict):
        raise ValueError(f"{name} must be an object")
    canonical = (json.dumps(payload, sort_keys=True, separators=(",", ":"), allow_nan=False) + "\n").encode()
    if content != canonical:
        raise ValueError(f"{name} bytes are not canonical")
    return payload


def _required_string(value: Any, name: str) -> str:
    if not isinstance(value, str) or not value or value.strip() != value:
        raise ValueError(f"{name} must be a non-empty canonical string")
    return value


def _sha_string(value: Any, length: int, name: str) -> str:
    text = _required_string(value, name)
    if len(text) != length or any(char not in "0123456789abcdef" for char in text):
        raise ValueError(f"{name} must be lowercase hexadecimal")
    return text


def _digits(value: Any, name: str) -> str:
    text = _required_string(value, name)
    if not text.isdigit() or text.startswith("0"):
        raise ValueError(f"{name} must be canonical decimal digits")
    return text


def _sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def _generator_dependency_identity(*, read_bytes=None) -> tuple[list[dict[str, str]], str]:
    loader = read_bytes or (lambda path: path.read_bytes())
    python_paths = {path.relative_to(_PROJECT_ROOT).as_posix() for path in (_PROJECT_ROOT / "zenquant").rglob("*.py")}
    dependency_paths = sorted(python_paths | set(_STATIC_DEPENDENCY_PATHS))
    files = []
    for relative in dependency_paths:
        path = _PROJECT_ROOT / relative
        files.append({"path": relative, "sha256": hashlib.sha256(loader(path)).hexdigest()})
    return files, canonical_sha256({"files": files})


def _canonical_sha256(value: Any) -> str:
    encoded = json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode("utf-8")
    return hashlib.sha256(encoded).hexdigest()


def _artifact_sha256(value: Any) -> str:
    encoded = (json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False) + "\n").encode("utf-8")
    return hashlib.sha256(encoded).hexdigest()
